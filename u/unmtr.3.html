sous-routine cunmtr (character side, character uplo, character trans, integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer lwork, integer info)

CUNMTR

But :

!>
!> CUNMTR écrase la matrice complexe générale M par N, C, avec
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'C':      Q**H * C       C * Q**H
!>
!> où Q est une matrice unitaire complexe d'ordre nq, avec nq = m si
!> SIDE = 'L' et nq = n si SIDE = 'R'. Q est définie comme le produit de
!> nq-1 réflecteurs élémentaires, comme retourné par CHETRD :
!>
!> si UPLO = 'U', Q = H(nq-1) . . . H(2) H(1);
!>
!> si UPLO = 'L', Q = H(1) H(2) . . . H(nq-1).
!>

Paramètres

SIDE 
!>          SIDE est CHARACTER*1
!>          = 'L' : appliquer Q ou Q**H depuis la gauche ;
!>          = 'R' : appliquer Q ou Q**H depuis la droite.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U' : le triangle supérieur de A contient les réflecteurs
!>                 élémentaires de CHETRD ;
!>          = 'L' : le triangle inférieur de A contient les réflecteurs
!>                 élémentaires de CHETRD.
!>

TRANS

!>          TRANS est CHARACTER*1
!>          = 'N' : Pas de transposée, appliquer Q ;
!>          = 'C' : Transposée conjuguée, appliquer Q**H.
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice C. M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice C. N >= 0.
!>

A

!>          A est un tableau COMPLEXE, dimension
!>                               (LDA,M) si SIDE = 'L'
!>                               (LDA,N) si SIDE = 'R'
!>          Les vecteurs qui définissent les réflecteurs élémentaires, tels que
!>          retournés par CHETRD.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.
!>          LDA >= max(1,M) si SIDE = 'L' ; LDA >= max(1,N) si SIDE = 'R'.
!>

TAU

!>          TAU est un tableau COMPLEXE, dimension
!>                               (M-1) si SIDE = 'L'
!>                               (N-1) si SIDE = 'R'
!>          TAU(i) doit contenir le facteur scalaire du réflecteur
!>          élémentaire H(i), tel que retourné par CHETRD.
!>

C

!>          C est un tableau COMPLEXE, dimension (LDC,N)
!>          À l'entrée, la matrice M par N, C.
!>          À la sortie, C est écrasé par Q*C ou Q**H*C ou C*Q**H ou C*Q.
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale du tableau C. LDC >= max(1,M).
!>

WORK

!>          WORK est un tableau COMPLEXE, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) contient le LWORK optimal.
!>

LWORK

!>          LWORK est INTEGER
!>          La dimension du tableau WORK.
!>          Si SIDE = 'L', LWORK >= max(1,N) ;
!>          si SIDE = 'R', LWORK >= max(1,M).
!>          Pour une performance optimale, LWORK >= N*NB si SIDE = 'L', et
!>          LWORK >= M*NB si SIDE = 'R', où NB est la taille de bloc optimale.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, renvoie
!>          cette valeur dans la première entrée du tableau WORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, alors le i-ème argument avait une valeur illégale
!>
