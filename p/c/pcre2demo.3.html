CODE SOURCE

/*************************************************
*           PROGRAMME DE DÉMONSTRATION PCRE2         *
*************************************************/
/* Ceci est un programme de démonstration pour illustrer une méthode simple
d'utilisation de la bibliothèque d'expressions régulières PCRE2 à partir
d'un programme C. Voir la documentation de pcre2sample pour une courte discussion
("man pcre2sample" si vous avez les pages de manuel PCRE2 installées). PCRE2 est
une API révisée de la bibliothèque, et est incompatible avec l'API PCRE originale.
Il existe en fait trois bibliothèques, chacune supportant une largeur d'unités
de code différente. Ce programme de démonstration utilise la bibliothèque 8 bits.
Par défaut, chaque unité de code est traitée comme un caractère séparé, mais si
le motif commence par "(*UTF)", lui-même et le sujet sont traités comme des chaînes
UTF-8, où les caractères peuvent occuper plusieurs unités de code.
Dans les environnements de type Unix, si PCRE2 est installé dans vos bibliothèques
système standard, vous devriez pouvoir compiler ce programme en utilisant cette
commande :
cc -Wall pcre2demo.c -lpcre2-8 -o pcre2demo
Si PCRE2 n'est pas installé dans un endroit standard, il est probable qu'il soit
installé avec le support du mécanisme pkg-config. Si vous avez pkg-config, vous
pouvez compiler ce programme en utilisant cette commande :
cc -Wall pcre2demo.c `pkg-config --cflags --libs libpcre2-8` -o pcre2demo
Si vous n'avez pas pkg-config, vous devrez peut-être utiliser quelque chose comme ceci :
cc -Wall pcre2demo.c -I/usr/local/include -L/usr/local/lib \
  -R/usr/local/lib -lpcre2-8 -o pcre2demo
Remplacez "/usr/local/include" et "/usr/local/lib" par l'emplacement des fichiers
d'en-tête et de bibliothèque de PCRE2 sur votre système. Seuls certains systèmes
d'exploitation (Solaris en fait partie) utilisent l'option -R.
Compilation sous Windows :
Si vous souhaitez lier statiquement ce programme à un fichier .a non-dll, vous devez
définir PCRE2_STATIC avant d'inclure pcre2.h, donc dans cet environnement,
décommentez la ligne suivante. */
/* #define PCRE2_STATIC */
/* La macro PCRE2_CODE_UNIT_WIDTH doit être définie avant d'inclure pcre2.h.
Pour un programme qui utilise une seule largeur d'unités de code, la définir à 8, 16 ou 32
permet d'utiliser des noms de fonctions génériques tels que pcre2_compile(). Notez
que changer simplement 8 en 16 (par exemple) n'est pas suffisant pour convertir
ce programme afin qu'il traite des caractères de 16 bits. Même dans un environnement
entièrement en 16 bits, où les fonctions de manipulation de chaînes telles que strcmp()
et printf() fonctionnent avec des caractères de 16 bits, le code de gestion de la
table des sous-chaînes nommées devra toujours être modifié. */
#define PCRE2_CODE_UNIT_WIDTH 8
#include <stdio.h>
#include <string.h>
#include <pcre2.h>
/**************************************************************************
* Voici le programme. L'API inclut le concept de "contextes" pour          *
* configurer des exigences d'interface inhabituelles pour la compilation et *
* la correspondance, telles que des gestionnaires de mémoire personnalisés et *
* des définitions de nouvelle ligne non standard.                            *
* Ce programme ne fait rien de tout cela, il n'utilise donc aucun contexte,  *
* passant toujours NULL là où un contexte pourrait être donné.             *
**************************************************************************/
int main(int argc, char **argv)
{
pcre2_code *re;
PCRE2_SPTR pattern;     /* PCRE2_SPTR est un pointeur vers des unités de code non signées */
PCRE2_SPTR subject;     /* de la largeur appropriée (dans ce cas, 8 bits). */
PCRE2_SPTR name_table;
int errornumber;
int find_all, caseless_match;
int i;
int rc;
uint32_t namecount;
uint32_t name_entry_size;
PCRE2_SIZE erroroffset;
PCRE2_SIZE *ovector;
PCRE2_SIZE ovector_last[2];
PCRE2_SIZE subject_length;
pcre2_match_data *match_data;
/**************************************************************************
* Tout d'abord, traitons la ligne de commande. Options :                   *
* - "-g" pour demander une correspondance répétée afin de trouver toutes les     *
*   occurrences, comme l'option /g de Perl. Nous définissons la variable       *
*   find_all à une valeur non nulle si l'option -g est présente.              *
* - "-i" pour demander une correspondance sans distinction de casse, comme      *
*   l'option /i de Perl. Nous définissons la variable caseless_match à          *
*   PCRE2_CASELESS si l'option -i est présente.                               *
**************************************************************************/
find_all = 0;
caseless_match = 0;
for (i = 1; i < argc; i++)
  {
  if (strcmp(argv[i], "-g") == 0) find_all = 1;
  else if (strcmp(argv[i], "-i") == 0) caseless_match = PCRE2_CASELESS;
  else if (argv[i][0] == '-')
    {
    printf("Option non reconnue %s\n", argv[i]);
    return 1;
    }
  else break;
  }
/* Après les options, nous exigeons exactement deux arguments, qui sont le motif
et la chaîne de sujet. */
if (argc - i != 2)
  {
  printf("Exactement deux arguments requis : une expression régulière et une chaîne de sujet\n");
  return 1;
  }
/* Le motif et le sujet sont des arguments char, ils peuvent donc être simplement
convertis en PCRE2_SPTR car nous travaillons avec des unités de code 8 bits. La
longueur du sujet est convertie en PCRE2_SIZE pour la forme, bien que PCRE2_SIZE
soit en fait défini comme size_t. */
pattern = (PCRE2_SPTR)argv[i];
subject = (PCRE2_SPTR)argv[i+1];
subject_length = (PCRE2_SIZE)strlen((char *)subject);
/*************************************************************************
* Nous allons maintenant compiler le motif d'expression régulière, et gérer *
* toute erreur détectée.                                                *
*************************************************************************/
re = pcre2_compile(
  pattern,               /* le motif */
  PCRE2_ZERO_TERMINATED, /* indique que le motif est terminé par zéro */
  caseless_match,        /* éventuellement activer la correspondance sans casse */
  &errornumber,          /* pour le numéro d'erreur */
  &erroroffset,          /* pour le décalage d'erreur */
  NULL);                 /* utiliser le contexte de compilation par défaut */
/* La compilation a échoué : afficher le message d'erreur et quitter. */
if (re == NULL)
  {
  PCRE2_UCHAR buffer[256];
  pcre2_get_error_message(errornumber, buffer, sizeof(buffer));
  printf("La compilation PCRE2 a échoué à l'offset %d : %s\n", (int)erroroffset,
    buffer);
  return 1;
  }
/*************************************************************************
* Si la compilation a réussi, nous appelons PCRE2 à nouveau, afin de faire une *
* correspondance de motif avec la chaîne de sujet. Ceci ne fait qu'UNE     *
* correspondance. Si une correspondance supplémentaire est nécessaire, elle sera *
* effectuée ci-dessous. Avant d'exécuter la correspondance, nous devons     *
* configurer un bloc match_data pour stocker le résultat.                   *
*************************************************************************/
match_data = pcre2_match_data_create_from_pattern(re, NULL);
if (match_data == NULL)
  {
  printf("La création de match_data a échoué\n");
  pcre2_code_free(re);  /* Libérer la mémoire du code compilé */
  return 1;
  }
/* Nous pouvons obtenir la table des noms et le nombre de noms, au cas où
une correspondance serait réussie. */
rc = pcre2_pattern_info(re, PCRE2_INFO_NAMECOUNT, &namecount, sizeof(namecount));
if (rc == 0)
  {
  rc = pcre2_pattern_info(re, PCRE2_INFO_NAMETABLE, &name_table,
    sizeof(name_table));
  if (rc != 0) name_table = NULL; /* L'échec signifie pas de noms */
  }
else
  {
  namecount = 0;
  name_table = NULL;
  }
/* Pour la table des sous-chaînes nommées, nous avons besoin de savoir quelle est
la taille de chaque entrée. */
rc = pcre2_pattern_info(re, PCRE2_INFO_NAMEENTRYSIZE, &name_entry_size,
  sizeof(name_entry_size));
if (rc != 0) name_entry_size = 0;
/*************************************************************************
* Maintenant, exécutez le motif contre le sujet. Notez que le sujet est passé *
* en tant que PCRE2_SPTR, et sa longueur comme PCRE2_SIZE.                 *
*************************************************************************/
rc = pcre2_match(
  re,                   /* le compilé regex */
  subject,              /* le sujet à rechercher */
  subject_length,       /* la longueur du sujet */
  0,                    /* décalage de départ dans le sujet */
  0,                    /* options de correspondance */
  match_data,           /* bloc de données de correspondance */
  NULL);                /* utiliser le contexte de correspondance par défaut */
/* Si une correspondance a été trouvée, imprimez les détails. */
if (rc < 0)
{
  if (rc == PCRE2_ERROR_NOMATCH)
    {
    printf("Aucune correspondance trouvée\n");
    }
  else
    {
    PCRE2_UCHAR buffer[256];
    pcre2_get_error_message(rc, buffer, sizeof(buffer));
    printf("Erreur lors de la correspondance : %s\n", buffer);
    }
}
else
{
  PCRE2_SIZE *offsets;
  PCRE2_SPTR names;
  ovector = pcre2_get_ovector_pointer(match_data);
  ovector_last[0] = ovector[0];  /* Stocker le début de la première correspondance */
  ovector_last[1] = ovector[1];  /* Stocker la fin de la première correspondance */
  printf("Correspondance trouvée !\n");
  printf("Correspondance de la chaîne : ");
  for (i = 0; i < rc; i++)
    {
    printf("%d/%d=%.*s\n", i, namecount,
      (int)(ovector[2*i+1] - ovector[2*i]),
      subject + ovector[2*i]);
    /* Si nous avons des noms, essayons de les imprimer. */
    if (namecount > 0)
      {
      names = name_table + (i * name_entry_size);
      printf("  (Nommé : %.*s)\n", (int)names[0], (PCRE2_SPTR)(names + 1));
      }
    }
  /* Si find_all est activé, continuez à rechercher des correspondances. */
  if (find_all)
    {
    PCRE2_SIZE subject_offset = ovector_last[1]; /* Commencez après la dernière correspondance */
    for (;;)
      {
      rc = pcre2_match(
        re,                /* le compilé regex */
        subject,           /* le sujet à rechercher */
        subject_length,    /* la longueur du sujet */
        subject_offset,    /* décalage de départ dans le sujet */
        0,                 /* options de correspondance */
        match_data,        /* bloc de données de correspondance */
        NULL);             /* utiliser le contexte de correspondance par défaut */
      if (rc < 0)
        {
        if (rc == PCRE2_ERROR_NOMATCH)
          {
          printf("Plus de correspondances.\n");
          break;
          }
        else
          {
          PCRE2_UCHAR buffer[256];
          pcre2_get_error_message(rc, buffer, sizeof(buffer));
          printf("Erreur lors de la correspondance répétée : %s\n", buffer);
          break;
          }
        }
      /* Nous avons trouvé une autre correspondance. */
      ovector = pcre2_get_ovector_pointer(match_data);
      printf("Correspondance suivante : ");
      for (i = 0; i < rc; i++)
        {
        printf("%d/%d=%.*s\n", i, namecount,
          (int)(ovector[2*i+1] - ovector[2*i]),
          subject + ovector[2*i]);
        /* Si nous avons des noms, essayons de les imprimer. */
        if (namecount > 0)
          {
          names = name_table + (i * name_entry_size);
          printf("  (Nommé : %.*s)\n", (int)names[0], (PCRE2_SPTR)(names + 1));
          }
        }
      subject_offset = ovector[1]; /* Commencez après cette correspondance */
      }
    }
}
/* Libérez la mémoire allouée pour le bloc de données de correspondance et le
code compilé. */
pcre2_match_data_free(match_data);
pcre2_code_free(re);
return 0;
}
/* End of pcre2demo.c */
