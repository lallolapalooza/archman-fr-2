DESCRIPTION

Ces fonctions servent à étendre et/ou modifier le comportement de la famille des fonctions printf(3).

register_printf_specifier()

VALEUR DE RETOUR

register_printf_specifier(), register_printf_modifier(), et register_printf_type() retournent zéro en cas de succès, ou -1 en cas d'erreur.

Callbacks

Le callback de type printf_function doit retourner le nombre de caractères écrits, ou -1 en cas d'erreur.

Le callback de type printf_arginfo_size_function doit retourner le nombre d'arguments à analyser par ce spécificateur. Il passe également des informations sur le type de ces arguments à l'appelant via argtypes. En cas d'erreur, il doit retourner -1.

ERREURS

EINVAL
Le spécificateur n'était pas un caractère valide.

NAME

register_printf_specifier - register a printf format specifier

SYNOPSIS

#include <stdio.h>
int register_printf_specifier(int c, printf_function func,
    printf_arginfo_function arginfo);
void unregister_printf_specifier(int c);

DESCRIPTION

The register_printf_specifier() function allows an application to register a new format specifier character, c, for use with the standard printf(3) family of functions. The function pointer func points to the function that will be called to print the argument associated with the specifier c. The function pointer arginfo points to a function that will be called to determine the type and number of arguments required by func.

The function func has the signature:

typedef int (*printf_function)(FILE *stream,
    const struct printf_info *info, const void *const args[]);

The function arginfo has the signature:

typedef int (*printf_arginfo_function)(const struct printf_info *info,
    size_t n, int argtypes[n], int size[n]);

The unregister_printf_specifier() function removes a previously registered specifier c.

The printf_info structure contains information about the format specifier, including flags, width, precision, and the specifier character itself. The args array contains pointers to the arguments that need to be printed. The argtypes array should be filled by the arginfo function with the types of the arguments required by func. The size array can be used to specify the size of each argument in bytes, which is primarily useful for varargs functions.

The func function should return the number of characters written to the stream, or a negative value if an output error occurred. The arginfo function should return the number of arguments required by func, or -1 if an error occurred.

The following constants are defined in <printf.h>:

  • PA_INT, PA_FLOAT, PA_DOUBLE, PA_LONG, PA_LONG_LONG, PA_CHAR, PA_SHORT, PA_POINTER, PA_STRING.
  • PA_FLAG_LONG, PA_FLAG_LONG_LONG, PA_FLAG_UNSIGNED.

These constants can be combined using the bitwise OR operator (|) to specify the type and modifiers of the arguments.

The printf_info structure contains the following members:

  • spec: The specifier character.
  • width: The minimum field width.
  • prec: The precision.
  • alt: True if the alternate form is requested (e.g., '#').
  • is_leading_zero: True if padding is with zeros (e.g., '0').
  • left: True if left-alignment is requested (e.g., '-').
  • group: True if grouping is requested (e.g., '' in some locales).
  • is_long_double: True if the argument is a long double.
  • is_long: True if the argument is a long int.
  • is_short: True if the argument is a short int.
  • is_char: True if the argument is a char.

The printf_arginfo_function is responsible for determining the types of arguments that the printf_function will consume. It should return the number of arguments consumed. The argtypes array should be filled with the appropriate type specifiers. The size array is optional and can be used to indicate the size of each argument in bytes.

STANDARDS

GNU.

HISTORY

register_printf_function(3) is an older function similar to register_printf_specifier(), and is now deprecated. That function can't handle user-defined types.

register_printf_specifier() supersedes register_printf_function(3).

EXAMPLES

The following example program registers the 'b' and 'B' specifiers to print integers in binary format, mirroring rules for other unsigned conversion specifiers like 'x' and 'u'. This can be used to print in binary prior to C23.

/* This code is in the public domain */
#include <err.h>
#include <limits.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/param.h>
#include <printf.h>
#define GROUP_SEP  '''
struct Printf_Pad {
    char    ch;
    size_t  len;
};
static int b_printf(FILE *stream, const struct printf_info *info,
                    const void *const args[]);
static int b_arginf_sz(const struct printf_info *info,
                    size_t n, int argtypes[n], int size[n]);
static uintmax_t b_value(const struct printf_info *info,
                    const void *arg);
static size_t b_bin_repr(char bin[UINTMAX_WIDTH],
                    const struct printf_info *info, const void *arg);
static size_t b_bin_len(const struct printf_info *info,
                    ptrdiff_t min_len);
static size_t b_pad_len(const struct printf_info *info,
                    ptrdiff_t bin_len);
static ssize_t b_print_prefix(FILE *stream,
                    const struct printf_info *info);
static ssize_t b_pad_zeros(FILE *stream, const struct printf_info *info,
                    ptrdiff_t min_len);
static ssize_t b_print_number(FILE *stream,
                    const struct printf_info *info,
                    const char bin[UINTMAX_WIDTH],
                    size_t min_len, size_t bin_len);
static char pad_ch(const struct printf_info *info);
static ssize_t pad_spaces(FILE *stream, size_t pad_len);

static int
b_printf(FILE *stream, const struct printf_info *info,
         const void *const args[])
{
    char               bin[UINTMAX_WIDTH];
    size_t             min_len, bin_len;
    ssize_t            len, tmp;
    struct Printf_Pad  pad = {0};
    len = 0;
    min_len = b_bin_repr(bin, info, args[0]);
    bin_len = b_bin_len(info, min_len);
    pad.ch = pad_ch(info);
    if (pad.ch == ' ')
        pad.len = b_pad_len(info, bin_len);
    /* Padding with ' ' (right aligned) */
    if ((pad.ch == ' ') && !info->left) {
        tmp = pad_spaces(stream, pad.len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* "0b"/"0B" prefix */
    if (info->alt) {
        tmp = b_print_prefix(stream, info);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* Padding with '0' */
    if (pad.ch == '0') {
        tmp = b_pad_zeros(stream, info, min_len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* Print number (including leading 0s to fill precision) */
    tmp = b_print_number(stream, info, bin, min_len, bin_len);
    if (tmp == EOF)
        return EOF;
    len += tmp;
    /* Padding with ' ' (left aligned) */
    if (info->left) {
        tmp = pad_spaces(stream, pad.len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    return len;
}
static int
b_arginf_sz(const struct printf_info *info, size_t n, int argtypes[n],
            [[maybe_unused]] int size[n])
{
    if (n < 1)
        return -1;
    if (info->is_long_double)
        argtypes[0] = PA_INT | PA_FLAG_LONG_LONG;
    else if (info->is_long)
        argtypes[0] = PA_INT | PA_FLAG_LONG;
    else
        argtypes[0] = PA_INT;
    return 1;
}
static uintmax_t
b_value(const struct printf_info *info, const void *arg)
{
    if (info->is_long_double)
        return *(const unsigned long long *)arg;
    if (info->is_long)
        return *(const unsigned long *)arg;
    /* short and char are both promoted to int */
    return *(const unsigned int *)arg;
}
static size_t
b_bin_repr(char bin[UINTMAX_WIDTH],
           const struct printf_info *info, const void *arg)
{
    size_t     min_len;
    uintmax_t  val;
    val = b_value(info, arg);
    bin[0] = '0';
    for (min_len = 0; val; min_len++) {
        bin[min_len] = '0' + (val % 2);
        val >>= 1;
    }
    return MAX(min_len, 1);
}
static size_t
b_bin_len(const struct printf_info *info, ptrdiff_t min_len)
{
    return MAX(info->prec, min_len);
}
static size_t
b_pad_len(const struct printf_info *info, ptrdiff_t bin_len)
{
    ptrdiff_t  pad_len;
    pad_len = info->width - bin_len;
    if (info->alt)
        pad_len -= 2;
    if (info->group)
        pad_len -= (bin_len - 1) / 4;
    return MAX(pad_len, 0);
}
static ssize_t
b_print_prefix(FILE *stream, const struct printf_info *info)
{
    ssize_t len;
    len = 0;
    if (fputc('0', stream) == EOF)
        return EOF;
    len++;
    if (fputc(info->spec, stream) == EOF)
        return EOF;
    len++;
    return len;
}
static ssize_t
b_pad_zeros(FILE *stream, const struct printf_info *info,
            ptrdiff_t min_len)
{
    ssize_t    len;
    ptrdiff_t  tmp;
    len = 0;
    tmp = info->width - (info->alt * 2);
    if (info->group)
        tmp -= tmp / 5 - !(tmp % 5);
    for (ptrdiff_t i = tmp - 1; i > min_len - 1; i--) {
        if (fputc('0', stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4))
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
static ssize_t
b_print_number(FILE *stream, const struct printf_info *info,
               const char bin[UINTMAX_WIDTH],
               size_t min_len, size_t bin_len)
{
    ssize_t  len;
    len = 0;
    /* Print leading zeros to fill precision */
    for (size_t i = bin_len - 1; i > min_len - 1; i--) {
        if (fputc('0', stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4))
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    /* Print number */
    for (size_t i = min_len - 1; i < min_len; i--) {
        if (fputc(bin[i], stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4) || !i)
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
static char
pad_ch(const struct printf_info *info)
{
    if ((info->prec != -1) || (info->pad == ' ') || info->left)
        return ' ';
    return '0';
}
static ssize_t
pad_spaces(FILE *stream, size_t pad_len)
{
    ssize_t  len;
    len = 0;
    for (size_t i = pad_len - 1; i < pad_len; i--) {
        if (fputc(' ', stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
