NOM

pacutils-mtree - lire les données mtree pour les paquets installés

SYNOPSIS

#include <pacutils/mtree.h>
typedef struct pu_mtree_t {
  char *path;
  char *type;
  uid_t uid;
  gid_t gid;
  mode_t mode;
  off_t size;
  char *md5digest;
  char *sha256digest;
} pu_mtree_t;
typedef struct {
  FILE *stream;
  int eof;
  pu_mtree_t defaults;
} pu_mtree_reader_t;
pu_mtree_reader_t *pu_mtree_reader_open_stream(FILE *stream);
pu_mtree_reader_t *pu_mtree_reader_open_package(alpm_handle_t *h, alpm_pkg_t *p);
pu_mtree_t *pu_mtree_reader_next(pu_mtree_reader_t *reader, pu_mtree_t *dest);
void pu_mtree_reader_free(pu_mtree_reader_t *reader);
void pu_mtree_free(pu_mtree_t *mtree);
/* déprécié */
alpm_list_t *pu_mtree_load_pkg_mtree(alpm_handle_t *handle, alpm_pkg_t *pkg);

DESCRIPTION

Les fonctions de lecture mtree des paquets sont fournies pour combler les lacunes du lecteur mtree de libarchive, notamment "md5digest" et "sha256digest" qui ne sont actuellement pas lus par libarchive. Le support est incomplet et ne devrait être utilisé que pour compléter le lecteur mtree natif de libalpm.

pu_mtree_reader_t *pu_mtree_reader_open_stream(FILE *stream);
Ouvre un flux de fichier pour l'analyse.
pu_mtree_reader_t *pu_mtree_reader_open_package(alpm_handle_t *h, alpm_pkg_t *p);
Ouvre le fichier mtree d'un paquet installé pour l'analyse. Les résultats sont indéfinis si "p" n'est pas un paquet installé localement.
pu_mtree_t *pu_mtree_reader_next(pu_mtree_reader_t *reader, pu_mtree_t *dest);
Lit et retourne la prochaine entrée dans le fichier mtree. Si "dest" est "NULL" un nouveau objet "pu_mtree_t" sera alloué et devra être libéré par l'appelant. Sinon, il sera rempli avec les données analysées. La mémoire allouée en interne sera automatiquement libérée si nécessaire.
void pu_mtree_free(pu_mtree_t *mtree);
Libère une structure "pu_mtree_t".
void pu_mtree_reader_free(pu_mtree_reader_t *reader);
Libère un objet "pu_mtree_reader_t".
alpm_list_t *pu_mtree_load_pkg_mtree(alpm_handle_t *handle, alpm_pkg_t *pkg);
Retourne une liste d'entrées mtree pour "pkg". DÉPRÉCIÉ : utilisez "pu_mtree_reader_open_package" à la place.

EXEMPLES

Afficher les sommes md5 des fichiers pour un paquet :
pu_mtree_t *m;
pu_mtree_reader_t *r;
if((r = pu_mtree_reader_open_package(handle, pkg)) == NULL) {
    fprintf(stderr, "erreur : impossible de charger les données mtree pour '%s'\n",
       alpm_pkg_get_name(pkg));
    return;
}
while((m = pu_mtree_reader_next(r, NULL))) {
    const char *md5 = m->md5digest;
    printf("%s: %s\n", m->path,
        md5 && md5[0] != '\0' ? md5 : "(aucune somme md5 fournie)");
    pu_mtree_free(m);
}
if(!reader->eof) {
    fprintf(stderr, "erreur : impossible de lire les données mtree pour '%s'\n",
       alpm_pkg_get_name(pkg));
}
pu_mtree_reader_free(r);
