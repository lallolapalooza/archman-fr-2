DESCRIPTIO

N

La fonction fopencookie() permet au programmeur de créer une implémentation personnalisée pour un flux d'E/S standard. Cette implémentation peut stocker les données du flux à un emplacement de son choix ; par exemple, fopencookie() est utilisé pour implémenter fmemopen(3), qui fournit une interface de flux pour des données stockées dans un tampon en

mémoire.

Afin de créer un flux personnalisé, le programmeur doit

:
•
Implémenter quatre fonctions « hook » utilisées en interne par la bibliothèque d'E/S standard lors de l'exécution d'E/S sur le
flux.
•
Définir un type de données « cookie », une structure qui fournit des informations de suivi (par exemple, où stocker les données) utilisées par les fonctions hook susmentionnées. Le paquet d'E/S standard ne connaît rien du contenu de ce cookie (il est donc typé void * lorsqu'il est passé à fopencookie()), mais fournit automatiquement le cookie comme premier argument lors de l'appel des fonctions
hook.
•
Appeler fopencookie() pour ouvrir un nouveau flux et ass
ocier le cookie et les f
onctions hook à ce flux.

La fonction fopencookie() remplit un rôle similaire à celui de fopen(3) : elle ouvre un nouveau flux et renvoie un pointeur vers un objet FILE utilisé pour manipuler ce

flux.

L'argument cookie est un pointeur vers la structure cookie de l'appelant qui doit être associée au nouveau flux. Ce pointeur est fourni comme premier argument lorsque la bibliothèque d'E/S standard invoque l'une des fonctions hook décrites

ci-dessous.

L'argument mode sert le même but que pour fopen(3). Les modes suivants sont pris en charge : r, w, a, r+, w+, et a+. Voir fopen(3) pour les

détails.

L'argument io_funcs est une structure contenant quatre champs pointant vers les fonctions hook définies par le programmeur et utilisées pour implémenter ce flux. La structure est définie comme

suit 


typedef struct {
    cookie_read_function_t  *read;
    cookie_write_function_t *write;
    cookie_seek_function_t  *seek;
    cookie_close_function_t *close;
} cookie_io_functions_t
;

EXEMPLE

S

Le programme ci-dessous implémente un flux personnalisé dont les fonctionnalités sont similaires (mais pas identiques) à celles disponibles via fmemopen(3). Il implémente un flux dont les données sont stockées dans un tampon mémoire. Le programme écrit ses arguments de ligne de commande dans le flux, puis parcourt le flux en lisant deux caractères sur cinq et en les écrivant sur la sortie standard. La session de shell suivante illustre l'utilisation du programme

: 


$ ./a.out 'hello world'
/he/
/ w/
/d/
Reached end of
file

Notez qu'une version plus générale du programme ci-dessous pourrait être améliorée pour gérer plus robustement diverses situations d'erreur (par exemple, ouvrir un flux avec un cookie qui a déjà un flux ouvert ; fermer un

de stream qui a déjà été fermé).

Source du programme

#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/types.h>
#include <unistd.h>
#define INIT_BUF_SIZE 4
struct memfile_cookie {
    char   *buf;        /* Buffer de taille dynamique pour les données */
    size_t  allocated;  /* Taille de buf */
    size_t  endpos;     /* Nombre de caractères dans buf */
    off_t   offset;     /* Décalage de fichier courant dans buf */
};
static ssize_t
memfile_write(void *c, const char *buf, size_t size)
{
    char *new_buff;
    struct memfile_cookie *cookie = c;
    /* Tampon trop petit ? Continuer à doubler la taille jusqu'à ce qu'elle soit suffisante.  */
    while (size + cookie->offset > cookie->allocated) {
        new_buff = realloc(cookie->buf, cookie->allocated * 2);
        if (new_buff == NULL)
            return -1;
        cookie->allocated *= 2;
        cookie->buf = new_buff;
    }
    memcpy(cookie->buf + cookie->offset, buf, size);
    cookie->offset += size;
    if (cookie->offset > cookie->endpos)
        cookie->endpos = cookie->offset;
    return size;
}
static ssize_t
memfile_read(void *c, char *buf, size_t size)
{
    ssize_t xbytes;
    struct memfile_cookie *cookie = c;
    /* Récupérer le minimum d'octets demandés et d'octets disponibles.  */
    xbytes = size;
    if (cookie->offset + size > cookie->endpos)
        xbytes = cookie->endpos - cookie->offset;
    if (xbytes < 0)     /* offset peut être après endpos */
        xbytes = 0;
    memcpy(buf, cookie->buf + cookie->offset, xbytes);
    cookie->offset += xbytes;
    return xbytes;
}
static int
memfile_seek(void *c, off_t *offset, int whence)
{
    off_t new_offset;
    struct memfile_cookie *cookie = c;
    if (whence == SEEK_SET)
        new_offset = *offset;
    else if (whence == SEEK_END)
        new_offset = cookie->endpos + *offset;
    else if (whence == SEEK_CUR)
        new_offset = cookie->offset + *offset;
    else
        return -1;
    if (new_offset < 0)
        return -1;
    cookie->offset = new_offset;
    *offset = new_offset;
    return 0;
}
static int
memfile_close(void *c)
{
    struct memfile_cookie *cookie = c;
    free(cookie->buf);
    cookie->allocated = 0;
    cookie->buf = NULL;
    return 0;
}
int
main(int argc, char *argv[])
{
    cookie_io_functions_t  memfile_func = {
        .read  = memfile_read,
        .write = memfile_write,
        .seek  = memfile_seek,
        .close = memfile_close
    };
    FILE *stream;
    struct memfile_cookie mycookie;
    size_t nread;
    char buf[1000];
    /* Configurer le cookie avant d'appeler fopencookie().  */
    mycookie.buf = malloc(INIT_BUF_SIZE);
    if (mycookie.buf == NULL) {
        perror("malloc");
        exit(EXIT_FAILURE);
    }
    mycookie.allocated = INIT_BUF_SIZE;
    mycookie.offset = 0;
    mycookie.endpos = 0;
    stream = fopencookie(&mycookie, "w+", memfile_func);
    if (stream == NULL) {
        perror("fopencookie");
        exit(EXIT_FAILURE);
    }
    /* Écrire les arguments de la ligne de commande dans notre fichier.  */
    for (size_t j = 1; j < argc; j++)
        if (fputs(argv[j], stream) == EOF) {
            perror("fputs");
            exit(EXIT_FAILURE);
        }
    /* Lire deux octets sur cinq, jusqu'à la fin du fichier.  */
    for (long p = 0; ; p += 5) {
        if (fseek(stream, p, SEEK_SET) == -1) {
            perror("fseek");
            exit(EXIT_FAILURE);
        }
        nread = fread(buf, 1, 2, stream);
        if (nread == 0) {
            if (ferror(stream) != 0) {
                fprintf(stderr, "fread failed\n");
                exit(EXIT_FAILURE);
            }
            printf("Reached end of file\n");
            break;
        }
        printf("/%.*s/\n", (int) nread, buf);
    }
    free(mycookie.buf);
    exit(EXIT_SUCCESS);
}

NOTES

_FILE_OFFSET_BITS doit être défini à 64 dans le code qui utilise un seek non nul ou qui prend l'adresse de fopencookie, si le code est destiné à être portable sur les plateformes traditionnelles 32 bits x86 et ARM où la largeur de off_t est par défaut de 32 bits.
