google-chrome.pop-up.forcedClose: 1
google-chrome.pop-up.ignorePagerPreview: 1
google-chrome.pop-up.ignoreUrgentHint: 1
google-chrome.pop-up.layer: Below

google-chrome.pop-up.noFocusOnMap: 1

fixe.
