IBV_QUERY_DEVICE_EX(3) Manuel du programmeur Libibverbs IBV_QUERY_DEVICE_EX(3)

NOM

ibv_query_device_ex - interroger les attributs d'un périphérique RDMA, y compris les propriétés étendues du périphérique.

SYNOPIS

#include <infiniband/verbs.h>

int ibv_query_device_ex(struct ibv_context *context,
                        struct ibv_query_device_ex_input *input,
                        struct ibv_device_attr_ex *attr);

DESCRIPTION

ibv_query_device_ex() renvoie les attributs du périphérique avec le contexte context. L'argument input est un pointeur vers une structure ibv_query_device_ex_input, utilisée pour des extensions futures. L'argument attr est un pointeur vers une structure ibv_device_attr_ex, telle que définie dans <infiniband/verbs.h>.

struct ibv_device_attr_ex {
struct ibv_device_attr orig_attr;
uint32_t               comp_mask;                  /* Masque de compatibilité qui définit quelles variables suivantes sont valides */
struct ibv_odp_caps    odp_caps;                   /* Capacités de pagination à la demande (On-Demand Paging) */
uint64_t               completion_timestamp_mask;  /* Masque d'horodatage de complétion (0 = non supporté) */
uint64_t               hca_core_clock;             /* La fréquence (en kHz) du HCA (0 = non supporté) */
uint64_t               device_cap_flags_ex;        /* Indicateurs de capacité de périphérique étendue */
struct ibv_tso_caps    tso_caps;                   /* Capacités de déchargement de segmentation TCP (TSO) */
struct ibv_rss_caps    rss_caps;                   /* Capacités RSS (Receive Side Scaling) */
uint32_t               max_wq_type_rq;             /* File d'attente de travail maximale de type RQ */
struct ibv_packet_pacing_caps packet_pacing_caps; /* Capacités de régulation de paquets */
uint32_t               raw_packet_caps;            /* Capacités de paquets bruts, utiliser enum ibv_raw_packet_caps */
struct ibv_tm_caps     tm_caps;                    /* Capacités d'appariement de tags */
struct ibv_cq_moderation_caps  cq_mod_caps;        /* Capacités maximales de modération de CQ */
uint64_t     	       max_dm_size;		   /* Taille maximale de la mémoire périphérique (en octets) disponible pour l'allocation */
struct ibv_pci_atomic_caps atomic_caps;            /* Capacités d'opérations atomiques PCI, utiliser enum ibv_pci_atomic_op_size */
uint32_t               xrc_odp_caps;               /* Masque avec enum ibv_odp_transport_cap_bits pour savoir quelles opérations sont supportées. */
uint32_t	       phys_port_cnt_ex		   /* Nombre étendu de ports physiques, permet d'exposer plus de 255 ports sur le périphérique */
};
struct ibv_odp_caps {
        uint64_t general_odp_caps;    /* Masque avec enum ibv_odp_general_cap_bits */
        struct {
                uint32_t rc_odp_caps; /* Masque avec enum ibv_odp_tranport_cap_bits pour savoir quelles opérations sont supportées. */
                uint32_t uc_odp_caps; /* Masque avec enum ibv_odp_tranport_cap_bits pour savoir quelles opérations sont supportées. */
                uint32_t ud_odp_caps; /* Masque avec enum ibv_odp_tranport_cap_bits pour savoir quelles opérations sont supportées. */
        } per_transport_caps;
};
enum ibv_odp_general_cap_bits {
        IBV_ODP_SUPPORT = 1 << 0, /* La pagination à la demande est supportée */
        IBV_ODP_SUPPORT_IMPLICIT = 1 << 1, /* La pagination implicite à la demande est supportée */
};
enum ibv_odp_transport_cap_bits {
        IBV_ODP_SUPPORT_SEND     = 1 << 0, /* Les opérations d'envoi supportent la pagination à la demande */
        IBV_ODP_SUPPORT_RECV     = 1 << 1, /* Les opérations de réception supportent la pagination à la demande */
        IBV_ODP_SUPPORT_WRITE    = 1 << 2, /* Les opérations RDMA-Write supportent la pagination à la demande */
        IBV_ODP_SUPPORT_READ     = 1 << 3, /* Les opérations RDMA-Read supportent la pagination à la demande */
        IBV_ODP_SUPPORT_ATOMIC   = 1 << 4, /* Les opérations RDMA-Atomic supportent la pagination à la demande */
        IBV_ODP_SUPPORT_SRQ_RECV = 1 << 5, /* Les opérations de réception SRQ supportent la pagination à la demande */
        IBV_ODP_SUPPORT_FLUSH    = 1 << 6, /* Les opérations RDMA-Flush supportent la pagination à la demande */
        IBV_ODP_SUPPORT_ATOMIC_WRITE     = 1 << 7, /* Les opérations RDMA-Atomic-write supportent la pagination à la demande */
};
struct ibv_tso_caps {
        uint32_t max_tso;        /* Taille maximale de la charge utile en octets supportée pour la segmentation par le moteur TSO. */
        uint32_t supported_qpts; /* Masque binaire indiquant quels types de QP sont supportés par l'opération TSO. */
};
struct ibv_rss_caps {
        uint32_t supported_qpts;                   /* Masque binaire indiquant quels types de QP sont supportés par RSS */
        uint32_t max_rwq_indirection_tables;       /* Nombre maximal de tables d'indirection de file d'attente de réception */
        uint32_t max_rwq_indirection_table_size;   /* Taille maximale d'une table d'indirection de file d'attente de réception */
        uint64_t rx_hash_fields_mask;              /* Masque avec enum ibv_rx_hash_fields pour savoir quel champ du paquet entrant peut participer au hachage RX */
        uint8_t  rx_hash_function;                 /* Masque avec enum ibv_rx_hash_function_flags pour savoir quelles fonctions de hachage sont supportées */
};
struct ibv_packet_pacing_caps {
       uint32_t qp_rate_limit_min; /* Limite de débit minimale en kbps */
       uint32_t qp_rate_limit_max; /* Limite de débit maximale en kbps */
       uint32_t supported_qpts;    /* Masque binaire indiquant quels types de QP sont supportés. */
};
enum ibv_raw_packet_caps {
IBV_RAW_PACKET_CAP_CVLAN_STRIPPING	= 1 << 0, /* La suppression CVLAN est supportée */
IBV_RAW_PACKET_CAP_SCATTER_FCS		= 1 << 1, /* La diffusion FCS est supportée */
IBV_RAW_PACKET_CAP_IP_CSUM		= 1 << 2, /* Le déchargement du somme de contrôle IP est supporté */
};
enum ibv_tm_cap_flags {
IBV_TM_CAP_RC   = 1 << 0,            /* Support de l'appariement de tags sur le transport RC */
};
struct ibv_tm_caps {
uint32_t        max_rndv_hdr_size;   /* Taille maximale de l'en-tête de la requête de rendez-vous */
uint32_t        max_num_tags;        /* Nombre maximal de tampons tagués dans une liste d'appariement TM-SRQ */
uint32_t        flags;    	     /* De enum ibv_tm_cap_flags */
uint32_t        max_ops;             /* Nombre maximal d'opérations de liste en attente */
uint32_t        max_sge;             /* Nombre maximal de SGE dans un tampon tagué */
};
struct ibv_cq_moderation_caps {
	uint16_t max_cq_count;
	uint16_t max_cq_period;
};
enum ibv_pci_atomic_op_size {
IBV_PCI_ATOMIC_OPERATION_4_BYTE_SIZE_SUP = 1 << 0,
IBV_PCI_ATOMIC_OPERATION_8_BYTE_SIZE_SUP = 1 << 1,
IBV_PCI_ATOMIC_OPERATION_16_BYTE_SIZE_SUP = 1 << 2,
};
struct ibv_pci_atomic_caps {
uint16_t fetch_add;	/* Tailles supportées pour une opération atomique fetch and add, utiliser enum ibv_pci_atomic_op_size */
uint16_t swap;		/* Tailles supportées pour une opération atomique de swap inconditionnel, utiliser enum ibv_pci_atomic_op_size */
uint16_t compare_swap;	/* Tailles supportées pour une opération atomique de comparaison et swap, utiliser enum ibv_pci_atomic_op_size */
};

Indicateurs de capacité de périphérique étendue (device_cap_flags_ex) :

IBV_DEVICE_PCI_WRITE_END_PADDING

Indique que le périphérique prend en charge le remplissage des écritures PCI jusqu'à une ligne de cache complète.

Le remplissage des paquets jusqu'à des lignes de cache complètes réduit la quantité de trafic requise au contrôleur mémoire, au détriment de la création de plus de trafic sur le port PCI-E.

Les charges de travail ayant une forte utilisation de la mémoire CPU et une faible utilisation du PCI-E bénéficieront de cette fonctionnalité, tandis que les charges de travail ayant une forte utilisation du PCI-E et de petits paquets seront pénalisées.

Par exemple, avec une taille de ligne de cache de 128 octets, le transfert de paquets inférieurs à 128 octets nécessitera un transfert complet de 128 octets sur le PCI, doublant potentiellement la bande passante PCI-E requise.

Cette fonctionnalité peut être activée par QP ou WQ via les indicateurs IBV_QP_CREATE_PCI_WRITE_END_PADDING ou IBV_WQ_FLAGS_PCI_WRITE_END_PADDING.

VALEUR RENVOYÉE

ibv_query_device_ex() renvoie 0 en cas de succès, ou la valeur d'errno en cas d'échec (qui indique la raison de l'échec).

NOTES

Les valeurs maximales renvoyées par cette fonction sont les limites supérieures des ressources supportées par le périphérique. Cependant, il peut ne pas être possible d'utiliser ces valeurs maximales, car le nombre réel de toute ressource pouvant être créée peut être limité par la configuration de la machine, la quantité de mémoire hôte, les permissions utilisateur et la quantité de ressources déjà utilisée par d'autres utilisateurs/processus.
