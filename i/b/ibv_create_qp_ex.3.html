SYNOPSI

S 
#include <infiniband/verbs.h>

struct ibv_qp *ibv_create_qp_ex(struct ibv_context *context,
                                struct ibv_qp_init_attr_ex *qp_init_attr);

int ibv_destroy_qp(struct ibv_qp *qp)
;

DESCRIPTIO

N

ibv_create_qp_ex() crée une paire de files d'attente (QP) associée au domaine de protection pd. L'argument qp_init_attr_ex est une structure ibv_qp_init_attr_ex, telle que définie dans <infiniband/verbs.h>

. 
struct ibv_qp_init_attr_ex {
void                   *qp_context;     /* Contexte associé de la QP */
struct ibv_cq          *send_cq;        /* CQ à associer à la file d'attente d'envoi (SQ) */
struct ibv_cq          *recv_cq;        /* CQ à associer à la file d'attente de réception (RQ) */
struct ibv_srq         *srq;            /* Poignée SRQ si la QP doit être associée à un SRQ, sinon NULL */
struct ibv_qp_cap       cap;            /* Capacités de la QP */
enum ibv_qp_type        qp_type;        /* Type de service de transport de la QP : IBV_QPT_RC, IBV_QPT_UC, IBV_QPT_UD, IBV_QPT_RAW_PACKET ou IBV_QPT_DRIVER */
int                     sq_sig_all;     /* Si défini, chaque requête de travail (WR) soumise à la SQ génère une entrée d'achèvement */
uint32_t                comp_mask;	/* Identifie les champs valides */
struct ibv_pd          *pd;		/* PD à associer à la QP */
struct ibv_xrcd        *xrcd;		/* Domaine XRC à associer à la QP cible */
enum ibv_qp_create_flags create_flags;	/* Drapeaux de création pour cette QP */
uint16_t                max_tso_header; /* Taille maximale de l'en-tête TSO */
struct ibv_rwq_ind_table *rwq_ind_tbl;  /* Table d'indirection à associer à la QP */
struct ibv_rx_hash_conf  rx_hash_conf;  /* Configuration de hachage RX à utiliser */
uint32_t                source_qpn;     /* Numéro de QP source, le drapeau de création IBV_QP_CREATE_SOURCE_QPN doit être défini, quelques NOTES ci-dessous */
uint64_t                send_ops_flags; /* Sélectionne les opérations d'envoi de la QP qui seront définies dans struct ibv_qp_ex. Utiliser l'enum ibv_qp_create_send_ops_flags */
};
struct ibv_qp_cap {
uint32_t                max_send_wr;    /* Nombre maximal demandé de WR en attente dans la SQ */
uint32_t                max_recv_wr;    /* Nombre maximal demandé de WR en attente dans la RQ */
uint32_t                max_send_sge;   /* Nombre maximal demandé d'éléments scatter/gather (s/g) dans une WR dans la SQ */
uint32_t                max_recv_sge;   /* Nombre maximal demandé d'éléments s/g dans une WR dans la RQ */
uint32_t                max_inline_data;/* Nombre maximal demandé de données (octets) pouvant être postées en ligne à la SQ, sinon 0 */
};
enum ibv_qp_create_flags {
IBV_QP_CREATE_BLOCK_SELF_MCAST_LB       = 1 << 1, /* Empêcher la bouclage de multicast vers soi-même */
IBV_QP_CREATE_SCATTER_FCS               = 1 << 8, /* Le champ FCS sera dispersé vers la mémoire hôte */
IBV_QP_CREATE_CVLAN_STRIPPING           = 1 << 9, /* Le champ CVLAN sera extrait des paquets entrants */
IBV_QP_CREATE_SOURCE_QPN                = 1 << 10, /* La QP créée utilisera le source_qpn comme son numéro de QP sur le fil */
IBV_QP_CREATE_PCI_WRITE_END_PADDING     = 1 << 11, /* Les paquets entrants seront alignés sur la taille de ligne de cache */
}
; 
struct ibv_rx_hash_conf {
uint8_t                rx_hash_function;       /* Fonction de hachage RX, utiliser l'enum ibv_rx_hash_function_flags */
uint8_t                rx_hash_key_len;        /* Longueur de la clé de hachage RX */
uint8_t                *rx_hash_key;           /* Données de la clé de hachage RX */
uint64_t               rx_hash_fields_mask;    /* Champs RX devant participer au hachage, utiliser l'enum ibv_rx_hash_fields */
}
; 
enum ibv_rx_hash_fields {
IBV_RX_HASH_SRC_IPV4            = 1 << 0,
IBV_RX_HASH_DST_IPV4            = 1 << 1,
IBV_RX_HASH_SRC_IPV6            = 1 << 2,
IBV_RX_HASH_DST_IPV6            = 1 << 3,
IBV_RX_HASH_SRC_PORT_TCP        = 1 << 4,
IBV_RX_HASH_DST_PORT_TCP        = 1 << 5,
IBV_RX_HASH_SRC_PORT_UDP        = 1 << 6,
IBV_RX_HASH_DST_PORT_UDP        = 1 << 7,
IBV_RX_HASH_IPSEC_SPI           = 1 << 8,
/* Lors de l'utilisation d'un protocole de tunnel, p. ex. VXLAN, nous avons un paquet intérieur (encapsulé) et extérieur.
 * Pour appliquer le RSS sur le paquet intérieur, le champ suivant doit être défini avec l'un des champs L3/L4.
*/
IBV_RX_HASH_INNER		= (1UL << 31),
}
; 
struct ibv_qp_create_send_ops_flags {
IBV_QP_EX_WITH_RDMA_WRITE		= 1 << 0,
IBV_QP_EX_WITH_RDMA_WRITE_WITH_IMM	= 1 << 1,
IBV_QP_EX_WITH_SEND			= 1 << 2,
IBV_QP_EX_WITH_SEND_WITH_IMM		= 1 << 3,
IBV_QP_EX_WITH_RDMA_READ		= 1 << 4,
IBV_QP_EX_WITH_ATOMIC_CMP_AND_SWP	= 1 << 5,
IBV_QP_EX_WITH_ATOMIC_FETCH_AND_ADD	= 1 << 6,
IBV_QP_EX_WITH_LOCAL_INV		= 1 << 7,
IBV_QP_EX_WITH_BIND_MW			= 1 << 8,
IBV_QP_EX_WITH_SEND_WITH_INV		= 1 << 9,
IBV_QP_EX_WITH_TSO			= 1 << 10,
}
;

La fonction ibv_create_qp_ex() mettra à jour la structure qp_init_attr_ex->cap avec les valeurs réelles de la QP de la QP qui a été créée ; les valeurs seront supérieures ou égales aux valeurs

demandées.

ibv_destroy_qp() détruit la QP q

p
.

VALEUR

RENVOYÉE

ibv_create_qp_ex() renvoie un pointeur vers la QP créée, ou NULL si la requête échoue. Vérifier le numéro de QP (qp_num) dans la QP

retournée.

ibv_destroy_qp() renvoie 0 en cas de succès, ou la valeur de

errno en cas d'échec (ce qui
indique la raison de l'échec).

NOTE

S

Les attributs max_recv_wr et max_recv_sge sont ignorés par ibv_create_qp_ex() si la QP doit être associée à un

SRQ.

L'attribut source_qpn n'est pris en charge que sur les QP UD, sans pilotage de flux la réception (RX) ne devrait pas être

possible.

Utiliser ibv_qp_to_qp_ex() pour obtenir le ibv_qp_ex pour accéder à l'interface d'itérateur des opérations d'envoi, lorsque l'attribut de création de QP IBV_QP_INIT_ATTR_SEND_OPS_FLAGS est

utilisé.

ibv_destroy_qp() échoue si la QP est attachée à un groupe de

multicast.

IBV_QPT_DRIVER ne représente pas un service spécifique et est

utilisé pour la logique de
QP spécifique au fournisseur.
