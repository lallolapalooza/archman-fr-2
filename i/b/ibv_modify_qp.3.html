ibv_modify_qp - modifier

les attributs d'une pair
e de files d'attente (QP)

SYNOPSI

S 
#include <infiniband/verbs.h>

int ibv_modify_qp(struct ibv_qp *qp, struct ibv_qp_attr *attr,
                  int attr_mask)
;

DESCRIPTIO

N

ibv_modify_qp() modifie les attributs de la QP qp avec les attributs dans attr selon le masque attr_mask. L'argument attr est une structure ibv_qp_attr, telle que définie dans <infiniband/verbs.h>

. 
struct ibv_qp_attr {
enum ibv_qp_state       qp_state;               /* Déplacer la QP vers cet état */
enum ibv_qp_state       cur_qp_state;           /* Supposer que c'est l'état actuel de la QP */
enum ibv_mtu            path_mtu;               /* MTU du chemin (valide uniquement pour les QP RC/UC) */
enum ibv_mig_state      path_mig_state;         /* État de migration du chemin (valide si la HCA prend en charge APM) */
uint32_t                qkey;                   /* Q_Key pour la QP (valide uniquement pour les QP UD) */
uint32_t                rq_psn;                 /* PSN pour la file d'attente de réception (valide uniquement pour les QP RC/UC) */
uint32_t                sq_psn;                 /* PSN pour la file d'attente d'envoi */
uint32_t                dest_qp_num;            /* Numéro de QP de destination (valide uniquement pour les QP RC/UC) */
unsigned int            qp_access_flags;        /* Masque des opérations d'accès distant activées (valide uniquement pour les QP RC/UC) */
struct ibv_qp_cap       cap;                    /* Capacités de la QP (valide si la HCA prend en charge le redimensionnement des QP) */
struct ibv_ah_attr      ah_attr;                /* Vecteur d'adresse du chemin principal (valide uniquement pour les QP RC/UC) */
struct ibv_ah_attr      alt_ah_attr;            /* Vecteur d'adresse du chemin alternatif (valide uniquement pour les QP RC/UC) */
uint16_t                pkey_index;             /* Index P_Key principal */
uint16_t                alt_pkey_index;         /* Index P_Key alternatif */
uint8_t                 en_sqd_async_notify;    /* Activer la notification asynchrone SQD.drained (valide uniquement si qp_state est SQD) */
uint8_t                 sq_draining;            /* La QP est-elle en cours de vidage ? Sans pertinence pour ibv_modify_qp() */
uint8_t                 max_rd_atomic;          /* Nombre de lectures RDMA et opérations atomiques en attente sur la QP de destination (valide uniquement pour les QP RC) */
uint8_t                 max_dest_rd_atomic;     /* Nombre de ressources du répondeur pour traiter les lectures RDMA et opérations atomiques entrantes (valide uniquement pour les QP RC) */
uint8_t                 min_rnr_timer;          /* Minuteur NAK RNR minimum (valide uniquement pour les QP RC) */
uint8_t                 port_num;               /* Numéro de port principal */
uint8_t                 timeout;                /* Délai d'accusé de réception local pour le chemin principal (valide uniquement pour les QP RC) */
uint8_t                 retry_cnt;              /* Nombre de tentatives (valide uniquement pour les QP RC) */
uint8_t                 rnr_retry;              /* Nouvelle tentative RNR (valide uniquement pour les QP RC) */
uint8_t                 alt_port_num;           /* Numéro de port alternatif */
uint8_t                 alt_timeout;            /* Délai d'accusé de réception local pour le chemin alternatif (valide uniquement pour les QP RC) */
uint32_t                rate_limit;             /* Limite de débit en kbps pour le rythmage des paquets */
}
;

VALEUR

RENVOYÉE

ibv_modify_qp() renvoie 0 en cas de succès, ou la valeur de er

rno en cas d'échec (ce qui
indique la raison de l'échec).

REMARQUE

S

Si l'un des attributs à modifier ou le masque de modification est invalide, aucun des attributs ne sera modifié (y compris l'état de la QP)

.

Tous les dispositifs ne prennent pas en charge le redimensionnement des QP. Pour vérifier si un dispositif le prend en charge, vérifiez si le bit IBV_DEVICE_RESIZE_MAX_WR est défini dans les drapeaux de capacités du

dispositif.

Tous les dispositifs ne prennent pas en charge les chemins alternatifs. Pour vérifier si un dispositif le prend en charge, vérifiez si le bit IBV_DEVICE_AUTO_PATH_MIG est défini dans les drapeaux de capacités du

dispositif.
