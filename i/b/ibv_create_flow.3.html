enum ibv_flow_attr_type {
IBV_FLOW_ATTR_NORMAL		= 0x0,		/* Guidage selon les spécifications de la règle */
IBV_FLOW_ATTR_ALL_DEFAULT	= 0x1,		/* Règle par défaut unicast et multicast - recevoir tout le trafic Ethernet qui n'est pas guidé vers un QP */
IBV_FLOW_ATTR_MC_DEFAULT 	= 0x2,		/* Règle multicast par défaut - recevoir tout le trafic multicast Ethernet qui n'est pas guidé vers un QP */

};
enum ibv_flow_flags {

Structure générale de spécification de flux :
struct ibv_flow_spec_xxx {
enum ibv_flow_spec_type  type;
uint16_t  size;                  /* Taille de la spécification de flux = sizeof(struct ibv_flow_spec_xxx) */
struct ibv_flow_xxx_filter val;
struct ibv_flow_xxx_filter mask; /* Définit quels bits de la valeur du filtre sont applicables lors de la recherche d'une correspondance dans le paquet entrant */
}
; 
Chaque structure de spécification contient les paramètres de couche réseau pertinents pour la correspondance. Pour imposer la correspondance, l'utilisateur définit un masque pour chaque paramètre.
Les paquets provenant du fil sont mis en correspondance avec la spécification de flux. Si une correspondance est trouvée, les actions de flux associées sont exécutées sur le paquet.
Dans les flux d'entrée, le paramètre QP est traité comme une autre action de dispersion du paquet vers le QP concerné.
Si le bit est défini dans le masque, le bit correspondant dans la valeur doit correspondre.
Notez que la plupart des fournisseurs prennent en charge soit un masque complet (tous les « 1 »), soit un masque nul (tous les « 0 »).

Domaines de flux et

priorité

Le guidage de flux définit le concept de domaine et de priorité. Chaque domaine représente une application qui peut rattacher un flux. Les domaines sont priorisés. Un domaine de priorité plus élevée prévaudra toujours sur un domaine de priorité inférieure lorsque leurs spécifications de flux se chevauchent.
Les verbes IB ont le domaine de priorité la plus élevée.
En plus du domaine, il y a une priorité au sein de chacun des domaines. Une valeur numérique de priorité plus basse (priorité plus élevée) prime sur les règles de correspondance ayant une valeur numérique de priorité pl

us élevée (priorité inférieure). Il est important de noter que la valeur de priorité d'une spécification de flux est utilisée non seulement pour établir la prééminence des correspondances de flux conflictuelles, mais
aussi comme moyen d'abstraire l'ordre selon lequel les spécifications de flux sont testées pour des correspondances. Les flux avec des priorités plus élevées seront testés avant les flux avec des priorités plus basses.

Ordre de définition des

règles

Une application peut fournir les règles ibv_flow_spec_xxx selon un schéma non ordonné. Dans ce cas, chaque spécification doit être bien définie et correspondre à une couche d'en-tête réseau spécifique. Dans certains cas, lorsque certains types de spécification de flux sont présents dans la liste de spécifications, il est requis de fournir la liste de manière ordonnée afin que la position de ce type de spécification de flux dans la pile de protocoles soit strictement définie. Lorsque le certain type de spécification, qui requiert l'ordre, se trouve dans la pile de protocoles réseau interne (dans les protocoles de tunnel), l'ordre doit être appliqué aux spécifications réseau internes et doit être combiné avec l'indication de spécification interne à l'aide du drapeau IBV_FLOW_SPEC_INNER. Par exemple : Une spécification MPLS qui tente de faire correspondre une balise MPLS dans le réseau interne doit avoir le drapeau IBV_FLOW_SPEC_INNER défini, et il en va de même pour le reste des spécifications réseau internes. En plus de cela, toutes les spécifications réseau internes doivent être fournies de manière ordonnée. C'est essentiel pour représenter de nombreux protocoles de tunnel d'encapsulation

.

Ci-dessous, flow_attr définit une règle de priorité 0 pour faire correspondre une adresse mac de destination et une adresse ipv4 source. Pour cela, les spécifications L2 et L3 sont utilisées.
S'il y a une correspondance sur cette règle, ce qui signifie que le paquet reçu a la destination mac : 66:11:22:33:44:55 et l'ip source : 0x0B86C806, le paquet est acheminé vers son qp

struct raw_eth_flow_attr {
struct ibv_flow_attr            attr;
struct ibv_flow_spec_eth        spec_eth;
struct ibv_flow_spec_ipv4       spec_ipv4;
} __attribute__((packed));
struct raw_eth_flow_attr flow_attr = {
        .attr = {
                .comp_mask      = 0,
                .type           = IBV_FLOW_ATTR_NORMAL,
                .size           = sizeof(flow_attr),
                .priority       = 0,
                .num_of_specs   = 2,
                .port           = 1,
                .flags          = 0,
        },
        .spec_eth = {
                .type   = IBV_FLOW_SPEC_ETH,
                .size   = sizeof(struct ibv_flow_spec_eth),
                .val = {
                        .dst_mac = {0x66, 0x11, 0x22, 0x33, 0x44, 0x55},
                        .src_mac = { 0x00, 0x00, 0x00, 0x00, 0x00, 0x00},
                        .ether_type = 0,
                        .vlan_tag = 0,
                },
                .mask = {
                        .dst_mac = { 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF},
                        .src_mac = { 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF},
                        .ether_type = 0,
                        .vlan_tag = 0,
                }
        },
        .spec_ipv4 = {
                .type   = IBV_FLOW_SPEC_IPV4,
                .size   = sizeof(struct ibv_flow_spec_ipv4),
                .val = {
                        .src_ip = 0x0B86C806,
                        .dst_ip = 0,
                },
                .mask = {
                        .src_ip = 0xFFFFFFFF,
                        .dst_ip = 0,
                }
        }
};
