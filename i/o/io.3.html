io(3) Linux AIO io(3)

NOM

io - Entrées/Sorties Asynchrones

SYNTAXE

#include <errno.h>

#include <libaio.h>

DESCRIPTION

La bibliothèque libaio définit un nouvel ensemble d'opérations d'E/S qui peuvent considérablement réduire le temps passé par une application à attendre les E/S. Les nouvelles fonctions permettent à un programme d'initier une ou plusieurs opérations d'E/S, puis de reprendre immédiatement son travail normal pendant que les opérations d'E/S sont exécutées en parallèle.

Ces fonctions font partie de la bibliothèque d'E/S temps réel nommée libaio. Elles ne font pas réellement partie du binaire libc. L'implémentation de ces fonctions peut être réalisée à l'aide du support du noyau.

Toutes les opérations d'E/S s'effectuent sur des fichiers qui ont été ouverts précédemment. Il peut y avoir un nombre arbitraire d'opérations en cours pour un même fichier. Les opérations d'E/S asynchrones sont contrôlées à l'aide d'une structure de données nommée struct iocb. Elle est définie dans libaio.h comme suit.

typedef struct io_context *io_context_t;
typedef enum io_iocb_cmd {
        IO_CMD_PREAD = 0,
        IO_CMD_PWRITE = 1,
        IO_CMD_FSYNC = 2,
        IO_CMD_FDSYNC = 3,
        IO_CMD_POLL = 5,
        IO_CMD_NOOP = 6,
} io_iocb_cmd_t;
struct io_iocb_common {
        void            *buf;
        unsigned        __pad1;
        long            nbytes;
        unsigned        __pad2;
        long long       offset;
        long long       __pad3, __pad4;
};      /* le code de résultat est la quantité lue ou un errno négatif */
struct iocb {
        void            *data;
        unsigned        key;
        short           aio_lio_opcode;
        short           aio_reqprio;
        int             aio_fildes;
        union {
                struct io_iocb_common           c;
                struct io_iocb_vector           v;
                struct io_iocb_poll             poll;
                struct io_iocb_sockaddr saddr;
        } u;
};
int aio_fildes
Cet élément spécifie le descripteur de fichier à utiliser pour l'opération. Il doit s'agir d'un descripteur valide, sinon l'opération échouera.

Le périphérique sur lequel le fichier est ouvert doit autoriser l'opération de décalage. C'est-à-dire qu'il n'est pas possible d'utiliser les opérations d'E/S sur des périphériques comme les terminaux où un appel à lseek(2) entraînerait une erreur.

long u.c.offset
Cet élément spécifie le décalage dans le fichier où l'opération (lecture ou écriture) est effectuée. Comme les opérations sont effectuées dans un ordre arbitraire et que plusieurs opérations peuvent être démarrées pour un même descripteur de fichier, on ne peut pas s'attendre à une position de lecture/écriture actuelle du descripteur de fichier.
void *buf
Il s'agit d'un pointeur vers le tampon contenant les données à écrire ou l'endroit où les données lues sont stockées.
long u.c.nbytes
Cet élément spécifie la longueur du tampon pointé par io_buf.
int aio_reqprio
N'est actuellement pas utilisé.
IO_CMD_PREAD
Démarre une opération de lecture. Lit depuis le fichier à la position u.c.offset et stocke les u.c.nbytes octets suivants dans le tampon pointé par buf.
IO_CMD_PWRITE
Démarre une opération d'écriture. Écrit u.c.nbytes octets à partir de buf dans le fichier à partir de la position u.c.offset.
IO_CMD_NOP
Ne fait rien pour ce bloc de contrôle. Cette valeur est parfois utile lorsqu'un tableau de valeurs struct iocb contient des "trous", c'est-à-dire que certaines valeurs ne doivent pas être traitées bien que l'ensemble du tableau soit présenté à la fonction io_submit(3).
IO_CMD_FSYNC
IO_CMD_POLL
Ceci est expérimental.

EXEMPLE

/*
 * Version simplifiée de la commande copy utilisant les E/S asynchrones
 *
 * De :	Stephen Hemminger <shemminger@osdl.org>
 * Copie un fichier en utilisant une machine d'état d'E/S asynchrones.
 * 1. Démarre la requête de lecture
 * 2. Lorsque la lecture est terminée, la transforme en requête d'écriture
 * 3. Lorsque l'écriture est terminée, décrémente le compteur et libère les ressources
 *
 *
 * Utilisation : aiocp fichier(s) destination
 */
#include <unistd.h>
#include <stdio.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <sys/param.h>
#include <fcntl.h>
#include <errno.h>
#include <libaio.h>
#define AIO_BLKSIZE	(64*1024)
#define AIO_MAXIO	32
static int busy = 0;		// Nombre d'E/S en cours
static int tocopy = 0;		// Nombre de blocs restants à copier
static int dstfd = -1;		// descripteur de fichier de destination
static const char *dstname = NULL;
static const char *srcname = NULL;
/* Gestionnaire d'erreur fatale */
static void io_error(const char *func, int rc)
{
    if (rc == -ENOSYS)
	fprintf(stderr, "AIO non présent dans ce noyau\n");
    else if (rc < 0 && -rc < sys_nerr)
	fprintf(stderr, "%s: %s\n", func, sys_errlist[-rc]);
    else
	fprintf(stderr, "%s: erreur %d\n", func, rc);
    if (dstfd > 0)
	close(dstfd);
    if (dstname)
	unlink(dstname);
    exit(1);
}
/*
 * Rappel pour écriture terminée.
 * Ajuste les compteurs et libère les ressources
 */
static void wr_done(io_context_t ctx, struct iocb *iocb, long res, long res2)
{
    if (res2 != 0) {
	io_error("aio write", res2);
    }
    if (res != iocb->u.c.nbytes) {
	fprintf(stderr, "écriture manquée octets attendus %d obtenus %d\n", iocb->u.c.nbytes, res2);
	exit(1);
    }
    --tocopy;
    --busy;
    free(iocb->u.c.buf);
    memset(iocb, 0xff, sizeof(iocb));	// paranoïa
    free(iocb);
    write(2, "w", 1);
}
/*
 * Rappel pour lecture terminée.
 * Transforme l'iocb de lecture en iocb d'écriture et le démarre.
 */
static void rd_done(io_context_t ctx, struct iocb *iocb, long res, long res2)
{
    /* la bibliothèque a-t-elle besoin d'accesseurs pour examiner l'iocb ? */
    int iosize = iocb->u.c.nbytes;
    char *buf = iocb->u.c.buf;
    off_t offset = iocb->u.c.offset;
    if (res2 != 0)
	io_error("aio read", res2);
    if (res != iosize) {
	fprintf(stderr, "lecture manquée octets attendus %d obtenus %d\n", iocb->u.c.nbytes, res);
	exit(1);
    }
    /* transformer la lecture en écriture */
    io_prep_pwrite(iocb, dstfd, buf, iosize, offset);
    io_set_callback(iocb, wr_done);
    if (1 != (res = io_submit(ctx, 1, &iocb)))
	io_error("io_submit write", res);
    write(2, "r", 1);
}
int main(int argc, char *const *argv)
{
    int srcfd;
    struct stat st;
    off_t length = 0, offset = 0;
    io_context_t myctx;
    if (argc != 3 || argv[1][0] == '-') {
	fprintf(stderr, "Utilisation : aiocp SOURCE DEST\n");
	exit(1);
    }
    if ((srcfd = open(srcname = argv[1], O_RDONLY)) < 0) {
	perror(srcname);
	exit(1);
    }
    if (fstat(srcfd, &st) < 0) {
	perror("fstat");
	exit(1);
    }
    length = st.st_size;
    if ((dstfd = open(dstname = argv[2], O_WRONLY | O_CREAT, 0666)) < 0) {
	close(srcfd);
	perror(dstname);
	exit(1);
    }
    /* initialisation de la machine d'état */
    memset(&myctx, 0, sizeof(myctx));
    io_queue_init(AIO_MAXIO, &myctx);
    tocopy = howmany(length, AIO_BLKSIZE);
    while (tocopy > 0) {
	int i, rc;
	/* Soumettre autant de lectures que possible à la fois, jusqu'à AIO_MAXIO */
	int n = MIN(MIN(AIO_MAXIO - busy, AIO_MAXIO / 2),
		    howmany(length - offset, AIO_BLKSIZE));
	if (n > 0) {
	    struct iocb *ioq[n];
	    for (i = 0; i < n; i++) {
		struct iocb *io = (struct iocb *) malloc(sizeof(struct iocb));
		int iosize = MIN(length - offset, AIO_BLKSIZE);
		char *buf = (char *) malloc(iosize);
		if (NULL == buf || NULL == io) {
		    fprintf(stderr, "mémoire insuffisante\n");
		    exit(1);
		}
		io_prep_pread(io, srcfd, buf, iosize, offset);
		io_set_callback(io, rd_done);
		ioq[i] = io;
		offset += iosize;
	    }
	    rc = io_submit(myctx, n, ioq);
	    if (rc < 0)
		io_error("io_submit", rc);
	    busy += n;
	}
	// Gérer les E/S qui sont terminées
	rc = io_queue_run(myctx);
	if (rc < 0)
	    io_error("io_queue_run", rc);
	// si nous avons le nombre maximum d'E/S en cours
	// alors attendez qu'une se termine
	if (busy == AIO_MAXIO) {
	    rc = io_queue_wait(myctx, NULL);
	    if (rc < 0)
		io_error("io_queue_wait", rc);
	}
    }
    close(srcfd);
    close(dstfd);
    exit(0);
}
