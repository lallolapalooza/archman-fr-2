subroutine ssytri_rook (character uplo, integer n, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real, dimension( * ) work, integer info)

SSYTRI_ROOK

But:

!>
!> SSYTRI_ROOK calcule l'inverse d'une matrice réelle symétrique
!> A en utilisant la factorisation A = U*D*U**T ou A = L*D*L**T
!> calculée par SSYTRF_ROOK.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si les détails de la factorisation sont stockés
!>          sous forme de matrice triangulaire supérieure ou inférieure.
!>          = 'U': Triangulaire supérieure, forme A = U*D*U**T ;
!>          = 'L': Triangulaire inférieure, forme A = L*D*L**T.
!>

N

!>          N est INTEGER
!>          Ordre de la matrice A.  N >= 0.
!>

A

!>          A est tableau de REELS, dimension (LDA,N)
!>          À l'entrée, la matrice diagonale par blocs D et les multiplicateurs
!>          utilisés pour obtenir le facteur U ou L calculés par SSYTRF_ROOK.
!>
!>          À la sortie, si INFO = 0, l'inverse (symétrique) de la matrice
!>          originale. Si UPLO = 'U', la partie triangulaire supérieure de
!>          l'inverse est formée et la partie de A sous la diagonale n'est
!>          pas référencée ; si UPLO = 'L', la partie triangulaire inférieure
!>          de l'inverse est formée et la partie de A au-dessus de la diagonale
!>          n'est pas référencée.
!>

LDA

!>          LDA est INTEGER
!>          Dimension principale du tableau A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV est tableau d'ENTIERS, dimension (N)
!>          Détails des échanges et de la structure par blocs de D
!>          tels que déterminés par SSYTRF_ROOK.
!>

WORK

!>          WORK est tableau de REELS, dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0: succès de l'exécution
!>          < 0: si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0: si INFO = i, D(i,i) = 0 ; la matrice est singulière et son
!>               inverse n'a pas pu être calculé.
!>
