sous-routine ssyrk (caractère uplo, caractère trans, entier n, entier k, réel alpha, réel, dimension(lda,*) a, entier lda, réel beta, réel, dimension(ldc,*) c, entier ldc)

SSYRK

But :

!>
!> SSYRK  effectue une des opérations de rang k symétrique
!>
!>    C := alpha*A*A**T + beta*C,
!>
!> ou
!>
!>    C := alpha*A**T*A + beta*C,
!>
!> où  alpha et beta  sont des scalaires, C est une matrice symétrique de taille n par n
!> et  A  est une matrice de taille n par k dans le premier cas et de taille k par n
!> dans le second cas.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>           À l'entrée,   UPLO  spécifie si la partie triangulaire supérieure ou inférieure
!>           du tableau  C  doit être référencée comme suit :
!>
!>              UPLO = 'U' ou 'u'   Seule la partie triangulaire supérieure de C
!>                                  est à référencer.
!>
!>              UPLO = 'L' ou 'l'   Seule la partie triangulaire inférieure de C
!>                                  est à référencer.
!>

TRANS

!>          TRANS est CHARACTER*1
!>           À l'entrée,  TRANS  spécifie l'opération à effectuer comme suit :
!>
!>              TRANS = 'N' ou 'n'   C := alpha*A*A**T + beta*C.
!>
!>              TRANS = 'T' ou 't'   C := alpha*A**T*A + beta*C.
!>
!>              TRANS = 'C' ou 'c'   C := alpha*A**T*A + beta*C.
!>

N

!>          N est INTEGER
!>           À l'entrée,  N spécifie l'ordre de la matrice C.  N doit être
!>           au moins zéro.
!>

K

!>          K est INTEGER
!>           À l'entrée,  avec  TRANS = 'N' ou 'n',  K  spécifie le nombre
!>           de colonnes   de la matrice   A, et à l'entrée   avec
!>           TRANS = 'T' ou 't' ou 'C' ou 'c',  K  spécifie le nombre
!>           de lignes de la matrice  A.  K doit être au moins zéro.
!>

ALPHA

!>          ALPHA est REAL
!>           À l'entrée, ALPHA spécifie le scalaire alpha.
!>

A

!>          A est un tableau REAL, de dimension ( LDA, ka ), où ka est
!>           k  quand  TRANS = 'N' ou 'n',  et n sinon.
!>           Avant l'entrée avec  TRANS = 'N' ou 'n',  la partie principale n par k
!>           du tableau  A  doit contenir la matrice  A, sinon
!>           la partie principale k par n du tableau  A  doit contenir la
!>           matrice A.
!>

LDA

!>          LDA est INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle que déclarée
!>           dans le programme appelant (sous-programme).   Quand  TRANS = 'N' ou 'n'
!>           alors  LDA doit être au moins  max( 1, n ), sinon  LDA doit
!>           être au moins  max( 1, k ).
!>

BETA

!>          BETA est REAL
!>           À l'entrée, BETA spécifie le scalaire beta.
!>

C

!>          C est un tableau REAL, de dimension ( LDC, N )
!>           Avant l'entrée avec  UPLO = 'U' ou 'u',  la partie principale n par n
!>           du tableau C doit contenir la partie triangulaire supérieure
!>           de la matrice symétrique  et la partie strictement inférieure de C n'est pas référencée.  À la sortie, la
!>           partie triangulaire supérieure du tableau  C est écrasée par la
!>           partie triangulaire supérieure de la matrice mise à jour.
!>           Avant l'entrée avec  UPLO = 'L' ou 'l',  la partie principale n par n
!>           du tableau C doit contenir la partie triangulaire inférieure
!>           de la matrice symétrique  et la partie strictement supérieure de C n'est pas référencée.  À la sortie, la
!>           partie triangulaire inférieure du tableau  C est écrasée par la
!>           partie triangulaire inférieure de la matrice mise à jour.
!>

LDC

!>          LDC est INTEGER
!>           À l'entrée, LDC spécifie la première dimension de C telle que déclarée
!>           dans le programme appelant (sous-programme).   LDC  doit
!>           être au moins  max( 1, n ).
!>
