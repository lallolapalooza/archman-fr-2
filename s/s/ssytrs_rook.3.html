sous-routine ssytrs_rook (caractère uplo, entier n, entier nrhs, réel, dimension( lda, * ) a, entier lda, entier, dimension( * ) ipiv, réel, dimension( ldb, * ) b, entier ldb, entier info)

SSYTRS_ROOK

Objectif:

!>
!> SSYTRS_ROOK résout un système d'équations linéaires A*X = B avec
!> une matrice symétrique réelle A en utilisant la factorisation A = U*D*U**T ou
!> A = L*D*L**T calculée par SSYTRF_ROOK.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si les détails de la factorisation sont stockés
!>          sous forme de matrice triangulaire supérieure ou inférieure.
!>          = 'U':  Triangulaire supérieure, forme A = U*D*U**T;
!>          = 'L':  Triangulaire inférieure, forme A = L*D*L**T.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!>

A

!>          A est un tableau de type REAL, dimension (LDA,N)
!>          La matrice diagonale par blocs D et les multiplicateurs utilisés pour
!>          obtenir le facteur U ou L calculés par SSYTRF_ROOK.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV est un tableau de type INTEGER, dimension (N)
!>          Détails des échanges et de la structure par blocs de D
!>          tels que déterminés par SSYTRF_ROOK.
!>

B

!>          B est un tableau de type REAL, dimension (LDB,NRHS)
!>          À l'entrée, la matrice de second membre B.
!>          À la sortie, la matrice solution X.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

INFO

!>          INFO est INTEGER
!>          = 0:  sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>
