sousroutine ssyt01_3 (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldafac, * ) afac, integer ldafac, real, dimension( * ) e, integer, dimension( * ) ipiv, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) rwork, real resid)

SSYT01_3

But :

!>
!> SSYT01_3 reconstruit une matrice indéfinie symétrique A à partir de sa
!> factorisation par blocs L*D*L' ou U*D*U' calculée par SSYTRF_RK
!> (ou SSYTRF_BK) et calcule le résidu
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> où C est la matrice reconstruite et EPS est le epsilon machine.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si la partie triangulaire supérieure ou inférieure de la
!>          matrice symétrique A est stockée :
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

N

!>          N est INTEGER
!>          Le nombre de lignes et de colonnes de la matrice A.  N >= 0.
!>

A

!>          A est un tableau DOUBLE PRECISION, dimension (LDA,N)
!>          La matrice symétrique originale A.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC est un tableau DOUBLE PRECISION, dimension (LDAFAC,N)
!>          Diagonale de la matrice diagonale par blocs D et facteurs U ou L
!>          tels que calculés par SSYTRF_RK et SSYTRF_BK :
!>            a) SEULEMENT les éléments diagonaux de la matrice diagonale par
!>               blocs symétrique D sur la diagonale de A, c.-à-d. D(k,k) = A(k,k);
!>               (les éléments superdiagonaux (ou sous-diagonaux) de D
!>                doivent être fournis en entrée dans le tableau E), et
!>            b) Si UPLO = 'U': facteur U dans la partie superdiagonale de A.
!>               Si UPLO = 'L': facteur L dans la partie sous-diagonale de A.
!>

LDAFAC

!>          LDAFAC est INTEGER
!>          La dimension principale du tableau AFAC.
!>          LDAFAC >= max(1,N).
!>

E

!>          E est un tableau DOUBLE PRECISION, dimension (N)
!>          En entrée, contient les éléments superdiagonaux (ou sous-diagonaux)
!>          de la matrice diagonale par blocs symétrique
!>          avec des blocs diagonaux de taille 1x1 ou 2x2, où
!>          Si UPLO = 'U': E(i) = D(i-1,i),i=2:N, E(1) non référencé;
!>          Si UPLO = 'L': E(i) = D(i+1,i),i=1:N-1, E(N) non référencé.
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (N)
!>          Les indices de pivot de SSYTRF_RK (ou SSYTRF_BK).
!>

C

!>          C est un tableau DOUBLE PRECISION, dimension (LDC,N)
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale du tableau C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, dimension (N)
!>

RESID

!>          RESID est un DOUBLE PRECISION
!>          Si UPLO = 'L', norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          Si UPLO = 'U', norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
