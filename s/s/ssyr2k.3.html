Documentation de la fonction/sous-programme

subroutine ssyr2k (character uplo, character trans, integer n, integer k, real alpha, real, dimension(lda,*) a, integer lda, real, dimension(ldb,*) b, integer ldb, real beta, real, dimension(ldc,*) c, integer ldc)

SSYR2K

But:

!>
!> SSYR2K  effectue l'une des opérations symétriques de rang 2k
!>
!>    C := alpha*A*B**T + alpha*B*A**T + beta*C,
!>
!> ou
!>
!>    C := alpha*A**T*B + alpha*B**T*A + beta*C,
!>
!> où  alpha et beta  sont des scalaires, C est une matrice symétrique de taille  n par n
!> et  A et B  sont des matrices de taille  n par k  dans le premier cas et  k par n
!> dans le second cas.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>           À l'entrée,   UPLO  spécifie si la partie  supérieure ou inférieure
!>           du tableau  C  doit être référencée comme suit :
!>
!>              UPLO = 'U' ou 'u'   Seule la partie triangulaire supérieure de C
!>                                  doit être référencée.
!>
!>              UPLO = 'L' ou 'l'   Seule la partie triangulaire inférieure de C
!>                                  doit être référencée.
!>

TRANS

!>          TRANS est CHARACTER*1
!>           À l'entrée,  TRANS  spécifie l'opération à effectuer comme suit :
!>
!>              TRANS = 'N' ou 'n'   C := alpha*A*B**T + alpha*B*A**T +
!>                                        beta*C.
!>
!>              TRANS = 'T' ou 't'   C := alpha*A**T*B + alpha*B**T*A +
!>                                        beta*C.
!>
!>              TRANS = 'C' ou 'c'   C := alpha*A**T*B + alpha*B**T*A +
!>                                        beta*C.
!>

N

!>          N est INTEGER
!>           À l'entrée,  N spécifie l'ordre de la matrice C.  N doit être
!>           au moins zéro.
!>

K

!>          K est INTEGER
!>           À l'entrée, avec  TRANS = 'N' ou 'n',  K  spécifie le nombre
!>           de colonnes des matrices  A et B, et à l'entrée avec
!>           TRANS = 'T' ou 't' ou 'C' ou 'c',  K  spécifie le nombre
!>           de lignes des matrices  A et B.  K doit être au moins  zéro.
!>

ALPHA

!>          ALPHA est REAL
!>           À l'entrée, ALPHA spécifie le scalaire alpha.
!>

A

!>          A est un tableau de REAL, dimension ( LDA, ka ), où ka est
!>           k  quand  TRANS = 'N' ou 'n',  et est  n  sinon.
!>           Avant l'entrée avec  TRANS = 'N' ou 'n',  la partie principale n par k
!>           du tableau  A  doit contenir la matrice  A,  sinon
!>           la partie principale k par n du tableau  A  doit contenir
!>           la matrice A.
!>

LDA

!>          LDA est INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle que déclarée
!>           dans le programme (sub) appelant.   Quand TRANS = 'N' ou 'n'
!>           alors  LDA doit être au moins  max( 1, n ), sinon  LDA doit
!>           être au moins  max( 1, k ).
!>

B

!>          B est un tableau de REAL, dimension ( LDB, kb ), où kb est
!>           k  quand  TRANS = 'N' ou 'n',  et est  n  sinon.
!>           Avant l'entrée avec  TRANS = 'N' ou 'n',  la partie principale n par k
!>           du tableau  B  doit contenir la matrice  B,  sinon
!>           la partie principale k par n du tableau  B  doit contenir
!>           la matrice B.
!>

LDB

!>          LDB est INTEGER
!>           À l'entrée, LDB spécifie la première dimension de B telle que déclarée
!>           dans le programme (sub) appelant.   Quand TRANS = 'N' ou 'n'
!>           alors  LDB doit être au moins  max( 1, n ), sinon  LDB doit
!>           être au moins  max( 1, k ).
!>

BETA

!>          BETA est REAL
!>           À l'entrée, BETA spécifie le scalaire beta.
!>

C

!>          C est un tableau de REAL, dimension ( LDC, N )
!>           Avant l'entrée avec  UPLO = 'U' ou 'u',  la partie principale n par n
!>           en triangle supérieur du tableau C doit contenir la partie
!>           triangulaire supérieure  de la matrice symétrique  et la partie strictement
!>           inférieure triangulaire de C n'est pas référencée.  À la sortie, la
!>           partie triangulaire supérieure du tableau  C est écrasée par la partie
!>           triangulaire supérieure de la matrice mise à jour.
!>           Avant l'entrée avec  UPLO = 'L' ou 'l',  la partie principale n par n
!>           en triangle inférieur du tableau C doit contenir la partie
!>           triangulaire inférieure  de la matrice symétrique  et la partie strictement
!>           supérieure triangulaire de C n'est pas référencée.  À la sortie, la
!>           partie triangulaire inférieure du tableau  C est écrasée par la partie
!>           triangulaire inférieure de la matrice mise à jour.
!>

LDC

!>          LDC est INTEGER
!>           À l'entrée, LDC spécifie la première dimension de C telle que déclarée
!>           dans le programme (sub) appelant.   LDC doit être au moins
!>           max( 1, n ).
!>
