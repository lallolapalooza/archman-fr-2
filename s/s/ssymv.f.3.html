sous-routine ssymv (caractere uplo, entier n, reel alpha, reel, dimension(lda,*) a, entier lda, reel, dimension(*) x, entier incx, reel beta, reel, dimension(*) y, entier incy)

SSYMV

But :

!>
!> SSYMV  effectue l'opération matricielle-vectorielle
!>
!>    y := alpha*A*x + beta*y,
!>
!> où alpha et beta sont des scalaires, x et y sont des vecteurs de n éléments et
!> A est une matrice symétrique de taille n par n.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>           À l'entrée, UPLO spécifie si la partie triangulaire supérieure ou
!>           inférieure du tableau A doit être référencée comme suit :
!>
!>              UPLO = 'U' ou 'u'   Seule la partie triangulaire supérieure de A
!>                                  doit être référencée.
!>
!>              UPLO = 'L' ou 'l'   Seule la partie triangulaire inférieure de A
!>                                  doit être référencée.
!>

N

!>          N est INTEGER
!>           À l'entrée, N spécifie l'ordre de la matrice A.
!>           N doit être au moins zéro.
!>

ALPHA

!>          ALPHA est REAL
!>           À l'entrée, ALPHA spécifie le scalaire alpha.
!>

A

!>          A est un tableau REAL, dimension ( LDA, N )
!>           Avant l'entrée avec UPLO = 'U' ou 'u', la partie supérieure
!>           triangulaire principale de taille n par n du tableau A doit contenir
!>           la partie triangulaire supérieure de la matrice symétrique et la partie
!>           strictement triangulaire inférieure de A n'est pas référencée.
!>           Avant l'entrée avec UPLO = 'L' ou 'l', la partie inférieure
!>           triangulaire principale de taille n par n du tableau A doit contenir
!>           la partie triangulaire inférieure de la matrice symétrique et la partie
!>           strictement triangulaire supérieure de A n'est pas référencée.
!>

LDA

!>          LDA est INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle que déclarée
!>           dans le programme appelant (subroutine). LDA doit être au moins
!!           max( 1, n ).
!>

X

!>          X est un tableau REAL, dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           Avant l'entrée, le tableau X incrémenté doit contenir le vecteur
!>           x de n éléments.
!>

INCX

!>          INCX est INTEGER
!>           À l'entrée, INCX spécifie l'incrément pour les éléments de
!>           X. INCX ne doit pas être zéro.
!>

BETA

!>          BETA est REAL
!>           À l'entrée, BETA spécifie le scalaire beta. Lorsque BETA est
!>           fourni comme zéro, Y n'a pas besoin d'être défini à l'entrée.
!>

Y

!>          Y est un tableau REAL, dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           Avant l'entrée, le tableau Y incrémenté doit contenir le vecteur
!>           y de n éléments. À la sortie, Y est écrasé par le vecteur
!>           y mis à jour.
!>

INCY

!>          INCY est INTEGER
!>           À l'entrée, INCY spécifie l'incrément pour les éléments de
!>           Y. INCY ne doit pas être zéro.
!>
