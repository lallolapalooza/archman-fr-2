sous-programme ssgt01 (entier itype, caractere uplo, entier n, entier m, reel, dimension( lda, * ) a, entier lda, reel, dimension( ldb, * ) b, entier ldb, reel, dimension( ldz, * ) z, entier ldz, reel, dimension( * ) d, reel, dimension( * ) work, reel, dimension( * ) result)

SSGT01

Objectif :

!>
!> SSGT01 vérifie une décomposition de la forme
!>
!>    A Z   =  B Z D ou
!>    A B Z =  Z D ou
!>    B A Z =  Z D
!>
!> où A est une matrice symétrique, B est
!> symétrique définie positive, Z est orthogonale, et D est diagonale.
!>
!> L'un des ratios de test suivants est calculé :
!>
!> ITYPE = 1:  RESULT(1) = | A Z - B Z D | / ( |A| |Z| n ulp )
!>
!> ITYPE = 2:  RESULT(1) = | A B Z - Z D | / ( |A| |Z| n ulp )
!>
!> ITYPE = 3:  RESULT(1) = | B A Z - Z D | / ( |A| |Z| n ulp )
!>

Paramètres

ITYPE 
!>          ITYPE est ENTIER
!>          La forme du problème de valeurs propres généralisé symétrique.
!>          = 1:  A*z = (lambda)*B*z
!>          = 2:  A*B*z = (lambda)*z
!>          = 3:  B*A*z = (lambda)*z
!>

UPLO

!>          UPLO est CARACTERE*1
!>          Spécifie si la partie triangulaire supérieure ou inférieure des
!>          matrices symétriques A et B est stockée.
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

N

!>          N est ENTIER
!>          L'ordre de la matrice A.  N >= 0.
!>

M

!>          M est ENTIER
!>          Le nombre de valeurs propres trouvées.  0 <= M <= N.
!>

A

!>          A est un tableau REEL, dimension (LDA, N)
!>          La matrice symétrique originale A.
!>

LDA

!>          LDA est ENTIER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

B

!>          B est un tableau REEL, dimension (LDB, N)
!>          La matrice symétrique définie positive originale B.
!>

LDB

!>          LDB est ENTIER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

Z

!>          Z est un tableau REEL, dimension (LDZ, M)
!>          Les vecteurs propres calculés du problème de valeurs propres généralisé.
!>

LDZ

!>          LDZ est ENTIER
!>          La dimension principale du tableau Z.  LDZ >= max(1,N).
!>

D

!>          D est un tableau REEL, dimension (M)
!>          Les valeurs propres calculées du problème de valeurs propres généralisé.
!>

WORK

!>          WORK est un tableau REEL, dimension (N*N)
!>

RESULT

!>          RESULT est un tableau REEL, dimension (1)
!>          Le ratio de test tel que décrit ci-dessus.
!>
