subroutine ssyconv (character uplo, character way, integer n, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real, dimension( * ) e, integer info)

SSYCONV

But :

!>
!> SSYCONV convertit A donné par TRF en L et D et vice-versa.
!> Obtient les éléments non diagonaux de D (retournés dans l'espace de travail) et
!> applique ou inverse la permutation effectuée dans TRF.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si les détails de la factorisation sont stockés
!>          sous forme de matrice triangulaire supérieure ou inférieure.
!>          = 'U':  Triangulaire supérieure, forme est A = U*D*U**T;
!>          = 'L':  Triangulaire inférieure, forme est A = L*D*L**T.
!>

WAY

!>          WAY est CHARACTER*1
!>          = 'C': Convertir
!>          = 'R': Revertir
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

A

!>          A est un tableau REAL, dimension (LDA,N)
!>          La matrice diagonale par blocs D et les multiplicateurs utilisés pour
!>          obtenir le facteur U ou L calculés par SSYTRF.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (N)
!>          Détails des échanges et de la structure par blocs de D
!>          tels que déterminés par SSYTRF.
!>

E

!>          E est un tableau REAL, dimension (N)
!>          E stocke la partie supradiagonale/sous-diagonale de la matrice diagonale par blocs symétrique 1x1
!>          ou 2x2 de D dans LDLT.
!>

INFO

!>          INFO est INTEGER
!>          = 0:  sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>
