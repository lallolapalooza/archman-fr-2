subroutine ssyr2 (character uplo, integer n, real alpha, real, dimension(*) x, integer incx, real, dimension(*) y, integer incy, real, dimension(lda,*) a, integer lda)

SSYR2

Objectif :

!>
!> SSYR2  effectue l'opération symétrique de rang 2
!>
!>    A := alpha*x*y**T + alpha*y*x**T + A,
!>
!> où alpha est un scalaire, x et y sont des vecteurs de n éléments et A est une matrice
!> symétrique de taille n x n.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>           En entrée, UPLO spécifie si la partie triangulaire supérieure ou inférieure
!>           du tableau A est à référencer comme suit :
!>
!>              UPLO = 'U' ou 'u'   Seule la partie triangulaire supérieure de A
!>                                  est à référencer.
!>
!>              UPLO = 'L' ou 'l'   Seule la partie triangulaire inférieure de A
!>                                  est à référencer.
!>

N

!>          N est INTEGER
!>           En entrée, N spécifie l'ordre de la matrice A.
!>           N doit être supérieur ou égal à zéro.
!>

ALPHA

!>          ALPHA est REAL
!>           En entrée, ALPHA spécifie le scalaire alpha.
!>

X

!>          X est un tableau REAL, de dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           Avant l'entrée, le tableau X incrémenté doit contenir le vecteur
!>           x de n éléments.
!>

INCX

!>          INCX est INTEGER
!>           En entrée, INCX spécifie l'incrément pour les éléments de X.
!>           INCX ne doit pas être nul.
!>

Y

!>          Y est un tableau REAL, de dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           Avant l'entrée, le tableau Y incrémenté doit contenir le vecteur
!>           y de n éléments.
!>

INCY

!>          INCY est INTEGER
!>           En entrée, INCY spécifie l'incrément pour les éléments de Y.
!>           INCY ne doit pas être nul.
!>

A

!>          A est un tableau REAL, de dimension ( LDA, N )
!>           Avant l'entrée avec UPLO = 'U' ou 'u', la partie triangulaire
!>           supérieure n x n du tableau A doit contenir la partie triangulaire
!>           supérieure de la matrice symétrique et la partie triangulaire
!>           strictement inférieure de A n'est pas référencée. En sortie, la
!>           partie triangulaire supérieure du tableau A est écrasée par la
!>           partie triangulaire supérieure de la matrice mise à jour.
!>           Avant l'entrée avec UPLO = 'L' ou 'l', la partie triangulaire
!>           inférieure n x n du tableau A doit contenir la partie triangulaire
!>           inférieure de la matrice symétrique et la partie triangulaire
!>           strictement supérieure de A n'est pas référencée. En sortie, la
!>           partie triangulaire inférieure du tableau A est écrasée par la
!>           partie triangulaire inférieure de la matrice mise à jour.
!>

LDA

!>          LDA est INTEGER
!>           En entrée, LDA spécifie la première dimension de A telle que déclarée
!>           dans le (sous-)programme appelant. LDA doit être au moins
!>           max( 1, n ).
!>
