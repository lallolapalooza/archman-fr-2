sous-routine ssbgvx (caractère jobz, caractère range, caractère uplo, entier n, entier ka, entier kb, réel, dimension( ldab, * ) ab, entier ldab, réel, dimension( ldbb, * ) bb, entier ldbb, réel, dimension( ldq, * ) q, entier ldq, réel vl, réel vu, entier il, entier iu, réel abstol, entier m, réel, dimension( * ) w, réel, dimension( ldz, * ) z, entier ldz, réel, dimension( * ) work, entier, dimension( * ) iwork, entier, dimension( * ) ifail, entier info)

SSBGVX

Objectif :

!>
!> SSBGVX calcule les valeurs propres sélectionnées, et optionnellement, les vecteurs propres
!> d'un problème aux valeurs propres généralisé symétrique-défini et bande réel
!> de la forme A*x=(lambda)*B*x. Ici, A et B sont supposées symétriques
!> et bande, et B est également définie positive. Les valeurs propres et les vecteurs propres
!> peuvent être sélectionnés en spécifiant soit toutes les valeurs propres,
!> une plage de valeurs ou une plage d'indices pour les valeurs propres désirées.
!>

Paramètres

JOBZ 
!>          JOBZ est CHARACTER*1
!>          = 'N':  Calcul des valeurs propres seulement ;
!>          = 'V':  Calcul des valeurs propres et des vecteurs propres.
!>

RANGE

!>          RANGE est CHARACTER*1
!>          = 'A': toutes les valeurs propres seront trouvées.
!>          = 'V': toutes les valeurs propres dans l'intervalle semi-ouvert (VL,VU]
!>                 seront trouvées.
!>          = 'I': les valeurs propres de l'indice IL à IU seront trouvées.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U':  Les triangles supérieurs de A et B sont stockés ;
!>          = 'L':  Les triangles inférieurs de A et B sont stockés.
!>

N

!>          N est INTEGER
!>          L'ordre des matrices A et B.  N >= 0.
!>

KA

!>          KA est INTEGER
!>          Le nombre de super-diagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'.  KA >= 0.
!>

KB

!>          KB est INTEGER
!>          Le nombre de super-diagonales de la matrice B si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'.  KB >= 0.
!>

AB

!>          AB est un tableau REAL, dimension (LDAB, N)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice bande symétrique
!>          A, stocké dans les KA+1 premières lignes du tableau. La
!>          colonne j de A est stockée dans la colonne j du tableau AB
!>          comme suit :
!>          si UPLO = 'U', AB(ka+1+i-j,j) = A(i,j) pour max(1,j-ka)<=i<=j ;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+ka).
!>
!>          À la sortie, le contenu de AB est détruit.
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KA+1.
!>

BB

!>          BB est un tableau REAL, dimension (LDBB, N)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice bande symétrique
!>          B, stocké dans les KB+1 premières lignes du tableau. La
!>          colonne j de B est stockée dans la colonne j du tableau BB
!>          comme suit :
!>          si UPLO = 'U', BB(ka+1+i-j,j) = B(i,j) pour max(1,j-kb)<=i<=j ;
!>          si UPLO = 'L', BB(1+i-j,j)    = B(i,j) pour j<=i<=min(n,j+kb).
!>
!>          À la sortie, le facteur S de la factorisation de Cholesky scindée
!>          B = S**T*S, tel que retourné par SPBSTF.
!>

LDBB

!>          LDBB est INTEGER
!>          La dimension principale du tableau BB.  LDBB >= KB+1.
!>

Q

!>          Q est un tableau REAL, dimension (LDQ, N)
!>          Si JOBZ = 'V', la matrice n x n utilisée dans la réduction de
!>          A*x = (lambda)*B*x à la forme standard, c'est-à-dire C*x = (lambda)*x,
!>          et par conséquent la réduction de C en forme tridiagonale.
!>          Si JOBZ = 'N', le tableau Q n'est pas référencé.
!>

LDQ

!>          LDQ est INTEGER
!>          La dimension principale du tableau Q.  Si JOBZ = 'N',
!>          LDQ >= 1. Si JOBZ = 'V', LDQ >= max(1,N).
!>

VL

!>          VL est REAL
!>
!>          Si RANGE='V', la borne inférieure de l'intervalle à
!>          rechercher pour les valeurs propres. VL < VU.
!>          Non référencé si RANGE = 'A' ou 'I'.
!>

VU

!>          VU est REAL
!>
!>          Si RANGE='V', la borne supérieure de l'intervalle à
!>          rechercher pour les valeurs propres. VL < VU.
!>          Non référencé si RANGE = 'A' ou 'I'.
!>

IL

!>          IL est INTEGER
!>
!>          Si RANGE='I', l'indice de la
!>          plus petite valeur propre à retourner.
!>          1 <= IL <= IU <= N, si N > 0 ; IL = 1 et IU = 0 si N = 0.
!>          Non référencé si RANGE = 'A' ou 'V'.
!>

IU

!>          IU est INTEGER
!>
!>          Si RANGE='I', l'indice de la
!>          plus grande valeur propre à retourner.
!>          1 <= IL <= IU <= N, si N > 0 ; IL = 1 et IU = 0 si N = 0.
!>          Non référencé si RANGE = 'A' ou 'V'.
!>

ABSTOL

!>          ABSTOL est REAL
!>          La tolérance d'erreur absolue pour les valeurs propres.
!>          Une valeur propre approximative est acceptée comme convergée
!>           lorsqu'elle est déterminée comme se situant dans un intervalle [a,b]
!>          de largeur inférieure ou égale à
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          où EPS est la précision de la machine. Si ABSTOL est inférieur ou
!>          égal à zéro, alors EPS*|T| sera utilisé à sa place,
!>          où |T| est la norme 1 de la matrice tridiagonale obtenue
!>          en réduisant A à une forme tridiagonale.
!>
!>          Les valeurs propres seront calculées le plus précisément lorsque ABSTOL est
!>          défini à deux fois le seuil de sous-flux 2*SLAMCH('S'), pas zéro.
!>          Si cette routine retourne avec INFO>0, indiquant que certains
!>          vecteurs propres n'ont pas convergé, essayez de définir ABSTOL à
!>          2*SLAMCH('S').
!>

M

!>          M est INTEGER
!>          Le nombre total de valeurs propres trouvées.  0 <= M <= N.
!>          Si RANGE = 'A', M = N, et si RANGE = 'I', M = IU-IL+1.
!>

W

!>          W est un tableau REAL, dimension (N)
!>          Si INFO = 0, les valeurs propres dans l'ordre croissant.
!>

Z

!>          Z est un tableau REAL, dimension (LDZ, N)
!>          Si JOBZ = 'V', alors si INFO = 0, Z contient la matrice Z des
!>          vecteurs propres, avec la i-ème colonne de Z contenant le
!>          vecteur propre associé à W(i). Les vecteurs propres sont
!>          normalisés de sorte que Z**T*B*Z = I.
!>          Si JOBZ = 'N', alors Z n'est pas référencé.
!>

LDZ

!>          LDZ est INTEGER
!>          La dimension principale du tableau Z.  LDZ >= 1, et si
!>          JOBZ = 'V', LDZ >= max(1,N).
!>

WORK

!>          WORK est un tableau REAL, dimension (7*N)
!>

IWORK

!>          IWORK est un tableau INTEGER, dimension (5*N)
!>

IFAIL

!>          IFAIL est un tableau INTEGER, dimension (M)
!>          Si JOBZ = 'V', alors si INFO = 0, les M premiers éléments de
!>          IFAIL sont zéro. Si INFO > 0, alors IFAIL contient les
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie

!>          <= N : si INFO = i, alors i vecteurs propres n'ont pas convergé.
