subroutine ssyt01_rook (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) rwork, real resid)

SSYT01_ROOK

But:

!>
!> SSYT01_ROOK reconstruit une matrice symétrique indéfinie A à partir de sa
!> factorisation bloc L*D*L' ou U*D*U' et calcule la norme résiduelle
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> où C est la matrice reconstruite et EPS est le epsilon machine.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si la partie triangulaire supérieure ou inférieure de la
!>          matrice symétrique A est stockée:
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

N

!>          N est INTEGER
!>          Le nombre de lignes et de colonnes de la matrice A.  N >= 0.
!>

A

!>          A est un tableau REAL, dimension (LDA,N)
!>          La matrice symétrique originale A.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC est un tableau REAL, dimension (LDAFAC,N)
!>          La forme factorisée de la matrice A. AFAC contient la matrice
!>          diagonale bloc D et les multiplicateurs utilisés pour obtenir la
!>          facteur L ou U à partir de la factorisation bloc L*D*L' ou U*D*U'
!>          telle que calculée par SSYTRF_ROOK.
!>

LDAFAC

!>          LDAFAC est INTEGER
!>          La dimension principale du tableau AFAC. LDAFAC >= max(1,N).
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (N)
!>          Les indices de pivot de SSYTRF_ROOK.
!>

C

!>          C est un tableau REAL, dimension (LDC,N)
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale du tableau C. LDC >= max(1,N).
!>

RWORK

!>          RWORK est un tableau REAL, dimension (N)
!>

RESID

!>          RESID est REAL
!>          Si UPLO = 'L', norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          Si UPLO = 'U', norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
