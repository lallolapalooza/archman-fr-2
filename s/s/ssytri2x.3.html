subroutine ssytri2x (character uplo, integer n, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real, dimension( n+nb+1,* ) work, integer nb, integer info)

SSYTRI2X

But :

!>
!> SSYTRI2X calcule l'inverse d'une matrice réelle symétrique indéfinie
!> A en utilisant la factorisation A = U*D*U**T ou A = L*D*L**T calculée par
!> SSYTRF.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si les détails de la factorisation sont stockés
!>          sous forme de matrice triangulaire supérieure ou inférieure.
!>          = 'U':  Triangulaire supérieure, forme est A = U*D*U**T;
!>          = 'L':  Triangulaire inférieure, forme est A = L*D*L**T.
!>

N

!>          N est INTEGER
!>          Ordre de la matrice A.  N >= 0.
!>

A

!>          A est un tableau REAL, dimension (LDA,N)
!>          En entrée, la matrice diagonale D de taille NNB et les multiplicateurs
!>          utilisés pour obtenir le facteur U ou L calculés par SSYTRF.
!>
!>          En sortie, si INFO = 0, l'inverse (symétrique) de la matrice originale.
!>          Si UPLO = 'U', la partie triangulaire supérieure de l'inverse est formée
!>          et la partie de A sous la diagonale n'est pas référencée ; si UPLO = 'L',
!>          la partie triangulaire inférieure de l'inverse est formée et la partie de A
!>          au-dessus de la diagonale n'est pas référencée.
!>

LDA

!>          LDA est INTEGER
!>          Dimension principale du tableau A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (N)
!>          Détails des échanges et de la structure NNB de D
!>          tels que déterminés par SSYTRF.
!>

WORK

!>          WORK est un tableau REAL, dimension (N+NB+1,NB+3)
!>

NB

!>          NB est INTEGER
!>          Taille du bloc
!>

INFO

!>          INFO est INTEGER
!>          = 0: succès
!>          < 0: si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0: si INFO = i, D(i,i) = 0 ; la matrice est singulière et son
!>               inverse n'a pas pu être calculé.
!>
