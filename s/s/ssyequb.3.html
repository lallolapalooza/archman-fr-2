subroutine ssyequb (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) s, real scond, real amax, real, dimension( * ) work, integer info)

SSYEQUB

But:

!>
!> SSYEQUB calcule les mises à l'échelle des lignes et des colonnes destinées à équilibrer une
!> matrice symétrique A (par rapport à la norme Euclidienne) et à réduire
!> son nombre de conditionnement. Les facteurs d'échelle S sont calculés par l'algorithme BIN
!> (voir références) de sorte que la matrice mise à l'échelle B avec les éléments
!> B(i,j) = S(i)*A(i,j)*S(j) ait un nombre de conditionnement dans un facteur N
!> du plus petit nombre de conditionnement possible sur toutes les mises à l'échelle diagonales possibles.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U':  Le triangle supérieur de A est stocké ;
!>          = 'L':  Le triangle inférieur de A est stocké.
!>

N

!>          N est un ENTIER
!>          L'ordre de la matrice A. N >= 0.
!>

A

!>          A est un tableau REEL, dimension (LDA,N)
!>          La matrice symétrique N x N dont les facteurs d'échelle doivent être
!>          calculés.
!>

LDA

!>          LDA est un ENTIER
!>          La dimension principale du tableau A. LDA >= max(1,N).
!>

S

!>          S est un tableau REEL, dimension (N)
!>          Si INFO = 0, S contient les facteurs d'échelle pour A.
!>

SCOND

!>          SCOND est REEL
!>          Si INFO = 0, S contient le rapport du plus petit S(i) au
!>          plus grand S(i). Si SCOND >= 0.1 et AMAX n'est ni trop
!>          grand ni trop petit, il n'est pas utile de mettre à l'échelle par S.
!>

AMAX

!>          AMAX est REEL
!>          La plus grande valeur absolue de tout élément de la matrice. Si AMAX est
!>          très proche du dépassement de capacité ou très proche du sous-dépassement,
!>          la matrice doit être mise à l'échelle.
!>

WORK

!>          WORK est un tableau REEL, dimension (2*N)
!>

INFO

!>          INFO est un ENTIER
!>          = 0:  Sortie réussie
!>          < 0:  Si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0:  Si INFO = i, le i-ème élément diagonal est non positif.
!>
