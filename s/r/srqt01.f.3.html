Documentation de la fonction/sous-programme

subroutine srqt01 (integer m, integer n, real, dimension( lda, * ) a, real, dimension( lda, * ) af, real, dimension( lda, * ) q, real, dimension( lda, * ) r, integer lda, real, dimension( * ) tau, real, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

SRQT01

Purpose:

!>
!> SRQT01 teste SGERQF, qui calcule la factorisation RQ d'une matrice m-par-n
!> A, et teste partiellement SORGRQ qui forme la matrice orthogonale n-par-n
!> Q.
!>
!> SRQT01 compare R avec A*Q', et vérifie que Q est orthogonale.
!>

Paramètres

M 
!>          M est un entier (INTEGER)
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est un entier (INTEGER)
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

A

!>          A est un tableau REEL (REAL), de dimension (LDA,N)
!>          La matrice A de taille m-par-n.
!>

AF

!>          AF est un tableau REEL (REAL), de dimension (LDA,N)
!>          Détails de la factorisation RQ de A, tels que retournés par SGERQF.
!>          Voir SGERQF pour plus de détails.
!>

Q

!>          Q est un tableau REEL (REAL), de dimension (LDA,N)
!>          La matrice orthogonale Q de taille n-par-n.
!>

R

!>          R est un tableau REEL (REAL), de dimension (LDA,max(M,N))
!>

LDA

!>          LDA est un entier (INTEGER)
!>          La dimension principale des tableaux A, AF, Q et L.
!>          LDA >= max(M,N).
!>

TAU

!>          TAU est un tableau REEL (REAL), de dimension (min(M,N))
!>          Les facteurs scalaires des réflecteurs élémentaires, tels que
!>          retournés par SGERQF.
!>

WORK

!>          WORK est un tableau REEL (REAL), de dimension (LWORK)
!>

LWORK

!>          LWORK est un entier (INTEGER)
!>          La dimension du tableau WORK.
!>

RWORK

!>          RWORK est un tableau REEL (REAL), de dimension (max(M,N))
!>

RESULT

!>          RESULT est un tableau REEL (REAL), de dimension (2)
!>          Les rapports de test :
!>          RESULT(1) = norm( R - A*Q' ) / ( N * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q*Q' ) / ( N * EPS )
!>
