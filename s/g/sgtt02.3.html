subroutine sgtt02 (trans, n, nrhs, dl, d, du, x, ldx, b, ldb, resid)
SGTT02

Documentation de fonction/sous programme

sous programme sgtt02 (character trans, integer n, integer nrhs, real, dimension( * ) dl, real, dimension( * ) d, real, dimension( * ) du, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldb, * ) b, integer ldb, real resid)

SGTT02

Objectif :

!>
!> SGTT02 calcule le résidu pour la solution d'un système d'équations
!> tridiagonales :
!>    RESID = norm(B - op(A)*X) / (norm(op(A)) * norm(X) * EPS),
!> où EPS est l'epsilon machine.
!> La norme utilisée est la 1-norme.
!>

Paramètres

TRANS 
!>          TRANS est CHARACTER
!>          Spécifie la forme du résidu.
!>          = 'N' :  B - A    * X  (Pas de transposée)
!>          = 'T' :  B - A**T * X  (Transposée)
!>          = 'C' :  B - A**H * X  (Conjuguée transposée = Transposée)
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de côtés droits, c'est-à-dire le nombre de colonnes
!>          des matrices B et X.  NRHS >= 0.
!>

DL

!>          DL est tableau REAL, dimension (N-1)
!>          Les éléments sous-diagonaux (n-1) de A.
!>

D

!>          D est tableau REAL, dimension (N)
!>          Les éléments diagonaux de A.
!>

DU

!>          DU est tableau REAL, dimension (N-1)
!>          Les éléments super-diagonaux (n-1) de A.
!>

X

!>          X est tableau REAL, dimension (LDX,NRHS)
!>          La solution calculée des vecteurs X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

B

!>          B est tableau REAL, dimension (LDB,NRHS)
!>          À l'entrée, les vecteurs du côté droit pour le système
!>          d'équations linéaires.
!>          À la sortie, B est écrasé par la différence B - op(A)*X.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

RESID

!>          RESID est REAL
!>          norm(B - op(A)*X) / (norm(op(A)) * norm(X) * EPS)
!>
