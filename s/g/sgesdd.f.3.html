subroutine sgesdd (jobz, m, n, a, lda, s, u, ldu, vt, ldvt, work, lwork, iwork, info)
SGESDD

Documentation de la fonction/sous-programme

sous-routine sgesdd (caractere jobz, entier m, entier n, reel, dimension( lda, * ) a, entier lda, reel, dimension( * ) s, reel, dimension( ldu, * ) u, entier ldu, reel, dimension( ldvt, * ) vt, entier ldvt, reel, dimension( * ) work, entier lwork, entier, dimension( * ) iwork, entier info)

SGESDD

Objectif :

!>
!> SGESDD calcule la décomposition en valeurs singulières (SVD) d'une matrice réelle
!> M par N, en calculant éventuellement les vecteurs singuliers gauches et droits.
!> Si des vecteurs singuliers sont désirés, un algorithme de diviser pour régner est utilisé.
!>
!> La SVD est écrite comme :
!>
!>      A = U * SIGMA * transpose(V)
!>
!> où SIGMA est une matrice M par N qui est nulle à l'exception de ses
!> min(m,n) éléments diagonaux, U est une matrice orthogonale M par M, et
!> V est une matrice orthogonale N par N. Les éléments diagonaux de SIGMA
!> sont les valeurs singulières de A ; elles sont réelles et non négatives, et
!> sont retournées dans l'ordre décroissant. Les premières min(m,n) colonnes de
!> U et V sont les vecteurs singuliers gauches et droits de A.
!>
!> Notez que la routine retourne VT = V**T, pas V.
!>
!>

Paramètres

JOBZ 
!>          JOBZ est CHARACTER*1
!>          Spécifie les options pour le calcul de tout ou partie de la matrice U :
!>          = 'A':  toutes les M colonnes de U et toutes les N lignes de V**T sont
!>                  retournées dans les tableaux U et VT ;
!>          = 'S':  les premières min(M,N) colonnes de U et les premières
!>                  min(M,N) lignes de V**T sont retournées dans les tableaux U
!>                  et VT ;
!>          = 'O':  Si M >= N, les premières N colonnes de U sont écrasées sur
!>                  le tableau A et toutes les lignes de V**T sont retournées dans
!>                  le tableau VT ;
!>                  sinon, toutes les colonnes de U sont retournées dans
!>                  le tableau U et les premières M lignes de V**T sont écrasées
!>                  sur le tableau A ;
!>          = 'N':  aucune colonne de U ou ligne de V**T n'est calculée.
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice d'entrée A. M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice d'entrée A. N >= 0.
!>

A

!>          A est un tableau REAL, dimension (LDA,N)
!>          À l'entrée, la matrice M par N A.
!>          À la sortie,
!>          si JOBZ = 'O', A est écrasée par les N premières colonnes
!>                          de U (les vecteurs singuliers gauches, stockés
!>                          par colonnes) si M >= N ;
!>                          A est écrasée par les M premières lignes
!>                          de V**T (les vecteurs singuliers droits, stockés
!>                          par lignes) sinon.
!>          si JOBZ .ne. 'O', le contenu de A est détruit.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,M).
!>

S

!>          S est un tableau REAL, dimension (min(M,N))
!>          Les valeurs singulières de A, triées de telle sorte que S(i) >= S(i+1).
!>

U

!>          U est un tableau REAL, dimension (LDU,UCOL)
!>          UCOL = M si JOBZ = 'A' ou JOBZ = 'O' et M < N ;
!>          UCOL = min(M,N) si JOBZ = 'S'.
!>          Si JOBZ = 'A' ou JOBZ = 'O' et M < N, U contient la matrice orthogonale M par M
!>          U ;
!>          si JOBZ = 'S', U contient les premières min(M,N) colonnes de
!>          U (les vecteurs singuliers gauches, stockés par colonnes) ;
!>          si JOBZ = 'O' et M >= N, ou JOBZ = 'N', U n'est pas référencé.
!>

LDU

!>          LDU est INTEGER
!>          La dimension principale du tableau U. LDU >= 1 ; si
!>          JOBZ = 'S' ou 'A' ou JOBZ = 'O' et M < N, LDU >= M.
!>

VT

!>          VT est un tableau REAL, dimension (LDVT,N)
!>          Si JOBZ = 'A' ou JOBZ = 'O' et M >= N, VT contient la
!>          matrice orthogonale N par N V**T ;
!>          si JOBZ = 'S', VT contient les premières min(M,N) lignes de
!>          V**T (les vecteurs singuliers droits, stockés par lignes) ;
!>          si JOBZ = 'O' et M < N, ou JOBZ = 'N', VT n'est pas référencé.
!>

LDVT

!>          LDVT est INTEGER
!>          La dimension principale du tableau VT. LDVT >= 1 ;
!>          si JOBZ = 'A' ou JOBZ = 'O' et M >= N, LDVT >= N ;
!>          si JOBZ = 'S', LDVT >= min(M,N).
!>

WORK

!>          WORK est un tableau REAL, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) retourne le LWORK optimal ;
!>

LWORK

!>          LWORK est INTEGER
!>          La dimension du tableau WORK. LWORK >= 1.
!>          Si LWORK = -1, une requête d'espace de travail est supposée. La
!>          taille optimale pour le tableau WORK est calculée et stockée dans WORK(1),
!>          et aucun autre travail que la vérification des arguments n'est effectué.
!>
!>          Soit mx = max(M,N) et mn = min(M,N).
!>          Si JOBZ = 'N', LWORK >= 3*mn + max( mx, 7*mn ).
!>          Si JOBZ = 'O', LWORK >= 3*mn + max( mx, 5*mn*mn + 4*mn ).
!>          Si JOBZ = 'S', LWORK >= 4*mn*mn + 7*mn.
!>          Si JOBZ = 'A', LWORK >= 4*mn*mn + 6*mn + mx.
!>          Ce ne sont pas des minimums stricts dans tous les cas ; voir les commentaires dans le code.
!>          Pour de bonnes performances, LWORK devrait généralement être plus grand ;
!>          une requête est recommandée.
!>

IWORK

!>          IWORK est un tableau INTEGER, dimension (8*min(M,N))
!>

INFO

!>          INFO est INTEGER
!>          <  0 : Si INFO = -i, le i-ème argument avait une valeur illégale.
!>          = -4 : Si A contenait une entrée NAN.
!>          >  0 : SGESDD n'a pas convergé, le processus de mise à jour a échoué.
!>          =  0 : Sortie réussie.
!>
