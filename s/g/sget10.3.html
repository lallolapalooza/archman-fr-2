sous-routine sget10 (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, real result)

SGET10

Objectif :

!>
!> SGET10 compare deux matrices A et B et calcule le rapport
!> RESULT = norm( A - B ) / ( norm(A) * M * EPS )
!>

Paramètres

M 
!>          M est un ENTIER
!>          Le nombre de lignes des matrices A et B.
!>

N

!>          N est un ENTIER
!>          Le nombre de colonnes des matrices A et B.
!>

A

!>          A est un tableau REEL, dimension (LDA,N)
!>          La matrice A de taille m par n.
!>

LDA

!>          LDA est un ENTIER
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!>

B

!>          B est un tableau REEL, dimension (LDB,N)
!>          La matrice B de taille m par n.
!>

LDB

!>          LDB est un ENTIER
!>          La dimension principale du tableau B.  LDB >= max(1,M).
!>

WORK

!>          WORK est un tableau REEL, dimension (M)
!>

RESULT

!>          RESULT est un REEL
!>          RESULT = norm( A - B ) / ( norm(A) * M * EPS )
!>
