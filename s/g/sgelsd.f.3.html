subroutine sgelsd (integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) s, real rcond, integer rank, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer info)

SGELSD calcule la solution de norme minimale d'un problème de moindres carrés linéaires pour matrices GE

Objectif :

!>
!> SGELSD calcule la solution de norme minimale d'un problème linéaire de moindres
!> carrés réels :
!>     minimiser la norme 2 de | b - A*x |)
!> en utilisant la décomposition en valeurs singulières (SVD) de A. A est une
!> matrice M par N qui peut être de rang déficient.
!>
!> Plusieurs vecteurs second membre b et vecteurs solution x peuvent être
!> gérés en un seul appel ; ils sont stockés comme colonnes de la matrice
!> second membre B de taille M par NRHS et de la matrice solution X de taille
!> N par NRHS.
!>
!> Le problème est résolu en trois étapes :
!> (1) Réduire la matrice coefficient A en forme bidiagonale avec des
!>     transformations de Householder, ramenant le problème original à un
!>     problème BLS (Bidiagonal Least Squares)
!> (2) Résoudre le BLS en utilisant une approche de type diviser pour régner.
!> (3) Appliquer en retour toutes les transformations de Householder pour résoudre
!>     le problème original de moindres carrés.
!>
!> Le rang effectif de A est déterminé en considérant comme zéro les valeurs
!> singulières inférieures à RCOND fois la plus grande valeur singulière.
!>
!>

Paramètres

M 
!>          M est un INTEGER
!>          Le nombre de lignes de A. M >= 0.
!>

N

!>          N est un INTEGER
!>          Le nombre de colonnes de A. N >= 0.
!>

NRHS

!>          NRHS est un INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de colonnes
!>          des matrices B et X. NRHS >= 0.
!>

A

!>          A est un tableau REAL de dimension (LDA,N)
!>          En entrée, la matrice A de taille M par N.
!>          En sortie, A a été détruit.
!>

LDA

!>          LDA est un INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,M).
!>

B

!>          B est un tableau REAL de dimension (LDB,NRHS)
!>          En entrée, la matrice second membre B de taille M par NRHS.
!>          En sortie, B est écrasé par la matrice solution X de taille N par NRHS.
!>          Si m >= n et RANK = n, la somme des carrés des résidus pour la
!>          solution dans la i-ème colonne est donnée par la somme des carrés
!>          des éléments n+1:m de cette colonne.
!>

LDB

!>          LDB est un INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,max(M,N)).
!>

S

!>          S est un tableau REAL de dimension (min(M,N))
!>          Les valeurs singulières de A en ordre décroissant.
!>          Le conditionnement de A en norme 2 = S(1)/S(min(m,n)).
!>

RCOND

!>          RCOND est un REAL
!>          RCOND est utilisé pour déterminer le rang effectif de A.
!>          Les valeurs singulières S(i) <= RCOND*S(1) sont traitées comme zéro.
!>          Si RCOND < 0, la précision machine est utilisée à la place.
!>

RANK

!>          RANK est un INTEGER
!>          Le rang effectif de A, c'est-à-dire le nombre de valeurs singulières
!>          supérieures à RCOND*S(1).
!>

WORK

!>          WORK est un tableau REAL de dimension (MAX(1,LWORK))
!>          En sortie, si INFO = 0, WORK(1) contient la taille optimale de LWORK.
!>

LWORK

!>          LWORK est un INTEGER
!>          La dimension du tableau WORK. LWORK doit être au moins 1.
!>          La quantité exacte minimale d'espace de travail nécessaire dépend de M,
!>          N et NRHS. Tant que LWORK est au moins
!>              12*N + 2*N*SMLSIZ + 8*N*NLVL + N*NRHS + (SMLSIZ+1)**2,
!>          si M est supérieur ou égal à N, ou
!>              12*M + 2*M*SMLSIZ + 8*M*NLVL + M*NRHS + (SMLSIZ+1)**2,
!>          si M est inférieur à N, le code s'exécutera correctement.
!>          SMLSIZ est retourné par ILAENV et est égal à la taille maximale
!>          des sous-problèmes au bas de l'arbre de calcul (généralement environ 25),
!>          et
!>             NLVL = MAX( 0, INT( LOG_2( MIN( M,N )/(SMLSIZ+1) ) ) + 1 )
!>          Pour de bonnes performances, LWORK devrait généralement être plus grand.
!>
!>          Si LWORK = -1, alors une requête de dimension de travail est supposée ; la routine
!>          calcule seulement la taille optimale du tableau WORK et la
!>          taille minimale du tableau IWORK, et retourne ces valeurs comme
!>          les premières entrées des tableaux WORK et IWORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!>

IWORK

!>          IWORK est un tableau INTEGER de dimension (MAX(1,LIWORK))
!>          LIWORK >= max(1, 3*MINMN*NLVL + 11*MINMN),
!>          où MINMN = MIN( M,N ).
!>          En sortie, si INFO = 0, IWORK(1) contient la taille minimale de LIWORK.
!>

INFO

!>          INFO est un INTEGER
!>          = 0 : exécution réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale.
!>          > 0 : l'algorithme de calcul de la SVD n'a pas convergé ;
!>                si INFO = i, i éléments hors diagonale d'une forme bidiagonale
!>                intermédiaire n'ont pas convergé vers zéro.
!>
