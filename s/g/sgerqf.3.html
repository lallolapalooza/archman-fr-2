subroutine sgerqf (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( * ) work, integer lwork, integer info)

SGERQF

Objectif :

!>
!> SGERQF calcule une factorisation RQ d'une matrice M-par-N réelle A :
!> A = R * Q.
!>

Paramètres

M 
!>          M est un entier (INTEGER)
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est un entier (INTEGER)
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

A

!>          A est un tableau de réels (REAL), de dimension (LDA,N)
!>          En entrée, la matrice M-par-N A.
!>          En sortie,
!>          si m <= n, la partie triangulaire supérieure du sous-tableau
!>          A(1:m,n-m+1:n) contient la matrice triangulaire supérieure R de taille M-par-M ;
!>          si m >= n, les éléments sur et au-dessus de la (m-n)-ième sous-diagonale
!>          contiennent la matrice trapézoïdale supérieure R de taille M-par-N ;
!>          les éléments restants, avec le tableau TAU, représentent la
!>          matrice orthogonale Q comme un produit de min(m,n) réflecteurs
!>          élémentaires (voir Détails supplémentaires).
!>

LDA

!>          LDA est un entier (INTEGER)
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!>

TAU

!>          TAU est un tableau de réels (REAL), de dimension (min(M,N))
!>          Les facteurs scalaires des réflecteurs élémentaires (voir Détails
!>          supplémentaires).
!>

WORK

!>          WORK est un tableau de réels (REAL), de dimension (MAX(1,LWORK))
!>          En sortie, si INFO = 0, WORK(1) contient la taille optimale de LWORK.
!>

LWORK

!>          LWORK est un entier (INTEGER)
!>          La dimension du tableau WORK.
!>          LWORK >= 1, si MIN(M,N) = 0, et LWORK >= M, sinon.
!>          Pour une performance optimale, LWORK >= M*NB, où NB est
!>          la taille de bloc optimale.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est un entier (INTEGER)
!>          = 0 : succès
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>

Détails supplémentaires :

!>
!>  La matrice Q est représentée comme un produit de réflecteurs élémentaires
!>
!>     Q = H(1) H(2) . . . H(k), où k = min(m,n).
!>
!>  Chaque H(i) a la forme
!>
!>     H(i) = I - tau * v * v**T
!>
!>  où tau est un scalaire réel, et v est un vecteur réel avec
!>  v(n-k+i+1:n) = 0 et v(n-k+i) = 1 ; v(1:n-k+i-1) est stocké en sortie dans
!>  A(m-k+i,1:n-k+i-1), et tau dans TAU(i).
!>
