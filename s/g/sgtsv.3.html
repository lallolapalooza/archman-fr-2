subroutine sgtsv (integer n, integer nrhs, real, dimension( * ) dl, real, dimension( * ) d, real, dimension( * ) du, real, dimension( ldb, * ) b, integer ldb, integer info)

SGTSV calcule la solution du système d'équations linéaires A * X = B pour les matrices GT

Objectif :

!>
!> SGTSV  résout l'équation
!>
!>    A*X = B,
!>
!> où A est une matrice tridiagonale de taille n par n, par élimination
!> de Gauss avec pivotage partiel.
!>
!> Notez que l'équation  A**T*X = B  peut être résolue en échangeant l'ordre
!> des arguments DU et DL.
!>

Paramètres

N 
!>          N est un entier
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est un entier
!>          Le nombre de seconds membres, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!>

DL

!>          DL est un tableau de réels, de dimension (N-1)
!>          À l'entrée, DL doit contenir les (n-1) éléments sous-diagonaux
!>          de A.
!>
!>          À la sortie, DL est écrasé par les (n-2) éléments de la
!>          deuxième super-diagonale de la matrice triangulaire supérieure U
!>          de la factorisation LU de A, dans DL(1), ..., DL(n-2).
!>

D

!>          D est un tableau de réels, de dimension (N)
!>          À l'entrée, D doit contenir les éléments diagonaux de A.
!>
!>          À la sortie, D est écrasé par les n éléments diagonaux de U.
!>

DU

!>          DU est un tableau de réels, de dimension (N-1)
!>          À l'entrée, DU doit contenir les (n-1) éléments super-diagonaux
!>          de A.
!>
!>          À la sortie, DU est écrasé par les (n-1) éléments de la première
!>          super-diagonale de U.
!>

B

!>          B est un tableau de réels, de dimension (LDB,NRHS)
!>          À l'entrée, la matrice des seconds membres B de taille N par NRHS.
!>          À la sortie, si INFO = 0, la matrice solution X de taille N par NRHS.
!>

LDB

!>          LDB est un entier
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

INFO

!>          INFO est un entier
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 : si INFO = i, U(i,i) est exactement zéro, et la solution
!>               n'a pas été calculée. La factorisation n'est pas complétée
!>               sauf si i = N.
!>
