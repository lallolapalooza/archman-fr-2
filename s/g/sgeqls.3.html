Documentation de la fonction/sous-programme

subroutine sgeqls (integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( ldb, * ) b, integer ldb, real, dimension( lwork ) work, integer lwork, integer info)

SGEQLS

Purpose:

!>
!> Résout le problème des moindres carrés
!>     min || A*X - B ||
!> en utilisant la factorisation QL
!>     A = Q*L
!> calculée par SGEQLF.
!>

Paramètres

M 
!>          M est un entier (INTEGER)
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est un entier (INTEGER)
!>          Le nombre de colonnes de la matrice A.  M >= N >= 0.
!>

NRHS

!>          NRHS est un entier (INTEGER)
!>          Le nombre de colonnes de B.  NRHS >= 0.
!>

A

!>          A est un tableau REAL, de dimension (LDA,N)
!>          Détails de la factorisation QL de la matrice originale A,
!>          renvoyée par SGEQLF.
!>

LDA

!>          LDA est un entier (INTEGER)
!>          La dimension principale du tableau A.  LDA >= M.
!>

TAU

!>          TAU est un tableau REAL, de dimension (N)
!>          Détails de la matrice orthogonale Q.
!>

B

!>          B est un tableau REAL, de dimension (LDB,NRHS)
!>          À l'entrée, la matrice B de membre droit de dimension m-par-nrhs.
!>          À la sortie, la matrice solution X de dimension n-par-nrhs,
!>          stockée dans les lignes m-n+1:m.
!>

LDB

!>          LDB est un entier (INTEGER)
!>          La dimension principale du tableau B. LDB >= M.
!>

WORK

!>          WORK est un tableau REAL, de dimension (LWORK)
!>

LWORK

!>          LWORK est un entier (INTEGER)
!>          La longueur du tableau WORK.  LWORK doit être au moins NRHS,
!>          et devrait être au moins NRHS*NB, où NB est la taille du bloc
!>          pour cet environnement.
!>

INFO

!>          INFO est un entier (INTEGER)
!>          = 0: sortie réussie
!>          < 0: si INFO = -i, le i-ème argument était illégal
!>
