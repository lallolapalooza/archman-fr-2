sous-routine sgemlqt (character side, character trans, integer m, integer n, integer k, integer mb, real, dimension( ldv, * ) v, integer ldv, real, dimension( ldt, * ) t, integer ldt, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer info)

SGEMLQT

But :

!>
!> SGEMLQT écrase la matrice réelle générale M par N de C avec
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q C            C Q
!> TRANS = 'T':   Q**T C            C Q**T
!>
!> où Q est une matrice orthogonale réelle définie comme le produit de K
!> réflecteurs élémentaires :
!>
!>       Q = H(1) H(2) . . . H(K) = I - V T V**T
!>
!> générée en utilisant la représentation compacte WY telle que retournée par SGELQT.
!>
!> Q est d'ordre M si SIDE = 'L' et d'ordre N si SIDE = 'R'.
!>

Paramètres

SIDE 
!>          SIDE est CHARACTER*1
!>          = 'L': applique Q ou Q**T par la Gauche ;
!>          = 'R': applique Q ou Q**T par la Droite.
!>

TRANS

!>          TRANS est CHARACTER*1
!>          = 'N': Pas de transposée, applique Q ;
!>          = 'C': Transposée, applique Q**T.
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice C. M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice C. N >= 0.
!>

K

!>          K est INTEGER
!>          Le nombre de réflecteurs élémentaires dont le produit définit
!>          la matrice Q.
!>          Si SIDE = 'L', M >= K >= 0 ;
!>          si SIDE = 'R', N >= K >= 0.
!>

MB

!>          MB est INTEGER
!>          La taille de bloc utilisée pour le stockage de T. K >= MB >= 1.
!>          Ce doit être la même valeur de MB utilisée pour générer T
!>          dans SGELQT.
!>

V

!>          V est tableau REAL, dimension
!>                               (LDV,M) si SIDE = 'L',
!>                               (LDV,N) si SIDE = 'R'
!>          La i-ème ligne doit contenir le vecteur qui définit le
!>          réflecteur élémentaire H(i), pour i = 1,2,...,k, tel que retourné par
!>          SGELQT dans les K premières lignes de son argument tableau A.
!>

LDV

!>          LDV est INTEGER
!>          La dimension principale du tableau V. LDV >= max(1,K).
!>

T

!>          T est tableau REAL, dimension (LDT,K)
!>          Les facteurs triangulaires supérieurs des réflecteurs de bloc
!>          tels que retournés par SGELQT, stockés sous forme de matrice MB par K.
!>

LDT

!>          LDT est INTEGER
!>          La dimension principale du tableau T. LDT >= MB.
!>

C

!>          C est tableau REAL, dimension (LDC,N)
!>          À l'entrée, la matrice M par N C.
!>          À la sortie, C est écrasé par Q C, Q**T C, C Q**T ou C Q.
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale du tableau C. LDC >= max(1,M).
!>

WORK

!>          WORK est tableau REAL. La dimension de
!>          WORK est N*MB si SIDE = 'L', ou M*MB si SIDE = 'R'.
!>

INFO

!>          INFO est INTEGER
!>          = 0: sortie réussie
!>          < 0: si INFO = -i, le i-ème argument avait une valeur illégale
!>
