subroutine sget53 (real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real scale, real wr, real wi, real result, integer info)

SGET53

But:

!>
!> SGET53  vérifie les valeurs propres généralisées calculées par SLAG2.
!>
!> Le test de base pour une valeur propre est :
!>
!>                              | det( s A - w B ) |
!>     RESULT =  ---------------------------------------------------
!>               ulp max( s norm(A), |w| norm(B) )*norm( s A - w B )
!>
!> Deux  sont effectués :
!>
!> (1)  ulp*max( s*norm(A), |w|*norm(B) )  doit être au moins
!>      safe_minimum.  Ceci garantit que le test effectué n'est pas
!>      essentiellement  det(0*A + 0*B)=0.
!>
!> (2)  s*norm(A) + |w|*norm(B) doit être inférieur à 1/safe_minimum.
!>      Ceci garantit que  s*A - w*B  ne provoquera pas de débordement.
!>
!> Si ces tests ne sont pas réussis, alors  s  et  w  sont mis à l'échelle et
!> testés quand même, si possible.
!>

Paramètres

A 
!>          A est un tableau REAL, de dimension (LDA, 2)
!>          La matrice A 2x2.
!>

LDA

!>          LDA est un INTEGER
!>          La dimension principale de A. Elle doit être au moins 2.
!>

B

!>          B est un tableau REAL, de dimension (LDB, N)
!>          La matrice triangulaire supérieure B 2x2.
!>

LDB

!>          LDB est un INTEGER
!>          La dimension principale de B. Elle doit être au moins 2.
!>

SCALE

!>          SCALE est un REAL
!>          Le  s  dans la formule  s A - w B .  Il est
!>          supposé être non négatif.
!>

WR

!>          WR est un REAL
!>          La partie réelle de la valeur propre  w  dans la formule
!>          s A - w B .
!>

WI

!>          WI est un REAL
!>          La partie imaginaire de la valeur propre  w  dans la formule
!>          s A - w B .
!>

RESULT

!>          RESULT est un REAL
!>          Si INFO est 2 ou moins, la valeur calculée par le test
!>             décrit ci-dessus.
!>          Si INFO=3, ce sera simplement 1/ulp.
!>

INFO

!>          INFO est un INTEGER
!>          =0:  Les données d'entrée passent les .
!>          =1:  s*norm(A) + |w|*norm(B) > 1/safe_minimum.
!>          =2:  ulp*max( s*norm(A), |w|*norm(B) ) < safe_minimum
!>          =3:  Identique à INFO=2, mais  s  et  w  n'ont pas pu être mis
!>               à l'échelle pour calculer le test.
!>
