subroutine sget36 (real rmax, integer lmax, integer, dimension( 3 ) ninfo, integer knt, integer nin)

SGET36

Objectif :

!>
!> SGET36 teste STREXC, une routine pour déplacer des blocs (soit 1 par 1, soit
!> 2 par 2) sur la diagonale d'une matrice sous forme de Schur réelle. Ainsi, SLAEXC
!> calcule une matrice orthogonale Q telle que
!>
!>    Q' * T1 * Q  = T2
!>
!> et où l'un des blocs diagonaux de T1 (celui de la ligne IFST) a été déplacé à la position ILST.
!>
!> Le code de test vérifie que le résidu Q'*T1*Q-T2 est petit, que T2
!> est sous forme de Schur, et que la position finale du bloc IFST est
!> ILST (à +-1 près).
!>
!> Les matrices de test sont lues à partir d'un fichier avec le numéro logique NIN.
!>

Paramètres

RMAX 
!>          RMAX est REAL
!>          Valeur du plus grand ratio de test.
!>

LMAX

!>          LMAX est INTEGER
!>          Numéro de l'exemple où le plus grand ratio de test est atteint.
!>

NINFO

!>          NINFO est un tableau INTEGER, de dimension (3)
!>          NINFO(J) est le nombre d'exemples où INFO=J.
!>

KNT

!>          KNT est INTEGER
!>          Nombre total d'exemples testés.
!>

NIN

!>          NIN est INTEGER
!>          Numéro logique d'entrée.
!>
