sous-routine sgsvts3 (integer m, integer p, integer n, real, dimension( lda, * ) a, real, dimension( lda, * ) af, integer lda, real, dimension( ldb, * ) b, real, dimension( ldb, * ) bf, integer ldb, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldv, * ) v, integer ldv, real, dimension( ldq, * ) q, integer ldq, real, dimension( * ) alpha, real, dimension( * ) beta, real, dimension( ldr, * ) r, integer ldr, integer, dimension( * ) iwork, real, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( 6 ) result)

SGSVTS3

Objectif :

!>
!> SGSVTS3 teste SGGSVD3, qui calcule la GSVD d'une matrice M par N, A
!> et une matrice P par N, B :
!>              U'*A*Q = D1*R et V'*B*Q = D2*R.
!>

Paramètres

M 
!>          M est un entier (INTEGER)
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

P

!>          P est un entier (INTEGER)
!>          Le nombre de lignes de la matrice B.  P >= 0.
!>

N

!>          N est un entier (INTEGER)
!>          Le nombre de colonnes des matrices A et B.  N >= 0.
!>

A

!>          A est un tableau de réels (REAL), dimension (LDA,M)
!>          La matrice A de dimensions M par N.
!>

AF

!>          AF est un tableau de réels (REAL), dimension (LDA,N)
!>          Détails de la GSVD de A et B, retournés par SGGSVD3,
!>          voir SGGSVD3 pour plus de détails.
!>

LDA

!>          LDA est un entier (INTEGER)
!>          La dimension principale des tableaux A et AF.
!>          LDA >= max( 1,M ).
!>

B

!>          B est un tableau de réels (REAL), dimension (LDB,P)
!>          À l'entrée, la matrice B de dimensions P par N.
!>

BF

!>          BF est un tableau de réels (REAL), dimension (LDB,N)
!>          Détails de la GSVD de A et B, retournés par SGGSVD3,
!>          voir SGGSVD3 pour plus de détails.
!>

LDB

!>          LDB est un entier (INTEGER)
!>          La dimension principale des tableaux B et BF.
!>          LDB >= max(1,P).
!>

U

!>          U est un tableau de réels (REAL), dimension (LDU,M)
!>          La matrice orthogonale U de dimensions M par M.
!>

LDU

!>          LDU est un entier (INTEGER)
!>          La dimension principale du tableau U. LDU >= max(1,M).
!>

V

!>          V est un tableau de réels (REAL), dimension (LDV,M)
!>          La matrice orthogonale V de dimensions P par P.
!>

LDV

!>          LDV est un entier (INTEGER)
!>          La dimension principale du tableau V. LDV >= max(1,P).
!>

Q

!>          Q est un tableau de réels (REAL), dimension (LDQ,N)
!>          La matrice orthogonale Q de dimensions N par N.
!>

LDQ

!>          LDQ est un entier (INTEGER)
!>          La dimension principale du tableau Q. LDQ >= max(1,N).
!>

ALPHA

!>          ALPHA est un tableau de réels (REAL), dimension (N)
!>

BETA

!>          BETA est un tableau de réels (REAL), dimension (N)
!>
!>          Les paires de valeurs singulières généralisées de A et B, les
!>          matrices "diagonales" D1 et D2 sont construites à partir de
!>          ALPHA et BETA, voir la sous-routine SGGSVD3 pour les détails.
!>

R

!>          R est un tableau de réels (REAL), dimension (LDR,N)
!>          La matrice triangulaire supérieure R.
!>

LDR

!>          LDR est un entier (INTEGER)
!>          La dimension principale du tableau R. LDR >= max(1,N).
!>

IWORK

!>          IWORK est un tableau d'entiers (INTEGER), dimension (N)
!>

WORK

!>          WORK est un tableau de réels (REAL), dimension (LWORK)
!>

LWORK

!>          LWORK est un entier (INTEGER)
!>          La dimension du tableau WORK,
!>          LWORK >= max(M,P,N)*max(M,P,N).
!>

RWORK

!>          RWORK est un tableau de réels (REAL), dimension (max(M,P,N))
!>

RESULT

!>          RESULT est un tableau de réels (REAL), dimension (6)
!>          Les rapports de test :
!>          RESULT(1) = norm( U'*A*Q - D1*R ) / ( MAX(M,N)*norm(A)*ULP)
!>          RESULT(2) = norm( V'*B*Q - D2*R ) / ( MAX(P,N)*norm(B)*ULP)
!>          RESULT(3) = norm( I - U'*U ) / ( M*ULP )
!>          RESULT(4) = norm( I - V'*V ) / ( P*ULP )
!>          RESULT(5) = norm( I - Q'*Q ) / ( N*ULP )
!>          RESULT(6) = 0        si ALPHA est dans l'ordre décroissant ;
!>                    = ULPINV   sinon.
!>
