subroutine sgerqf (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( * ) work, integer lwork, integer info)

SGERQF

But:

!>
!> SGERQF calcule une factorisation RQ d'une matrice réelle M-par-N A:
!> A = R * Q.
!>

Paramètres

M 
!>          M est un entier (INTEGER)
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est un entier (INTEGER)
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

A

!>          A est un tableau réel (REAL), de dimension (LDA,N)
!>          À l'entrée, la matrice M-par-N A.
!>          À la sortie,
!>          si m <= n, la triangle supérieur de la sous-matrice
!>          A(1:m,n-m+1:n) contient la matrice triangulaire supérieure M-par-M R;
!>          si m >= n, les éléments sur et au-dessus de la (m-n)-ième sous-diagonale
!>          contiennent la matrice trapézoïdale supérieure M-par-N R;
!>          les éléments restants, avec le tableau TAU, représentent la
!>          matrice orthogonale Q comme un produit de min(m,n) réflexions
!>          élémentaires (voir Détails supplémentaires).
!>

LDA

!>          LDA est un entier (INTEGER)
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!>

TAU

!>          TAU est un tableau réel (REAL), de dimension (min(M,N))
!>          Les facteurs scalaires des réflexions élémentaires (voir Détails
!>          supplémentaires).
!>

WORK

!>          WORK est un tableau réel (REAL), de dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie le LWORK optimal.
!>

LWORK

!>          LWORK est un entier (INTEGER)
!>          La dimension du tableau WORK.
!>          LWORK >= 1, si MIN(M,N) = 0, et LWORK >= M sinon.
!>          Pour des performances optimales, LWORK >= M*NB, où NB est
!>          la taille de bloc optimale.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la routine
!>          calcule seulement la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est un entier (INTEGER)
!>          = 0:  Succès de l'exécution
!>          < 0:  Si INFO = -i, le i-ème argument avait une valeur illégale
!>

Détails supplémentaires:

!>
!>  La matrice Q est représentée comme un produit de réflexions élémentaires
!>
!>     Q = H(1) H(2) . . . H(k), où k = min(m,n).
!>
!>  Chaque H(i) a la forme
!>
!>     H(i) = I - tau * v * v**T
!>
!>  où tau est un scalaire réel, et v est un vecteur réel avec
!>  v(n-k+i+1:n) = 0 et v(n-k+i) = 1 ; v(1:n-k+i-1) est stocké à la sortie dans
!>  A(m-k+i,1:n-k+i-1), et tau dans TAU(i).
!>
