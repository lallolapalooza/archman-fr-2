subroutine sgehrd (integer n, integer ilo, integer ihi, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( * ) work, integer lwork, integer info)

SGEHRD

But :

!>
!> SGEHRD réduit une matrice générale réelle A en forme quasi-triangulaire supérieure H par
!> une transformation de similarité orthogonale :  Q**T * A * Q = H .
!>

Paramètres

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

ILO

!>          ILO est INTEGER
!>

IHI

!>          IHI est INTEGER
!>
!>          On suppose que A est déjà triangulaire supérieure en lignes
!>          et colonnes 1:ILO-1 et IHI+1:N. ILO et IHI sont normalement
!>          définis par un appel précédent à SGEBAL ; sinon ils devraient être
!>          définis à 1 et N respectivement. Voir Détails supplémentaires.
!>          1 <= ILO <= IHI <= N, si N > 0 ; ILO=1 et IHI=0, si N=0.
!>

A

!>          A est un tableau REAL, dimension (LDA,N)
!>          En entrée, la matrice générale N-par-N à réduire.
!>          En sortie, le triangle supérieur et la première sous-diagonale de A
!>          sont écrasés par la matrice quasi-triangulaire supérieure H, et les
!>          éléments sous la première sous-diagonale, avec le tableau TAU,
!>          représentent la matrice orthogonale Q comme un produit de réflexions élémentaires. Voir Détails supplémentaires.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

TAU

!>          TAU est un tableau REAL, dimension (N-1)
!>          Les facteurs scalaires des réflexions élémentaires (voir Détails
!>          supplémentaires). Les éléments 1:ILO-1 et IHI:N-1 de TAU sont définis à
!>          zéro.
!>

WORK

!>          WORK est un tableau REAL, dimension (MAX(1,LWORK))
!>          En sortie, si INFO = 0, WORK(1) contient la taille optimale LWORK.
!>

LWORK

!>          LWORK est INTEGER
!>          La longueur du tableau WORK.  LWORK >= max(1,N).
!>          Pour de bonnes performances, LWORK devrait généralement être plus grand.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la routine
!>          calcule seulement la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0:  succès de l'exécution
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale.
!>

Détails supplémentaires :

!>
!>  La matrice Q est représentée comme un produit de (ihi-ilo) réflexions élémentaires
!>
!>     Q = H(ilo) H(ilo+1) . . . H(ihi-1).
!>
!>  Chaque H(i) a la forme
!>
!>     H(i) = I - tau * v * v**T
!>
!>  où tau est un scalaire réel, et v est un vecteur réel avec
!>  v(1:i) = 0, v(i+1) = 1 et v(ihi+1:n) = 0 ; v(i+2:ihi) est stocké à la sortie
!>  dans A(i+2:ihi,i), et tau dans TAU(i).
!>
!>  Le contenu de A est illustré par l'exemple suivant, avec
!>  n = 7, ilo = 2 et ihi = 6 :
!>
!>  à l'entrée,                       à la sortie,
!>
!>  ( a   a   a   a   a   a   a )    (  a   a   h   h   h   h   a )
!>  (     a   a   a   a   a   a )    (      a   h   h   h   h   a )
!>  (     a   a   a   a   a   a )    (      h   h   h   h   h   h )
!>  (     a   a   a   a   a   a )    (      v2  h   h   h   h   h )
!>  (     a   a   a   a   a   a )    (      v2  v3  h   h   h   h )
!>  (     a   a   a   a   a   a )    (      v2  v3  v4  h   h   h )
!>  (                         a )    (                          a )
!>
!>  où a désigne un élément de la matrice originale A, h désigne un
!>  élément modifié de la matrice quasi-triangulaire supérieure H, et vi désigne un
!>  élément du vecteur définissant H(i).
!>
!>  Ce fichier est une légère modification de la sous-routine SGEHRD de LAPACK-3.0
!>  incorporant des améliorations proposées par Quintana-Orti et
!>  Van de Geijn (2006). (Voir SLAHR2.)
!>
