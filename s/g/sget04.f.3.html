sous-routine sget04 (integer n, integer nrhs, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldxact, * ) xact, integer ldxact, real rcond, real resid)

SGET04

Objectif:

!>
!> SGET04 calcule la différence entre une solution calculée et la
!> vraie solution d'un système d'équations linéaires.
!>
!> RESID =  ( norm(X-XACT) * RCOND ) / ( norm(XACT) * EPS ),
!> où RCOND est l'inverse du nombre de condition et EPS est l'epsilon machine.
!>

Paramètres

N 
!>          N est INTEGER
!>          Le nombre de lignes des matrices X et XACT.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de colonnes des matrices X et XACT.  NRHS >= 0.
!>

X

!>          X est un tableau de type REAL, dimension (LDX,NRHS)
!>          Les vecteurs solutions calculés. Chaque vecteur est stocké comme une
!>          colonne de la matrice X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

XACT

!>          XACT est un tableau de type REAL, dimension( LDX, NRHS )
!>          Les vecteurs solutions exacts. Chaque vecteur est stocké comme une
!>          colonne de la matrice XACT.
!>

LDXACT

!>          LDXACT est INTEGER
!>          La dimension principale du tableau XACT.  LDXACT >= max(1,N).
!>

RCOND

!>          RCOND est REAL
!>          L'inverse du nombre de condition de la matrice des coefficients
!>          dans le système d'équations.
!>

RESID

!>          RESID est REAL
!>          Le maximum sur les NRHS vecteurs solutions de
!>          ( norm(X-XACT) * RCOND ) / ( norm(XACT) * EPS )
!>
