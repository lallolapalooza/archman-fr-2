subroutine sggev3 (character jobvl, character jobvr, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) alphar, real, dimension( * ) alphai, real, dimension( * ) beta, real, dimension( ldvl, * ) vl, integer ldvl, real, dimension( ldvr, * ) vr, integer ldvr, real, dimension( * ) work, integer lwork, integer info)

SGGEV3 calcule les valeurs propres et, éventuellement, les vecteurs propres gauches et/ou droits pour les matrices GE (algorithme bloqué)

Objectif :

!>
!> SGGEV3 calcule pour une paire de matrices réelles non symétriques N x N (A,B)
!> les valeurs propres généralisées, et éventuellement, les vecteurs propres généralisés gauches et/ou droits.
!>
!> Une valeur propre généralisée pour une paire de matrices (A,B) est un scalaire
!> lambda ou un rapport alpha/beta = lambda, tel que A - lambda*B est
!> singulier. Elle est généralement représentée par la paire (alpha,beta),
!> car il existe une interprétation raisonnable pour beta=0, et même pour les deux
!> étant nuls.
!>
!> Le vecteur propre droit v(j) correspondant à la valeur propre lambda(j)
!> de (A,B) satisfait :
!>
!>                  A * v(j) = lambda(j) * B * v(j).
!>
!> Le vecteur propre gauche u(j) correspondant à la valeur propre lambda(j)
!> de (A,B) satisfait :
!>
!>                  u(j)**H * A  = lambda(j) * u(j)**H * B .
!>
!> où u(j)**H est la transposée conjuguée de u(j).
!>
!>

Paramètres

JOBVL 
!>          JOBVL est CHARACTER*1
!>          = 'N':  ne pas calculer les vecteurs propres généralisés gauches ;
!>          = 'V':  calculer les vecteurs propres généralisés gauches.
!>

JOBVR

!>          JOBVR est CHARACTER*1
!>          = 'N':  ne pas calculer les vecteurs propres généralisés droits ;
!>          = 'V':  calculer les vecteurs propres généralisés droits.
!>

N

!>          N est INTEGER
!>          L'ordre des matrices A, B, VL, et VR.  N >= 0.
!>

A

!>          A est un tableau REAL, de dimension (LDA, N)
!>          À l'entrée, la matrice A dans la paire (A,B).
!>          À la sortie, A a été écrasée.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale de A.  LDA >= max(1,N).
!>

B

!>          B est un tableau REAL, de dimension (LDB, N)
!>          À l'entrée, la matrice B dans la paire (A,B).
!>          À la sortie, B a été écrasée.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale de B.  LDB >= max(1,N).
!>

ALPHAR

!>          ALPHAR est un tableau REAL, de dimension (N)
!>

ALPHAI

!>          ALPHAI est un tableau REAL, de dimension (N)
!>

BETA

!>          BETA est un tableau REAL, de dimension (N)
!>          À la sortie, (ALPHAR(j) + ALPHAI(j)*i)/BETA(j), j=1,...,N, seront
!>          les valeurs propres généralisées. Si ALPHAI(j) est zéro, alors
!>          la j-ème valeur propre est réelle ; si elle est positive, alors la j-ème et
!>          la (j+1)-ème valeurs propres sont une paire de nombres complexes conjugués, avec
!>          ALPHAI(j+1) négatif.
!>
!>          Note : les quotients ALPHAR(j)/BETA(j) et ALPHAI(j)/BETA(j)
!>          peuvent facilement déborder ou sous-déborder, et BETA(j) peut même être zéro.
!>          Ainsi, l'utilisateur doit éviter de calculer naïvement le rapport
!>          alpha/beta. Cependant, ALPHAR et ALPHAI seront toujours inférieurs
!>          et généralement comparables à norm(A) en magnitude, et
!>          BETA toujours inférieur et généralement comparable à norm(B).
!>

VL

!>          VL est un tableau REAL, de dimension (LDVL,N)
!>          Si JOBVL = 'V', les vecteurs propres généralisés gauches u(j) sont stockés
!>          colonne par colonne dans VL, dans le même ordre que
!>          leurs valeurs propres. Si la j-ème valeur propre est réelle, alors
!>          u(j) = VL(:,j), la j-ème colonne de VL. Si la j-ème et
!>          la (j+1)-ème valeurs propres forment une paire de nombres complexes conjugués, alors
!>          u(j) = VL(:,j)+i*VL(:,j+1) et u(j+1) = VL(:,j)-i*VL(:,j+1).
!>          Chaque vecteur propre est normalisé de sorte que le composant le plus grand ait
!>          abs(partie réelle)+abs(partie imaginaire)=1.
!>          Non référencé si JOBVL = 'N'.
!>

LDVL

!>          LDVL est INTEGER
!>          La dimension principale de la matrice VL. LDVL >= 1, et
!>          si JOBVL = 'V', LDVL >= N.
!>

VR

!>          VR est un tableau REAL, de dimension (LDVR,N)
!>          Si JOBVR = 'V', les vecteurs propres généralisés droits v(j) sont stockés
!>          colonne par colonne dans VR, dans le même ordre que
!>          leurs valeurs propres. Si la j-ème valeur propre est réelle, alors
!>          v(j) = VR(:,j), la j-ème colonne de VR. Si la j-ème et
!>          la (j+1)-ème valeurs propres forment une paire de nombres complexes conjugués, alors
!>          v(j) = VR(:,j)+i*VR(:,j+1) et v(j+1) = VR(:,j)-i*VR(:,j+1).
!>          Chaque vecteur propre est normalisé de sorte que le composant le plus grand ait
!>          abs(partie réelle)+abs(partie imaginaire)=1.
!>          Non référencé si JOBVR = 'N'.
!>

LDVR

!>          LDVR est INTEGER
!>          La dimension principale de la matrice VR. LDVR >= 1, et
!>          si JOBVR = 'V', LDVR >= N.
!>

WORK

!>          WORK est un tableau REAL, de dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) contient la taille optimale de LWORK.
!>

LWORK

!>          LWORK est INTEGER
!>          La dimension du tableau WORK. LWORK >= MAX(1,8*N).
!>          Pour de bonnes performances, LWORK devrait généralement être plus grand.
!>
!>          Si LWORK = -1, alors une requête de mémoire est supposée ; la routine
!>          calcule seulement la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0:  fin réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale.
!>          = 1,...,N:
!>                L'itération QZ a échoué. Aucun vecteur propre n'a été
!>                calculé, mais ALPHAR(j), ALPHAI(j), et BETA(j)
!>                devraient être corrects pour j=INFO+1,...,N.
!>          > N:  =N+1: échec de l'itération QZ dans SLAQZ0.
!>                =N+2: retour d'erreur de STGEVC.
!>
