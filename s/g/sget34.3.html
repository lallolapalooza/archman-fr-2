subroutine sget34 (real rmax, integer lmax, integer, dimension( 2 ) ninfo, integer knt)

SGET34

Objectif :

!>
!> SGET34 teste SLAEXC, une routine pour échanger des blocs adjacents (soit
!> 1 par 1, soit 2 par 2) sur la diagonale d'une matrice sous forme de Schur réelle.
!> Ainsi, SLAEXC calcule une matrice orthogonale Q telle que
!>
!>     Q' * [ A B ] * Q  = [ C1 B1 ]
!>          [ 0 C ]        [ 0  A1 ]
!>
!> où C1 est similaire à C et A1 est similaire à A. A et C sont supposés
!> être sous forme standard (entrées diagonales égales et
!> hors diagonale avec des signes différents) et A1 et C1 sont retournés avec les
!> mêmes propriétés.
!>
!> Le code de test vérifie ces dernières assertions, ainsi que le fait que
!> le résidu dans l'équation ci-dessus est petit.
!>

Paramètres

RMAX 
!>          RMAX est REAL
!>          Valeur du plus grand ratio de test.
!>

LMAX

!>          LMAX est INTEGER
!>          Numéro de l'exemple où le plus grand ratio de test a été atteint.
!>

NINFO

!>          NINFO est un tableau INTEGER, dimension (2)
!>          NINFO(J) est le nombre d'exemples où INFO=J s'est produit.
!>

KNT

!>          KNT est INTEGER
!>          Nombre total d'exemples testés.
!>
