subroutine sgtcon (norm, n, dl, d, du, du2, ipiv, anorm, rcond, work, iwork, info)
SGTCON

Documentation de fonction/sous-programme

sous-programme sgtcon (caractère norm, entier n, réel, dimension( * ) dl, réel, dimension( * ) d, réel, dimension( * ) du, réel, dimension( * ) du2, entier, dimension( * ) ipiv, réel anorm, réel rcond, réel, dimension( * ) work, entier, dimension( * ) iwork, entier info)

SGTCON

Objectif :

!>
!> SGTCON estime l'inverse du nombre de condition d'une matrice tridiagonale réelle A
!> en utilisant la factorisation LU calculée par SGTTRF.
!>
!> Une estimation est obtenue pour norm(inv(A)), et l'inverse du nombre de condition
!> est calculé comme RCOND = 1 / (ANORM * norm(inv(A))).
!>

Paramètres

NORM 
!>          NORM est CHARACTER*1
!>          Spécifie si le nombre de condition en norme 1 ou en norme
!>          infinie est requis :
!>          = '1' ou 'O' :  norme 1 ;
!>          = 'I'        :  norme infinie.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

DL

!>          DL est REAL tableau, dimension (N-1)
!>          Les (n-1) multiplicateurs qui définissent la matrice L de la
!>          factorisation LU de A calculée par SGTTRF.
!>

D

!>          D est REAL tableau, dimension (N)
!>          Les n éléments diagonaux de la matrice triangulaire supérieure U de la
!>          factorisation LU de A.
!>

DU

!>          DU est REAL tableau, dimension (N-1)
!>          Les (n-1) éléments de la première superdiagonale de U.
!>

DU2

!>          DU2 est REAL tableau, dimension (N-2)
!>          Les (n-2) éléments de la deuxième superdiagonale de U.
!>

IPIV

!>          IPIV est INTEGER tableau, dimension (N)
!>          Les indices de pivot ; pour 1 <= i <= n, la ligne i de la matrice a été
!>          échangée avec la ligne IPIV(i).  IPIV(i) sera toujours égal à
!>          i ou i+1 ; IPIV(i) = i indique qu'aucun échange de ligne n'a été
!>          requis.
!>

ANORM

!>          ANORM est REAL
!>          Si NORM = '1' ou 'O', la norme 1 de la matrice originale A.
!>          Si NORM = 'I', la norme infinie de la matrice originale A.
!>

RCOND

!>          RCOND est REAL
!>          L'inverse du nombre de condition de la matrice A,
!>          calculé comme RCOND = 1/(ANORM * AINVNM), où AINVNM est une
!>          estimation de la norme 1 de inv(A) calculée dans cette routine.
!>

WORK

!>          WORK est REAL tableau, dimension (2*N)
!>

IWORK

!>          IWORK est INTEGER tableau, dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!>
