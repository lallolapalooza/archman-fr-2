subroutine sgghrd (character compq, character compz, integer n, integer ilo, integer ihi, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldq, * ) q, integer ldq, real, dimension( ldz, * ) z, integer ldz, integer info)

SGGHRD

But :

!>
!> SGGHRD réduit une paire de matrices réelles (A,B) à la forme généralisée
!> supérieure de Hessenberg en utilisant des transformations orthogonales, où A est une
!> matrice générale et B est triangulaire supérieure. La forme du problème
!> généralisé des valeurs propres est
!>    A*x = lambda*B*x,
!> et B est typiquement rendue triangulaire supérieure en calculant sa factorisation QR
!> et en déplaçant la matrice orthogonale Q du côté gauche de l'équation.
!>
!> Cette sous-routine réduit simultanément A en une matrice de Hessenberg H :
!>    Q**T*A*Z = H
!> et transforme B en une autre matrice triangulaire supérieure T :
!>    Q**T*B*Z = T
!> afin de réduire le problème à sa forme standard
!>    H*y = lambda*T*y
!> où y = Z**T*x.
!>
!> Les matrices orthogonales Q et Z sont déterminées comme des produits de
!> rotations de Givens. Elles peuvent être formées explicitement, ou elles peuvent être
!> post-multipliées dans les matrices d'entrée Q1 et Z1, de sorte que
!>
!>      Q1 * A * Z1**T = (Q1*Q) * H * (Z1*Z)**T
!>
!>      Q1 * B * Z1**T = (Q1*Q) * T * (Z1*Z)**T
!>
!> Si Q1 est la matrice orthogonale de la factorisation QR de B dans l'équation
!> originale A*x = lambda*B*x, alors SGGHRD réduit le problème original
!> à la forme de Hessenberg généralisée.
!>

Paramètres

COMPQ 
!>          COMPQ est CHARACTER*1
!>          = 'N': ne pas calculer Q ;
!>          = 'I': Q est initialisée à la matrice unité, et la
!>                 matrice orthogonale Q est retournée ;
!>          = 'V': Q doit contenir une matrice orthogonale Q1 à l'entrée,
!>                 et le produit Q1*Q est retourné.
!>

COMPZ

!>          COMPZ est CHARACTER*1
!>          = 'N': ne pas calculer Z ;
!>          = 'I': Z est initialisée à la matrice unité, et la
!>                 matrice orthogonale Z est retournée ;
!>          = 'V': Z doit contenir une matrice orthogonale Z1 à l'entrée,
!>                 et le produit Z1*Z est retourné.
!>

N

!>          N est INTEGER
!>          L'ordre des matrices A et B. N >= 0.
!>

ILO

!>          ILO est INTEGER
!>

IHI

!>          IHI est INTEGER
!>
!>          ILO et IHI marquent les lignes et colonnes de A qui doivent être
!>          réduites. On suppose que A est déjà triangulaire supérieure
!>          dans les lignes et colonnes 1:ILO-1 et IHI+1:N. ILO et IHI sont
!>          normalement définis par un appel précédent à SGGBAL ; sinon, ils
!>          devraient être définis à 1 et N respectivement.
!>          1 <= ILO <= IHI <= N, si N > 0 ; ILO=1 et IHI=0, si N=0.
!>

A

!>          A est tableau REAL, dimension (LDA, N)
!>          À l'entrée, la matrice générale N-par-N à réduire.
!>          À la sortie, le triangle supérieur et la première sous-diagonale de A
!>          sont recouverts par la matrice de Hessenberg supérieure H, et le
!>          reste est mis à zéro.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,N).
!>

B

!>          B est tableau REAL, dimension (LDB, N)
!>          À l'entrée, la matrice triangulaire supérieure N-par-N B.
!>          À la sortie, la matrice triangulaire supérieure T = Q**T B Z. Les
!>          éléments sous la diagonale sont mis à zéro.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,N).
!>

Q

!>          Q est tableau REAL, dimension (LDQ, N)
!>          À l'entrée, si COMPQ = 'V', la matrice orthogonale Q1,
!>          typiquement issue de la factorisation QR de B.
!>          À la sortie, si COMPQ='I', la matrice orthogonale Q, et si
!>          COMPQ = 'V', le produit Q1*Q.
!>          Non référencé si COMPQ='N'.
!>

LDQ

!>          LDQ est INTEGER
!>          La dimension principale du tableau Q.
!>          LDQ >= N si COMPQ='V' ou 'I' ; LDQ >= 1 sinon.
!>

Z

!>          Z est tableau REAL, dimension (LDZ, N)
!>          À l'entrée, si COMPZ = 'V', la matrice orthogonale Z1.
!>          À la sortie, si COMPZ='I', la matrice orthogonale Z, et si
!>          COMPZ = 'V', le produit Z1*Z.
!>          Non référencé si COMPZ='N'.
!>

LDZ

!>          LDZ est INTEGER
!>          La dimension principale du tableau Z.
!>          LDZ >= N si COMPZ='V' ou 'I' ; LDZ >= 1 sinon.
!>

INFO

!>          INFO est INTEGER
!>          = 0:  fin de l'exécution réussie.
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale.
!>
