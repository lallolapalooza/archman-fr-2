subroutine sgerqs (integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( ldb, * ) b, integer ldb, real, dimension( lwork ) work, integer lwork, integer info)

SGERQS

Objectif :

!>
!> Calcule une solution de norme minimale
!>     min || A*X - B ||
!> en utilisant la factorisation RQ
!>     A = R*Q
!> calculée par SGERQF.
!>

Paramètres

M 
!>          M est un ENTIER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est un ENTIER
!>          Le nombre de colonnes de la matrice A.  N >= M >= 0.
!>

NRHS

!>          NRHS est un ENTIER
!>          Le nombre de colonnes de B.  NRHS >= 0.
!>

A

!>          A est un tableau REEL, de dimension (LDA,N)
!>          Détails de la factorisation RQ de la matrice originale A retournée
!>          par SGERQF.
!>

LDA

!>          LDA est un ENTIER
!>          La dimension principale du tableau A.  LDA >= M.
!>

TAU

!>          TAU est un tableau REEL, de dimension (M)
!>          Détails de la matrice orthogonale Q.
!>

B

!>          B est un tableau REEL, de dimension (LDB,NRHS)
!>          À l'entrée, les vecteurs du second membre pour le système linéaire.
!>          À la sortie, les vecteurs solution X. Chaque vecteur solution
!>          est contenu dans les lignes 1:N d'une colonne de B.
!>

LDB

!>          LDB est un ENTIER
!>          La dimension principale du tableau B. LDB >= max(1,N).
!>

WORK

!>          WORK est un tableau REEL, de dimension (LWORK)
!>

LWORK

!>          LWORK est un ENTIER
!>          La longueur du tableau WORK.  LWORK doit être au moins NRHS,
!>          et devrait être au moins NRHS*NB, où NB est la taille du bloc
!>          pour cet environnement.
!>

INFO

!>          INFO est un ENTIER
!>          = 0: sortie réussie
!>          < 0: si INFO = -i, le i-ème argument avait une valeur illégale
!>
