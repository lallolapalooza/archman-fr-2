subroutine slagsy (integer n, integer k, real, dimension( * ) d, real, dimension( lda, * ) a, integer lda, integer, dimension( 4 ) iseed, real, dimension( * ) work, integer info)

SLAGSY

But :

!>
!> SLAGSY génère une matrice symétrique réelle A, en pré- et post-
!> multipliant une matrice diagonale D réelle par une matrice orthogonale
!> aléatoire : A = U*D*U'. La semi-bande peut alors être réduite à k par des
!> transformations orthogonales supplémentaires.
!>

Paramètres

N 
!>          N est un ENTIER
!>          L'ordre de la matrice A.  N >= 0.
!>

K

!>          K est un ENTIER
!>          Le nombre de sous-diagonales non nulles dans la bande de A.
!>          0 <= K <= N-1.
!>

D

!>          D est un tableau REEL, de dimension (N)
!>          Les éléments diagonaux de la matrice diagonale D.
!>

A

!>          A est un tableau REEL, de dimension (LDA,N)
!>          La matrice symétrique générée n x n A (la matrice complète est
!>          stockée).
!>

LDA

!>          LDA est un ENTIER
!>          La dimension principale du tableau A.  LDA >= N.
!>

ISEED

!>          ISEED est un tableau ENTIER, de dimension (4)
!>          À l'entrée, la graine du générateur de nombres aléatoires ; les
!>          éléments du tableau doivent être compris entre 0 et 4095, et
!>          ISEED(4) doit être impair.
!>          À la sortie, la graine est mise à jour.
!>

WORK

!>          WORK est un tableau REEL, de dimension (2*N)
!>

INFO

!>          INFO est un ENTIER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument était invalide
!>
