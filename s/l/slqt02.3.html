subroutine slqt02 (integer m, integer n, integer k, real, dimension( lda, * ) a, real, dimension( lda, * ) af, real, dimension( lda, * ) q, real, dimension( lda, * ) l, integer lda, real, dimension( * ) tau, real, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

SLQT02

But :

!>
!> SLQT02 teste SORGLQ, qui génère une matrice Q de dimension m x n avec
!> des lignes orthogonales qui est définie comme le produit de k
!> réflecteurs élémentaires.
!>
!> Étant donné la factorisation LQ d'une matrice A de dimension m x n, SLQT02 génère
!> la matrice orthogonale Q définie par la factorisation des k premières
!> lignes de A ; il compare L(1:k,1:m) à A(1:k,1:n)*Q(1:m,1:n)', et
!> vérifie que les lignes de Q sont orthogonales.
!>

Paramètres

M 
!>          M est un entier
!>          Le nombre de lignes de la matrice Q à générer.  M >= 0.
!>

N

!>          N est un entier
!>          Le nombre de colonnes de la matrice Q à générer.
!>          N >= M >= 0.
!>

K

!>          K est un entier
!>          Le nombre de réflecteurs élémentaires dont le produit définit
!>          la matrice Q. M >= K >= 0.
!>

A

!>          A est un tableau REEL, de dimension (LDA,N)
!>          La matrice A de dimension m x n qui a été factorisée par SLQT01.
!>

AF

!>          AF est un tableau REEL, de dimension (LDA,N)
!>          Détails de la factorisation LQ de A, tels que retournés par SGELQF.
!>          Voir SGELQF pour plus de détails.
!>

Q

!>          Q est un tableau REEL, de dimension (LDA,N)
!>

L

!>          L est un tableau REEL, de dimension (LDA,M)
!>

LDA

!>          LDA est un entier
!>          La dimension principale des tableaux A, AF, Q et L. LDA >= N.
!>

TAU

!>          TAU est un tableau REEL, de dimension (M)
!>          Les facteurs scalaires des réflecteurs élémentaires correspondant
!>          à la factorisation LQ dans AF.
!>

WORK

!>          WORK est un tableau REEL, de dimension (LWORK)
!>

LWORK

!>          LWORK est un entier
!>          La dimension du tableau WORK.
!>

RWORK

!>          RWORK est un tableau REEL, de dimension (M)
!>

RESULT

!>          RESULT est un tableau REEL, de dimension (2)
!>          Les ratios de test :
!>          RESULT(1) = norm( L - A*Q' ) / ( N * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q*Q' ) / ( N * EPS )
!>
