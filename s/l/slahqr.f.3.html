subroutine slahqr (logical wantt, logical wantz, integer n, integer ilo, integer ihi, real, dimension( ldh, * ) h, integer ldh, real, dimension( * ) wr, real, dimension( * ) wi, integer iloz, integer ihiz, real, dimension( ldz, * ) z, integer ldz, integer info)

SLAHQR calcule les valeurs propres et la factorisation de Schur d'une matrice supérieure de Hessenberg, en utilisant l'algorithme QR à double-décalage/simple-décalage.

Objectif :

!>
!>    SLAHQR est une sous-routine auxiliaire appelée par SHSEQR pour mettre à jour les
!>    valeurs propres et la décomposition de Schur déjà calculées par SHSEQR, en
!>    traitant la sous-matrice de Hessenberg dans les lignes et colonnes ILO à IHI.
!>

Paramètres

WANTT 
!>          WANTT est LOGICAL
!>          = .TRUE. : la forme de Schur complète T est requise ;
!>          = .FALSE. : seules les valeurs propres sont requises.
!>

WANTZ

!>          WANTZ est LOGICAL
!>          = .TRUE. : la matrice des vecteurs de Schur Z est requise ;
!>          = .FALSE. : les vecteurs de Schur ne sont pas requis.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice H. N >= 0.
!>

ILO

!>          ILO est INTEGER
!>

IHI

!>          IHI est INTEGER
!>          On suppose que H est déjà quasi-triangulaire supérieure dans
!>          les lignes et colonnes IHI+1:N, et que H(ILO,ILO-1) = 0 (sauf
!>          si ILO = 1). SLAHQR travaille principalement avec la sous-matrice
!>          de Hessenberg dans les lignes et colonnes ILO à IHI, mais applique
!>          des transformations à tout H si WANTT est .TRUE..
!>          1 <= ILO <= max(1,IHI) ; IHI <= N.
!>

H

!>          H est un tableau REAL, dimension (LDH,N)
!>          En entrée, la matrice de Hessenberg supérieure H.
!>          En sortie, si INFO est zéro et si WANTT est .TRUE., H est quasi-triangulaire
!>          supérieure dans les lignes et colonnes ILO:IHI, avec tout bloc diagonal
!>          de 2x2 sous forme standard. Si INFO est zéro et WANTT est .FALSE.,
!>          le contenu de H n'est pas spécifié en sortie. L'état de sortie de H
!>          si INFO est non nul est décrit ci-dessous sous la description de INFO.
!>

LDH

!>          LDH est INTEGER
!>          La dimension principale du tableau H. LDH >= max(1,N).
!>

WR

!>          WR est un tableau REAL, dimension (N)
!>

WI

!>          WI est un tableau REAL, dimension (N)
!>          Les parties réelle et imaginaire, respectivement, des valeurs propres
!>          calculées ILO à IHI sont stockées dans les éléments correspondants de WR et WI.
!>          Si deux valeurs propres sont calculées comme une paire conjuguée complexe,
!>          elles sont stockées dans des éléments consécutifs de WR et WI, par exemple
!>          le i-ème et le (i+1)-ème, avec WI(i) > 0 et WI(i+1) < 0. Si WANTT est .TRUE.,
!>          les valeurs propres sont stockées dans le même ordre que sur la diagonale de
!>          la forme de Schur retournée dans H, avec WR(i) = H(i,i), et, si
!>          H(i:i+1,i:i+1) est un bloc diagonal de 2x2,
!>          WI(i) = sqrt(H(i+1,i)*H(i,i+1)) et WI(i+1) = -WI(i).
!>

ILOZ

!>          ILOZ est INTEGER
!>

IHIZ

!>          IHIZ est INTEGER
!>          Spécifiez les lignes de Z auxquelles les transformations doivent être
!>          appliquées si WANTZ est .TRUE..
!>          1 <= ILOZ <= ILO ; IHI <= IHIZ <= N.
!>

Z

!>          Z est un tableau REAL, dimension (LDZ,N)
!>          Si WANTZ est .TRUE., en entrée Z doit contenir la matrice Z courante des
!>          transformations accumulées par SHSEQR, et en sortie Z a été mise à jour ;
!>          les transformations sont appliquées uniquement à la sous-matrice Z(ILOZ:IHIZ,ILO:IHI).
!>          Si WANTZ est .FALSE., Z n'est pas référencé.
!>

LDZ

!>          LDZ est INTEGER
!>          La dimension principale du tableau Z. LDZ >= max(1,N).
!>

INFO

!>          INFO est INTEGER
!>           = 0 : sortie réussie
!>           > 0 : Si INFO = i, SLAHQR n'a pas réussi à calculer toutes les
!>                 valeurs propres ILO à IHI en un total de 30 itérations
!>                 par valeur propre ; les éléments i+1:ihi de WR et WI
!>                 contiennent les valeurs propres qui ont été
!>                 calculées avec succès.
!>
!>                 Si INFO > 0 et WANTT est .FALSE., alors en sortie,
!>                 les valeurs propres non convergentes restantes sont les
!>                 valeurs propres de la matrice de Hessenberg supérieure des
!>                 lignes et colonnes ILO à INFO de la valeur finale, de sortie, de H.
!>
!>                 Si INFO > 0 et WANTT est .TRUE., alors en sortie
!>         (*)       (valeur initiale de H)*U  = U*(valeur finale de H)
!>                 où U est une matrice orthogonale. La valeur finale de H est de Hessenberg
!>                 supérieure et triangulaire dans les lignes et colonnes INFO+1 à IHI.
!>
!>                 Si INFO > 0 et WANTZ est .TRUE., alors en sortie
!>                     (valeur finale de Z)  = (valeur initiale de Z)*U
!>                 où U est la matrice orthogonale dans (*)
!>                 (quelle que soit la valeur de WANTT.)
!>
