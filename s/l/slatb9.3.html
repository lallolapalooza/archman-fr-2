sous-routine slatb9 (character*3 path, integer imat, integer m, integer p, integer n, character type, integer kla, integer kua, integer klb, integer kub, real anorm, real bnorm, integer modea, integer modeb, real cndnma, real cndnmb, character dista, character distb)

SLATB9

But :

!>
!> SLATB9 définit les paramètres pour le générateur de matrices en fonction du type
!> de matrice à générer.
!>

Paramètres

PATH 
!>          PATH est CHARACTER*3
!>          Le nom du chemin LAPACK.
!>

IMAT

!>          IMAT est INTEGER
!>          Une clé entière décrivant quelle matrice générer pour ce
!>          chemin.
!>          = 1 :   A : diagonale, B : triangulaire supérieure
!>          = 2 :   A : triangulaire supérieure, B : triangulaire supérieure
!>          = 3 :   A : triangulaire inférieure, B : triangulaire supérieure
!>          Sinon :   A : dense générale, B : dense générale
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice à générer.
!>

P

!>          P est INTEGER
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice à générer.
!>

TYPE

!>          TYPE est CHARACTER*1
!>          Le type de la matrice à générer :
!>          = 'S' :  matrice symétrique ;
!>          = 'P' :  matrice symétrique définie (positive) ;
!>          = 'N' :  matrice non symétrique.
!>

KLA

!>          KLA est INTEGER
!>          La largeur de bande inférieure de la matrice à générer.
!>

KUA

!>          KUA est INTEGER
!>          La largeur de bande supérieure de la matrice à générer.
!>

KLB

!>          KLB est INTEGER
!>          La largeur de bande inférieure de la matrice à générer.
!>

KUB

!>          KUA est INTEGER
!>          La largeur de bande supérieure de la matrice à générer.
!>

ANORM

!>          ANORM est REAL
!>          La norme désirée de la matrice à générer. La matrice diagonale
!>          des valeurs singulières ou des valeurs propres est mise à l'échelle
!>          par cette valeur.
!>

BNORM

!>          BNORM est REAL
!>          La norme désirée de la matrice à générer. La matrice diagonale
!>          des valeurs singulières ou des valeurs propres est mise à l'échelle
!>          par cette valeur.
!>

MODEA

!>          MODEA est INTEGER
!>          Une clé indiquant comment choisir le vecteur des valeurs propres.
!>

MODEB

!>          MODEB est INTEGER
!>          Une clé indiquant comment choisir le vecteur des valeurs propres.
!>

CNDNMA

!>          CNDNMA est REAL
!>          Le conditionnement désiré.
!>

CNDNMB

!>          CNDNMB est REAL
!>          Le conditionnement désiré.
!>

DISTA

!>          DISTA est CHARACTER*1
!>          Le type de distribution à utiliser par le générateur de nombres
!>          aléatoires.
!>

DISTB

!>          DISTB est CHARACTER*1
!>          Le type de distribution à utiliser par le générateur de nombres
!>          aléatoires.
!>
