subroutine slatm5 (entier prtype, entier m, entier n, réel, dimension( lda, * ) a, entier lda, réel, dimension( ldb, * ) b, entier ldb, réel, dimension( ldc, * ) c, entier ldc, réel, dimension( ldd, * ) d, entier ldd, réel, dimension( lde, * ) e, entier lde, réel, dimension( ldf, * ) f, entier ldf, réel, dimension( ldr, * ) r, entier ldr, réel, dimension( ldl, * ) l, entier ldl, réel alpha, entier qblcka, entier qblckb)

SLATM5

Objectif :

!>
!> SLATM5 génère les matrices impliquées dans l'équation de Sylvester généralisée :
!>
!>     A * R - L * B = C
!>     D * R - L * E = F
!>
!> Elles satisfont également (la condition de diagonalisation)
!>
!>  [ I -L ] ( [ A  -C ], [ D -F ] ) [ I  R ] = ( [ A    ], [ D    ] )
!>  [    I ] ( [     B ]  [    E ] ) [    I ]   ( [    B ]  [    E ] )
!>
!>

Paramètres

PRTYPE 
!>          PRTYPE est un ENTIER
!>           vers un certain type de matrices à générer
!>          (voir plus de détails).
!>

M

!>          M est un ENTIER
!>          Spécifie l'ordre de A et D et le nombre de lignes de
!>          C, F, R et L.
!>

N

!>          N est un ENTIER
!>          Spécifie l'ordre de B et E et le nombre de colonnes de
!>          C, F, R et L.
!>

A

!>          A est un tableau REEL, de dimension (LDA, M).
!>          En sortie, A de taille M par M est initialisé selon PRTYPE.
!>

LDA

!>          LDA est un ENTIER
!>          La dimension principale de A.
!>

B

!>          B est un tableau REEL, de dimension (LDB, N).
!>          En sortie, B de taille N par N est initialisé selon PRTYPE.
!>

LDB

!>          LDB est un ENTIER
!>          La dimension principale de B.
!>

C

!>          C est un tableau REEL, de dimension (LDC, N).
!>          En sortie, C de taille M par N est initialisé selon PRTYPE.
!>

LDC

!>          LDC est un ENTIER
!>          La dimension principale de C.
!>

D

!>          D est un tableau REEL, de dimension (LDD, M).
!>          En sortie, D de taille M par M est initialisé selon PRTYPE.
!>

LDD

!>          LDD est un ENTIER
!>          La dimension principale de D.
!>

E

!>          E est un tableau REEL, de dimension (LDE, N).
!>          En sortie, E de taille N par N est initialisé selon PRTYPE.
!>

LDE

!>          LDE est un ENTIER
!>          La dimension principale de E.
!>

F

!>          F est un tableau REEL, de dimension (LDF, N).
!>          En sortie, F de taille M par N est initialisé selon PRTYPE.
!>

LDF

!>          LDF est un ENTIER
!>          La dimension principale de F.
!>

R

!>          R est un tableau REEL, de dimension (LDR, N).
!>          En sortie, R de taille M par N est initialisé selon PRTYPE.
!>

LDR

!>          LDR est un ENTIER
!>          La dimension principale de R.
!>

L

!>          L est un tableau REEL, de dimension (LDL, N).
!>          En sortie, L de taille M par N est initialisé selon PRTYPE.
!>

LDL

!>          LDL est un ENTIER
!>          La dimension principale de L.
!>

ALPHA

!>          ALPHA est REEL
!>          Paramètre utilisé dans la génération des matrices de type PRTYPE = 1 et 5.
!>

QBLCKA

!>          QBLCKA est un ENTIER
!>          Lorsque PRTYPE = 3, spécifie la distance entre les blocs de taille 2x2
!>          sur la diagonale dans A. Sinon, QBLCKA n'est pas référencé. QBLCKA > 1.
!>

QBLCKB

!>          QBLCKB est un ENTIER
!>          Lorsque PRTYPE = 3, spécifie la distance entre les blocs de taille 2x2
!>          sur la diagonale dans B. Sinon, QBLCKB n'est pas référencé. QBLCKB > 1.
!>

Détails supplémentaires :

!>
!>  PRTYPE = 1 : A et B sont des blocs de Jordan, D et E sont des matrices identité
!>
!>             A : si (i == j) alors A(i, j) = 1.0
!>                 si (j == i + 1) alors A(i, j) = -1.0
!>                 sinon A(i, j) = 0.0,            i, j = 1...M
!>
!>             B : si (i == j) alors B(i, j) = 1.0 - ALPHA
!>                 si (j == i + 1) alors B(i, j) = 1.0
!>                 sinon B(i, j) = 0.0,            i, j = 1...N
!>
!>             D : si (i == j) alors D(i, j) = 1.0
!>                 sinon D(i, j) = 0.0,            i, j = 1...M
!>
!>             E : si (i == j) alors E(i, j) = 1.0
!>                 sinon E(i, j) = 0.0,            i, j = 1...N
!>
!>             L = R sont choisis dans [-10...10],
!>                  ce qui spécifie les membres de droite (C, F).
!>
!>  PRTYPE = 2 ou 3 : Triangulaire et/ou quasi-triangulaire.
!>
!>             A : si (i <= j) alors A(i, j) = [-1...1]
!>                 sinon A(i, j) = 0.0,             i, j = 1...M
!>
!>                 si (PRTYPE = 3) alors
!>                    A(k + 1, k + 1) = A(k, k)
!>                    A(k + 1, k) = [-1...1]
!>                    sign(A(k, k + 1) = -(sin(A(k + 1, k))
!>                        k = 1, M - 1, QBLCKA
!>
!>             B : si (i <= j) alors B(i, j) = [-1...1]
!>                 sinon B(i, j) = 0.0,            i, j = 1...N
!>
!>                 si (PRTYPE = 3) alors
!>                    B(k + 1, k + 1) = B(k, k)
!>                    B(k + 1, k) = [-1...1]
!>                    sign(B(k, k + 1) = -(sign(B(k + 1, k))
!>                        k = 1, N - 1, QBLCKB
!>
!>             D : si (i <= j) alors D(i, j) = [-1...1].
!>                 sinon D(i, j) = 0.0,            i, j = 1...M
!>
!>
!>             E : si (i <= j) alors D(i, j) = [-1...1]
!>                 sinon E(i, j) = 0.0,            i, j = 1...N
!>
!>                 L, R sont choisis dans [-10...10],
!>                 ce qui spécifie les membres de droite (C, F).
!>
!>  PRTYPE = 4 Plein
!>             A(i, j) = [-10...10]
!>             D(i, j) = [-1...1]    i,j = 1...M
!>             B(i, j) = [-10...10]
!>             E(i, j) = [-1...1]    i,j = 1...N
!>             R(i, j) = [-10...10]
!>             L(i, j) = [-1...1]    i = 1..M ,j = 1...N
!>
!>             L, R spécifient les membres de droite (C, F).
!>
!>  PRTYPE = 5 cas spécial cas communs et/ou valeurs propres proches.
!>
