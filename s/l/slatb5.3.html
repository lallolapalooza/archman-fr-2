subroutine slatb5 (character*3 path, integer imat, integer n, character type, integer kl, integer ku, real anorm, integer mode, real cndnum, character dist)

SLATB5

Objectif :

!>
!> SLATB5 définit les paramètres du générateur de matrices en fonction du type
!> de matrice à générer.
!>

Paramètres

PATH 
!>          PATH est CHARACTER*3
!>          Le nom du chemin LAPACK.
!>

IMAT

!>          IMAT est INTEGER
!>          Une clé entière décrivant quelle matrice générer pour ce
!>          chemin.
!>

N

!>          N est INTEGER
!>          Le nombre de lignes et de colonnes de la matrice à générer.
!>

TYPE

!>          TYPE est CHARACTER*1
!>          Le type de la matrice à générer :
!>          = 'S' : matrice symétrique
!>          = 'P' : matrice symétrique positive (semi)définie
!>          = 'N' : matrice non symétrique
!>

KL

!>          KL est INTEGER
!>          La largeur de bande inférieure de la matrice à générer.
!>

KU

!>          KU est INTEGER
!>          La largeur de bande supérieure de la matrice à générer.
!>

ANORM

!>          ANORM est REAL
!>          La norme désirée de la matrice à générer. La matrice diagonale
!>          des valeurs singulières ou propres est mise à l'échelle par cette
!>          valeur.
!>

MODE

!>          MODE est INTEGER
!>          Une clé indiquant comment choisir le vecteur des valeurs propres.
!>

CNDNUM

!>          CNDNUM est REAL
!>          Le nombre de conditionnement désiré.
!>

DIST

!>          DIST est CHARACTER*1
!>          Le type de distribution à utiliser par le générateur de nombres
!>          aléatoires.
!>
