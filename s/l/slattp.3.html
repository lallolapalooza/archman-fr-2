subroutine slattp (integer imat, character uplo, character trans, character diag, integer, dimension( 4 ) iseed, integer n, real, dimension( * ) a, real, dimension( * ) b, real, dimension( * ) work, integer info)

SLATTP

But :

!>
!> SLATTP génère une matrice triangulaire de test en stockage compacté.
!> IMAT et UPLO spécifient de manière unique les propriétés de la matrice de test,
!> qui est retournée dans le tableau AP.
!>

Paramètres

IMAT 
!>          IMAT est INTEGER
!>          Une clé entière décrivant quelle matrice générer pour ce chemin.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          Spécifie si la matrice A sera triangulaire supérieure ou inférieure.
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

TRANS

!>          TRANS est CHARACTER*1
!>          Spécifie si la matrice ou sa transposée sera utilisée.
!>          = 'N':  Pas de transposée
!>          = 'T':  Transposée
!>          = 'C':  Transposée conjuguée (= Transposée)
!>

DIAG

!>          DIAG est CHARACTER*1
!>          Spécifie si la matrice A est unitaire ou non.
!>          = 'N':  Non unitaire
!>          = 'U':  Unitaire
!>

ISEED

!>          ISEED est un tableau INTEGER, dimension (4)
!>          Le vecteur de graine pour le générateur de nombres aléatoires (utilisé dans
!>          SLATMS). Modifié à la sortie.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice à générer.
!>

A

!>          A est un tableau REAL, dimension (N*(N+1)/2)
!>          La matrice triangulaire supérieure ou inférieure A, compactée par colonnes
!>          dans un tableau linéaire. La j-ème colonne de A est stockée dans le tableau
!>          AP comme suit :
!>          si UPLO = 'U', AP((j-1)*j/2 + i) = A(i,j) pour 1<=i<=j;
!>          si UPLO = 'L',
!>             AP((j-1)*(n-j) + j*(j+1)/2 + i-j) = A(i,j) pour j<=i<=n.
!>

B

!>          B est un tableau REAL, dimension (N)
!>          Le vecteur du membre de droite, si IMAT > 10.
!>

WORK

!>          WORK est un tableau REAL, dimension (3*N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -k, le k-ème argument avait une valeur illégale
!>
