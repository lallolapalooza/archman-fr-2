subroutine slattr (integer imat, character uplo, character trans, character diag, integer, dimension( 4 ) iseed, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) b, real, dimension( * ) work, integer info)

SLATTR

But :

!>
!> SLATTR génère une matrice triangulaire de test.
!> IMAT et UPLO spécifient de manière unique les propriétés de la matrice de test,
!> qui est retournée dans le tableau A.
!>

Paramètres

IMAT 
!>          IMAT est un entier
!>          Une clé entière décrivant quelle matrice générer pour ce chemin.
!>

UPLO

!>          UPLO est un caractère de longueur 1
!>          Spécifie si la matrice A sera triangulaire supérieure ou inférieure.
!>          = 'U' : Supérieure triangulaire
!>          = 'L' : Inférieure triangulaire
!>

TRANS

!>          TRANS est un caractère de longueur 1
!>          Spécifie si la matrice ou sa transposée sera utilisée.
!>          = 'N' : Pas de transposée
!>          = 'T' : Transposée
!>          = 'C' : Transposée conjuguée (= Transposée)
!>

DIAG

!>          DIAG est un caractère de longueur 1
!>          Spécifie si la matrice A est unitaire triangulaire ou non.
!>          = 'N' : Triangulaire non unitaire
!>          = 'U' : Triangulaire unitaire
!>

ISEED

!>          ISEED est un tableau d'entiers, de dimension (4)
!>          Le vecteur de germe pour le générateur de nombres aléatoires (utilisé dans
!>          SLATMS). Modifié à la sortie.
!>

N

!>          N est un entier
!>          L'ordre de la matrice à générer.
!>

A

!>          A est un tableau réel, de dimension (LDA,N)
!>          La matrice triangulaire A. Si UPLO = 'U', la partie triangulaire
!>          supérieure n x n de premier ordre du tableau A contient la matrice
!>          triangulaire supérieure, et la partie strictement triangulaire
!>          inférieure de A n'est pas référencée. Si UPLO = 'L', la partie
!>          triangulaire inférieure n x n de premier ordre du tableau A contient
!>          la matrice triangulaire inférieure, et la partie strictement
!>          triangulaire supérieure de A n'est pas référencée. Si DIAG = 'U',
!>          les éléments diagonaux de A sont définis de telle sorte que A(k,k) = k
!>          pour 1 <= k <= n.
!>

LDA

!>          LDA est un entier
!>          La dimension principale du tableau A. LDA >= max(1,N).
!>

B

!>          B est un tableau réel, de dimension (N)
!>          Le vecteur de membre droit, si IMAT > 10.
!>

WORK

!>          WORK est un tableau réel, de dimension (3*N)
!>

INFO

!>          INFO est un entier
!>          = 0 : Sortie réussie
!>          < 0 : si INFO = -k, le k-ième argument était une valeur illégale
!>
