subroutine slapmt (logical forwrd, integer m, integer n, real, dimension( ldx, * ) x, integer ldx, integer, dimension( * ) k)

SLAPMT effectue une permutation avant ou arrière des colonnes de la matrice.

Objectif :

!>
!> SLAPMT réorganise les colonnes de la matrice M par N, X, comme spécifié
!> par la permutation K(1),K(2),...,K(N) des entiers 1,...,N.
!> Si FORWRD = .VRAI., permutation avant :
!>
!>      X(*,K(J)) est déplacé vers X(*,J) pour J = 1,2,...,N.
!>
!> Si FORWRD = .FAUX., permutation arrière :
!>
!>      X(*,J) est déplacé vers X(*,K(J)) pour J = 1,2,...,N.
!>

Paramètres

FORWRD 
!>          FORWRD est LOGIQUE
!>          = .VRAI., permutation avant
!>          = .FAUX., permutation arrière
!>

M

!>          M est ENTIER
!>          Le nombre de lignes de la matrice X. M >= 0.
!>

N

!>          N est ENTIER
!>          Le nombre de colonnes de la matrice X. N >= 0.
!>

X

!>          X est tableau REEL, dimension (LDX,N)
!>          À l'entrée, la matrice M par N, X.
!>          À la sortie, X contient la matrice permutée X.
!>

LDX

!>          LDX est ENTIER
!>          La dimension principale du tableau X, LDX >= MAX(1,M).
!>

K

!>          K est tableau ENTIER, dimension (N)
!>          À l'entrée, K contient le vecteur de permutation. K est utilisé
!>          comme espace de travail interne, mais remis à sa valeur
!>          d'origine à la sortie.
!>
