subroutine slasq6 (integer i0, integer n0, real, dimension( * ) z, integer pp, real dmin, real dmin1, real dmin2, real dn, real dnm1, real dnm2)

SLASQ6 calcule une transformation dqd en forme ping-pong, avec protection contre le sous-dépassement et le dépassement de capacité. Utilisé par sbdsqr et sstegr.

But :

!>
!> SLASQ6 calcule une transformation dqd (décalage égal à zéro) en
!> forme ping-pong, avec protection contre le sous-dépassement et le
!> dépassement de capacité.
!>

Paramètres

I0 
!>          I0 est un ENTIER
!>        Premier indice.
!>

N0

!>          N0 est un ENTIER
!>        Dernier indice.
!>

Z

!>          Z est un tableau REEL, de dimension ( 4*N )
!>        Z contient le tableau qd. EMIN est stocké dans Z(4*N0) pour éviter
!>        un argument supplémentaire.
!>

PP

!>          PP est un ENTIER
!>        PP=0 pour ping, PP=1 pour pong.
!>

DMIN

!>          DMIN est un REEL
!>        Valeur minimale de d.
!>

DMIN1

!>          DMIN1 est un REEL
!>        Valeur minimale de d, excluant D( N0 ).
!>

DMIN2

!>          DMIN2 est un REEL
!>        Valeur minimale de d, excluant D( N0 ) et D( N0-1 ).
!>

DN

!>          DN est un REEL
!>        d(N0), la dernière valeur de d.
!>

DNM1

!>          DNM1 est un REEL
!>        d(N0-1).
!>

DNM2

!>          DNM2 est un REEL
!>        d(N0-2).
!>
