subroutine sdrges3 (integer nsizes, integer, dimension( * ) nn, integer ntypes, logical, dimension( * ) dotype, integer, dimension( 4 ) iseed, real thresh, integer nounit, real, dimension( lda, * ) a, integer lda, real, dimension( lda, * ) b, real, dimension( lda, * ) s, real, dimension( lda, * ) t, real, dimension( ldq, * ) q, integer ldq, real, dimension( ldq, * ) z, real, dimension( * ) alphar, real, dimension( * ) alphai, real, dimension( * ) beta, real, dimension( * ) work, integer lwork, real, dimension( 13 ) result, logical, dimension( * ) bwork, integer info)

SDRGES3

Objectif :

!>
!> SDRGES3 vérifie le pilote de problème de valeurs propres généralisées non symétriques
!> (forme de Schur) SGGES3.
!>
!> SGGES3 factorise A et B en Q S Z'  et Q T Z' , où ' signifie
!> transposé, T est triangulaire supérieure, S est sous forme de Schur généralisée
!> (triangulaire supérieure par blocs, avec des blocs 1x1 et 2x2 sur la diagonale,
!> les blocs 2x2 correspondant à des paires de valeurs propres généralisées complexes conjuguées),
!> et Q et Z sont orthogonales. Elle calcule également les valeurs propres généralisées
!> (alpha(j),beta(j)), j=1,...,n,
!> Ainsi, w(j) = alpha(j)/beta(j) est une racine de l'équation caractéristique
!>                 det( A - w(j) B ) = 0
!> Optionnellement, elle réorganise également les valeurs propres afin qu'un
!> cluster sélectionné de valeurs propres apparaisse dans le bloc diagonal principal des
!> formes de Schur.
!>
!> Lorsque SDRGES3 est appelée, un certain nombre de matrices  () et un
!> certain nombre de matrices  sont spécifiés. Pour chaque taille ()
!> et chaque TYPE de matrice, une paire de matrices (A, B) sera générée
!> et utilisée pour les tests. Pour chaque paire de matrices, les 13 tests suivants
!> seront effectués et comparés au seuil THRESH, à l'exception des tests (5), (11) et (13).
!>
!>
!> (1)   | A - Q S Z' | / ( |A| n ulp ) (pas de tri des valeurs propres)
!>
!>
!> (2)   | B - Q T Z' | / ( |B| n ulp ) (pas de tri des valeurs propres)
!>
!>
!> (3)   | I - QQ' | / ( n ulp ) (pas de tri des valeurs propres)
!>
!>
!> (4)   | I - ZZ' | / ( n ulp ) (pas de tri des valeurs propres)
!>
!> (5)   si A est sous forme de Schur (c'est-à-dire forme quasi-triangulaire)
!>       (pas de tri des valeurs propres)
!>
!> (6)   si les valeurs propres = blocs diagonaux de la forme de Schur (S, T),
!>       c'est-à-dire tester le maximum sur j de D(j)  où :
!>
!>       si alpha(j) est réel :
!>                     |alpha(j) - S(j,j)|        |beta(j) - T(j,j)|
!>           D(j) = ------------------------ + -----------------------
!>                  max(|alpha(j)|,|S(j,j)|)   max(|beta(j)|,|T(j,j)|)
!>
!>       si alpha(j) est complexe :
!>                                 | det( s S - w T ) |
!>           D(j) = ---------------------------------------------------
!>                  ulp max( s norm(S), |w| norm(T) )*norm( s S - w T )
!>
!>       et S et T sont ici les blocs diagonaux 2x2 de S et T
!>       correspondant aux j-ème et (j+1)-ème valeurs propres.
!>       (pas de tri des valeurs propres)
!>
!> (7)   | (A,B) - Q (S,T) Z' | / ( | (A,B) | n ulp )
!>            (avec tri des valeurs propres).
!>
!> (8)   | I - QQ' | / ( n ulp ) (avec tri des valeurs propres).
!>
!> (9)   | I - ZZ' | / ( n ulp ) (avec tri des valeurs propres).
!>
!> (10)  si A est sous forme de Schur (c'est-à-dire forme quasi-triangulaire)
!>       (avec tri des valeurs propres).
!>
!> (11)  si les valeurs propres = blocs diagonaux de la forme de Schur (S, T),
!>       c'est-à-dire tester le maximum sur j de D(j)  où :
!>
!>       si alpha(j) est réel :
!>                     |alpha(j) - S(j,j)|        |beta(j) - T(j,j)|
!>           D(j) = ------------------------ + -----------------------
!>                  max(|alpha(j)|,|S(j,j)|)   max(|beta(j)|,|T(j,j)|)
!>
!>       si alpha(j) est complexe :
!>                                 | det( s S - w T ) |
!>           D(j) = ---------------------------------------------------
!>                  ulp max( s norm(S), |w| norm(T) )*norm( s S - w T )
!>
!>       et S et T sont ici les blocs diagonaux 2x2 de S et T
!>       correspondant aux j-ème et (j+1)-ème valeurs propres.
!>       (avec tri des valeurs propres).
!>
!> (12)  si le tri a fonctionné et SDIM est le nombre de valeurs propres
!>       qui ont été SÉLECTIONNÉES.
!>
!> Matrices de test
!> =============
!>
!> Les tailles des matrices de test sont spécifiées par un tableau
!> NN(1:NSIZES) ; la valeur de chaque élément NN(j) spécifie une taille.
!> Les  sont spécifiés par un tableau logique DOTYPE( 1:NTYPES ) ; si
!> DOTYPE(j) est .VRAI., alors le type de matrice  sera généré.
!> Actuellement, la liste des types possibles est :
!>
!> (1)  ( 0, 0 )         (une paire de matrices nulles)
!>
!> (2)  ( I, 0 )         (une identité et une matrice nulle)
!>
!> (3)  ( 0, I )         (une identité et une matrice nulle)
!>
!> (4)  ( I, I )         (une paire de matrices identités)
!>
!>         t   t
!> (5)  ( J , J  )       (une paire de blocs de Jordan transposés)
!>
!>                                     t                ( I   0  )
!> (6)  ( X, Y )         où  X = ( J   0  )  et Y = (      t )
!>                                  ( 0   I  )          ( 0   J  )
!>                       et I est une identité k x k et J un bloc de Jordan (k+1)x(k+1)
!>                       ; k=(N-1)/2
!>
!> (7)  ( D, I )         où D est diag( 0, 1,..., N-1 ) (une matrice diagonale
!>                       avec ces entrées diagonales.)
!> (8)  ( I, D )
!>
!> (9)  ( big*D, small*I ) où  est proche du dépassement de capacité et small=1/big
!>
!> (10) ( small*D, big*I )
!>
!> (11) ( big*I, small*D )
!>
!> (12) ( small*I, big*D )
!>
!> (13) ( big*D, big*I )
!>
!> (14) ( small*D, small*I )
!>
!> (15) ( D1, D2 )        où D1 est diag( 0, 0, 1, ..., N-3, 0 ) et
!>                        D2 est diag( 0, N-3, N-4,..., 1, 0, 0 )
!>           t   t
!> (16) Q ( J , J ) Z     où Q et Z sont des matrices orthogonales aléatoires.
!>
!> (17) Q ( T1, T2 ) Z    où T1 et T2 sont des matrices triangulaires supérieures
!>                        avec des entrées aléatoires O(1) au-dessus de la diagonale
!>                        et des entrées diagonales diag(T1) =
!>                        ( 0, 0, 1, ..., N-3, 0 ) et diag(T2) =
!>                        ( 0, N-3, N-4,..., 1, 0, 0 )
!>
!> (18) Q ( T1, T2 ) Z    diag(T1) = ( 0, 0, 1, 1, s, ..., s, 0 )
!>                        diag(T2) = ( 0, 1, 0, 1,..., 1, 0 )
!>                        s = précision machine.
!>
!> (19) Q ( T1, T2 ) Z    diag(T1)=( 0,0,1,1, 1-d, ..., 1-(N-5)*d=s, 0 )
!>                        diag(T2) = ( 0, 1, 0, 1, ..., 1, 0 )
!>
!>                                                        N-5
!> (20) Q ( T1, T2 ) Z    diag(T1)=( 0, 0, 1, 1, a, ..., a   =s, 0 )
!>                        diag(T2) = ( 0, 1, 0, 1, ..., 1, 0, 0 )
!>
!> (21) Q ( T1, T2 ) Z    diag(T1)=( 0, 0, 1, r1, r2, ..., r(N-4), 0 )
!>                        diag(T2) = ( 0, 1, 0, 1, ..., 1, 0, 0 )
!>                        où r1,..., r(N-4) sont aléatoires.
!>
!> (22) Q ( big*T1, small*T2 ) Z    diag(T1) = ( 0, 0, 1, ..., N-3, 0 )
!>                                  diag(T2) = ( 0, 1, ..., 1, 0, 0 )
!>
!> (23) Q ( small*T1, big*T2 ) Z    diag(T1) = ( 0, 0, 1, ..., N-3, 0 )
!>                                  diag(T2) = ( 0, 1, ..., 1, 0, 0 )
!>
!> (24) Q ( small*T1, small*T2 ) Z  diag(T1) = ( 0, 0, 1, ..., N-3, 0 )
!>                                  diag(T2) = ( 0, 1, ..., 1, 0, 0 )
!>
!> (25) Q ( big*T1, big*T2 ) Z      diag(T1) = ( 0, 0, 1, ..., N-3,

Paramètres

NSIZES 
!>          NSIZES est un entier
!>          Le nombre de tailles de matrices à utiliser. S'il est nul,
!>          SDRGES3 ne fait rien. NSIZES >= 0.
!>

NN

!>          NN est un tableau d'entiers, dimensionné (NSIZES)
!>          Un tableau contenant les tailles à utiliser pour les matrices.
!>          Les valeurs nulles seront ignorées. NN >= 0.
!>

NTYPES

!>          NTYPES est un entier
!>          Le nombre d'éléments dans DOTYPE. S'il est nul, SDRGES3
!>          ne fait rien. Il doit être supérieur ou égal à zéro. S'il est égal à MAXTYP+1
!>          et NSIZES est 1, alors un type supplémentaire, MAXTYP+1, est
!>          défini, qui consiste à utiliser la matrice présente dans A en entrée.
!>          Ceci n'est utile que si DOTYPE(1:MAXTYP) est .FALSE. et
!>          DOTYPE(MAXTYP+1) est .TRUE. .
!>

DOTYPE

!>          DOTYPE est un tableau logique, dimensionné (NTYPES)
!>          Si DOTYPE(j) est .TRUE., alors pour chaque taille dans NN une
!>          matrice de cette taille et de type j sera générée.
!>          Si NTYPES est inférieur au nombre maximal de types
!>          définis (PARAMETER MAXTYP), alors les types NTYPES+1 à
!>          MAXTYP ne seront pas générés. Si NTYPES est supérieur
!>          à MAXTYP, DOTYPE(MAXTYP+1) à DOTYPE(NTYPES)
!>          seront ignorés.
!>

ISEED

!>          ISEED est un tableau d'entiers, dimensionné (4)
!>          En entrée, ISEED spécifie la graine du générateur de nombres
!>          aléatoires. Les éléments du tableau doivent être compris entre 0 et 4095 ;
!>          sinon, ils seront réduits modulo 4096. De plus, ISEED(4) doit
!>          être impair. Le générateur de nombres aléatoires utilise une séquence
!>          linéaire congruente limitée à de petits entiers, et doit donc
!>          produire des nombres aléatoires indépendants de la machine. Les
!>          valeurs de ISEED sont modifiées en sortie et peuvent être utilisées
!>          dans le prochain appel à SDRGES3 pour continuer la même séquence
!>          de nombres aléatoires.
!>

THRESH

!>          THRESH est un réel
!>          Un test sera considéré comme réussi si le rapport d'erreur, calculé
!>          comme décrit ci-dessus, ne dépasse pas THRESH. Notez que l'erreur est
!>          normalisée pour être de l'ordre de 1 (O(1)), THRESH devrait donc être un multiple
!>          raisonnablement petit de 1, par exemple 10 ou 100. En particulier, il ne doit pas
!>          dépendre de la précision (simple ou double) ou de la taille
!>          de la matrice. THRESH >= 0.
!>

NOUNIT

!>          NOUNIT est un entier
!>          Le numéro d'unité FORTRAN pour l'impression des messages d'erreur
!>          (par exemple, si une routine renvoie IINFO différent de 0).
!>

A

!>          A est un tableau réel,
!>                                       dimensionné(LDA, max(NN))
!>          Utilisé pour contenir la matrice A originale. Utilisé en entrée uniquement
!>          si NTYPES=MAXTYP+1, DOTYPE(1:MAXTYP)=.FALSE., et
!>          DOTYPE(MAXTYP+1)=.TRUE.
!>

LDA

!>          LDA est un entier
!>          La dimension principale de A, B, S, et T.
!>          Elle doit être au moins 1 et au moins max( NN ).
!>

B

!>          B est un tableau réel,
!>                                       dimensionné(LDA, max(NN))
!>          Utilisé pour contenir la matrice B originale. Utilisé en entrée uniquement
!>          si NTYPES=MAXTYP+1, DOTYPE(1:MAXTYP)=.FALSE., et
!>          DOTYPE(MAXTYP+1)=.TRUE.
!>

S

!>          S est un tableau réel, dimensionné (LDA, max(NN))
!>          La matrice de forme de Schur calculée à partir de A par SGGES3. En sortie, S
!>          contient la matrice de forme de Schur correspondant à la matrice
!>          dans A.
!>

T

!>          T est un tableau réel, dimensionné (LDA, max(NN))
!>          La matrice triangulaire supérieure calculée à partir de B par SGGES3.
!>

Q

!>          Q est un tableau réel, dimensionné (LDQ, max(NN))
!>          La matrice orthogonale (gauche) calculée par SGGES3.
!>

LDQ

!>          LDQ est un entier
!>          La dimension principale de Q et Z. Elle doit
!>          être au moins 1 et au moins max( NN ).
!>

Z

!>          Z est un tableau réel, dimensionné( LDQ, max(NN) )
!>          La matrice orthogonale (droite) calculée par SGGES3.
!>

ALPHAR

!>          ALPHAR est un tableau réel, dimensionné (max(NN))
!>

ALPHAI

!>          ALPHAI est un tableau réel, dimensionné (max(NN))
!>

BETA

!>          BETA est un tableau réel, dimensionné (max(NN))
!>
!>          Les valeurs propres généralisées de (A,B) calculées par SGGES3.
!>          ( ALPHAR(k)+ALPHAI(k)*i ) / BETA(k) est la k-ième
!>          valeur propre généralisée de A et B.
!>

WORK

!>          WORK est un tableau réel, dimensionné (LWORK)
!>

LWORK

!>          LWORK est un entier
!>          La dimension du tableau WORK.
!>          LWORK >= MAX( 10*(N+1), 3*N*N ), où N est la plus grande
!>          dimension de matrice.
!>

RESULT

!>          RESULT est un tableau réel, dimensionné (15)
!>          Les valeurs calculées par les tests décrits ci-dessus.
!>          Les valeurs sont actuellement limitées à 1/ulp, pour éviter les dépassements.
!>

BWORK

!>          BWORK est un tableau logique, dimensionné (N)
!>

INFO

!>          INFO est un entier
!>          = 0: sortie réussie
!>          < 0: si INFO = -i, le i-ème argument avait une valeur illégale.
!>          > 0: une routine a renvoyé un code d'erreur. INFO est la
!>                valeur absolue de la valeur INFO renvoyée.
!>
