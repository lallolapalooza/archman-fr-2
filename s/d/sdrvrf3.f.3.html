sous-routine sdrvrf3 (entier nout, entier nn, entier, dimension( nn ) nval, reel thresh, reel, dimension( lda, * ) a, entier lda, reel, dimension( * ) arf, reel, dimension( lda, * ) b1, reel, dimension( lda, * ) b2, reel, dimension( * ) s_work_slange, reel, dimension( * ) s_work_sgeqrf, reel, dimension( * ) tau)

SDRVRF3

Objectif :

!>
!> SDRVRF3 teste les routines RFP de LAPACK :
!>     STFSM
!>

Paramètres

NOUT 
!>          NOUT est un ENTIER
!>                Le numéro d'unité pour la sortie.
!>

NN

!>          NN est un ENTIER
!>                Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau d'ENTIERS, dimension (NN)
!>                Les valeurs de la dimension de matrice N.
!>

THRESH

!>          THRESH est un REEL
!>                La valeur seuil pour les rapports de test. Un résultat est
!>                inclus dans le fichier de sortie si RESULT >= THRESH. Pour
!>                imprimer chaque rapport de test, utilisez THRESH = 0.
!>

A

!>          A est un tableau de REELS, dimension (LDA,NMAX)
!>

LDA

!>          LDA est un ENTIER
!>                La dimension principale du tableau A. LDA >= max(1,NMAX).
!>

ARF

!>          ARF est un tableau de REELS, dimension ((NMAX*(NMAX+1))/2).
!>

B1

!>          B1 est un tableau de REELS, dimension (LDA,NMAX)
!>

B2

!>          B2 est un tableau de REELS, dimension (LDA,NMAX)
!>

S_WORK_SLANGE

!>          S_WORK_SLANGE est un tableau de REELS, dimension (NMAX)
!>

S_WORK_SGEQRF

!>          S_WORK_SGEQRF est un tableau de REELS, dimension (NMAX)
!>

TAU

!>          TAU est un tableau de REELS, dimension (NMAX)
!>
