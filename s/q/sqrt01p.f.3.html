subroutine sqrt01p (m, n, a, af, q, r, lda, tau, work, lwork, rwork, result)
SQRT01P

Documentation de la fonction/sous-programme

sous-programme sqrt01p (integer m, integer n, real, dimension( lda, * ) a, real, dimension( lda, * ) af, real, dimension( lda, * ) q, real, dimension( lda, * ) r, integer lda, real, dimension( * ) tau, real, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

SQRT01P

Objectif :

!>
!> SQRT01P teste SGEQRFP, qui calcule la factorisation QR d'une matrice m-par-n
!> A, et teste partiellement SORGQR qui forme la matrice orthogonale m-par-m
!> Q.
!>
!> SQRT01P compare R avec Q'*A, et vérifie que Q est orthogonale.
!>

Paramètres

M 
!>          M est un ENTIER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est un ENTIER
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

A

!>          A est un tableau REEL, de dimension (LDA,N)
!>          La matrice m-par-n A.
!>

AF

!>          AF est un tableau REEL, de dimension (LDA,N)
!>          Détails de la factorisation QR de A, tels que retournés par SGEQRFP.
!>          Voir SGEQRFP pour plus de détails.
!>

Q

!>          Q est un tableau REEL, de dimension (LDA,M)
!>          La matrice orthogonale m-par-m Q.
!>

R

!>          R est un tableau REEL, de dimension (LDA,max(M,N))
!>

LDA

!>          LDA est un ENTIER
!>          La dimension principale des tableaux A, AF, Q et R.
!>          LDA >= max(M,N).
!>

TAU

!>          TAU est un tableau REEL, de dimension (min(M,N))
!>          Les facteurs scalaires des réflecteurs élémentaires, tels que retournés
!>          par SGEQRFP.
!>

WORK

!>          WORK est un tableau REEL, de dimension (LWORK)
!>

LWORK

!>          LWORK est un ENTIER
!>          La dimension du tableau WORK.
!>

RWORK

!>          RWORK est un tableau REEL, de dimension (M)
!>

RESULT

!>          RESULT est un tableau REEL, de dimension (2)
!>          Les ratios de test :
!>          RESULT(1) = norm( R - Q'*A ) / ( M * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q'*Q ) / ( M * EPS )
!>
