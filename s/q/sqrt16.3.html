Documentation de la fonction/sous-programme

subroutine sqrt16 (character trans, integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) rwork, real resid)

SQRT16

Objectif :

!>
!> SQRT16 calcule le résidu pour une solution d'un système d'équations linéaires
!>  A*x = b  ou  A'*x = b:
!>    RESID = norm(B - A*X) / ( max(m,n) * norm(A) * norm(X) * EPS ),
!> où EPS est l'epsilon machine.
!>

Paramètres

TRANS 
!>          TRANS est CHARACTER*1
!>          Spécifie la forme du système d'équations :
!>          = 'N':  A *x = b
!>          = 'T':  A'*x = b, où A' est la transposée de A
!>          = 'C':  A'*x = b, où A' est la transposée de A
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de colonnes de B, la matrice des seconds membres.
!>          NRHS >= 0.
!>

A

!>          A est un tableau de REAL, dimension (LDA,N)
!>          La matrice originale A de taille M x N.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!>

X

!>          X est un tableau de REAL, dimension (LDX,NRHS)
!>          Les vecteurs solutions calculés pour le système d'équations linéaires.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  Si TRANS = 'N',
!>          LDX >= max(1,N); si TRANS = 'T' ou 'C', LDX >= max(1,M).
!>

B

!>          B est un tableau de REAL, dimension (LDB,NRHS)
!>          En entrée, les vecteurs seconds membres pour le système d'équations linéaires.
!>          En sortie, B est écrasé par la différence B - A*X.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  Si TRANS = 'N',
!>          LDB >= max(1,M); si TRANS = 'T' ou 'C', LDB >= max(1,N).
!>

RWORK

!>          RWORK est un tableau de REAL, dimension (M)
!>

RESID

!>          RESID est REAL
!>          Le maximum sur le nombre de seconds membres de
!>          norm(B - A*X) / ( max(m,n) * norm(A) * norm(X) * EPS ).
!>
