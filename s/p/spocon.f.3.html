subroutine spocon (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real anorm, real rcond, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SPOCON

Objectif :

!>
!> SPOCON estime l'inverse du nombre de conditionnement (en norme 1)
!> d'une matrice symétrique réelle définie positive, en utilisant la
!> factorisation de Cholesky A = U**T*U ou A = L*L**T calculée par SPOTRF.
!>
!> Une estimation de norm(inv(A)) est obtenue, et l'inverse du nombre de
!> conditionnement est calculé comme RCOND = 1 / (ANORM * norm(inv(A))).
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U' : La partie triangulaire supérieure de A est stockée ;
!>          = 'L' : La partie triangulaire inférieure de A est stockée.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A. N >= 0.
!>

A

!>          A est un tableau REAL de dimension (LDA,N)
!>          Le facteur triangulaire U ou L de la factorisation de Cholesky
!>          A = U**T*U ou A = L*L**T, tel que calculé par SPOTRF.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,N).
!>

ANORM

!>          ANORM est REAL
!>          La norme 1 (ou norme infinie) de la matrice symétrique A.
!>

RCOND

!>          RCOND est REAL
!>          L'inverse du nombre de conditionnement de la matrice A,
!>          calculé comme RCOND = 1/(ANORM * AINVNM), où AINVNM est une
!>          estimation de la norme 1 de inv(A) calculée par cette routine.
!>

WORK

!>          WORK est un tableau REAL de dimension (3*N)
!>

IWORK

!>          IWORK est un tableau INTEGER de dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, alors le i-ème argument avait une valeur illégale
!>
