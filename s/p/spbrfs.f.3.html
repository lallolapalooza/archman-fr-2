sous-routine spbrfs (character uplo, integer n, integer kd, integer nrhs, real, dimension( ldab, * ) ab, integer ldab, real, dimension( ldafb, * ) afb, integer ldafb, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SPBRFS

But :

!>
!> SPBRFS améliore la solution calculée d'un système d'équations linéaires
!> lorsque la matrice des coefficients est symétrique et définie positive,
!> et à bande, et fournit des bornes d'erreur ainsi que des estimations d'erreur arrière
!> pour la solution.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U':  La partie triangulaire supérieure de A est stockée ;
!>          = 'L':  La partie triangulaire inférieure de A est stockée.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

KD

!>          KD est INTEGER
!>          Le nombre de super-diagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          des matrices B et X.  NRHS >= 0.
!>

AB

!>          AB est un tableau REAL, dimension (LDAB,N)
!>          La partie triangulaire supérieure ou inférieure de la matrice bande symétrique A,
!>          stockée dans les KD+1 premières lignes du tableau. La j-ème colonne
!>          de A est stockée dans la j-ème colonne du tableau AB comme suit :
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) pour max(1,j-kd)<=i<=j ;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KD+1.
!>

AFB

!>          AFB est un tableau REAL, dimension (LDAFB,N)
!>          Le facteur triangulaire U ou L de la factorisation de Cholesky
!>          A = U**T*U ou A = L*L**T de la matrice bande A calculée par
!>          SPBTRF, dans le même format de stockage que A (voir AB).
!>

LDAFB

!>          LDAFB est INTEGER
!>          La dimension principale du tableau AFB.  LDAFB >= KD+1.
!>

B

!>          B est un tableau REAL, dimension (LDB,NRHS)
!>          La matrice des membres de droite B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est un tableau REAL, dimension (LDX,NRHS)
!>          En entrée, la matrice solution X, telle que calculée par SPBTRS.
!>          En sortie, la matrice solution améliorée X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

FERR

!>          FERR est un tableau REAL, dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur solution
!>          X(j) (la j-ème colonne de la matrice solution X).
!>          Si XTRUE est la vraie solution correspondant à X(j), FERR(j)
!>          est une borne supérieure estimée pour la magnitude du plus grand
!>          élément dans (X(j) - XTRUE) divisée par la magnitude du plus grand
!>          élément dans X(j). L'estimation est aussi fiable que
!>          l'estimation pour RCOND, et est presque toujours une légère
!>          surestimation de l'erreur réelle.
!>

BERR

!>          BERR est un tableau REAL, dimension (NRHS)
!>          L'erreur arrière relative composante par composante de chaque vecteur solution
!>          X(j) (c'est-à-dire le plus petit changement relatif dans
!>          n'importe quel élément de A ou B qui fait de X(j) une solution exacte).
!>

WORK

!>          WORK est un tableau REAL, dimension (3*N)
!>

IWORK

!>          IWORK est un tableau INTEGER, dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0:  sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>

Paramètres internes :

!>  ITMAX est le nombre maximum d'étapes de raffinement itératif.
!>
