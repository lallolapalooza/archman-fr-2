subroutine spbt05 (character uplo, integer n, integer kd, integer nrhs, real, dimension( ldab, * ) ab, integer ldab, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldxact, * ) xact, integer ldxact, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) reslts)

SPBT05

Purpose:

!>
!> SPBT05 teste les bornes d'erreur du raffinement itératif pour la
!> solution calculée d'un système d'équations A*X = B, où A est une
!> matrice bande symétrique.
!>
!> RESLTS(1) = test de la borne d'erreur
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> Une grande valeur est retournée si ce ratio n'est pas inférieur à un.
!>
!> RESLTS(2) = résidu de la routine de raffinement itératif
!>           = le maximum de BERR / ( NZ*EPS + (*) ), où
!>             (*) = NZ*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>             et NZ = nombre max. de non-zéros dans n'importe quelle ligne de A, plus 1
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si la partie triangulaire supérieure ou inférieure de la
!>          matrice symétrique A est stockée.
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

N

!>          N est INTEGER
!>          Le nombre de lignes des matrices X, B et XACT, et l'ordre de la
!>          matrice A.  N >= 0.
!>

KD

!>          KD est INTEGER
!>          Le nombre de super-diagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de colonnes des matrices X, B et XACT.
!>          NRHS >= 0.
!>

AB

!>          AB est tableau REAL, dimension (LDAB,N)
!>          Le triangle supérieur ou inférieur de la matrice bande symétrique A,
!>          stocké dans les KD+1 premières lignes du tableau.  La j-ième colonne
!>          de A est stockée dans la j-ième colonne du tableau AB comme suit:
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) pour max(1,j-kd)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KD+1.
!>

B

!>          B est tableau REAL, dimension (LDB,NRHS)
!>          Les vecteurs de second membre pour le système d'équations
!>          linéaires.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est tableau REAL, dimension (LDX,NRHS)
!>          Les vecteurs solution calculés.  Chaque vecteur est stocké comme une
!>          colonne de la matrice X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

XACT

!>          XACT est tableau REAL, dimension (LDX,NRHS)
!>          Les vecteurs solution exacts.  Chaque vecteur est stocké comme une
!>          colonne de la matrice XACT.
!>

LDXACT

!>          LDXACT est INTEGER
!>          La dimension principale du tableau XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR est tableau REAL, dimension (NRHS)
!>          Les bornes d'erreur avant estimées pour chaque vecteur solution
!>          X.  Si XTRUE est la vraie solution, FERR borne la magnitude
!>          de la plus grande entrée dans (X - XTRUE) divisée par la magnitude
!>          de la plus grande entrée dans X.
!>

BERR

!>          BERR est tableau REAL, dimension (NRHS)
!>          L'erreur arrière relative par composante de chaque vecteur solution
!>          (c'est-à-dire, le plus petit changement relatif dans n'importe quelle entrée de A
!>          ou B qui rend X une solution exacte).
!>

RESLTS

!>          RESLTS est tableau REAL, dimension (2)
!>          Le maximum sur les vecteurs solutions NRHS des ratios:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( NZ*EPS + (*) )
!>
