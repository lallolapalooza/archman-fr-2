subroutine sporfs (character uplo, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldaf, * ) af, integer ldaf, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SPORFS

Objectif :

!>
!> SPORFS améliore la solution calculée d'un système d'équations linéaires
!> lorsque la matrice des coefficients est symétrique définie positive,
!> et fournit des bornes d'erreur et des estimations d'erreur arrière pour la
!> solution.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U':  La partie triangulaire supérieure de A est stockée ;
!>          = 'L':  La partie triangulaire inférieure de A est stockée.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          des matrices B et X.  NRHS >= 0.
!>

A

!>          A est un tableau REAL, de dimension (LDA,N)
!>          La matrice symétrique A.  Si UPLO = 'U', la partie triangulaire
!>          supérieure N-par-N de tête de A contient la partie triangulaire
!>          supérieure de la matrice A, et la partie strictement triangulaire
!>          inférieure de A n'est pas référencée.  Si UPLO = 'L', la partie
!>          triangulaire inférieure N-par-N de tête de A contient la partie
!>          triangulaire inférieure de la matrice A, et la partie strictement
!>          triangulaire supérieure de A n'est pas référencée.
!>

LDA

!>          LDA est INTEGER
!>          La dimension de tête du tableau A.  LDA >= max(1,N).
!>

AF

!>          AF est un tableau REAL, de dimension (LDAF,N)
!>          Le facteur triangulaire U ou L de la factorisation de Cholesky
!>          A = U**T*U ou A = L*L**T, tel que calculé par SPOTRF.
!>

LDAF

!>          LDAF est INTEGER
!>          La dimension de tête du tableau AF.  LDAF >= max(1,N).
!>

B

!>          B est un tableau REAL, de dimension (LDB,NRHS)
!>          La matrice des membres de droite B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension de tête du tableau B.  LDB >= max(1,N).
!>

X

!>          X est un tableau REAL, de dimension (LDX,NRHS)
!>          En entrée, la matrice solution X, telle que calculée par SPOTRS.
!>          En sortie, la matrice solution améliorée X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension de tête du tableau X.  LDX >= max(1,N).
!>

FERR

!>          FERR est un tableau REAL, de dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur solution
!>          X(j) (la j-ième colonne de la matrice solution X).
!>          Si XTRUE est la vraie solution correspondant à X(j), FERR(j)
!>          est une borne supérieure estimée de la magnitude du plus grand
!>          élément de (X(j) - XTRUE) divisé par la magnitude du plus grand
!>          élément de X(j).  L'estimation est aussi fiable que
!>          l'estimation de RCOND, et est presque toujours une légère
!>          majoration de l'erreur réelle.
!>

BERR

!>          BERR est un tableau REAL, de dimension (NRHS)
!>          L'erreur arrière relative composante par composante de chaque
!>          vecteur solution X(j) (c'est-à-dire la plus petite variation relative
!>          dans n'importe quel élément de A ou B qui rend X(j) une solution
!>          exacte).
!>

WORK

!>          WORK est un tableau REAL, de dimension (3*N)
!>

IWORK

!>          IWORK est un tableau INTEGER, de dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0:  succès de l'exécution
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>

Paramètres internes :

!>  ITMAX est le nombre maximum d'étapes de raffinement itératif.
!>
