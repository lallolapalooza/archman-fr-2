subroutine spbt02 (character uplo, integer n, integer kd, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) rwork, real resid)

SPBT02

But:

!>
!> SPBT02 calcule le résidu pour une solution d'un système d'équations
!> symétrique bande A*x = b :
!>    RESID = norm( B - A*X ) / ( norm(A) * norm(X) * EPS)
!> où EPS est la précision de la machine.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si la partie triangulaire supérieure ou inférieure de
!>          la matrice symétrique A est stockée :
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

N

!>          N est INTEGER
!>          Le nombre de lignes et de colonnes de la matrice A.  N >= 0.
!>

KD

!>          KD est INTEGER
!>          Le nombre de super-diagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de seconds membres. NRHS >= 0.
!>

A

!>          A est un tableau de REAL, dimension (LDA,N)
!>          La matrice bande symétrique originale A.  Si UPLO = 'U', la
!>          partie triangulaire supérieure de A est stockée comme une
!>          matrice bande ; si UPLO = 'L', la partie triangulaire
!>          inférieure de A est stockée.  Les colonnes du triangle approprié
!>          sont stockées dans les colonnes de A et les diagonales du
!>          triangle sont stockées dans les lignes de A.  Voir SPBTRF
!>          pour plus de détails.
!>

LDA

!>          LDA est INTEGER.
!>          La dimension principale du tableau A.  LDA >= max(1,KD+1).
!>

X

!>          X est un tableau de REAL, dimension (LDX,NRHS)
!>          Les vecteurs solution calculés pour le système d'équations
!>          linéaires.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.   LDX >= max(1,N).
!>

B

!>          B est un tableau de REAL, dimension (LDB,NRHS)
!>          À l'entrée, les vecteurs seconds membres pour le système
!>          d'équations linéaires.
!>          À la sortie, B est écrasé par la différence B - A*X.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

RWORK

!>          RWORK est un tableau de REAL, dimension (N)
!>

RESID

!>          RESID est REAL
!>          Le maximum sur le nombre de seconds membres de
!>          norm(B - A*X) / ( norm(A) * norm(X) * EPS ).
!>
