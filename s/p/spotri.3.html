subroutine spotri (character uplo, integer n, real, dimension( lda, * ) a, integer lda, integer info)

SPOTRI

Objectif :

!>
!> SPOTRI calcule l'inverse d'une matrice symétrique définie positive réelle A
!> en utilisant la décomposition de Cholesky A = U**T*U ou A = L*L**T
!> calculée par SPOTRF.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U' : le triangle supérieur de A est stocké ;
!>          = 'L' : le triangle inférieur de A est stocké.
!>

N

!>          N est un ENTIER
!>          L'ordre de la matrice A. N >= 0.
!>

A

!>          A est un tableau RÉEL, de dimension (LDA,N)
!>          À l'entrée, le facteur triangulaire U ou L de la décomposition de
!>          Cholesky A = U**T*U ou A = L*L**T, tel que calculé par
!>          SPOTRF.
!>          À la sortie, le triangle supérieur ou inférieur de l'inverse
!>          (symétrique) de A, écrasant le facteur U ou L d'entrée.
!>

LDA

!>          LDA est un ENTIER
!>          La dimension principale du tableau A. LDA >= max(1,N).
!>

INFO

!>          INFO est un ENTIER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument était invalide
!>          > 0 : si INFO = i, l'élément (i,i) du facteur U ou L est
!>                nul, et l'inverse n'a pas pu être calculé.
!>
