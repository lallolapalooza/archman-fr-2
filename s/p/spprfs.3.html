subroutine spprfs (character uplo, integer n, integer nrhs, real, dimension( * ) ap, real, dimension( * ) afp, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SPPRFS

Objectif :

!>
!> SPPRFS améliore la solution calculée pour un système d'équations linéaires
!> lorsque la matrice des coefficients est symétrique définie positive
!> et empaquetée, et fournit des bornes d'erreur et des estimations d'erreur arrière
!> pour la solution.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U': le triangle supérieur de A est stocké ;
!>          = 'L': le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de côtés droits, c'est-à-dire le nombre de colonnes
!>          des matrices B et X.  NRHS >= 0.
!>

AP

!>          AP est un tableau REAL, dimension (N*(N+1)/2)
!>          Le triangle supérieur ou inférieur de la matrice symétrique A, empaqueté
!>          par colonnes dans un tableau linéaire.  La j-ième colonne de A est stockée
!>          dans le tableau AP comme suit :
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L', AP(i + (j-1)*(2n-j)/2) = A(i,j) pour j<=i<=n.
!>

AFP

!>          AFP est un tableau REAL, dimension (N*(N+1)/2)
!>          Le facteur triangulaire U ou L de la factorisation de Cholesky
!>          A = U**T*U ou A = L*L**T, comme calculé par SPPTRF/CPPTRF,
!>          empaqueté par colonnes dans un tableau linéaire, dans le même format que A
!>          (voir AP).
!>

B

!>          B est un tableau REAL, dimension (LDB,NRHS)
!>          La matrice du côté droit B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est un tableau REAL, dimension (LDX,NRHS)
!>          À l'entrée, la matrice solution X, comme calculée par SPPTRS.
!>          À la sortie, la matrice solution améliorée X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

FERR

!>          FERR est un tableau REAL, dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur solution
!>          X(j) (la j-ième colonne de la matrice solution X).
!>          Si XTRUE est la vraie solution correspondant à X(j), FERR(j)
!>          est une borne supérieure estimée pour la magnitude du plus grand
!>          élément dans (X(j) - XTRUE) divisé par la magnitude du plus grand
!>          élément dans X(j).  L'estimation est aussi fiable que
!>          l'estimation pour RCOND, et est presque toujours une légère
!>          surostimation de l'erreur réelle.
!>

BERR

!>          BERR est un tableau REAL, dimension (NRHS)
!>          L'erreur arrière relative composante par composante de chaque vecteur
!>          solution X(j) (c'est-à-dire le plus petit changement relatif dans
!>          n'importe quel élément de A ou B qui fait de X(j) une solution exacte).
!>

WORK

!>          WORK est un tableau REAL, dimension (3*N)
!>

IWORK

!>          IWORK est un tableau INTEGER, dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0:  sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument était illégal
!>

Paramètres internes :

!>  ITMAX est le nombre maximum d'étapes de raffinement itératif.
!>
