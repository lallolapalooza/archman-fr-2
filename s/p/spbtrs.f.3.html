sous-routine spbtrs (caractère uplo, entier n, entier kd, entier nrhs, réel, dimension( ldab, * ) ab, entier ldab, réel, dimension( ldb, * ) b, entier ldb, entier info)

SPBTRS

Objectif:

!>
!> SPBTRS résout un système d'équations linéaires A*X = B avec une matrice bande
!> symétrique définie positive A en utilisant la factorisation de Cholesky
!> A = U**T*U ou A = L*L**T calculée par SPBTRF.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U': Facteur triangulaire supérieur stocké dans AB;
!>          = 'L': Facteur triangulaire inférieur stocké dans AB.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

KD

!>          KD est INTEGER
!>          Le nombre de superdiagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de côtés droits, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!>

AB

!>          AB est un tableau REEL, dimension (LDAB,N)
!>          Le facteur triangulaire U ou L de la factorisation de Cholesky
!>          A = U**T*U ou A = L*L**T de la matrice bande A, stocké dans les
!>          KD+1 premières lignes du tableau. La j-ème colonne de U ou L est
!>          stockée dans la j-ème colonne du tableau AB comme suit :
!>          si UPLO ='U', AB(kd+1+i-j,j) = U(i,j) pour max(1,j-kd)<=i<=j;
!>          si UPLO ='L', AB(1+i-j,j)    = L(i,j) pour j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KD+1.
!>

B

!>          B est un tableau REEL, dimension (LDB,NRHS)
!>          En entrée, la matrice des seconds membres B.
!>          En sortie, la matrice solution X.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

INFO

!>          INFO est INTEGER
!>          = 0:  sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>
