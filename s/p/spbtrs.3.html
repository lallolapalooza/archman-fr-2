subroutine spbtrs (character uplo, integer n, integer kd, integer nrhs, real, dimension( ldab, * ) ab, integer ldab, real, dimension( ldb, * ) b, integer ldb, integer info)

SPBTRS

But :

!>
!> SPBTRS résout un système d'équations linéaires A*X = B avec une matrice bande symétrique
!> définie positive A en utilisant la factorisation de Cholesky
!> A = U**T*U ou A = L*L**T calculée par SPBTRF.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U' : Facteur triangulaire supérieur stocké dans AB ;
!>          = 'L' : Facteur triangulaire inférieur stocké dans AB.
!>

N

!>          N est un ENTIER
!>          L'ordre de la matrice A.  N >= 0.
!>

KD

!>          KD est un ENTIER
!>          Le nombre de super-diagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS est un ENTIER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!>

AB

!>          AB est un tableau de REELS, de dimension (LDAB,N)
!>          Le facteur triangulaire U ou L de la factorisation de Cholesky
!>          A = U**T*U ou A = L*L**T de la matrice bande A, stocké dans les
!>          KD+1 premières lignes du tableau. La j-ème colonne de U ou L est
!>          stockée dans la j-ème colonne du tableau AB comme suit :
!>          si UPLO ='U', AB(kd+1+i-j,j) = U(i,j) pour max(1,j-kd)<=i<=j ;
!>          si UPLO ='L', AB(1+i-j,j)    = L(i,j) pour j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB est un ENTIER
!>          La dimension principale du tableau AB.  LDAB >= KD+1.
!>

B

!>          B est un tableau de REELS, de dimension (LDB,NRHS)
!>          À l'entrée, la matrice B du second membre.
!>          À la sortie, la matrice solution X.
!>

LDB

!>          LDB est un ENTIER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

INFO

!>          INFO est un ENTIER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>
