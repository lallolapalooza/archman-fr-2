subroutine spst01 (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldafac, * ) afac, integer ldafac, real, dimension( ldperm, * ) perm, integer ldperm, integer, dimension( * ) piv, real, dimension( * ) rwork, real resid, integer rank)

SPST01

Purpose:

!>
!> SPST01 reconstruit une matrice symétrique semi-définie positive A
!> à partir de ses facteurs L ou U et de la matrice de permutation P, et calcule
!> le résidu
!>    norm( P*L*L'*P' - A ) / ( N * norm(A) * EPS ) ou
!>    norm( P*U'*U*P' - A ) / ( N * norm(A) * EPS ),
!> où EPS est l'epsilon machine.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si la partie triangulaire supérieure ou inférieure de la
!>          matrice symétrique A est stockée :
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

N

!>          N est INTEGER
!>          Le nombre de lignes et de colonnes de la matrice A.  N >= 0.
!>

A

!>          A est un tableau REAL, dimension (LDA,N)
!>          La matrice symétrique originale A.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC est un tableau REAL, dimension (LDAFAC,N)
!>          Le facteur L ou U de la factorisation L*L' ou U'*U
!>          de A.
!>

LDAFAC

!>          LDAFAC est INTEGER
!>          La dimension principale du tableau AFAC.  LDAFAC >= max(1,N).
!>

PERM

!>          PERM est un tableau REAL, dimension (LDPERM,N)
!>          Être écrasé par la matrice reconstruite, puis par la
!>          différence P*L*L'*P' - A (ou P*U'*U*P' - A)
!>

LDPERM

!>          LDPERM est INTEGER
!>          La dimension principale du tableau PERM.
!>          LDAPERM >= max(1,N).
!>

PIV

!>          PIV est un tableau INTEGER, dimension (N)
!>          PIV est tel que les entrées non nulles sont
!>          P( PIV( K ), K ) = 1.
!>

RWORK

!>          RWORK est un tableau REAL, dimension (N)
!>

RESID

!>          RESID est REAL
!>          Si UPLO = 'L', norm(L*L' - A) / ( N * norm(A) * EPS )
!>          Si UPLO = 'U', norm(U'*U - A) / ( N * norm(A) * EPS )
!>

RANK

!>          RANK est INTEGER
!>          nombre de valeurs singulières non nulles de A.
!>
