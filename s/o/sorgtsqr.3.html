subroutine sorgtsqr (integer m, integer n, integer mb, integer nb, real, dimension( lda, * ) a, integer lda, real, dimension( ldt, * ) t, integer ldt, real, dimension( * ) work, integer lwork, integer info)

SORGTSQR

But :

!>
!> SORGTSQR génère une matrice réelle M-par-N, Q_out, avec des colonnes
!> orthonormées, qui sont les N premières colonnes d'un produit de matrices
!> réelles orthogonales d'ordre M retournées par SLATSQR
!>
!>      Q_out = premières_N_colonnes( Q(1)_in * Q(2)_in * ... * Q(k)_in ).
!>
!> Voir la documentation de SLATSQR.
!>

Paramètres

M 
!>          M est un ENTIER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est un ENTIER
!>          Le nombre de colonnes de la matrice A. M >= N >= 0.
!>

MB

!>          MB est un ENTIER
!>          La taille de bloc des lignes utilisée par SLATSQR pour retourner
!>          les tableaux A et T. MB > N.
!>          (Notez que si MB > M, alors M est utilisé à la place de MB
!>          comme taille de bloc des lignes).
!>

NB

!>          NB est un ENTIER
!>          La taille de bloc des colonnes utilisée par SLATSQR pour retourner
!>          les tableaux A et T. NB >= 1.
!>          (Notez que si NB > N, alors N est utilisé à la place de NB
!>          comme taille de bloc des colonnes).
!>

A

!>          A est un tableau REEL, de dimension (LDA,N)
!>
!>          En entrée :
!>
!>             Les éléments sur et au-dessus de la diagonale ne sont pas accédés.
!>             Les éléments en dessous de la diagonale représentent la matrice
!>             trapézoïdale inférieure unitaire bloquée V calculée par SLATSQR
!>             qui définit les matrices d'entrée Q_in(k) (les uns sur la
!>             diagonale ne sont pas stockés) (même format que la sortie A
!>             en dessous de la diagonale dans SLATSQR).
!>
!>          En sortie :
!>
!>             Le tableau A contient une matrice orthonormée M-par-N, Q_out,
!>             c'est-à-dire que les colonnes de A sont des vecteurs unitaires
!>             orthogonaux.
!>

LDA

!>          LDA est un ENTIER
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!>

T

!>          T est un tableau REEL,
!>          de dimension (LDT, N * NIRB)
!>          où NIRB = Nombre_de_blocs_de_lignes_d_entrée
!>                     = MAX( 1, CEIL((M-N)/(MB-N)) )
!>          Soit NICB = Nombre_de_blocs_de_colonnes_d_entrée
!>                   = CEIL(N/NB)
!>
!>          Les réflecteurs de blocs triangulaires supérieurs utilisés pour définir les
!>          matrices d'entrée Q_in(k), k=(1:NIRB*NICB). Les réflecteurs de blocs
!>          sont stockés sous forme compacte en NIRB séquences de réflecteurs de blocs.
!>          Chacune des NIRB séquences de réflecteurs de blocs est stockée dans un
!>          bloc de colonnes plus grand de T, de taille NB-par-N, et consiste en
!>          NICB plus petits blocs de colonnes triangulaires supérieurs de taille NB-par-NB.
!>          (même format que la sortie T dans SLATSQR).
!>

LDT

!>          LDT est un ENTIER
!>          La dimension principale du tableau T.
!>          LDT >= max(1,min(NB1,N)).
!>

WORK

!>          (espace de travail) tableau REEL, de dimension (MAX(2,LWORK))
!>          En sortie, si INFO = 0, WORK(1) retourne la valeur optimale de LWORK.
!>

LWORK

!>          LWORK est un ENTIER
!>          La dimension du tableau WORK.  LWORK >= (M+NB)*N.
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée.
!>          La routine calcule seulement la taille optimale du tableau WORK,
!>          retourne cette valeur comme première entrée du tableau WORK, et aucun
!>          message d'erreur concernant LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est un ENTIER
!>          = 0:  succès
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>
