subroutine sormr3 (character side, character trans, integer m, integer n, integer k, integer l, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer info)

SORMR3 multiplie une matrice générale par la matrice orthogonale issue d'une factorisation RZ déterminée par stzrzf (algorithme non bloqué).

Objectif :

!>
!> SORMR3 écrase la matrice réelle générale m par n, C, avec
!>
!>       Q * C  si SIDE = 'L' et TRANS = 'N', ou
!>
!>       Q**T* C  si SIDE = 'L' et TRANS = 'C', ou
!>
!>       C * Q  si SIDE = 'R' et TRANS = 'N', ou
!>
!>       C * Q**T si SIDE = 'R' et TRANS = 'C',
!>
!> où Q est une matrice orthogonale réelle définie comme le produit de k
!> réflexions élémentaires
!>
!>       Q = H(1) H(2) . . . H(k)
!>
!> telles que retournées par STZRZF. Q est d'ordre m si SIDE = 'L' et d'ordre n
!> si SIDE = 'R'.
!>

Paramètres

SIDE 
!>          SIDE est CHARACTER*1
!>          = 'L': appliquer Q ou Q**T depuis la gauche
!>          = 'R': appliquer Q ou Q**T depuis la droite
!>

TRANS

!>          TRANS est CHARACTER*1
!>          = 'N': appliquer Q  (Pas de transposition)
!>          = 'T': appliquer Q**T (Transposée)
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice C. M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice C. N >= 0.
!>

K

!>          K est INTEGER
!>          Le nombre de réflexions élémentaires dont le produit définit
!>          la matrice Q.
!>          Si SIDE = 'L', M >= K >= 0 ;
!>          si SIDE = 'R', N >= K >= 0.
!>

L

!>          L est INTEGER
!>          Le nombre de colonnes de la matrice A contenant
!>          la partie significative des réflexions de Householder.
!>          Si SIDE = 'L', M >= L >= 0, si SIDE = 'R', N >= L >= 0.
!>

A

!>          A est un tableau REAL, de dimension
!>                               (LDA,M) si SIDE = 'L',
!>                               (LDA,N) si SIDE = 'R'
!>          La i-ème ligne doit contenir le vecteur qui définit la
!>          réflexion élémentaire H(i), pour i = 1,2,...,k, telle que retournée par
!>          STZRZF dans les k dernières lignes de son argument de tableau A.
!>          A est modifié par la routine mais restauré à la sortie.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,K).
!>

TAU

!>          TAU est un tableau REAL, de dimension (K)
!>          TAU(i) doit contenir le facteur scalaire de la réflexion
!>          élémentaire H(i), tel que retourné par STZRZF.
!>

C

!>          C est un tableau REAL, de dimension (LDC,N)
!>          À l'entrée, la matrice m par n C.
!>          À la sortie, C est écrasé par Q*C ou Q**T*C ou C*Q**T ou C*Q.
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale du tableau C. LDC >= max(1,M).
!>

WORK

!>          WORK est un tableau REAL, de dimension
!>                                   (N) si SIDE = 'L',
!>                                   (M) si SIDE = 'R'
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>
