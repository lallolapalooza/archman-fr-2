sous-programme sormql (caractère side, caractère trans, entier m, entier n, entier k, réel, dimension( lda, * ) a, entier lda, réel, dimension( * ) tau, réel, dimension( ldc, * ) c, entier ldc, réel, dimension( * ) work, entier lwork, entier info)

SORMQL

But :

!>
!> SORMQL écrase la matrice générale réelle M-par-N C avec
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'T':      Q**T * C       C * Q**T
!>
!> où Q est une matrice orthogonale réelle définie comme le produit de k
!> réflecteurs élémentaires
!>
!>       Q = H(k) . . . H(2) H(1)
!>
!> tels que renvoyés par SGEQLF. Q est d'ordre M si SIDE = 'L' et d'ordre N
!> si SIDE = 'R'.
!>

Paramètres

SIDE 
!>          SIDE est CHARACTER*1
!>          = 'L' : appliquer Q ou Q**T à Gauche ;
!>          = 'R' : appliquer Q ou Q**T à Droite.
!>

TRANS

!>          TRANS est CHARACTER*1
!>          = 'N' : Pas de transposer, appliquer Q ;
!>          = 'T' : Transposer, appliquer Q**T.
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice C. M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice C. N >= 0.
!>

K

!>          K est INTEGER
!>          Le nombre de réflecteurs élémentaires dont le produit définit
!>          la matrice Q.
!>          Si SIDE = 'L', M >= K >= 0 ;
!>          si SIDE = 'R', N >= K >= 0.
!>

A

!>          A est un tableau REEL, dimension (LDA,K)
!>          La i-ème colonne doit contenir le vecteur qui définit le
!>          réflecteur élémentaire H(i), pour i = 1,2,...,k, tel que renvoyé par
!>          SGEQLF dans les k dernières colonnes de son tableau argument A.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.
!>          Si SIDE = 'L', LDA >= max(1,M) ;
!>          si SIDE = 'R', LDA >= max(1,N).
!>

TAU

!>          TAU est un tableau REEL, dimension (K)
!>          TAU(i) doit contenir le facteur scalaire du réflecteur
!>          élémentaire H(i), tel que renvoyé par SGEQLF.
!>

C

!>          C est un tableau REEL, dimension (LDC,N)
!>          En entrée, la matrice M-par-N C.
!>          En sortie, C est écrasée par Q*C ou Q**T*C ou C*Q**T ou C*Q.
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale du tableau C. LDC >= max(1,M).
!>

WORK

!>          WORK est un tableau REEL, dimension (MAX(1,LWORK))
!>          En sortie, si INFO = 0, WORK(1) renvoie LWORK optimal.
!>

LWORK

!>          LWORK est INTEGER
!>          La dimension du tableau WORK.
!>          Si SIDE = 'L', LWORK >= max(1,N) ;
!>          si SIDE = 'R', LWORK >= max(1,M).
!>          Pour de bonnes performances, LWORK devrait généralement être plus grand.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>
