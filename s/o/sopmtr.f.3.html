subroutine sopmtr (side, uplo, trans, m, n, ap, tau, c, ldc, work, info)

Documentation de la fonction/sous-routine

sous-routine sopmtr (caractère side, caractère uplo, caractère trans, entier m, entier n, réel, dimension( * ) ap, réel, dimension( * ) tau, réel, dimension( ldc, * ) c, entier ldc, réel, dimension( * ) work, entier info)

SOPMTR

Objectif :

!>
!> SOPMTR écrase la matrice générale réelle M-par-N C avec
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'T':      Q**T * C       C * Q**T
!>
!> où Q est une matrice orthogonale réelle d'ordre nq, avec nq = m si
!> SIDE = 'L' et nq = n si SIDE = 'R'. Q est définie comme le produit de
!> nq-1 réflecteurs élémentaires, tels que retournés par SSPTRD en utilisant
!> le stockage compact :
!>
!> si UPLO = 'U', Q = H(nq-1) . . . H(2) H(1);
!>
!> si UPLO = 'L', Q = H(1) H(2) . . . H(nq-1).
!>

Paramètres

SIDE 
!>          SIDE est CHARACTER*1
!>          = 'L' : appliquer Q ou Q**T à gauche ;
!>          = 'R' : appliquer Q ou Q**T à droite.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U' : stockage triangulaire supérieur compact utilisé dans
!>                 l'appel précédent à SSPTRD ;
!>          = 'L' : stockage triangulaire inférieur compact utilisé dans
!>                 l'appel précédent à SSPTRD.
!>

TRANS

!>          TRANS est CHARACTER*1
!>          = 'N' : Pas de transposée, appliquer Q ;
!>          = 'T' : Transposée, appliquer Q**T.
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice C. M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice C. N >= 0.
!>

AP

!>          AP est un tableau REAL, dimension
!>                               (M*(M+1)/2) si SIDE = 'L'
!>                               (N*(N+1)/2) si SIDE = 'R'
!>          Les vecteurs qui définissent les réflecteurs élémentaires, tels que
!>          retournés par SSPTRD. AP est modifié par la routine mais
!>          restauré à la sortie.
!>

TAU

!>          TAU est un tableau REAL, dimension (M-1) si SIDE = 'L'
!>                                     ou (N-1) si SIDE = 'R'
!>          TAU(i) doit contenir le facteur scalaire du réflecteur élémentaire
!>          H(i), tel que retourné par SSPTRD.
!>

C

!>          C est un tableau REAL, dimension (LDC,N)
!>          À l'entrée, la matrice M-par-N C.
!>          À la sortie, C est écrasée par Q*C ou Q**T*C ou C*Q**T ou C*Q.
!>

LDC

!>          LDC est INTEGER
!>          La dimension de tête du tableau C. LDC >= max(1,M).
!>

WORK

!>          WORK est un tableau REAL, dimension
!>                                   (N) si SIDE = 'L'
!>                                   (M) si SIDE = 'R'
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>
