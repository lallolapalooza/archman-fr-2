subroutine sorghr (n, ilo, ihi, a, lda, tau, work, lwork, info)
SORGHR

Documentation de la fonction/sous-programme

sous-programme sorghr (integer n, integer ilo, integer ihi, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( * ) work, integer lwork, integer info)

SORGHR

Objectif:

!>
!> SORGHR génère une matrice orthogonale réelle Q définie comme le produit
!> de IHI-ILO réflecteurs élémentaires d'ordre N, tels que retournés par
!> SGEHRD:
!>
!> Q = H(ilo) H(ilo+1) . . . H(ihi-1).
!>

Paramètres

N 
!>          N est INTEGER
!>          L'ordre de la matrice Q. N >= 0.
!>

ILO

!>          ILO est INTEGER
!>

IHI

!>          IHI est INTEGER
!>
!>          ILO et IHI doivent avoir les mêmes valeurs que lors de l'appel
!>          précédent à SGEHRD. Q est égale à la matrice unité sauf dans la
!>          sous-matrice Q(ilo+1:ihi,ilo+1:ihi).
!>          1 <= ILO <= IHI <= N, si N > 0; ILO=1 et IHI=0, si N=0.
!>

A

!>          A est un tableau REAL, dimension (LDA,N)
!>          À l'entrée, les vecteurs qui définissent les réflecteurs élémentaires,
!>          tels que retournés par SGEHRD.
!>          À la sortie, la matrice orthogonale N-par-N Q.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,N).
!>

TAU

!>          TAU est un tableau REAL, dimension (N-1)
!>          TAU(i) doit contenir le facteur scalaire du réflecteur élémentaire
!>          H(i), tel que retourné par SGEHRD.
!>

WORK

!>          WORK est un tableau REAL, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) contient la taille optimale de LWORK.
!>

LWORK

!>          LWORK est INTEGER
!>          La dimension du tableau WORK. LWORK >= IHI-ILO.
!>          Pour une performance optimale LWORK >= (IHI-ILO)*NB, où NB est
!>          la taille de bloc optimale.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la routine
!>          calcule seulement la taille optimale du tableau WORK, retourne
!>          cette valeur comme première entrée du tableau WORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>
