subroutine sorgrq (integer m, integer n, integer k, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( * ) work, integer lwork, integer info)

SORGRQ

Objectif :

!>
!> SORGRQ génère une matrice réelle M par N, Q, avec des lignes
!> orthonormées, qui est définie comme les M dernières lignes d'un produit
!> de K réflecteurs élémentaires d'ordre N
!>
!>       Q  =  H(1) H(2) . . . H(k)
!>
!> tel que retourné par SGERQF.
!>

Paramètres

M 
!>          M est un entier (INTEGER)
!>          Le nombre de lignes de la matrice Q. M >= 0.
!>

N

!>          N est un entier (INTEGER)
!>          Le nombre de colonnes de la matrice Q. N >= M.
!>

K

!>          K est un entier (INTEGER)
!>          Le nombre de réflecteurs élémentaires dont le produit définit
!>          la matrice Q. M >= K >= 0.
!>

A

!>          A est un tableau de réels (REAL), de dimension (LDA,N)
!>          À l'entrée, la (m-k+i)-ème ligne doit contenir le vecteur qui
!>          définit le réflecteur élémentaire H(i), pour i = 1,2,...,k, tel
!>          que retourné par SGERQF dans les k dernières lignes de son
!>          argument tableau A.
!>          À la sortie, la matrice Q de M par N.
!>

LDA

!>          LDA est un entier (INTEGER)
!>          La première dimension du tableau A. LDA >= max(1,M).
!>

TAU

!>          TAU est un tableau de réels (REAL), de dimension (K)
!>          TAU(i) doit contenir le facteur scalaire du réflecteur
!>          élémentaire H(i), tel que retourné par SGERQF.
!>

WORK

!>          WORK est un tableau de réels (REAL), de dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) contient la taille optimale de LWORK.
!>

LWORK

!>          LWORK est un entier (INTEGER)
!>          La dimension du tableau WORK. LWORK >= max(1,M).
!>          Pour une performance optimale, LWORK >= M*NB, où NB est la
!>          taille de bloc optimale.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ;
!>          la routine calcule seulement la taille optimale du tableau WORK,
!>          retourne cette valeur comme première entrée du tableau WORK, et
!>          aucun message d'erreur lié à LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est un entier (INTEGER)
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument a une valeur illégale
!>
