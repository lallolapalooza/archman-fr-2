subroutine stpqrt (integer m, integer n, integer l, integer nb, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldt, * ) t, integer ldt, real, dimension( * ) work, integer info)

STPQRT

Objectif :

!>
!> STPQRT calcule une factorisation QR bloquée d'une matrice réelle
!>  C, composée d'un bloc triangulaire A et d'un bloc pentagonal B,
!>  en utilisant la représentation WY compacte pour Q.
!>

Paramètres

M 
!>          M est un ENTIER
!>          Le nombre de lignes de la matrice B.
!>          M >= 0.
!>

N

!>          N est un ENTIER
!>          Le nombre de colonnes de la matrice B, et l'ordre de la
!>          matrice triangulaire A.
!>          N >= 0.
!>

L

!>          L est un ENTIER
!>          Le nombre de lignes de la partie trapézoïdale supérieure de B.
!>          MIN(M,N) >= L >= 0. Voir Détails supplémentaires.
!>

NB

!>          NB est un ENTIER
!>          La taille du bloc à utiliser dans la QR bloquée.  N >= NB >= 1.
!>

A

!>          A est un tableau REEL, dimension (LDA,N)
!>          En entrée, la matrice triangulaire supérieure N-par-N A.
!>          En sortie, les éléments sur et au-dessus de la diagonale du tableau
!>          contiennent la matrice triangulaire supérieure R.
!>

LDA

!>          LDA est un ENTIER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

B

!>          B est un tableau REEL, dimension (LDB,N)
!>          En entrée, la matrice pentagonale M-par-N B. Les M-L premières lignes
!>          sont rectangulaires, et les L dernières lignes sont trapézoïdales supérieures.
!>          En sortie, B contient la matrice pentagonale V. Voir Détails supplémentaires.
!>

LDB

!>          LDB est un ENTIER
!>          La dimension principale du tableau B.  LDB >= max(1,M).
!>

T

!>          T est un tableau REEL, dimension (LDT,N)
!>          Les réflecteurs de bloc triangulaires supérieurs stockés sous forme compacte
!>          en tant que séquence de blocs triangulaires supérieurs. Voir Détails supplémentaires.
!>

LDT

!>          LDT est un ENTIER
!>          La dimension principale du tableau T.  LDT >= NB.
!>

WORK

!>          WORK est un tableau REEL, dimension (NB*N)
!>

INFO

!>          INFO est un ENTIER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>

Détails supplémentaires :

!>
!>  La matrice d'entrée C est une matrice (N+M)-par-N
!>
!>               C = [ A ]
!>                   [ B ]
!>
!>  où A est une matrice N-par-N triangulaire supérieure, et B est une matrice
!>  pentagonale M-par-N composée d'une matrice rectangulaire (M-L)-par-N, B1,
!>  au-dessus d'une matrice trapézoïdale supérieure L-par-N, B2 :
!>
!>               B = [ B1 ]  <- rectangulaire (M-L)-par-N
!>                   [ B2 ]  <- trapézoïdale supérieure     L-par-N.
!>
!>  La matrice trapézoïdale supérieure B2 est constituée des L premières lignes
!>  d'une matrice triangulaire supérieure N-par-N, où 0 <= L <= MIN(M,N). Si L=0,
!>  B est rectangulaire M-par-N ; si M=L=N, B est triangulaire supérieure.
!>
!>  La matrice W stocke les réflecteurs élémentaires H(i) dans la i-ème colonne
!>  en dessous de la diagonale (de A) dans la matrice d'entrée C de taille (N+M)-par-N
!>
!>               C = [ A ]  <- triangulaire supérieure N-par-N
!>                   [ B ]  <- pentagonale M-par-N
!>
!>  de sorte que W peut être représentée comme
!>
!>               W = [ I ]  <- identité, N-par-N
!>                   [ V ]  <- M-par-N, même forme que B.
!>
!>  Ainsi, toutes les informations nécessaires pour W sont contenues en sortie dans B,
!>  que nous appelons V ci-dessus. Notez que V a la même forme que B ; c'est-à-dire,
!>
!>               V = [ V1 ] <- rectangulaire (M-L)-par-N
!>                   [ V2 ] <- trapézoïdale supérieure     L-par-N.
!>
!>  Les colonnes de V représentent les vecteurs qui définissent les H(i).
!>
!>  Le nombre de blocs est B = ceiling(N/NB), où chaque
!>  bloc est d'ordre NB sauf le dernier bloc, qui est d'ordre
!>  IB = N - (B-1)*NB. Pour chacun des B blocs, un facteur de réflecteur
!>  de bloc triangulaire supérieur est calculé : T1, T2, ..., TB. Les T de taille
!>  NB-par-NB (et IB-par-IB pour le dernier bloc) sont stockés dans la matrice
!>  T de taille NB-par-N comme suit :
!>
!>               T = [T1 T2 ... TB].
!>
