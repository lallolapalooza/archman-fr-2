sous-routine stbt02 (character uplo, character trans, character diag, integer n, integer kd, integer nrhs, real, dimension( ldab, * ) ab, integer ldab, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, real resid)

STBT02

But :

!>
!> STBT02 calcule le résidu pour la solution calculée du système d'équations linéaires
!> triangulaire op(A)*X = B, où A est une matrice triangulaire bande.
!> Le ratio de test est le maximum sur
!>    norm(b - op(A)*x) / ( ||op(A)||_1 * norm(x) * EPS ),
!> où op(A) = A ou A**T, b est la colonne de B, x est le vecteur solution,
!> et EPS est l'epsilon machine.
!> La norme utilisée est la 1-norme.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si la matrice A est triangulaire supérieure ou inférieure.
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

TRANS

!>          TRANS est CHARACTER*1
!>          Spécifie l'opération appliquée à A.
!>          = 'N':  A    * X = B  (Pas de transposée)
!>          = 'T':  A**T * X = B  (Transposée)
!>          = 'C':  A**H * X = B  (Transposée conjuguée = Transposée)
!>

DIAG

!>          DIAG est CHARACTER*1
!>          Spécifie si la matrice A est unitaire triangulaire ou non.
!>          = 'N':  Triangulaire non unitaire
!>          = 'U':  Triangulaire unitaire
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

KD

!>          KD est INTEGER
!>          Le nombre de super-diagonales ou de sous-diagonales de la
!>          matrice bande triangulaire A.  KD >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de colonnes
!>          des matrices X et B.  NRHS >= 0.
!>

AB

!>          AB est un tableau REAL, dimension (LDAB,N)
!>          La matrice bande triangulaire supérieure ou inférieure A, stockée dans les
!>          kd+1 premières lignes du tableau. La j-ème colonne de A est stockée
!>          dans la j-ème colonne du tableau AB comme suit :
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) pour max(1,j-kd)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KD+1.
!>

X

!>          X est un tableau REAL, dimension (LDX,NRHS)
!>          Les vecteurs solutions calculés pour le système d'équations linéaires.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

B

!>          B est un tableau REAL, dimension (LDB,NRHS)
!>          Les vecteurs seconds membres pour le système d'équations linéaires.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

WORK

!>          WORK est un tableau REAL, dimension (N)
!>

RESID

!>          RESID est REAL
!>          Le maximum sur le nombre de seconds membres de
!>          norm(op(A)*x - b) / ( norm(op(A)) * norm(x) * EPS ).
!>
