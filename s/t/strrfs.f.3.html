sous-routine strrfs (character uplo, character trans, character diag, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

STRRFS

Objectif :

!>
!> STRRFS fournit des bornes d'erreur et des estimations d'erreur arrière pour la
!> solution d'un système d'équations linéaires avec une matrice de coefficient triangulaire.
!>
!> La matrice solution X doit être calculée par STRTRS ou par un autre
!> moyen avant d'appeler cette routine. STRRFS ne fait pas de raffinement itératif
!> car cela ne peut pas améliorer l'erreur arrière.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U':  A est triangulaire supérieure ;
!>          = 'L':  A est triangulaire inférieure.
!>

TRANS

!>          TRANS est CHARACTER*1
!>          Spécifie la forme du système d'équations :
!>          = 'N':  A * X = B  (Pas de transposée)
!>          = 'T':  A**T * X = B  (Transposée)
!>          = 'C':  A**H * X = B  (Conjuguée transposée = Transposée)
!>

DIAG

!>          DIAG est CHARACTER*1
!>          = 'N':  A est triangulaire non unitaire ;
!>          = 'U':  A est triangulaire unitaire.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A. N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          des matrices B et X. NRHS >= 0.
!>

A

!>          A est un tableau de type REAL, de dimension (LDA,N)
!>          La matrice triangulaire A. Si UPLO = 'U', la partie triangulaire
!>          supérieure de A de taille N par N dans le tableau A contient
!>          la matrice triangulaire supérieure, et la partie strictement
!>          triangulaire inférieure de A n'est pas référencée. Si UPLO = 'L',
!>          la partie triangulaire inférieure de A de taille N par N dans le
!>          tableau A contient la matrice triangulaire inférieure, et la partie
!>          strictement triangulaire supérieure de A n'est pas référencée.
!>          Si DIAG = 'U', les éléments diagonaux de A ne sont pas non plus
!>          référencés et sont supposés être 1.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,N).
!>

B

!>          B est un tableau de type REAL, de dimension (LDB,NRHS)
!>          La matrice des membres de droite B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,N).
!>

X

!>          X est un tableau de type REAL, de dimension (LDX,NRHS)
!>          La matrice solution X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X. LDX >= max(1,N).
!>

FERR

!>          FERR est un tableau de type REAL, de dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur solution
!>          X(j) (la j-ième colonne de la matrice solution X).
!>          Si XTRUE est la vraie solution correspondant à X(j), FERR(j)
!>          est une borne supérieure estimée de la magnitude du plus grand
!>          élément de (X(j) - XTRUE) divisée par la magnitude du plus grand
!>          élément de X(j). L'estimation est aussi fiable que
!>          l'estimation de RCOND, et est presque toujours une légère
!>          surestimation de l'erreur réelle.
!>

BERR

!>          BERR est un tableau de type REAL, de dimension (NRHS)
!>          L'erreur arrière relative composante par composante de chaque vecteur solution
!>          X(j) (c'est-à-dire le plus petit changement relatif dans
!>          n'importe quel élément de A ou B qui rend X(j) une solution exacte).
!>

WORK

!>          WORK est un tableau de type REAL, de dimension (3*N)
!>

IWORK

!>          IWORK est un tableau de type INTEGER, de dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>
