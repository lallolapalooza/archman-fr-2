sous-routine stbcon (caractère norm, caractère uplo, caractère diag, entier n, entier kd, réel, dimension( ldab, * ) ab, entier ldab, réel rcond, réel, dimension( * ) work, entier, dimension( * ) iwork, entier info)

STBCON

Objectif:

!>
!> STBCON estime l'inverse du nombre de condition d'une matrice bande
!> triangulaire A, soit en norme 1, soit en norme infini.
!>
!> La norme de A est calculée et une estimation de
!> norm(inv(A)) est obtenue, puis l'inverse du nombre de condition est
!> calculé comme suit :
!>    RCOND = 1 / ( norm(A) * norm(inv(A)) ).
!>

Paramètres

NORM 
!>          NORM est CHARACTER*1
!>          Spécifie si le nombre de condition en norme 1 ou en norme
!>          infini est requis :
!>          = '1' ou 'O' :  Norme 1 ;
!>          = 'I' :         Norme infini.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U' :  A est triangulaire supérieure ;
!>          = 'L' :  A est triangulaire inférieure.
!>

DIAG

!>          DIAG est CHARACTER*1
!>          = 'N' :  A est triangulaire non-unitaire ;
!>          = 'U' :  A est triangulaire unitaire.
!>

N

!>          N est INTEGER
!>          Ordre de la matrice A.  N >= 0.
!>

KD

!>          KD est INTEGER
!>          Le nombre de super-diagonales ou sous-diagonales de la
!>          matrice bande triangulaire A.  KD >= 0.
!>

AB

!>          AB est un tableau REAL de dimension (LDAB,N)
!>          La matrice bande triangulaire supérieure ou inférieure A, stockée
!>          dans les kd+1 premières lignes du tableau. La j-ième colonne de A
!>          est stockée dans la j-ième colonne du tableau AB comme suit :
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) pour max(1,j-kd)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+kd).
!>          Si DIAG = 'U', les éléments diagonaux de A ne sont pas référencés
!>          et sont supposés être 1.
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KD+1.
!>

RCOND

!>          RCOND est REAL
!>          L'inverse du nombre de condition de la matrice A,
!>          calculé comme RCOND = 1/(norm(A) * norm(inv(A))).
!>

WORK

!>          WORK est un tableau REAL de dimension (3*N)
!>

IWORK

!>          IWORK est un tableau INTEGER de dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!>
