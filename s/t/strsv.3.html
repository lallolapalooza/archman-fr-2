subroutine strsv (character uplo, character trans, character diag, integer n, real, dimension(lda,*) a, integer lda, real, dimension(*) x, integer incx)

STRSV

But:

!>
!> STRSV résout l'un des systèmes d'équations
!>
!>    A*x = b,   ou   A**T*x = b,
!>
!> où b et x sont des vecteurs de n éléments et A est une matrice
!> triangulaire unitaire, ou non unitaire, supérieure ou inférieure
!> de taille n par n.
!>
!> Aucun test de singularité ou de quasi-singularité n'est inclus dans
!> cette routine. De tels tests doivent être effectués avant d'appeler
!> cette routine.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>           À l'entrée, UPLO spécifie si la matrice est une matrice
!>           triangulaire supérieure ou inférieure comme suit :
!>
!>              UPLO = 'U' ou 'u'   A est une matrice triangulaire supérieure.
!>
!>              UPLO = 'L' ou 'l'   A est une matrice triangulaire inférieure.
!>

TRANS

!>          TRANS est CHARACTER*1
!>           À l'entrée, TRANS spécifie les équations à résoudre comme
!>           suit :
!>
!>              TRANS = 'N' ou 'n'   A*x = b.
!>
!>              TRANS = 'T' ou 't'   A**T*x = b.
!>
!>              TRANS = 'C' ou 'c'   A**T*x = b.
!>

DIAG

!>          DIAG est CHARACTER*1
!>           À l'entrée, DIAG spécifie si A est unitaire ou non
!>           triangulaire comme suit :
!>
!>              DIAG = 'U' ou 'u'   A est supposée être unitaire triangulaire.
!>
!>              DIAG = 'N' ou 'n'   A n'est pas supposée être unitaire
!>                                  triangulaire.
!>

N

!>          N est INTEGER
!>           À l'entrée, N spécifie l'ordre de la matrice A.
!>           N doit être supérieur ou égal à zéro.
!>

A

!>          A est un tableau de type REAL, de dimension ( LDA, N )
!>           Avant l'entrée avec UPLO = 'U' ou 'u', la partie supérieure
!>           triangulaire de taille n par n du tableau A doit contenir
!>           la matrice triangulaire supérieure et la partie strictement
!>           inférieure triangulaire de A n'est pas référencée.
!>           Avant l'entrée avec UPLO = 'L' ou 'l', la partie inférieure
!>           triangulaire de taille n par n du tableau A doit contenir
!>           la matrice triangulaire inférieure et la partie strictement
!>           supérieure triangulaire de A n'est pas référencée.
!>           Notez que lorsque DIAG = 'U' ou 'u', les éléments diagonaux
!>           de A ne sont pas non plus référencés, mais sont supposés être
!>           unitaires.
!>

LDA

!>          LDA est INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle
!>           que déclarée dans le programme appelant (ou sous-programme).
!>           LDA doit être supérieure ou égale à max( 1, n ).
!>

X

!>          X est un tableau de type REAL, de dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           Avant l'entrée, le tableau X incrémenté doit contenir le
!>           vecteur b du second membre à n éléments. À la sortie, X est
!>           écrasé par le vecteur solution x.
!>

INCX

!>          INCX est INTEGER
!>           À l'entrée, INCX spécifie l'incrément pour les éléments de
!>           X. INCX ne doit pas être zéro.
!>
