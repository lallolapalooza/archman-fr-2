sous-routine strsyl3 (caractère trana, caractère tranb, entier isgn, entier m, entier n, réel, dimension( lda, * ) a, entier lda, réel, dimension( ldb, * ) b, entier ldb, réel, dimension( ldc, * ) c, entier ldc, réel scale, entier, dimension( * ) iwork, entier liwork, réel, dimension( ldswork, * ) swork, entier ldswork, entier info)

STRSYL3

But :

!>
!>  STRSYL3 résout l'équation matricielle de Sylvester réelle :
!>
!>     op(A)*X + X*op(B) = scale*C ou
!>     op(A)*X - X*op(B) = scale*C,
!>
!>  où op(A) = A ou A**T, et A et B sont toutes deux quasi-triangulaires
!>  supérieures. A est de taille M par M et B est de taille N par N ; le
!>  membre de droite C et la solution X sont de taille M par N ; et scale
!>  est un facteur d'échelle en sortie, fixé à <= 1 pour éviter le
!>  dépassement de capacité dans X.
!>
!>  A et B doivent être sous forme canonique de Schur (telles que retournées
!>  par SHSEQR), c'est-à-dire, bloc triangulaire supérieur avec des blocs
!>  diagonaux de 1x1 et 2x2 ; chaque bloc diagonal 2x2 a ses éléments
!>  diagonaux égaux et ses éléments non diagonaux de signe opposé.
!>
!>  Ceci est la version bloc de l'algorithme.
!>

Paramètres

TRANA 
!>          TRANA est CHARACTER*1
!>          Spécifie l'option op(A) :
!>          = 'N': op(A) = A    (Pas de transposée)
!>          = 'T': op(A) = A**T (Transposée)
!>          = 'C': op(A) = A**H (Conjuguée transposée = Transposée)
!>

TRANB

!>          TRANB est CHARACTER*1
!>          Spécifie l'option op(B) :
!>          = 'N': op(B) = B    (Pas de transposée)
!>          = 'T': op(B) = B**T (Transposée)
!>          = 'C': op(B) = B**H (Conjuguée transposée = Transposée)
!>

ISGN

!>          ISGN est INTEGER
!>          Spécifie le signe dans l'équation :
!>          = +1: résout op(A)*X + X*op(B) = scale*C
!>          = -1: résout op(A)*X - X*op(B) = scale*C
!>

M

!>          M est INTEGER
!>          L'ordre de la matrice A, et le nombre de lignes des matrices
!>          X et C. M >= 0.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice B, et le nombre de colonnes des matrices
!>          X et C. N >= 0.
!>

A

!>          A est un tableau REAL, dimension (LDA,M)
!>          La matrice quasi-triangulaire supérieure A, sous forme canonique
!>          de Schur.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,M).
!>

B

!>          B est un tableau REAL, dimension (LDB,N)
!>          La matrice quasi-triangulaire supérieure B, sous forme canonique
!>          de Schur.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,N).
!>

C

!>          C est un tableau REAL, dimension (LDC,N)
!>          À l'entrée, la matrice C du membre de droite de taille M par N.
!>          À la sortie, C est écrasée par la matrice solution X.
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale du tableau C. LDC >= max(1,M)
!>

SCALE

!>          SCALE est REAL
!>          Le facteur d'échelle, scale, fixé à <= 1 pour éviter le
!>          dépassement de capacité dans X.
!>

IWORK

!>          IWORK est un tableau INTEGER, dimension (MAX(1,LIWORK))
!>          À la sortie, si INFO = 0, IWORK(1) retourne le LIWORK optimal.
!>

LIWORK

!>          IWORK est INTEGER
!>          La dimension du tableau IWORK. LIWORK >=  ((M + NB - 1) / NB + 1)
!>          + ((N + NB - 1) / NB + 1), où NB est la taille de bloc optimale.
!>
!>          Si LIWORK = -1, alors une requête d'espace de travail est supposée ;
!>          la routine calcule uniquement la dimension optimale du tableau IWORK,
!>          retourne cette valeur comme première entrée du tableau IWORK, et
!>          aucun message d'erreur lié à LIWORK n'est émis par XERBLA.
!>

SWORK

!>          SWORK est un tableau REAL, dimension (MAX(2, ROWS),
!>          MAX(1,COLS)).
!>          À la sortie, si INFO = 0, SWORK(1) retourne la valeur optimale ROWS
!>          et SWORK(2) retourne la valeur optimale COLS.
!>

LDSWORK

!>          LDSWORK est INTEGER
!>          LDSWORK >= MAX(2,ROWS), où ROWS = ((M + NB - 1) / NB + 1)
!>          et NB est la taille de bloc optimale.
!>
!>          Si LDSWORK = -1, alors une requête d'espace de travail est supposée ;
!>          la routine calcule uniquement les dimensions optimales de la matrice
!>          SWORK, retourne ces valeurs comme première et deuxième entrées de la
!>          matrice SWORK, et aucun message d'erreur lié à LWORK n'est émis par
!>          XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0: sortie réussie
!>          < 0: si INFO = -i, le i-ème argument avait une valeur illégale
!>          = 1: A et B ont des valeurs propres communes ou très proches ; des
!>               valeurs perturbées ont été utilisées pour résoudre l'équation
!>               (mais les matrices A et B ne sont pas modifiées).
!>
