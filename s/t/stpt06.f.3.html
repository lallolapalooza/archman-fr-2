subroutine stpt06 (real rcond, real rcondc, character uplo, character diag, integer n, real, dimension( * ) ap, real, dimension( * ) work, real rat)

STPT06

Objectif :

!>
!> STPT06 calcule un ratio de test comparant RCOND (le nombre de condition
!> réciproque d'une matrice triangulaire A) et RCONDC, l'estimation calculée
!> par STPCON. Les informations sur la matrice triangulaire A sont utilisées si
!> une estimation est nulle et l'autre non nulle pour décider si le sous-dépassement
!> dans l'estimation est justifié.
!>

Paramètres

RCOND 
!>          RCOND est REAL
!>          L'estimation du nombre de condition réciproque obtenue en formant
!>          l'inverse explicite de la matrice A et en calculant
!>          RCOND = 1/( norm(A) * norm(inv(A)) ).
!>

RCONDC

!>          RCONDC est REAL
!>          L'estimation du nombre de condition réciproque calculée par
!>          STPCON.
!>

UPLO

!>          UPLO est CHARACTER
!>          Spécifie si la matrice A est triangulaire supérieure ou inférieure.
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

DIAG

!>          DIAG est CHARACTER
!>          Spécifie si la matrice A est unitaire ou non.
!>          = 'N':  Non-unitaire
!>          = 'U':  Unitaire
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

AP

!>          AP est un tableau de REAL, de dimension (N*(N+1)/2)
!>          La matrice triangulaire supérieure ou inférieure A, rangée en colonne
!>          dans un tableau linéaire. La j-ème colonne de A est stockée dans le
!>          tableau AP comme suit :
!>          si UPLO = 'U', AP((j-1)*j/2 + i) = A(i,j) pour 1<=i<=j;
!>          si UPLO = 'L',
!>             AP((j-1)*(n-j) + j*(j+1)/2 + i-j) = A(i,j) pour j<=i<=n.
!>

WORK

!>          WORK est un tableau de REAL, de dimension (N)
!>

RAT

!>          RAT est REAL
!>          Le ratio de test.  Si RCOND et RCONDC sont tous deux non nuls,
!>             RAT = MAX( RCOND, RCONDC )/MIN( RCOND, RCONDC ) - 1.
!>          Si RAT = 0, les deux estimations sont exactement les mêmes.
!>
