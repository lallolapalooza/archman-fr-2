subroutine schktz (logical, dimension( * ) dotype, integer nm, integer, dimension( * ) mval, integer nn, integer, dimension( * ) nval, real thresh, logical tsterr, real, dimension( * ) a, real, dimension( * ) copya, real, dimension( * ) s, real, dimension( * ) tau, real, dimension( * ) work, integer nout)

SCHKTZ

Objectif :

!>
!> SCHKTZ teste STZRZF.
!>

Paramètres

DOTYPE 
!>          DOTYPE est un tableau LOGICAL, dimension (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., alors le type j n'est pas utilisé.
!>

NM

!>          NM est un INTEGER
!>          Le nombre de valeurs de M contenues dans le vecteur MVAL.
!>

MVAL

!>          MVAL est un tableau INTEGER, dimension (NM)
!>          Les valeurs de la dimension des lignes de la matrice M.
!>

NN

!>          NN est un INTEGER
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau INTEGER, dimension (NN)
!>          Les valeurs de la dimension des colonnes de la matrice N.
!>

THRESH

!>          THRESH est un REAL
!>          La valeur seuil pour les rapports de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>          chaque rapport de test soit imprimé, utilisez THRESH = 0.
!>

TSTERR

!>          TSTERR est un LOGICAL
!>          Indicateur indiquant si les sorties d'erreur doivent être testées.
!>

A

!>          A est un tableau REAL, dimension (MMAX*NMAX)
!>          où MMAX est la valeur maximale de M dans MVAL et NMAX est la
!>          valeur maximale de N dans NVAL.
!>

COPYA

!>          COPYA est un tableau REAL, dimension (MMAX*NMAX)
!>

S

!>          S est un tableau REAL, dimension
!>                      (min(MMAX,NMAX))
!>

TAU

!>          TAU est un tableau REAL, dimension (MMAX)
!>

WORK

!>          WORK est un tableau REAL, dimension
!>                      (MMAX*NMAX + 4*NMAX + MMAX)
!>

NOUT

!>          NOUT est un INTEGER
!>          Le numéro d'unité pour la sortie.
!>
