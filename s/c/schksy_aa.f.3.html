sous-routine schksy_aa (logique, dimension( * ) dotype, entier nn, entier, dimension( * ) nval, entier nnb, entier, dimension( * ) nbval, entier nns, entier, dimension( * ) nsval, réel thresh, logique tsterr, entier nmax, réel, dimension( * ) a, réel, dimension( * ) afac, réel, dimension( * ) ainv, réel, dimension( * ) b, réel, dimension( * ) x, réel, dimension( * ) xact, réel, dimension( * ) work, réel, dimension( * ) rwork, entier, dimension( * ) iwork, entier nout)

SCHKSY_AA

Objectif:

!>
!> SCHKSY_AA teste SSYTRF_AA, -TRS_AA.
!>

Paramètres

DOTYPE 
!>          DOTYPE est un tableau LOGIQUE, dimension (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .VRAI.; si DOTYPE(j) = .FAUX., alors le type j n'est pas utilisé.
!>

NN

!>          NN est un ENTIER
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau d'ENTIERS, dimension (NN)
!>          Les valeurs de la dimension de la matrice N.
!>

NNB

!>          NNB est un ENTIER
!>          Le nombre de valeurs de NB contenues dans le vecteur NBVAL.
!>

NBVAL

!>          NBVAL est un tableau d'ENTIERS, dimension (NNB)
!>          Les valeurs de la taille de bloc NB.
!>

NNS

!>          NNS est un ENTIER
!>          Le nombre de valeurs de NRHS contenues dans le vecteur NSVAL.
!>

NSVAL

!>          NSVAL est un tableau d'ENTIERS, dimension (NNS)
!>          Les valeurs du nombre de membres de droite NRHS.
!>

THRESH

!>          THRESH est un RÉEL
!>          La valeur seuil pour les ratios de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour
!>          imprimer tous les ratios de test, utiliser THRESH = 0.
!>

TSTERR

!>          TSTERR est LOGIQUE
!>          Indicateur indiquant si les sorties d'erreur doivent être testées.
!>

NMAX

!>          NMAX est un ENTIER
!>          La valeur maximale autorisée pour N, utilisée pour dimensionner les
!>          tableaux de travail.
!>

A

!>          A est un tableau RÉEL, dimension (NMAX*NMAX)
!>

AFAC

!>          AFAC est un tableau RÉEL, dimension (NMAX*NMAX)
!>

AINV

!>          AINV est un tableau RÉEL, dimension (NMAX*NMAX)
!>

B

!>          B est un tableau RÉEL, dimension (NMAX*NSMAX)
!>          où NSMAX est la plus grande entrée dans NSVAL.
!>

X

!>          X est un tableau RÉEL, dimension (NMAX*NSMAX)
!>

XACT

!>          XACT est un tableau RÉEL, dimension (NMAX*NSMAX)
!>

WORK

!>          WORK est un tableau RÉEL, dimension (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK est un tableau RÉEL, dimension (max(NMAX,2*NSMAX))
!>

IWORK

!>          IWORK est un tableau d'ENTIERS, dimension (2*NMAX)
!>

NOUT

!>          NOUT est un ENTIER
!>          Le numéro d'unité pour la sortie.
!>
