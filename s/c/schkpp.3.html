sous-routine schkpp (logique, dimension( * ) dotype, entier nn, entier, dimension( * ) nval, entier nns, entier, dimension( * ) nsval, reel thresh, logique tsterr, entier nmax, reel, dimension( * ) a, reel, dimension( * ) afac, reel, dimension( * ) ainv, reel, dimension( * ) b, reel, dimension( * ) x, reel, dimension( * ) xact, reel, dimension( * ) work, reel, dimension( * ) rwork, entier, dimension( * ) iwork, entier nout)

SCHKPP

But :

!>
!> SCHKPP teste SPPTRF, -TRI, -TRS, -RFS, et -CON
!>

Paramètres

DOTYPE 
!>          DOTYPE est un tableau LOGIQUE, de dimension (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .VRAI. ; si DOTYPE(j) = .FAUX., alors le type j n'est pas utilisé.
!>

NN

!>          NN est un ENTIER
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau ENTIER, de dimension (NN)
!>          Les valeurs de la dimension de matrice N.
!>

NNS

!>          NNS est un ENTIER
!>          Le nombre de valeurs de NRHS contenues dans le vecteur NSVAL.
!>

NSVAL

!>          NSVAL est un tableau ENTIER, de dimension (NNS)
!>          Les valeurs du nombre de côtés droits NRHS.
!>

THRESH

!>          THRESH est un REEL
!>          La valeur seuil pour les ratios de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour faire
!>          imprimer tous les ratios de test, utilisez THRESH = 0.
!>

TSTERR

!>          TSTERR est LOGIQUE
!>          Indicateur qui spécifie si les sorties d'erreur doivent être testées.
!>

NMAX

!>          NMAX est un ENTIER
!>          La valeur maximale autorisée pour N, utilisée dans la détermination de la
!>          taille des tableaux de travail.
!>

A

!>          A est un tableau REEL, de dimension
!>                      (NMAX*(NMAX+1)/2)
!>

AFAC

!>          AFAC est un tableau REEL, de dimension
!>                      (NMAX*(NMAX+1)/2)
!>

AINV

!>          AINV est un tableau REEL, de dimension
!>                      (NMAX*(NMAX+1)/2)
!>

B

!>          B est un tableau REEL, de dimension (NMAX*NSMAX)
!>          où NSMAX est la plus grande entrée de NSVAL.
!>

X

!>          X est un tableau REEL, de dimension (NMAX*NSMAX)
!>

XACT

!>          XACT est un tableau REEL, de dimension (NMAX*NSMAX)
!>

WORK

!>          WORK est un tableau REEL, de dimension
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK est un tableau REEL, de dimension
!>                      (max(NMAX,2*NSMAX))
!>

IWORK

!>          IWORK est un tableau ENTIER, de dimension (NMAX)
!>

NOUT

!>          NOUT est un ENTIER
!>          Le numéro de l'unité de sortie.
!>
