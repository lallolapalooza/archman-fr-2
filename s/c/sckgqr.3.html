subroutine sckgqr (integer nm, integer, dimension( * ) mval, integer np, integer, dimension( * ) pval, integer nn, integer, dimension( * ) nval, integer nmats, integer, dimension( 4 ) iseed, real thresh, integer nmax, real, dimension( * ) a, real, dimension( * ) af, real, dimension( * ) aq, real, dimension( * ) ar, real, dimension( * ) taua, real, dimension( * ) b, real, dimension( * ) bf, real, dimension( * ) bz, real, dimension( * ) bt, real, dimension( * ) bwk, real, dimension( * ) taub, real, dimension( * ) work, real, dimension( * ) rwork, integer nin, integer nout, integer info)

SCKGQR

Objectif :

!>
!> SCKGQR teste
!> SGGQRF : factorisation GQR pour la matrice N-par-M A et la matrice N-par-P B,
!> SGGRQF : factorisation GRQ pour la matrice M-par-N A et la matrice P-par-N B.
!>

Paramètres

NM 
!>          NM est un entier
!>          Le nombre de valeurs de M contenues dans le vecteur MVAL.
!>

MVAL

!>          MVAL est un tableau d'entiers, de dimension (NM)
!>          Les valeurs de la dimension des lignes (colonnes) de la matrice M.
!>

NP

!>          NP est un entier
!>          Le nombre de valeurs de P contenues dans le vecteur PVAL.
!>

PVAL

!>          PVAL est un tableau d'entiers, de dimension (NP)
!>          Les valeurs de la dimension des lignes (colonnes) de la matrice P.
!>

NN

!>          NN est un entier
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau d'entiers, de dimension (NN)
!>          Les valeurs de la dimension des colonnes (lignes) de la matrice N.
!>

NMATS

!>          NMATS est un entier
!>          Le nombre de types de matrices à tester pour chaque combinaison
!>          des dimensions de matrices. Si NMATS >= NTYPES (le nombre maximum
!>          de types de matrices), alors tous les différents types sont
!>          générés pour les tests. Si NMATS < NTYPES, une ligne d'entrée
!>          supplémentaire est lue pour obtenir les numéros des types de matrices
!>          à utiliser.
!>

ISEED

!>          ISEED est un tableau d'entiers, de dimension (4)
!>          À l'entrée, la graine du générateur de nombres aléatoires. Les éléments
!>          du tableau doivent être compris entre 0 et 4095, sinon ils seront
!>          réduits modulo 4096, et ISEED(4) doit être impair.
!>          À la sortie, la graine suivante dans la séquence de nombres aléatoires
!>          après la génération de toutes les matrices de test.
!>

THRESH

!>          THRESH est un réel
!>          La valeur seuil pour les ratios de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>          chaque ratio de test soit imprimé, utilisez THRESH = 0.
!>

NMAX

!>          NMAX est un entier
!>          La valeur maximale autorisée pour M ou N, utilisée pour dimensionner
!>          les tableaux de travail.
!>

A

!>          A est un tableau réel, de dimension (NMAX*NMAX)
!>

AF

!>          AF est un tableau réel, de dimension (NMAX*NMAX)
!>

AQ

!>          AQ est un tableau réel, de dimension (NMAX*NMAX)
!>

AR

!>          AR est un tableau réel, de dimension (NMAX*NMAX)
!>

TAUA

!>          TAUA est un tableau réel, de dimension (NMAX)
!>

B

!>          B est un tableau réel, de dimension (NMAX*NMAX)
!>

BF

!>          BF est un tableau réel, de dimension (NMAX*NMAX)
!>

BZ

!>          BZ est un tableau réel, de dimension (NMAX*NMAX)
!>

BT

!>          BT est un tableau réel, de dimension (NMAX*NMAX)
!>

BWK

!>          BWK est un tableau réel, de dimension (NMAX*NMAX)
!>

TAUB

!>          TAUB est un tableau réel, de dimension (NMAX)
!>

WORK

!>          WORK est un tableau réel, de dimension (NMAX*NMAX)
!>

RWORK

!>          RWORK est un tableau réel, de dimension (NMAX)
!>

NIN

!>          NIN est un entier
!>          Le numéro d'unité d'entrée.
!>

NOUT

!>          NOUT est un entier
!>          Le numéro d'unité de sortie.
!>

INFO

!>          INFO est un entier
!>          = 0 : sortie réussie
!>          > 0 : Si SLATMS renvoie un code d'erreur, sa valeur absolue est retournée.
!>
