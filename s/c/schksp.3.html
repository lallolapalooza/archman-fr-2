sous programme schksp (logique, dimension( * ) dotype, entier nn, entier, dimension( * ) nval, entier nns, entier, dimension( * ) nsval, reel thresh, logique tsterr, entier nmax, reel, dimension( * ) a, reel, dimension( * ) afac, reel, dimension( * ) ainv, reel, dimension( * ) b, reel, dimension( * ) x, reel, dimension( * ) xact, reel, dimension( * ) work, reel, dimension( * ) rwork, entier, dimension( * ) iwork, entier nout)

SCHKSP

Objectif :

!>
!> SCHKSP teste SSPTRF, -TRI, -TRS, -RFS, et -CON
!>

Paramètres

DOTYPE 
!>          DOTYPE est un tableau logique, dimensionné (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .VRAI. ; si DOTYPE(j) = .FAUX., alors le type j n'est pas utilisé.
!>

NN

!>          NN est un ENTIER
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau ENTIER, dimensionné (NN)
!>          Les valeurs de la dimension N de la matrice.
!>

NNS

!>          NNS est un ENTIER
!>          Le nombre de valeurs de NRHS contenues dans le vecteur NSVAL.
!>

NSVAL

!>          NSVAL est un tableau ENTIER, dimensionné (NNS)
!>          Les valeurs du nombre de seconds membres NRHS.
!>

THRESH

!>          THRESH est un REEL
!>          La valeur seuil pour les rapports de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>          chaque rapport de test soit imprimé, utilisez THRESH = 0.
!>

TSTERR

!>          TSTERR est un LOGIQUE
!>          Indicateur qui spécifie si les sorties d'erreur doivent être testées.
!>

NMAX

!>          NMAX est un ENTIER
!>          La valeur maximale autorisée pour N, utilisée pour dimensionner les
!>          tableaux de travail.
!>

A

!>          A est un tableau REEL, dimensionné
!>                      (NMAX*(NMAX+1)/2)
!>

AFAC

!>          AFAC est un tableau REEL, dimensionné
!>                      (NMAX*(NMAX+1)/2)
!>

AINV

!>          AINV est un tableau REEL, dimensionné
!>                      (NMAX*(NMAX+1)/2)
!>

B

!>          B est un tableau REEL, dimensionné (NMAX*NSMAX)
!>          où NSMAX est la plus grande entrée dans NSVAL.
!>

X

!>          X est un tableau REEL, dimensionné (NMAX*NSMAX)
!>

XACT

!>          XACT est un tableau REEL, dimensionné (NMAX*NSMAX)
!>

WORK

!>          WORK est un tableau REEL, dimensionné
!>                      (NMAX*max(2,NSMAX))
!>

RWORK

!>          RWORK est un tableau REEL,
!>                                 dimensionné (NMAX+2*NSMAX)
!>

IWORK

!>          IWORK est un tableau ENTIER, dimensionné (2*NMAX)
!>

NOUT

!>          NOUT est un ENTIER
!>          Le numéro de l'unité de sortie.
!>
