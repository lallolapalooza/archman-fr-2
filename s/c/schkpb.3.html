subroutine schkpb (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nnb, integer, dimension( * ) nbval, integer nns, integer, dimension( * ) nsval, real thresh, logical tsterr, integer nmax, real, dimension( * ) a, real, dimension( * ) afac, real, dimension( * ) ainv, real, dimension( * ) b, real, dimension( * ) x, real, dimension( * ) xact, real, dimension( * ) work, real, dimension( * ) rwork, integer, dimension( * ) iwork, integer nout)

SCHKPB

Objectif :

!>
!> SCHKPB teste SPBTRF, -TRS, -RFS, et -CON.
!>

Paramètres

DOTYPE 
!>          DOTYPE est un tableau LOGICAL, dimensionné (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .TRUE. ; si DOTYPE(j) = .FALSE., alors le type j n'est pas utilisé.
!>

NN

!>          NN est un entier
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau d'entiers, dimensionné (NN)
!>          Les valeurs de la dimension de la matrice N.
!>

NNB

!>          NNB est un entier
!>          Le nombre de valeurs de NB contenues dans le vecteur NBVAL.
!>

NBVAL

!>          NBVAL est un tableau d'entiers, dimensionné (NNB)
!>          Les valeurs de la taille de bloc NB.
!>

NNS

!>          NNS est un entier
!>          Le nombre de valeurs de NRHS contenues dans le vecteur NSVAL.
!>

NSVAL

!>          NSVAL est un tableau d'entiers, dimensionné (NNS)
!>          Les valeurs du nombre de côtés droits NRHS.
!>

THRESH

!>          THRESH est un réel
!>          La valeur seuil pour les rapports de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>          chaque rapport de test soit imprimé, utilisez THRESH = 0.
!>

TSTERR

!>          TSTERR est LOGICAL
!>          Indicateur qui signale si les sorties d'erreur doivent être testées.
!>

NMAX

!>          NMAX est un entier
!>          La valeur maximale autorisée pour N, utilisée pour dimensionner les
!>          tableaux de travail.
!>

A

!>          A est un tableau de réels, dimensionné (NMAX*NMAX)
!>

AFAC

!>          AFAC est un tableau de réels, dimensionné (NMAX*NMAX)
!>

AINV

!>          AINV est un tableau de réels, dimensionné (NMAX*NMAX)
!>

B

!>          B est un tableau de réels, dimensionné (NMAX*NSMAX)
!>          où NSMAX est la plus grande entrée de NSVAL.
!>

X

!>          X est un tableau de réels, dimensionné (NMAX*NSMAX)
!>

XACT

!>          XACT est un tableau de réels, dimensionné (NMAX*NSMAX)
!>

WORK

!>          WORK est un tableau de réels, dimensionné
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK est un tableau de réels, dimensionné
!>                      (max(NMAX,2*NSMAX))
!>

IWORK

!>          IWORK est un tableau d'entiers, dimensionné (NMAX)
!>

NOUT

!>          NOUT est un entier
!>          Le numéro d'unité de sortie.
!>
