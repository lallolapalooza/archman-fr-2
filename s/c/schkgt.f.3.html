sous-routine schkgt (logique, dimension( * ) dotype, entier nn, entier, dimension( * ) nval, entier nns, entier, dimension( * ) nsval, relatif thresh, logique tsterr, relatif, dimension( * ) a, relatif, dimension( * ) af, relatif, dimension( * ) b, relatif, dimension( * ) x, relatif, dimension( * ) xact, relatif, dimension( * ) work, relatif, dimension( * ) rwork, entier, dimension( * ) iwork, entier nout)

!>
!> SCHKGT teste SGTTRF, -TRS, -RFS, et -CON
!>

Paramètres

DOTYPE 
!>          DOTYPE est un tableau LOGIQUE, dimension (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., alors le type j n'est pas utilisé.
!>

NN

!>          NN est un ENTIER
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau ENTIER, dimension (NN)
!>          Les valeurs de la dimension de matrice N.
!>

NNS

!>          NNS est un ENTIER
!>          Le nombre de valeurs de NRHS contenues dans le vecteur NSVAL.
!>

NSVAL

!>          NSVAL est un tableau ENTIER, dimension (NNS)
!>          Les valeurs du nombre de côtés droits NRHS.
!>

THRESH

!>          THRESH est un RELATIF
!>          La valeur seuil pour les ratios de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>          chaque ratio de test soit imprimé, utilisez THRESH = 0.
!>

TSTERR

!>          TSTERR est LOGIQUE
!>          Indicateur indiquant si les sorties d'erreur doivent être testées.
!>

A

!>          A est un tableau RELATIF, dimension (NMAX*4)
!>

AF

!>          AF est un tableau RELATIF, dimension (NMAX*4)
!>

B

!>          B est un tableau RELATIF, dimension (NMAX*NSMAX)
!>          où NSMAX est la plus grande entrée dans NSVAL.
!>

X

!>          X est un tableau RELATIF, dimension (NMAX*NSMAX)
!>

XACT

!>          XACT est un tableau RELATIF, dimension (NMAX*NSMAX)
!>

WORK

!>          WORK est un tableau RELATIF, dimension
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK est un tableau RELATIF, dimension
!>                      (max(NMAX,2*NSMAX))
!>

IWORK

!>          IWORK est un tableau ENTIER, dimension (2*NMAX)
!>

NOUT

!>          NOUT est un ENTIER
!>          Le numéro de l'unité de sortie.
!>
