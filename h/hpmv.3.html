subroutine chpmv (character uplo, integer n, complex alpha, complex, dimension(*) ap, complex, dimension(*) x, integer incx, complex beta, complex, dimension(*) y, integer

incy)

CHPM

V

But

: 
!>
!> CHPMV effectue l'opération matrice-vecteur
!>
!>    y := alpha*A*x + beta*y,
!>
!> où alpha et beta sont des scalaires, x et y sont des vecteurs de n éléments et
!> A est une matrice hermitienne de taille n par n, fournie sous forme empaquetée.
!
>

Paramètre

s
UPLO 
!>          UPLO est CHARACTER*1
!>           À l'entrée, UPLO spécifie si la partie triangulaire supérieure ou
!>           inférieure de la matrice A est fournie dans le tableau empaqueté
!>           AP comme suit :
!>
!>              UPLO = 'U' ou 'u'   La partie triangulaire supérieure de A est
!>                                  fournie dans AP.
!>
!>              UPLO = 'L' ou 'l'   La partie triangulaire inférieure de A est
!>                                  fournie dans AP.
!
>

N 
!>          N est INTEGER
!>           À l'entrée, N spécifie l'ordre de la matrice A.
!>           N doit être au moins zéro.
!
>

ALPH

A 
!>          ALPHA est COMPLEX
!>           À l'entrée, ALPHA spécifie le scalaire alpha.
!
>

A

P 
!>          AP est un tableau COMPLEX, dimension au moins
!>           ( ( n*( n + 1 ) )/2 ).
!>           Avant l'entrée avec UPLO = 'U' ou 'u', le tableau AP doit
!>           contenir la partie triangulaire supérieure de la matrice hermitienne
!>           empaquetée séquentiellement, colonne par colonne, de sorte que AP( 1 )
!>           contient a( 1, 1 ), AP( 2 ) et AP( 3 ) contiennent a( 1, 2 )
!>           et a( 2, 2 ) respectivement, et ainsi de suite.
!>           Avant l'entrée avec UPLO = 'L' ou 'l', le tableau AP doit
!>           contenir la partie triangulaire inférieure de la matrice hermitienne
!>           empaquetée séquentiellement, colonne par colonne, de sorte que AP( 1 )
!>           contient a( 1, 1 ), AP( 2 ) et AP( 3 ) contiennent a( 2, 1 )
!>           et a( 3, 1 ) respectivement, et ainsi de suite.
!>           Notez que les parties imaginaires des éléments diagonaux n'ont pas
!>           besoin d'être définies et sont supposées être nulles.
!
>

X 
!>          X est un tableau COMPLEX, dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           Avant l'entrée, le tableau incrémenté X doit contenir le vecteur
!>           de n éléments x.
!
>

INC

X 
!>          INCX est INTEGER
!>           À l'entrée, INCX spécifie l'incrément pour les éléments de
!>           X. INCX ne doit pas être zéro.
!
>

BET

A 
!>          BETA est COMPLEX
!>           À l'entrée, BETA spécifie le scalaire beta. Lorsque BETA est
!>           fourni comme zéro, Y n'a pas besoin d'être défini en entrée.
!
>

Y 
!>          Y est un tableau COMPLEX, dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           Avant l'entrée, le tableau incrémenté Y doit contenir le vecteur
!>           de n éléments y. À la sortie, Y est écrasé par le vecteur y mis à jour.
!
>

INC

Y 
!>          INCY est INTEGER
!>           À l'entrée, INCY spécifie l'incrément pour les éléments de
!>           Y. INCY ne doit pas être zéro.
!
>
