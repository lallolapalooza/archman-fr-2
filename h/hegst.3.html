subroutine chegst (integer itype, character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, integer

info)

CHEGS

T

But

: 
!>
!> CHEGST réduit un problème aux valeurs propres généralisé
!> hermitien-défini à la forme standard.
!>
!> Si ITYPE = 1, le problème est A*x = lambda*B*x,
!> et A est écrasée par inv(U**H)*A*inv(U) ou inv(L)*A*inv(L**H)
!>
!> Si ITYPE = 2 ou 3, le problème est A*B*x = lambda*x ou
!> B*A*x = lambda*x, et A est écrasée par U*A*U**H ou L**H*A*L.
!>
!> B doit avoir été préalablement factorisée en U**H*U ou L*L**H par CPOTRF.
!
>

Paramètre

s
ITYPE 
!>          ITYPE est INTEGER
!>          = 1 : calculer inv(U**H)*A*inv(U) ou inv(L)*A*inv(L**H) ;
!>          = 2 ou 3 : calculer U*A*U**H ou L**H*A*L.
!
>

UPL

O 
!>          UPLO est CHARACTER*1
!>          = 'U' : le triangle supérieur de A est stocké et B est factorisé en
!>                  U**H*U ;
!>          = 'L' : le triangle inférieur de A est stocké et B est factorisé en
!>                  L*L**H.
!
>

N 
!>          N est INTEGER
!>          L'ordre des matrices A et B.  N >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          À l'entrée, la matrice hermitienne A.  Si UPLO = 'U', la partie
!>          triangulaire supérieure de A de taille N par N contient la partie
!>          triangulaire supérieure de la matrice A, et la partie
!>          triangulaire strictement inférieure de A n'est pas référencée.
!>          Si UPLO = 'L', la partie triangulaire inférieure de A de taille
!>          N par N contient la partie triangulaire inférieure de la matrice A,
!>          et la partie triangulaire strictement supérieure de A n'est pas
!>          référencée.
!>
!>          À la sortie, si INFO = 0, la matrice transformée, stockée dans le
!>          même format que A.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!
>

B 
!>          B est un tableau COMPLEX, dimension (LDB,N)
!>          Le facteur triangulaire de la factorisation de Cholesky de B,
!>          tel que renvoyé par CPOTRF.
!>          B est modifiée par la routine mais restaurée à la sortie.
!
>

LD

B 
!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!
>
