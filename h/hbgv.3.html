sous-routine chbgv (character jobz, character uplo, integer n, integer ka, integer kb, complex, dimension( ldab, * ) ab, integer ldab, complex, dimension( ldbb, * ) bb, integer ldbb, real, dimension( * ) w, complex, dimension( ldz, * ) z, integer ldz, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CHBGV

But :

!>
!> CHBGV calcule toutes les valeurs propres, et éventuellement les vecteurs propres
!> d'un problème aux valeurs propres généralisé hermitien-défini par bandes, de
!> la forme A*x=(lambda)*B*x. Ici, A et B sont supposés hermitiens
!> et à bandes, et B est également défini positif.
!>

Paramètres

JOBZ 
!>          JOBZ est CHARACTER*1
!>          = 'N' :  Calculer uniquement les valeurs propres ;
!>          = 'V' :  Calculer les valeurs propres et les vecteurs propres.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U' :  Les triangles supérieurs de A et B sont stockés ;
!>          = 'L' :  Les triangles inférieurs de A et B sont stockés.
!>

N

!>          N est INTEGER
!>          L'ordre des matrices A et B.  N >= 0.
!>

KA

!>          KA est INTEGER
!>          Le nombre de superdiagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'. KA >= 0.
!>

KB

!>          KB est INTEGER
!>          Le nombre de superdiagonales de la matrice B si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'. KB >= 0.
!>

AB

!>          AB est un tableau COMPLEX, dimension (LDAB, N)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice
!>          hermitienne par bandes A, stocké dans les premières ka+1 lignes du
!>          tableau. La j-ième colonne de A est stockée dans la j-ième colonne
!>          du tableau AB comme suit :
!>          si UPLO = 'U', AB(ka+1+i-j,j) = A(i,j) pour max(1,j-ka)<=i<=j ;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+ka).
!>
!>          À la sortie, le contenu de AB est détruit.
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KA+1.
!>

BB

!>          BB est un tableau COMPLEX, dimension (LDBB, N)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice
!>          hermitienne par bandes B, stocké dans les premières kb+1 lignes du
!>          tableau. La j-ième colonne de B est stockée dans la j-ième colonne
!>          du tableau BB comme suit :
!>          si UPLO = 'U', BB(kb+1+i-j,j) = B(i,j) pour max(1,j-kb)<=i<=j ;
!>          si UPLO = 'L', BB(1+i-j,j)    = B(i,j) pour j<=i<=min(n,j+kb).
!>
!>          À la sortie, le facteur S de la factorisation de Cholesky scindée
!>          B = S**H*S, tel que retourné par CPBSTF.
!>

LDBB

!>          LDBB est INTEGER
!>          La dimension principale du tableau BB.  LDBB >= KB+1.
!>

W

!>          W est un tableau REAL, dimension (N)
!>          Si INFO = 0, les valeurs propres en ordre croissant.
!>

Z

!>          Z est un tableau COMPLEX, dimension (LDZ, N)
!>          Si JOBZ = 'V', alors si INFO = 0, Z contient la matrice Z des
!>          vecteurs propres, avec la i-ième colonne de Z contenant le
!>          vecteur propre associé à W(i). Les vecteurs propres sont
!>          normalisés de sorte que Z**H*B*Z = I.
!>          Si JOBZ = 'N', Z n'est pas référencé.
!>

LDZ

!>          LDZ est INTEGER
!>          La dimension principale du tableau Z.  LDZ >= 1, et si
!>          JOBZ = 'V', LDZ >= N.
!>

WORK

!>          WORK est un tableau COMPLEX, dimension (N)
!>

RWORK

!>          RWORK est un tableau REAL, dimension (3*N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 :  si INFO = i, et i est :
!>             <= N :  l'algorithme n'a pas convergé :
!>                    i éléments hors diagonale d'une forme
!>                    intermédiaire tridiagonale n'ont pas convergé vers zéro ;
!>             > N :   si INFO = N + i, pour 1 <= i <= N, alors CPBSTF
!>                    a retourné INFO = i : B n'est pas défini positif.
!>                    La factorisation de B n'a pas pu être complétée et
!>                    aucune valeur propre ni vecteur propre n'a été calculé.
!>
