subroutine chetf2 (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, integer

info)

CHETF2 calcule la factorisation d'une matrice hermitienne complexe, en utilisant la méthode de pivotage diagonal (algorithme non parallélisé appelant les BLAS de niveau 2)

.

But

: 
!>
!> CHETF2 calcule la factorisation d'une matrice hermitienne complexe A
!> en utilisant la méthode de pivotage diagonal de Bunch-Kaufman :
!>
!>    A = U*D*U**H  ou  A = L*D*L**H
!>
!> où U (ou L) est un produit de matrices de permutation et de matrices
!> triangulaires unitaires supérieures (inférieures), U**H est la
!> transposée conjuguée de U, et D est hermitienne et bloc-diagonale
!> avec des blocs diagonaux de taille 1 par 1 et 2 par 2.
!>
!> Ceci est la version non parallélisée de l'algorithme, appelant les
!> BLAS de niveau 2.
!
>

Paramètre

s
UPLO 
!>          UPLO est CHARACTER*1
!>          Indique si la partie triangulaire supérieure ou inférieure
!>          de la matrice hermitienne A est stockée :
!>          = 'U' :  triangulaire supérieure
!>          = 'L' :  triangulaire inférieure
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          À l'entrée, la matrice hermitienne A.  Si UPLO = 'U', la
!>          partie triangulaire supérieure de taille n par n de A
!>          contient la partie triangulaire supérieure de la matrice A,
!>          et la partie strictement inférieure de A n'est pas
!>          référencée.  Si UPLO = 'L', la partie triangulaire inférieure
!>          de taille n par n de A contient la partie triangulaire
!>          inférieure de la matrice A, et la partie strictement
!>          supérieure de A n'est pas référencée.
!>
!>          À la sortie, la matrice bloc-diagonale D et les
!>          multiplicateurs utilisés pour obtenir le facteur U ou L
!>          (voir ci-dessous pour plus de détails).
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!
>

IPI

V 
!>          IPIV est un tableau INTEGER, dimension (N)
!>          Détails des interchanges et de la structure bloc de D.
!>
!>          Si UPLO = 'U' :
!>             Si IPIV(k) > 0, alors les lignes et les colonnes k et
!>             IPIV(k) ont été interchangées et D(k,k) est un bloc
!>             diagonal de taille 1 par 1.
!>
!>             Si IPIV(k) = IPIV(k-1) < 0, alors les lignes et les
!>             colonnes k-1 et -IPIV(k) ont été interchangées et
!>             D(k-1:k,k-1:k) est un bloc diagonal de taille 2 par 2.
!>
!>          Si UPLO = 'L' :
!>             Si IPIV(k) > 0, alors les lignes et les colonnes k et
!>             IPIV(k) ont été interchangées et D(k,k) est un bloc
!>             diagonal de taille 1 par 1.
!>
!>             Si IPIV(k) = IPIV(k+1) < 0, alors les lignes et les
!>             colonnes k+1 et -IPIV(k) ont été interchangées et
!>             D(k:k+1,k:k+1) est un bloc diagonal de taille 2 par 2.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -k, le k-ième argument avait une valeur
!>                illégale
!>          > 0 : si INFO = k, D(k,k) est exactement zéro.  La
!>                factorisation a été complétée, mais la matrice
!>                bloc-diagonale D est exactement singulière, et une
!>                division par zéro se produira si elle est utilisée
!>                pour résoudre un système d'équations.
!
>

!>  et A(k+1,k+1), et v écrase A(k+2:n,k:k+1).
!

subroutine   zsytf2 (character uplo, integer n, complex*16, dimension( lda, * ) a, integer   lda, integer, dimension( * ) ipiv, integer 
info)

ZSYTF2 calcule la factorisation d'une matrice symétrique réelle
    indéfinie, en utilisant la méthode de pivotage diagonal (algorithme
  non parallélisé)
.

But 
:



!>
!> ZSYTF2 calcule la factorisation d'une matrice symétrique complexe A
!> en utilisant la méthode de pivotage diagonal de Bunch-Kaufman :
!>
!>    A = U*D*U**T  ou  A = L*D*L**T
!>
!> où U (ou L) est un produit de matrices de permutation et de matrices
!> triangulaires supérieures (inférieures) unitaires, U**T est la
!> transposée de U, et D est symétrique et bloc diagonale avec des
!> blocs diagonaux de tailles 1x1 et 2x2.
!>
!> Il s'agit de la version non parallélisée de l'algorithme, qui fait
!> appel aux BLAS de niveau 2.

!
>

Paramètre
s

UPLO

!>          UPLO est CHARACTER*1
!>          Indique si la partie triangulaire supérieure ou inférieure
!>          de la matrice symétrique A est stockée :
!>          = 'U' :  Triangulaire supérieure
!>          = 'L' :  Triangulaire inférieure
!
>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>          À l'entrée, la matrice symétrique A.  Si UPLO = 'U', la
!>          partie triangulaire supérieure de taille n x n de A
!>          contient la partie triangulaire supérieure de la matrice A,
!>          et la partie strictement inférieure de A n'est pas
!>          référencée.  Si UPLO = 'L', la partie triangulaire
!>          inférieure de taille n x n de A contient la partie
!>          triangulaire inférieure de la matrice A, et la partie
!>          strictement supérieure de A n'est pas référencée.
!>
!>          À la sortie, la matrice bloc diagonale D et les
!>          multiplicateurs utilisés pour obtenir le facteur U ou L
!>          (voir ci-dessous pour plus de détails).
!
>

LD
A

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!
>

IPI
V

!>          IPIV est un tableau INTEGER, dimension (N)
!>          Détails des échanges et de la structure de blocs de D.
!>
!>          Si UPLO = 'U' :
!>             Si IPIV(k) > 0, alors les lignes et colonnes k et
!>             IPIV(k) ont été échangées et D(k,k) est un bloc
!>             diagonal de taille 1x1.
!>
!>             Si IPIV(k) = IPIV(k-1) < 0, alors les lignes et
!>             colonnes k-1 et -IPIV(k) ont été échangées et
!>             D(k-1:k,k-1:k) est un bloc diagonal de taille 2x2.
!>
!>          Si UPLO = 'L' :
!>             Si IPIV(k) > 0, alors les lignes et colonnes k et
!>             IPIV(k) ont été échangées et D(k,k) est un bloc
!>             diagonal de taille 1x1.
!>
!>             Si IPIV(k) = IPIV(k+1) < 0, alors les lignes et
!>             colonnes k+1 et -IPIV(k) ont été échangées et
!>             D(k:k+1,k:k+1) est un bloc diagonal de taille 2x2.
!
>

O

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -k, le k-ième argument avait une valeur
!>                illégale
!>          > 0 : si INFO = k, D(k,k) est exactement nul.  La

!>                diagonale D est exactement singulière, et une
!>                division par zéro se produira si elle est utilisée
!>                pour résoudre un système d'équations.
