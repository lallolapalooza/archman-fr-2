subroutine cheev (character jobz, character uplo, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) w, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, integer info)

CHEEV calcule les valeurs propres et, en option, les vecteurs propres gauche et/ou droit des matrices HE

But :

!>
!> CHEEV calcule toutes les valeurs propres et, en option, les vecteurs
!> propres d'une matrice hermitienne complexe A.
!>

Paramètres

JOBZ 
!>          JOBZ est CHARACTER*1
!>          = 'N' :  calcule uniquement les valeurs propres ;
!>          = 'V' :  calcule les valeurs propres et les vecteurs propres.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U' :  le triangle supérieur de A est stocké ;
!>          = 'L' :  le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

A

!>          A est un tableau COMPLEX, dimension (LDA, N)
!>          À l'entrée, la matrice hermitienne A.  Si UPLO = 'U', la
!>          partie triangulaire supérieure de taille N×N de A contient
!>          la partie triangulaire supérieure de la matrice A.  Si UPLO = 'L',
!>          la partie triangulaire inférieure de taille N×N de A contient
!>          la partie triangulaire inférieure de la matrice A.
!>          À la sortie, si JOBZ = 'V', alors si INFO = 0, A contient les
!>          vecteurs propres orthonormaux de la matrice A.
!>          Si JOBZ = 'N', alors à la sortie le triangle inférieur (si UPLO='L')
!>          ou le triangle supérieur (si UPLO='U') de A, y compris la
!>          diagonale, est détruit.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

W

!>          W est un tableau REAL, dimension (N)
!>          Si INFO = 0, les valeurs propres en ordre croissant.
!>

WORK

!>          WORK est un tableau COMPLEX, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie le LWORK optimal.
!>

LWORK

!>          LWORK est INTEGER
!>          La longueur du tableau WORK.  LWORK >= max(1,2*N-1).
!>          Pour une efficacité optimale, LWORK >= (NB+1)*N,
!>          où NB est la taille de bloc pour CHETRD renvoyée par ILAENV.
!>
!>          Si LWORK = -1, une interrogation d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message
!>          d'erreur lié à LWORK n'est émis par XERBLA.
!>

RWORK

!>          RWORK est un tableau REAL, dimension (max(1, 3*N-2))
!>

INFO

!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 :  si INFO = i, l'algorithme n'a pas convergé ; i
!>                éléments hors diagonale d'une forme tridiagonale
!>                intermédiaire n'ont pas convergé vers zéro.
!>
