subroutine chpevd (character jobz, character uplo, integer n, complex, dimension( * ) ap, real, dimension( * ) w, complex, dimension( ldz, * ) z, integer ldz, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, integer lrwork, integer, dimension( * ) iwork, integer liwork, integer

info)

CHPEVD calcule les valeurs propres et, en option, les vecteurs propres gauche et/ou droit pour les matrices

OTHER

But

: 
!>
!> CHPEVD calcule toutes les valeurs propres et, en option, les vecteurs propres
!> d'une matrice hermitienne complexe A en stockage compact. Si les vecteurs
!> propres sont souhaités, un algorithme de type divide and conquer est utilisé.
!>
!
>

Paramètre

s
JOBZ 
!>          JOBZ est CHARACTER*1
!>          = 'N' :  Calculer uniquement les valeurs propres ;
!>          = 'V' :  Calculer les valeurs propres et les vecteurs propres.
!
>

UPL

O 
!>          UPLO est CHARACTER*1
!>          = 'U' :  Le triangle supérieur de A est stocké ;
!>          = 'L' :  Le triangle inférieur de A est stocké.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

A

P 
!>          AP est un tableau COMPLEX, dimension (N*(N+1)/2)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice
!>          hermitienne A, compacté colonne par colonne dans un tableau
!>          linéaire. La j-ième colonne de A est stockée dans le tableau
!>          AP comme suit :
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L', AP(i + (j-1)*(2*n-j)/2) = A(i,j) pour j<=i<=n.
!>
!>          À la sortie, AP est écrasé par les valeurs générées pendant la
!>          réduction en forme tridiagonale. Si UPLO = 'U', la diagonale
!>          et la première surdiagonale de la matrice tridiagonale T
!>          écrasent les éléments correspondants de A, et si UPLO = 'L', la
!>          diagonale et la première sous-diagonale de T écrasent les
!>          éléments correspondants de A.
!
>

W 
!>          W est un tableau REAL, dimension (N)
!>          Si INFO = 0, les valeurs propres dans l'ordre croissant.
!
>

Z 
!>          Z est un tableau COMPLEX, dimension (LDZ, N)
!>          Si JOBZ = 'V', alors si INFO = 0, Z contient les vecteurs
!>          propres orthonormaux de la matrice A, la i-ième colonne de Z
!>          contenant le vecteur propre associé à W(i).
!>          Si JOBZ = 'N', Z n'est pas référencé.
!
>

LD

Z 
!>          LDZ est INTEGER
!>          La dimension principale du tableau Z.  LDZ >= 1, et si
!>          JOBZ = 'V', LDZ >= max(1,N).
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie le LWORK requis.
!
>

LWOR

K 
!>          LWORK est INTEGER
!>          La dimension du tableau WORK.
!>          Si N <= 1,               LWORK doit être au moins 1.
!>          Si JOBZ = 'N' et N > 1, LWORK doit être au moins N.
!>          Si JOBZ = 'V' et N > 1, LWORK doit être au moins 2*N.
!>
!>          Si LWORK = -1, une interrogation d'espace de travail est
!>          supposée ; la routine calcule uniquement les tailles requises
!>          des tableaux WORK, RWORK et IWORK, renvoie ces valeurs comme
!>          premières entrées des tableaux WORK, RWORK et IWORK, et aucun
!>          message d'erreur lié à LWORK ou LRWORK ou LIWORK n'est émis
!>          par XERBLA.
!
>

RWOR

K 
!>          RWORK est un tableau REAL, dimension (MAX(1,LRWORK))
!>          À la sortie, si INFO = 0, RWORK(1) renvoie le LRWORK requis.
!
>

LRWOR

K 
!>          LRWORK est INTEGER
!>          La dimension du tableau RWORK.
!>          Si N <= 1,               LRWORK doit être au moins 1.
!>          Si JOBZ = 'N' et N > 1, LRWORK doit être au moins N.
!>          Si JOBZ = 'V' et N > 1, LRWORK doit être au moins
!>                    1 + 5*N + 2*N**2.
!>
!>          Si LRWORK = -1, une interrogation d'espace de travail est
!>          supposée ; la routine calcule uniquement les tailles requises
!>          des tableaux WORK, RWORK et IWORK, renvoie ces valeurs comme
!>          premières entrées des tableaux WORK, RWORK et IWORK, et aucun
!>          message d'erreur lié à LWORK ou LRWORK ou LIWORK n'est émis
!>          par XERBLA.
!
>

IWOR

K 
!>          IWORK est un tableau INTEGER, dimension (MAX(1,LIWORK))
!>          À la sortie, si INFO = 0, IWORK(1) renvoie le LIWORK requis.
!
>

LIWOR

K 
!>          LIWORK est INTEGER
!>          La dimension du tableau IWORK.
!>          Si JOBZ  = 'N' ou N <= 1, LIWORK doit être au moins 1.
!>          Si JOBZ  = 'V' et N > 1, LIWORK doit être au moins 3 + 5*N.
!>
!>          Si LIWORK = -1, une interrogation d'espace de travail est
!>          supposée ; la routine calcule uniquement les tailles requises
!>          des tableaux WORK, RWORK et IWORK, renvoie ces valeurs comme
!>          premières entrées des tableaux WORK, RWORK et IWORK, et aucun
!>          message d'erreur lié à LWORK ou LRWORK ou LIWORK n'est émis
!>          par XERBLA.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale.
!>          > 0 :  si INFO = i, l'algorithme n'a pas convergé ; i
!>                éléments hors diagonale d'une forme tridiagonale
!>                intermédiaire n'ont pas convergé vers zéro.
!
>
