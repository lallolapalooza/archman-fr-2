subroutine chetrd (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) d, real, dimension( * ) e, complex, dimension( * ) tau, complex, dimension( * ) work, integer lwork, integer

info)

CHETR

D

But

: 
!>
!> CHETRD réduit une matrice hermitienne complexe A à une forme
!> tridiagonale réelle symétrique T par une transformation de
!> similarité unitaire :
!> Q**H * A * Q = T.
!
>

Paramètre

s
UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U' : le triangle supérieur de A est stocké ;
!>          = 'L' : le triangle inférieur de A est stocké.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          À l'entrée, la matrice hermitienne A.  Si UPLO = 'U', la
!>          partie triangulaire supérieure de A de taille N par N
!>          contient la partie triangulaire supérieure de la matrice A,
!>          et la partie strictement triangulaire inférieure de A n'est
!>          pas référencée.  Si UPLO = 'L', la partie triangulaire
!>          inférieure de A de taille N par N contient la partie
!>          triangulaire inférieure de la matrice A, et la partie
!>          strictement triangulaire supérieure de A n'est pas
!>          référencée.
!>          À la sortie, si UPLO = 'U', la diagonale et la première
!>          surdiagonale de A sont écrasées par les éléments
!>          correspondants de la matrice tridiagonale T, et les éléments
!>          au-dessus de la première surdiagonale, avec le tableau TAU,
!>          représentent la matrice unitaire Q comme un produit de
!>          réflecteurs élémentaires ; si UPLO = 'L', la diagonale et la
!>          première sous-diagonale de A sont écrasées par les éléments
!>          correspondants de la matrice tridiagonale T, et les éléments
!>          en dessous de la première sous-diagonale, avec le tableau
!>          TAU, représentent la matrice unitaire Q comme un produit de
!>          réflecteurs élémentaires. Voir Détails supplémentaires.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!
>

D 
!>          D est un tableau REAL, dimension (N)
!>          Les éléments diagonaux de la matrice tridiagonale T :
!>          D(i) = A(i,i).
!
>

E 
!>          E est un tableau REAL, dimension (N-1)
!>          Les éléments hors diagonale de la matrice tridiagonale T :
!>          E(i) = A(i,i+1) si UPLO = 'U', E(i) = A(i+1,i) si UPLO = 'L'.
!
>

TA

U 
!>          TAU est un tableau COMPLEX, dimension (N-1)
!>          Les facteurs scalaires des réflecteurs élémentaires (voir
!>          Détails supplémentaires).
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie le LWORK optimal.
!
>

LWOR

K 
!>          LWORK est INTEGER
!>          La dimension du tableau WORK.  LWORK >= 1.
!>          Pour des performances optimales LWORK >= N*NB, où NB est la
!>          taille de bloc optimale.
!>
!>          Si LWORK = -1, une interrogation d'espace de travail est
!>          supposée ; la routine calcule uniquement la taille optimale
!>          du tableau WORK, renvoie cette valeur comme première entrée
!>          du tableau WORK, et aucun message d'erreur lié à LWORK n'est
!>          émis par XERBLA.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur
!>          illégale
!
>

Détails supplémentaires

: 
!>
!>  Si UPLO = 'U', la matrice Q est représentée comme un produit de
!>  réflecteurs élémentaires
!>
!>     Q = H(n-1) . . . H(2) H(1).
!>
!>  Chaque H(i) a la forme
!>
!>     H(i) = I - tau * v * v**H
!>
!>  où tau est un scalaire complexe, et v est un vecteur complexe avec
!>  v(i+1:n) = 0 et v(i) = 1 ; v(1:i-1) est stocké à la sortie dans
!>  A(1:i-1,i+1), et tau dans TAU(i).
!>
!>  Si UPLO = 'L', la matrice Q est représentée comme un produit de
!>  réflecteurs élémentaires
!>
!>     Q = H(1) H(2) . . . H(n-1).
!>
!>  Chaque H(i) a la forme
!>
!>     H(i) = I - tau * v * v**H
!>
!>  où tau est un scalaire complexe, et v est un vecteur complexe avec
!>  v(1:i) = 0 et v(i+1) = 1 ; v(i+2:n) est stocké à la sortie dans
!>  A(i+2:n,i), et tau dans TAU(i).
!>
!>  Le contenu de A à la sortie est illustré par les exemples suivants
!>  avec n = 5 :
!>
!>  si UPLO = 'U' :                     si UPLO = 'L' :
!>
!>    (  d   e   v2  v3  v4 )              (  d                  )
!>    (      d   e   v3  v4 )              (  e   d              )
!>    (          d   e   v4 )              (  v1  e   d          )
!>    (              d   e  )              (  v1  v2  e   d      )
!>    (                  d  )              (  v1  v2  v3  e   d  )
!>
!>  où d et e désignent les éléments diagonaux et hors diagonale de T,
!>  et vi désigne un élément du vecteur définissant H(i).
!
>
