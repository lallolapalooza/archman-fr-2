subroutine cher (character uplo, integer n, real alpha, complex, dimension(*) x, integer incx, complex, dimension(lda,*) a, integer

lda)

CHE

R

But

: 
!>
!> CHER   effectue l'opération de rang 1 hermitienne
!>
!>    A := alpha*x*x**H + A,
!>
!> où alpha est un scalaire réel, x est un vecteur de n éléments et A est
!> une matrice hermitienne de taille n par n.
!
>

Paramètre

s
UPLO 
!>          UPLO est CHARACTER*1
!>           À l'entrée, UPLO spécifie si la partie triangulaire supérieure
!>           ou inférieure du tableau A doit être référencée comme suit :
!>
!>              UPLO = 'U' ou 'u'   Seule la partie triangulaire
!>                                  supérieure de A doit être référencée.
!>
!>              UPLO = 'L' ou 'l'   Seule la partie triangulaire
!>                                  inférieure de A doit être référencée.
!
>

N 
!>          N est INTEGER
!>           À l'entrée, N spécifie l'ordre de la matrice A.
!>           N doit être au moins égal à zéro.
!
>

ALPH

A 
!>          ALPHA est REAL
!>           À l'entrée, ALPHA spécifie le scalaire alpha.
!
>

X 
!>          X est un tableau COMPLEX, de dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           Avant l'entrée, le tableau incrémenté X doit contenir le
!>           vecteur x de n éléments.
!
>

INC

X 
!>          INCX est INTEGER
!>           À l'entrée, INCX spécifie l'incrément pour les éléments de
!>           X. INCX ne doit pas être égal à zéro.
!
>

A 
!>          A est un tableau COMPLEX, de dimension ( LDA, N )
!>           Avant l'entrée avec UPLO = 'U' ou 'u', la partie triangulaire
!>           supérieure de taille n par n en tête du tableau A doit
!>           contenir la partie triangulaire supérieure de la matrice
!>           hermitienne et la partie strictement inférieure de A n'est pas
!>           référencée. À la sortie, la partie triangulaire supérieure du
!>           tableau A est écrasée par la partie triangulaire supérieure de
!>           la matrice mise à jour.
!>           Avant l'entrée avec UPLO = 'L' ou 'l', la partie triangulaire
!>           inférieure de taille n par n en tête du tableau A doit
!>           contenir la partie triangulaire inférieure de la matrice
!>           hermitienne et la partie strictement supérieure de A n'est pas
!>           référencée. À la sortie, la partie triangulaire inférieure du
!>           tableau A est écrasée par la partie triangulaire inférieure de
!>           la matrice mise à jour.
!>           Notez que les parties imaginaires des éléments diagonaux ne
!>           doivent pas être définies, elles sont supposées être égales à
!>           zéro et à la sortie elles sont définies à zéro.
!
>

LD

A 
!>          LDA est INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle que
!>           déclarée dans le (sous-)programme appelant. LDA doit être au
!>           moins max( 1, n ).
!
>
