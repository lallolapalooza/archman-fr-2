subroutine cheev_2stage (character jobz, character uplo, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) w, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, integer

info)

CHEEV_2STAGE calcule les valeurs propres et, en option, les vecteurs propres gauche et/ou droit pour les matrices

HE

But

: 
!>
!> CHEEV_2STAGE calcule toutes les valeurs propres et, en option, les vecteurs propres d'une
!> matrice hermitienne complexe A en utilisant la technique en 2 étapes pour
!> la réduction en tridiagonale.
!
>

Paramètre

s
JOBZ 
!>          JOBZ est CHARACTER*1
!>          = 'N' :  Calcule les valeurs propres uniquement ;
!>          = 'V' :  Calcule les valeurs propres et les vecteurs propres.
!>                  Non disponible dans cette version.
!
>

UPL

O 
!>          UPLO est CHARACTER*1
!>          = 'U' :  Le triangle supérieur de A est stocké ;
!>          = 'L' :  Le triangle inférieur de A est stocké.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA, N)
!>          À l'entrée, la matrice hermitienne A.  Si UPLO = 'U', la
!>          partie triangulaire supérieure N×N de A contient la
!>          partie triangulaire supérieure de la matrice A.  Si UPLO = 'L',
!>          la partie triangulaire inférieure N×N de A contient
!>          la partie triangulaire inférieure de la matrice A.
!>          À la sortie, si JOBZ = 'V', alors si INFO = 0, A contient les
!>          vecteurs propres orthonormaux de la matrice A.
!>          Si JOBZ = 'N', alors à la sortie le triangle inférieur (si UPLO='L')
!>          ou le triangle supérieur (si UPLO='U') de A, y compris la
!>          diagonale, est détruit.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!
>

W 
!>          W est un tableau REAL, dimension (N)
!>          Si INFO = 0, les valeurs propres en ordre croissant.
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie le LWORK optimal.
!
>

LWOR

K 
!>          LWORK est INTEGER
!>          La longueur du tableau WORK. LWORK >= 1, quand N <= 1 ;
!>          sinon
!>          Si JOBZ = 'N' et N > 1, LWORK doit être interrogé.
!>                                   LWORK = MAX(1, dimension) où
!>                                   dimension = max(stage1,stage2) + (KD+1)*N + N
!>                                             = N*KD + N*max(KD+1,FACTOPTNB)
!>                                               + max(2*KD*KD, KD*NTHREADS)
!>                                               + (KD+1)*N + N
!>                                   où KD est la taille de bloc de la réduction,
!>                                   FACTOPTNB est le bloc utilisé par l'algorithme QR ou LQ,
!>                                   généralement FACTOPTNB=128 est un bon choix
!>                                   NTHREADS est le nombre de fils d'exécution utilisé quand
!>                                   la compilation openMP est activée, sinon =1.
!>          Si JOBZ = 'V' et N > 1, LWORK doit être interrogé. Pas encore disponible
!>
!>          Si LWORK = -1, alors une interrogation d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!
>

RWOR

K 
!>          RWORK est un tableau REAL, dimension (max(1, 3*N-2))
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 :  si INFO = i, l'algorithme n'a pas convergé ; i
!>                éléments hors diagonale d'une forme tridiagonale
!>                intermédiaire n'ont pas convergé vers zéro.
!
>

subroutine zheev_2stage (character jobz, character uplo, integer n, complex*16, dimension( lda, * ) a, inte ZHEEV_2STAGE(3)

NAME

ZHEEV_2STAGE

calcule les valeurs propres et, optionnellement, les vecteurs propres gauches et/ou droits pour les matrices HE

SYNOPSIS

SUBROUTINE ZHEEV_2STAGE(JOBZ, UPLO, N, A, LDA, W, WORK, LWORK, RWORK, INFO)

  CHARACTER JOBZ
  CHARACTER UPLO
  INTEGER N
  COMPLEX*16 A(LDA, *)
  INTEGER LDA
  DOUBLE PRECISION W(*)
  COMPLEX*16 WORK(*)
  INTEGER LWORK
  DOUBLE PRECISION RWORK(*)
  INTEGER INFO

Detailed Description

ZHEEV_2STAGE calcule les valeurs propres et, optionnellement, les vecteurs propres gauches et/ou droits pour les matrices HE

Objectif :

!>
!> ZHEEV_2STAGE calcule toutes les valeurs propres et, optionnellement, les vecteurs propres d'une matrice complexe hermitienne A en utilisant la technique 2stage pour la réduction à la forme tridiagonale.
!>

Paramètres

JOBZ 
!>          JOBZ est CHARACTER*1
!>          = 'N':  Calcule uniquement les valeurs propres ;
!>          = 'V':  Calcule les valeurs propres et les vecteurs propres.
!>                  Non disponible dans cette version.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U':  Le triangle supérieur de A est stocké ;
!>          = 'L':  Le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

A

!>          A est un tableau COMPLEX*16, de dimension (LDA, N)
!>          En entrée, la matrice hermitienne A. Si UPLO = 'U', la partie triangulaire supérieure N-par-N de tête de A contient la partie triangulaire supérieure de la matrice A. Si UPLO = 'L', la partie triangulaire inférieure N-par-N de tête de A contient la partie triangulaire inférieure de la matrice A.
!>          En sortie, si JOBZ = 'V', alors si INFO = 0, A contient les vecteurs propres orthonormés de la matrice A.
!>          Si JOBZ = 'N', alors en sortie le triangle inférieur (si UPLO='L') ou le triangle supérieur (si UPLO='U') de A, y compris la diagonale, est détruit.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

W

!>          W est un tableau DOUBLE PRECISION, de dimension (N)
!>          Si INFO = 0, les valeurs propres dans l'ordre croissant.
!>

WORK

!>          WORK est un tableau COMPLEX*16, de dimension (MAX(1,LWORK))
!>          En sortie, si INFO = 0, WORK(1) renvoie la valeur LWORK optimale.
!>

LWORK

!>          LWORK est INTEGER
!>          La longueur du tableau WORK. LWORK >= 1, lorsque N <= 1 ;
!>          sinon
!>          Si JOBZ = 'N' et N > 1, LWORK doit être interrogé.
!>                                   LWORK = MAX(1, dimension) où
!>                                   dimension = max(stage1,stage2) + (KD+1)*N + N
!>                                             = N*KD + N*max(KD+1,FACTOPTNB)
!>                                               + max(2*KD*KD, KD*NTHREADS)
!>                                               + (KD+1)*N + N
!>                                   où KD est la taille du bloc de la réduction,
!>                                   FACTOPTNB est le bloc utilisé par l'algorithme QR ou LQ,
!>                                   généralement FACTOPTNB=128 est un bon choix
!>                                   NTHREADS est le nombre de threads utilisés lorsque la compilation OpenMP est activée, sinon =1.
!>          Si JOBZ = 'V' et N > 1, LWORK doit être interrogé. Pas encore disponible
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, de dimension (max(1, 3*N-2))
!>

INFO

!>          INFO est INTEGER
!>          = 0:  succès de l'exécution
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0:  si INFO = i, l'algorithme n'a pas convergé ; i
!>                éléments hors diagonale d'une forme tridiagonale intermédiaire
!>                n'ont pas convergé vers zéro.
!>
