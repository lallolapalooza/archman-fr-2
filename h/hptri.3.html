subroutine chptri (character uplo, integer n, complex, dimension( * ) ap, integer, dimension( * ) ipiv, complex, dimension( * ) work, integer

info)

CHPTR

I

But

: 
!>
!> CHPTRI calcule l'inverse d'une matrice hermitienne indéfinie complexe
!> A en stockage compact en utilisant la factorisation A = U*D*U**H ou
!> A = L*D*L**H calculée par CHPTRF.
!
>

Paramètre

s
UPLO 
!>          UPLO est CHARACTER*1
!>          Indique si les détails de la factorisation sont stockés
!>          sous forme de matrice triangulaire supérieure ou inférieure.
!>          = 'U' : triangulaire supérieure, la forme est A = U*D*U**H ;
!>          = 'L' : triangulaire inférieure, la forme est A = L*D*L**H.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

A

P 
!>          AP est un tableau COMPLEX, dimension (N*(N+1)/2)
!>          À l'entrée, la matrice diagonale par blocs D et les
!>          multiplicateurs utilisés pour obtenir le facteur U ou L
!>          tels que calculés par CHPTRF, stockés sous forme de
!>          matrice triangulaire compacte.
!>
!>          À la sortie, si INFO = 0, l'inverse (hermitienne) de la
!>          matrice d'origine, stocké sous forme de matrice triangulaire
!>          compacte. La j-ième colonne de inv(A) est stockée dans le
!>          tableau AP comme suit :
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = inv(A)(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L',
!>             AP(i + (j-1)*(2n-j)/2) = inv(A)(i,j) pour j<=i<=n.
!
>

IPI

V 
!>          IPIV est un tableau INTEGER, dimension (N)
!>          Détails des échanges et de la structure par blocs de D
!>          tels que déterminés par CHPTRF.
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (N)
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 : si INFO = i, D(i,i) = 0 ; la matrice est singulière et son
!>               inverse n'a pas pu être calculé.
!
>
