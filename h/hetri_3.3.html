subroutine chetri_3 (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) e, integer, dimension( * ) ipiv, complex, dimension( * ) work, integer lwork, integer

info)

CHETRI_

3

But

: 
!> CHETRI_3 calcule l'inverse d'une matrice hermitienne indéfinie
!> complexe A en utilisant la factorisation calculée par CHETRF_RK ou CHETRF_BK :
!>
!>     A = P*U*D*(U**H)*(P**T) ou A = P*L*D*(L**H)*(P**T),
!>
!> où U (ou L) est une matrice triangulaire supérieure (ou inférieure)
!> unitaire, U**H (ou L**H) est le conjugué de U (ou L), P est une matrice
!> de permutation, P**T est la transposée de P, et D est hermitienne et
!> diagonale par blocs avec des blocs diagonaux de taille 1 par 1 et 2 par 2.
!>
!> CHETRI_3 définit la dimension principale de l'espace de travail avant
!> d'appeler CHETRI_3X qui calcule réellement l'inverse. Il s'agit de la
!> version par blocs de l'algorithme, appelant les BLAS de niveau 3.
!
>

Paramètre

s
UPLO 
!>          UPLO est CHARACTER*1
!>          Indique si les détails de la factorisation sont
!>          stockés sous forme de matrice triangulaire supérieure ou inférieure.
!>          = 'U' : le triangle supérieur de A est stocké ;
!>          = 'L' : le triangle inférieur de A est stocké.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A. N >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          À l'entrée, la diagonale de la matrice diagonale par blocs D et
!>          les facteurs U ou L tels que calculés par CHETRF_RK et CHETRF_BK :
!>            a) UNIQUEMENT les éléments diagonaux de la matrice hermitienne
!>               diagonale par blocs D sur la diagonale de A, c'est-à-dire
!>               D(k,k) = A(k,k) ;
!>               (les éléments superdiagonaux (ou subdiagonaux) de D
!>                doivent être fournis à l'entrée dans le tableau E), et
!>            b) Si UPLO = 'U' : le facteur U dans la partie superdiagonale de A.
!>               Si UPLO = 'L' : le facteur L dans la partie subdiagonale de A.
!>
!>          À la sortie, si INFO = 0, l'inverse hermitienne de la matrice
!>          originale.
!>             Si UPLO = 'U' : la partie triangulaire supérieure de l'inverse
!>             est formée et la partie de A sous la diagonale n'est pas
!>             référencée ;
!>             Si UPLO = 'L' : la partie triangulaire inférieure de l'inverse
!>             est formée et la partie de A au-dessus de la diagonale n'est pas
!>             référencée.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,N).
!
>

E 
!>          E est un tableau COMPLEX, dimension (N)
!>          À l'entrée, contient les éléments superdiagonaux (ou subdiagonaux)
!>          de la matrice hermitienne diagonale par blocs D
!>          avec des blocs diagonaux de taille 1 par 1 ou 2 par 2, où
!>          Si UPLO = 'U' : E(i) = D(i-1,i), i=2:N, E(1) non référencé ;
!>          Si UPLO = 'L' : E(i) = D(i+1,i), i=1:N-1, E(N) non référencé.
!>
!>          REMARQUE : Pour un bloc diagonal 1 par 1 D(k), où
!>          1 <= k <= N, l'élément E(k) n'est pas référencé dans les deux cas
!>          UPLO = 'U' ou UPLO = 'L'.
!
>

IPI

V 
!>          IPIV est un tableau INTEGER, dimension (N)
!>          Détails des interchanges et de la structure par blocs de D
!>          tels que déterminés par CHETRF_RK ou CHETRF_BK.
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (MAX(1,LWORK)).
!>          À la sortie, si INFO = 0, WORK(1) renvoie le LWORK optimal.
!
>

LWOR

K 
!>          LWORK est INTEGER
!>          La longueur de WORK.
!>          Si N = 0, LWORK >= 1, sinon LWORK >= (N+NB+1)*(NB+3).
!>
!>          Si LWORK = -1, une interrogation d'espace de travail est supposée ;
!>          la routine calcule uniquement la taille optimale du tableau
!>          WORK, renvoie cette valeur comme première entrée du tableau
!>          WORK, et aucun message d'erreur lié à LWORK n'est émis par XERBLA.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 : si INFO = i, D(i,i) = 0 ; la matrice est singulière et son
!>               inverse n'a pas pu être calculé.
!
>
