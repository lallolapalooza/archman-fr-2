subroutine chesv_aa_2stage (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tb, integer ltb, integer, dimension( * ) ipiv, integer, dimension( * ) ipiv2, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, integer lwork, integer

info)

CHESV_AA_2STAGE calcule la solution du système d'équations linéaires A * X = B pour les matrices

HE

But

: 
!>
!> CHESV_AA_2STAGE calcule la solution d'un système d'équations
!> linéaires complexes
!>    A * X = B,
!> où A est une matrice hermitienne N par N et X et B sont des matrices
!> N par NRHS.
!>
!> L'algorithme en 2 étapes de Aasen est utilisé pour factoriser A comme
!>    A = U**H * T * U,  si UPLO = 'U', ou
!>    A = L * T * L**H,  si UPLO = 'L',
!> où U (ou L) est un produit de matrices de permutation et de matrices
!> triangulaires supérieures (inférieures) unitaires, et T est hermitienne
!> et bande. La matrice T est ensuite factorisée LU avec pivotage partiel.
!> La forme factorisée de A est ensuite utilisée pour résoudre le système
!> d'équations A * X = B.
!>
!> Il s'agit de la version par blocs de l'algorithme, faisant appel aux
!> BLAS de niveau 3.
!
>

Paramètre

s
UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U' : le triangle supérieur de A est stocké ;
!>          = 'L' : le triangle inférieur de A est stocké.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

NRH

S 
!>          NRHS est INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de
!>          colonnes de la matrice B.  NRHS >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          À l'entrée, la matrice hermitienne A.  Si UPLO = 'U', la
!>          partie triangulaire supérieure N par N de A contient la partie
!>          triangulaire supérieure de la matrice A, et la partie
!>          triangulaire strictement inférieure de A n'est pas référencée.
!>          Si UPLO = 'L', la partie triangulaire inférieure N par N de A
!>          contient la partie triangulaire inférieure de la matrice A, et
!>          la partie triangulaire strictement supérieure de A n'est pas
!>          référencée.
!>
!>          À la sortie, L est stocké sous (ou au-dessus des) les blocs
!>          sous-diagonaux, lorsque UPLO vaut 'L' (ou 'U').
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!
>

T

B 
!>          TB est un tableau COMPLEX, dimension (MAX(1,LTB)).
!>          À la sortie, les détails de la factorisation LU de la matrice
!>          bande.
!
>

LT

B 
!>          LTB est INTEGER
!>          La taille du tableau TB. LTB >= MAX(1,4*N), utilisée
!>          internement pour sélectionner NB tel que LTB >= (3*NB+1)*N.
!>
!>          Si LTB = -1, une interrogation d'espace de travail est
!>          supposée ; la routine calcule uniquement la taille optimale de
!>          LTB, retourne cette valeur comme première entrée de TB, et
!>          aucun message d'erreur lié à LTB n'est émis par XERBLA.
!
>

IPI

V 
!>          IPIV est un tableau INTEGER, dimension (N)
!>          À la sortie, il contient les détails des interchanges, c'est-
!>          à-dire que la ligne et la colonne k de A ont été interchangées
!>          avec la ligne et la colonne IPIV(k).
!
>

IPIV

2 
!>          IPIV2 est un tableau INTEGER, dimension (N)
!>          À la sortie, il contient les détails des interchanges, c'est-
!>          à-dire que la ligne et la colonne k de T ont été interchangées
!>          avec la ligne et la colonne IPIV(k).
!
>

B 
!>          B est un tableau COMPLEX, dimension (LDB,NRHS)
!>          À l'entrée, la matrice du second membre B.
!>          À la sortie, la matrice solution X.
!
>

LD

B 
!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!
>

WOR

K 
!>          WORK est un espace de travail COMPLEX de taille (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) retourne le LWORK optimal.
!
>

LWOR

K 
!>          LWORK est INTEGER
!>          La taille de WORK. LWORK >= MAX(1,N), utilisée internement
!>          pour sélectionner NB tel que LWORK >= N*NB.
!>
!>          Si LWORK = -1, une interrogation d'espace de travail est
!>          supposée ; la routine calcule uniquement la taille optimale du
!>          tableau WORK, retourne cette valeur comme première entrée du
!>          tableau WORK, et aucun message d'erreur lié à LWORK n'est émis
!>          par XERBLA.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur
!>                 illégale.
!>          > 0 :  si INFO = i, la factorisation LU bande a échoué sur la
!>                 i-ème colonne
!
>

!> La forme factorisée de A est ensuite utilisée pour résoudre le système
!> d'équations A * X = B.
