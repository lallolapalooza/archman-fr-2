subroutine chetri_3x (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) e, integer, dimension( * ) ipiv, complex, dimension( n+nb+1, * ) work, integer nb, integer

info)

CHETRI_3

X

But

: 
!> CHETRI_3X calcule l'inverse d'une matrice hermitienne indéfinie
!> complexe A en utilisant la factorisation calculée par CHETRF_RK ou CHETRF_BK :
!>
!>     A = P*U*D*(U**H)*(P**T) ou A = P*L*D*(L**H)*(P**T),
!>
!> où U (ou L) est une matrice triangulaire supérieure (ou inférieure) unitaire,
!> U**H (ou L**H) est le conjugué de U (ou L), P est une matrice de permutation,
!> P**T est la transposée de P, et D est hermitienne et bloc diagonale
!> avec des blocs diagonaux de taille 1 par 1 et 2 par 2.
!>
!> Ceci est la version par blocs de l'algorithme, appelant les BLAS de niveau 3.
!
>

Paramètre

s
UPLO 
!>          UPLO est CHARACTER*1
!>          Indique si les détails de la factorisation sont
!>          stockés sous forme de matrice triangulaire supérieure ou inférieure.
!>          = 'U' : le triangle supérieur de A est stocké ;
!>          = 'L' : le triangle inférieur de A est stocké.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          À l'entrée, la diagonale de la matrice bloc diagonale D et
!>          les facteurs U ou L tels que calculés par CHETRF_RK et CHETRF_BK :
!>            a) UNIQUEMENT les éléments diagonaux de la matrice hermitienne
!>               bloc diagonale D sur la diagonale de A, c.-à-d. D(k,k) = A(k,k) ;
!>               (les éléments superdiagonaux (ou subdiagonaux) de D
!>                doivent être fournis à l'entrée dans le tableau E), et
!>            b) Si UPLO = 'U' : le facteur U dans la partie superdiagonale de A.
!>               Si UPLO = 'L' : le facteur L dans la partie subdiagonale de A.
!>
!>          À la sortie, si INFO = 0, l'inverse hermitienne de la matrice
!>          originale.
!>             Si UPLO = 'U' : la partie triangulaire supérieure de l'inverse
!>             est formée et la partie de A sous la diagonale n'est pas
!>             référencée ;
!>             Si UPLO = 'L' : la partie triangulaire inférieure de l'inverse
!>             est formée et la partie de A au-dessus de la diagonale n'est pas
!>             référencée.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!
>

E 
!>          E est un tableau COMPLEX, dimension (N)
!>          À l'entrée, contient les éléments superdiagonaux (ou subdiagonaux)
!>          de la matrice hermitienne bloc diagonale D
!>          avec des blocs diagonaux de taille 1 par 1 ou 2 par 2, où
!>          Si UPLO = 'U' : E(i) = D(i-1,i), i=2:N, E(1) non référencé ;
!>          Si UPLO = 'L' : E(i) = D(i+1,i), i=1:N-1, E(N) non référencé.
!>
!>          REMARQUE : Pour un bloc diagonal D(k) de taille 1 par 1, où
!>          1 <= k <= N, l'élément E(k) n'est pas référencé dans les deux cas
!>          UPLO = 'U' ou UPLO = 'L'.
!
>

IPI

V 
!>          IPIV est un tableau INTEGER, dimension (N)
!>          Détails des interchanges et de la structure de blocs de D
!>          tels que déterminés par CHETRF_RK ou CHETRF_BK.
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (N+NB+1,NB+3).
!
>

N

B 
!>          NB est INTEGER
!>          Taille des blocs.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 : si INFO = i, D(i,i) = 0 ; la matrice est singulière et son
!>               inverse n'a pas pu être calculé.
!
>

!> ZSYTRI_3X calcule l'inverse d'une matrice symétrique indéfinie
!> complexe A en utilisant la factorisation calculée par ZSYTRF_RK ou ZSYTRF_BK :
!>
!>     A = P*U*D*(U**T)*(P**T) ou A = P*L*D*(L**T)*(P**T),
!>
!> où U (ou L) est une matrice triangulaire supérieure (ou inférieure)
!> unitaire, U**T (ou L**T) est la transposée de U (ou L), P est une
!> matrice de permutation, P**T est la transposée de P, et D est
!> symétrique et bloc diagonale avec des blocs diagonaux 1-par-1 et
!> 2-par-2.
!>
!> Il s'agit de la version par blocs de l'algorithme, appelant les
!> BLAS de niveau 3.
!
>

Paramètre

s

!>          UPLO est de type CHARACTER*1
!>          Indique si les détails de la factorisation sont
!>          stockés sous forme de matrice triangulaire supérieure
!>          ou inférieure.
!>          = 'U' : le triangle supérieur de A est stocké ;
!>          = 'L' : le triangle inférieur de A est stocké.
!
>

N 
!>          N est de type INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

A 
!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>          À l'entrée, la diagonale de la matrice bloc diagonale D et
!>          les facteurs U ou L tels que calculés par ZSYTRF_RK et ZSYTRF_BK :
!>            a) UNIQUEMENT les éléments diagonaux de la matrice
!>               symétrique bloc diagonale D sur la diagonale de A,
!>               c.-à-d. D(k,k) = A(k,k) ;
!>               (les éléments superdiagonaux (ou subdiagonaux) de D
!>                doivent être fournis à l'entrée dans le tableau E), et
!>            b) Si UPLO = 'U' : facteur U dans la partie superdiagonale de A.
!>               Si UPLO = 'L' : facteur L dans la partie subdiagonale de A.
!>
!>          À la sortie, si INFO = 0, l'inverse symétrique de la matrice
!>          originale.
!>             Si UPLO = 'U' : la partie triangulaire supérieure de
!>             l'inverse est formée et la partie de A sous la diagonale
!>             n'est pas référencée ;
!>             Si UPLO = 'L' : la partie triangulaire inférieure de
!>             l'inverse est formée et la partie de A au-dessus de la
