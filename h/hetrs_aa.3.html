sous-routine chetrs_aa (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, integer lwork, integer

info)

CHETRS_A

A

But

: 
!>
!> CHETRS_AA résout un système d'équations linéaires A*X = B avec une
!> matrice hermitienne complexe A en utilisant la factorisation A = U**H*T*U ou
!> A = L*T*L**H calculée par CHETRF_AA.
!
>

Paramètre

s
UPLO 
!>          UPLO est CHARACTER*1
!>          Indique si les détails de la factorisation sont stockés
!>          sous forme de matrice triangulaire supérieure ou inférieure.
!>          = 'U' :  triangulaire supérieur, la forme est A = U**H*T*U ;
!>          = 'L' :  triangulaire inférieur, la forme est A = L*T*L**H.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

NRH

S 
!>          NRHS est INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          Détails des facteurs calculés par CHETRF_AA.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!
>

IPI

V 
!>          IPIV est un tableau INTEGER, dimension (N)
!>          Détails des interchanges tels que calculés par CHETRF_AA.
!
>

B 
!>          B est un tableau COMPLEX, dimension (LDB,NRHS)
!>          À l'entrée, la matrice du second membre B.
!>          À la sortie, la matrice solution X.
!
>

LD

B 
!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (MAX(1,LWORK))
!
>

LWOR

K 
!>          LWORK est INTEGER
!>          La dimension du tableau WORK.
!>          Si MIN(N,NRHS) = 0, LWORK >= 1, sinon LWORK >= 3*N-2.
!>
!>          Si LWORK = -1, une interrogation d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille minimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!
>
