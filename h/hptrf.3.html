subroutine chptrf (character uplo, integer n, complex, dimension( * ) ap, integer, dimension( * ) ipiv, integer

info)

CHPTR

F

But

: 
!>
!> CHPTRF calcule la factorisation d'une matrice hermitienne complexe
!> A au format emballé en utilisant la méthode de pivotage diagonal
!> Bunch-Kaufman :
!>
!>    A = U*D*U**H  ou  A = L*D*L**H
!>
!> où U (ou L) est un produit de matrices de permutation et de matrices
!> triangulaires supérieures (inférieures) unitaires, et D est
!> hermitienne et bloc diagonale avec des blocs diagonaux de taille
!> 1-par-1 et 2-par-2.
!
>

Paramètre

s
UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U' : le triangle supérieur de A est stocké ;
!>          = 'L' : le triangle inférieur de A est stocké.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

A

P 
!>          AP est un tableau COMPLEX, dimension (N*(N+1)/2)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice
!>          hermitienne A, emballé colonne par colonne dans un tableau
!>          linéaire.  La j-ième colonne de A est stockée dans le tableau
!>          AP comme suit :
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L', AP(i + (j-1)*(2n-j)/2) = A(i,j) pour j<=i<=n.
!>
!>          À la sortie, la matrice bloc diagonale D et les multiplicateurs
!>          utilisés pour obtenir le facteur U ou L, stockés sous forme de
!>          matrice triangulaire emballée en écrasant A (voir ci-dessous
!>          pour plus de détails).
!
>

IPI

V 
!>          IPIV est un tableau INTEGER, dimension (N)
!>          Détails des interchanges et de la structure bloc de D.
!>          Si IPIV(k) > 0, alors les lignes et colonnes k et IPIV(k)
!>          ont été interchangées et D(k,k) est un bloc diagonal de
!>          taille 1-par-1.
!>          Si UPLO = 'U' et IPIV(k) = IPIV(k-1) < 0, alors les lignes
!>          et colonnes k-1 et -IPIV(k) ont été interchangées et
!>          D(k-1:k,k-1:k) est un bloc diagonal de taille 2-par-2.
!>          Si UPLO = 'L' et IPIV(k) = IPIV(k+1) < 0, alors les lignes
!>          et colonnes k+1 et -IPIV(k) ont été interchangées et
!>          D(k:k+1,k:k+1) est un bloc diagonal de taille 2-par-2.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur
!>                illégale
!>          > 0 : si INFO = i, D(i,i) est exactement nul.  La
!>               factorisation a été complétée, mais la matrice bloc
!>               diagonale D est exactement singulière, et une division
!>               par zéro se produira si elle est utilisée pour résoudre
!>               un système d'équations.
!
>

Détails supplémentaires

: 
!>
!>  Si UPLO = 'U', alors A = U*D*U**H, où
!>     U = P(n)*U(n)* ... *P(k)U(k)* ...,
!>  c.-à-d. U est un produit de termes P(k)*U(k), où k décroît de n à
!>  1 par pas de 1 ou 2, et D est une matrice bloc diagonale avec des
!>  blocs diagonaux de taille 1-par-1 et 2-par-2 D(k).  P(k) est une
!>  matrice de permutation telle que définie par IPIV(k), et U(k) est
!>  une matrice triangulaire supérieure unitaire, telle que si le bloc
!>  diagonal D(k) est d'ordre s (s = 1 ou 2), alors
!>
!>             (   I    v    0   )   k-s
!>     U(k) =  (   0    I    0   )   s
!>             (   0    0    I   )   n-k
!>                k-s   s   n-k
!>
!>  Si s = 1, D(k) écrase A(k,k), et v écrase A(1:k-1,k).
!>  Si s = 2, le triangle supérieur de D(k) écrase A(k-1,k-1),
!>  A(k-1,k), et A(k,k), et v écrase A(1:k-2,k-1:k).
!>
!>  Si UPLO = 'L', alors A = L*D*L**H, où
!>     L = P(1)*L(1)* ... *P(k)*L(k)* ...,
!>  c.-à-d. L est un produit de termes P(k)*L(k), où k croît de 1 à
!>  n par pas de 1 ou 2, et D est une matrice bloc diagonale avec des
!>  blocs diagonaux de taille 1-par-1 et 2-par-2 D(k).  P(k) est une
!>  matrice de permutation telle que définie par IPIV(k), et L(k) est
!>  une matrice triangulaire inférieure unitaire, telle que si le bloc
!>  diagonal D(k) est d'ordre s (s = 1 ou 2), alors
!>
!>             (   I    0     0   )  k-1
!>     L(k) =  (   0    I     0   )  s
!>             (   0    v     I   )  n-k-s+1
!>                k-1   s  n-k-s+1
!>
!>  Si s = 1, D(k) écrase A(k,k), et v écrase A(k+1:n,k).
!>  Si s = 2, le triangle inférieur de D(k) écrase A(k,k), A(k+1,k),
!>  et A(k+1,k+1), et v écrase A(k+2:n,k:k+1).
!
>

Contributeurs

:
J. Lewis, Boeing Computer Services
Company

!>

!>          À l'entrée, le triangle supérieur ou inférieur de la matrice
!>          hermitienne A, stocké colonne par colonne dans un tableau
!>          linéaire. La j-ième colonne de A est stockée dans le tableau
!>          AP comme suit :
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L', AP(i + (j-1)*(2n-j)/2) = A(i,j) pour j<=i<=n.
!>
