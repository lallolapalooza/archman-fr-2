subroutine checon (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real anorm, real rcond, complex, dimension( * ) work, integer

info)

CHECO

N

But

: 
!>
!> CHECON estime l'inverse du nombre de conditionnement d'une matrice
!> hermitienne complexe A en utilisant la factorisation A = U*D*U**H ou
!> A = L*D*L**H calculée par CHETRF.
!>
!> Une estimation de norm(inv(A)) est obtenue, et l'inverse du
!> nombre de conditionnement est calculé comme RCOND = 1 / (ANORM * norm(inv(A))).
!
>

Paramètre

s
UPLO 
!>          UPLO est CHARACTER*1
!>          Indique si les détails de la factorisation sont stockés
!>          sous forme de matrice triangulaire supérieure ou inférieure.
!>          = 'U' :  Triangulaire supérieure, la forme est A = U*D*U**H ;
!>          = 'L' :  Triangulaire inférieure, la forme est A = L*D*L**H.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          La matrice diagonale par blocs D et les multiplicateurs utilisés pour
!>          obtenir le facteur U ou L tels que calculés par CHETRF.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!
>

IPI

V 
!>          IPIV est un tableau INTEGER, dimension (N)
!>          Détails des interchanges et de la structure par blocs de D
!>          tels que déterminés par CHETRF.
!
>

ANOR

M 
!>          ANORM est REAL
!>          La 1-norme de la matrice originale A.
!
>

RCON

D 
!>          RCOND est REAL
!>          L'inverse du nombre de conditionnement de la matrice A,
!>          calculé comme RCOND = 1/(ANORM * AINVNM), où AINVNM est une
!>          estimation de la 1-norme de inv(A) calculée dans cette routine.
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (2*N)
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!
>
