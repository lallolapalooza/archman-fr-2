subroutine chetri_rook (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( * ) work, integer

info)

CHETRI_ROOK calcule l'inverse de la matrice HE en utilisant la factorisation obtenue avec la méthode de pivotement diagonal Bunch-Kaufman ("rook")

bornée.

But

: 
!>
!> CHETRI_ROOK calcule l'inverse d'une matrice hermitienne indéfinie complexe
!> A en utilisant la factorisation A = U*D*U**H ou A = L*D*L**H calculée par
!> CHETRF_ROOK.
!
>

Paramètre

s
UPLO 
!>          UPLO est CHARACTER*1
!>          Indique si les détails de la factorisation sont stockés
!>          sous forme de matrice triangulaire supérieure ou inférieure.
!>          = 'U' : triangulaire supérieur, forme A = U*D*U**H ;
!>          = 'L' : triangulaire inférieur, forme A = L*D*L**H.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          À l'entrée, la matrice diagonale par blocs D et les multiplicateurs
!>          utilisés pour obtenir le facteur U ou L tels que calculés par CHETRF_ROOK.
!>
!>          À la sortie, si INFO = 0, l'inverse (hermitien) de la matrice
!>          originale.  Si UPLO = 'U', la partie triangulaire supérieure de
!>          l'inverse est formée et la partie de A sous la diagonale n'est pas
!>          référencée ; si UPLO = 'L', la partie triangulaire inférieure de
!>          l'inverse est formée et la partie de A au-dessus de la diagonale
!>          n'est pas référencée.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!
>

IPI

V 
!>          IPIV est un tableau INTEGER, dimension (N)
!>          Détails des échanges et de la structure par blocs de D
!>          tels que déterminés par CHETRF_ROOK.
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (N)
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 : si INFO = i, D(i,i) = 0 ; la matrice est singulière et son
!>               inverse n'a pas pu être calculé.
!
>
