subroutine chpgvx (integer itype, character jobz, character range, character uplo, integer n, complex, dimension( * ) ap, complex, dimension( * ) bp, real vl, real vu, integer il, integer iu, real abstol, integer m, real, dimension( * ) w, complex, dimension( ldz, * ) z, integer ldz, complex, dimension( * ) work, real, dimension( * ) rwork, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer

CHPGV

X

But

: 
!>
!> CHPGVX calcule les valeurs propres sélectionnées et, éventuellement, les vecteurs propres
!> d'un problème de valeurs propres généralisé hermitien-défini positif, de la forme
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  ou B*A*x=(lambda)*x.  Ici, A et
!> B sont supposés hermitiens, stockés au format compact, et B est également
!> défini positif.  Les valeurs propres et les vecteurs propres peuvent être sélectionnés en
!> spécifiant soit une plage de valeurs soit une plage d'indices pour les
!> valeurs propres souhaitées.
!
>

Paramètre

s
ITYPE 
!>          ITYPE est INTEGER
!>          Spécifie le type de problème à résoudre :
!>          = 1 :  A*x = (lambda)*B*x
!>          = 2 :  A*B*x = (lambda)*x
!>          = 3 :  B*A*x = (lambda)*x
!
>

JOB

Z 
!>          JOBZ est CHARACTER*1
!>          = 'N' :  calcule uniquement les valeurs propres ;
!>          = 'V' :  calcule les valeurs propres et les vecteurs propres.
!
>

RANG

E 
!>          RANGE est CHARACTER*1
!>          = 'A' : toutes les valeurs propres seront trouvées ;
!>          = 'V' : toutes les valeurs propres dans l'intervalle semi-ouvert (VL,VU]
!>                 seront trouvées ;
!>          = 'I' : les valeurs propres de la IL-ième à la IU-ième seront trouvées.
!
>

UPL

O 
!>          UPLO est CHARACTER*1
!>          = 'U' :  les triangles supérieurs de A et B sont stockés ;
!>          = 'L' :  les triangles inférieurs de A et B sont stockés.
!
>

N 
!>          N est INTEGER
!>          L'ordre des matrices A et B.  N >= 0.
!
>

A

P 
!>          AP est un tableau COMPLEX, dimension (N*(N+1)/2)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice hermitienne
!>          A, compacté colonne par colonne dans un tableau linéaire.  La j-ième colonne de A
!>          est stockée dans le tableau AP comme suit :
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L', AP(i + (j-1)*(2*n-j)/2) = A(i,j) pour j<=i<=n.
!>
!>          À la sortie, le contenu de AP est détruit.
!
>

B

P 
!>          BP est un tableau COMPLEX, dimension (N*(N+1)/2)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice hermitienne
!>          B, compacté colonne par colonne dans un tableau linéaire.  La j-ième colonne de B
!>          est stockée dans le tableau BP comme suit :
!>          si UPLO = 'U', BP(i + (j-1)*j/2) = B(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L', BP(i + (j-1)*(2*n-j)/2) = B(i,j) pour j<=i<=n.
!>
!>          À la sortie, le facteur triangulaire U ou L de la factorisation
!>          de Cholesky B = U**H*U ou B = L*L**H, dans le même format de stockage
!>          que B.
!
>

V

L 
!>          VL est REAL
!>
!>          Si RANGE='V', la borne inférieure de l'intervalle à
!>          parcourir pour trouver les valeurs propres.  VL < VU.
!>          Non référencé si RANGE = 'A' ou 'I'.
!
>

V

U 
!>          VU est REAL
!>
!>          Si RANGE='V', la borne supérieure de l'intervalle à
!>          parcourir pour trouver les valeurs propres.  VL < VU.
!>          Non référencé si RANGE = 'A' ou 'I'.
!
>

I

L 
!>          IL est INTEGER
!>
!>          Si RANGE='I', l'indice de la
!>          plus petite valeur propre à retourner.
!>          1 <= IL <= IU <= N, si N > 0 ; IL = 1 et IU = 0 si N = 0.
!>          Non référencé si RANGE = 'A' ou 'V'.
!
>

I

U 
!>          IU est INTEGER
!>
!>          Si RANGE='I', l'indice de la
!>          plus grande valeur propre à retourner.
!>          1 <= IL <= IU <= N, si N > 0 ; IL = 1 et IU = 0 si N = 0.
!>          Non référencé si RANGE = 'A' ou 'V'.
!
>

ABSTO

L 
!>          ABSTOL est REAL
!>          La tolérance d'erreur absolue pour les valeurs propres.
!>          Une valeur propre approchée est acceptée comme convergée
!>          lorsqu'il est déterminé qu'elle se situe dans un intervalle [a,b]
!>          de largeur inférieure ou égale à
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          où EPS est la précision machine.  Si ABSTOL est inférieur ou
!>          égal à zéro, alors  EPS*|T|  sera utilisé à la place,
!>          où |T| est la 1-norme de la matrice tridiagonale obtenue
!>          en réduisant AP à une forme tridiagonale.
!>
!>          Les valeurs propres seront calculées avec le plus d'exactitude lorsque ABSTOL est
!>          réglé sur deux fois le seuil de débordement inférieur 2*SLAMCH('S'), et non zéro.
!>          Si cette routine retourne avec INFO>0, indiquant que certains
!>          vecteurs propres n'ont pas convergé, essayez de régler ABSTOL sur
!>          2*SLAMCH('S').
!
>

M 
!>          M est INTEGER
!>          Le nombre total de valeurs propres trouvées.  0 <= M <= N.
!>          Si RANGE = 'A', M = N, et si RANGE = 'I', M = IU-IL+1.
!
>

W 
!>          W est un tableau REAL, dimension (N)
!>          À la sortie normale, les M premiers éléments contiennent les valeurs
!>          propres sélectionnées dans l'ordre croissant.
!
>

Z 
!>          Z est un tableau COMPLEX, dimension (LDZ, N)
!>          Si JOBZ = 'N', alors Z n'est pas référencé.
!>          Si JOBZ = 'V', alors si INFO = 0, les M premières colonnes de Z
!>          contiennent les vecteurs propres orthonormaux de la matrice A
!>          correspondant aux valeurs propres sélectionnées, avec la i-ième
!>          colonne de Z contenant le vecteur propre associé à W(i).
!>          Les vecteurs propres sont normalisés comme suit :
!>          si ITYPE = 1 ou 2, Z**H*B*Z = I ;
!>          si ITYPE = 3, Z**H*inv(B)*Z = I.
!>
!>          Si un vecteur propre ne converge pas, alors cette colonne de Z
!>          contient la dernière approximation du vecteur propre, et l'indice
!>          du vecteur propre est retourné dans IFAIL.
!>          Note : l'utilisateur doit s'assurer qu'au moins max(1,M) colonnes sont
!>          fournies dans le tableau Z ; si RANGE = 'V', la valeur exacte de M
!>          n'est pas connue à l'avance et une borne supérieure doit être utilisée.
!
>

LD

Z 
!>          LDZ est INTEGER
!>          La dimension principale du tableau Z.  LDZ >= 1, et si
!>          JOBZ = 'V', LDZ >= max(1,N).
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (2*N)
!
>

RWOR

K 
!>          RWORK est un tableau REAL, dimension (7*N)
!
>

IWOR

K 
!>          IWORK est un tableau INTEGER, dimension (5*N)
!
>

IFAI

L 
!>          IFAIL est un tableau INTEGER, dimension (N)
!>          Si JOBZ = 'V', alors si INFO = 0, les M premiers éléments de
!>          IFAIL sont nuls. Si INFO > 0, alors IFAIL contient les
!>          indices des vecteurs propres qui n'ont pas convergé.
!>          Si JOBZ = 'N', alors IFAIL n'est pas référencé.
!>

INFO

!>          INFO est INTEGER
!>          = 0:  sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0:  CPPTRF ou CHPEVX a retourné un code d'erreur :
!>             <= N:  si INFO = i, CHPEVX n'a pas convergé ;
!>                    i vecteurs propres n'ont pas convergé. Leurs indices
!>                    sont stockés dans le tableau IFAIL.
!>             > N:   si INFO = N + i, pour 1 <= i <= n, alors le mineur
!>                    principal d'ordre i de B n'est pas positif.
!>                    La factorisation de B n'a pas pu être complétée et
!>                    aucune valeur propre ou vecteur propre n'a été calculé.
!>

sous_routine dspgvx (integer itype, character jobz, character range, character uplo, integer n, double precision, dimension( * ) ap, double precision, dimension( * ) bp, double precision vl, double precision vu, integer il, integer iu, double precision abstol, integer m, double precision, dimension( * ) w, double precision, dimension( ldz, * ) z, integer ldz, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

DSPGVX

Objectif :

!>
!> DSPGVX calcule les valeurs propres sélectionnées, et optionnellement, les vecteurs propres
!> d'un problème généralisé symétrique défini réel, de la forme
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  ou B*A*x=(lambda)*x. Ici A
!> et B sont supposés être symétriques, stockés en stockage compacté, et B
!> est également défini positif. Les valeurs propres et les vecteurs propres peuvent être
!> sélectionnés en spécifiant soit une plage de valeurs, soit une plage d'indices
!> pour les valeurs propres désirées.
!>

Paramètres

ITYPE 
!>          ITYPE est INTEGER
!>          Spécifie le type de problème à résoudre :
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ est CHARACTER*1
!>          = 'N':  Calculer uniquement les valeurs propres ;
!>          = 'V':  Calculer les valeurs propres et les vecteurs propres.
!>

RANGE

!>          RANGE est CHARACTER*1
!>          = 'A': toutes les valeurs propres seront trouvées.
!>          = 'V': toutes les valeurs propres dans l'intervalle ouvert à droite (VL,VU]
!>                 seront trouvées.
!>          = 'I': les valeurs propres d'indice IL à IU seront trouvées.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U':  Le triangle supérieur de A et B est stocké ;
!>          = 'L':  Le triangle inférieur de A et B est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre du pencil matriciel (A,B).  N >= 0.
!>

AP

!>          AP est un tableau DOUBLE PRECISION, dimension (N*(N+1)/2)
!>          En entrée, le triangle supérieur ou inférieur de la matrice symétrique
!>          A, stocké compacté par colonne dans un tableau linéaire. La j-ème colonne de A
!>          est stockée dans le tableau AP comme suit :
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L', AP(i + (j-1)*(2*n-j)/2) = A(i,j) pour j<=i<=n.
!>
!>          En sortie, le contenu de AP est détruit.
!>

BP

!>          BP est un tableau DOUBLE PRECISION, dimension (N*(N+1)/2)
!>          En entrée, le triangle supérieur ou inférieur de la matrice symétrique
!>          B, stocké compacté par colonne dans un tableau linéaire. La j-ème colonne de B
!>          est stockée dans le tableau BP comme suit :
!>          si UPLO = 'U', BP(i + (j-1)*j/2) = B(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L', BP(i + (j-1)*(2*n-j)/2) = B(i,j) pour j<=i<=n.
!>
!>          En sortie, le facteur triangulaire U ou L de la factorisation de Cholesky
!!>          B = U**T*U ou B = L*L**T, dans le même format de stockage que B.
!>

VL

!>          VL est DOUBLE PRECISION
!>
!>          Si RANGE='V', la borne inférieure de l'intervalle à
!>          rechercher pour les valeurs propres. VL < VU.
!>          Non référencé si RANGE = 'A' ou 'I'.
!>

VU

!>          VU est DOUBLE PRECISION
!>
!>          Si RANGE='V', la borne supérieure de l'intervalle à
!>          rechercher pour les valeurs propres. VL < VU.
!>          Non référencé si RANGE = 'A' ou 'I'.
!>

IL

!>          IL est INTEGER
!>
!>          Si RANGE='I', l'indice de la
!>          plus petite valeur propre à retourner.
!>          1 <= IL <= IU <= N, si N > 0 ; IL = 1 et IU = 0 si N = 0.
!>          Non référencé si RANGE = 'A' ou 'V'.
!>

IU

!>          IU est INTEGER
!>
!>          Si RANGE='I', l'indice de la
!>          plus grande valeur propre à retourner.
!>          1 <= IL <= IU <= N, si N > 0 ; IL = 1 et IU = 0 si N = 0.
!>          Non référencé si RANGE = 'A' ou 'V'.
!>

ABSTOL

!>          ABSTOL est DOUBLE PRECISION
!>          La tolérance d'erreur absolue pour les valeurs propres.
!>          Une valeur propre approximative est acceptée comme convergée
!>          lorsqu'elle est déterminée comme se situant dans un intervalle [a,b]
!>          de largeur inférieure ou égale à
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          où EPS est la précision machine. Si ABSTOL est inférieur ou
!>          égal à zéro, alors EPS * |T| sera utilisé à sa place,
!>          où |T| est la norme 1 de la matrice tridiagonale obtenue
!>          en réduisant A à une forme tridiagonale.
!>
!>          Les valeurs propres seront calculées le plus précisément lorsque ABSTOL est
!>          défini à deux fois le seuil de sous-dépassement 2*DLAMCH('S'), pas zéro.
!>          Si cette routine retourne avec INFO>0, indiquant que certains
!>          vecteurs propres n'ont pas convergé, essayez de définir ABSTOL à
!>          2*DLAMCH('S').
!>

M

!>          M est INTEGER
!>          Le nombre total de valeurs propres trouvées.  0 <= M <= N.
!>          Si RANGE = 'A', M = N, et si RANGE = 'I', M = IU-IL+1.
!>

W

!>          W est un tableau DOUBLE PRECISION, dimension (N)
!>          En cas de sortie normale, les M premiers éléments contiennent les valeurs propres sélectionnées
!>          en ordre croissant.
!>

Z

!>          Z est un tableau DOUBLE PRECISION, dimension (LDZ, max(1,M))
!>          Si JOBZ = 'N', alors Z n'est pas référencé.
!>          Si JOBZ = 'V', alors si INFO = 0, les M premières colonnes de Z
!>          contiennent les vecteurs propres orthonormaux de la matrice A
!>          correspondant aux valeurs propres sélectionnées, la i-ème
!>          colonne de Z contenant le vecteur propre associé à W(i).
!>          Les vecteurs propres sont normalisés comme suit :
!>          si ITYPE = 1 ou 2, Z**T*B*Z = I ;
!>          si ITYPE = 3, Z**T*inv(B)*Z = I.
!>
!>          Si un vecteur propre échoue à converger, alors cette colonne de Z
!>          contient la dernière approximation du vecteur propre, et
!>          l'indice du vecteur propre est retourné dans IFAIL.
!>          Note : l'utilisateur doit s'assurer qu'au moins max(1,M) colonnes sont
!>          fournies dans le tableau Z ; si RANGE = 'V', la valeur exacte de M
!>          n'est pas connue à l'avance et une borne supérieure doit être utilisée.
!>

LDZ

!>          LDZ est INTEGER
!>          La dimension principale du tableau Z.  LDZ >= 1, et si
!>          JOBZ = 'V', LDZ >= max(1,N).
!>

WORK

!>          WORK est un tableau DOUBLE PRECISION, dimension (8*N)
!>

IWORK

!>          IWORK est un tableau INTEGER, dimension (5*N)
!>

IFAIL

!>          IFAIL est un tableau INTEGER, dimension (N)
!>          Si JOBZ = 'V', alors si INFO = 0, les M premiers éléments de
!>          IFAIL sont nuls. Si INFO > 0, alors IFAIL contient les
!>          indices des vecteurs propres qui n'ont pas convergé.
!>          Si JOBZ = 'N', alors IFAIL n'est pas référencé.
!>

INFO

!>          INFO est INTEGER
!>          = 0:  sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0:  DPPTRF ou DSPEVX a retourné un code d'erreur :
!>             <= N:  si INFO = i, DSPEVX n'a pas convergé ;
!>                    i vecteurs propres n'ont pas convergé. Leurs indices
!>                    sont stockés dans le tableau IFAIL.
!>             > N:   si INFO = N + i, pour 1 <= i <= N, alors le mineur
!>                    principal d'ordre i de B n'est pas positif.
!>                    La factorisation de B n'a pas pu être complétée et
!>                    aucune valeur propre ou vecteur propre n'a été calculé.
!>
