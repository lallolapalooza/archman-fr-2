subroutine chbevd_2stage (character jobz, character uplo, integer n, integer kd, complex, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) w, complex, dimension( ldz, * ) z, integer ldz, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, integer lrwork, integer, dimension( * ) iwork, integer liwork, integer

CHBEVD_2STAGE calcule les valeurs propres et, optionnellement, les vecteurs propres gauche et/ou droit pour les matrices

OTHER

But

: 
!>
!> CHBEVD_2STAGE calcule toutes les valeurs propres et, optionnellement, les vecteurs propres
!> d'une matrice hermitienne par bandes A en utilisant la technique 2stage pour
!> la réduction en tridiagonale. Si des vecteurs propres sont souhaités, elle
!> utilise un algorithme de type divide and conquer.
!>
!
>

Paramètre

s
JOBZ 
!>          JOBZ est CHARACTER*1
!>          = 'N' :  calcule uniquement les valeurs propres ;
!>          = 'V' :  calcule les valeurs propres et les vecteurs propres.
!>                   Non disponible dans cette version.
!
>

UPL

O 
!>          UPLO est CHARACTER*1
!>          = 'U' :  le triangle supérieur de A est stocké ;
!>          = 'L' :  le triangle inférieur de A est stocké.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

K

D 
!>          KD est INTEGER
!>          Le nombre de surdiagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'.  KD >= 0.
!
>

A

B 
!>          AB est un tableau COMPLEX, dimension (LDAB, N)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice
!>          hermitienne par bandes A, stocké dans les premières KD+1 lignes du
!>          tableau. La j-ième colonne de A est stockée dans la j-ième colonne
!>          du tableau AB comme suit :
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) pour max(1,j-kd)<=i<=j ;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+kd).
!>
!>          À la sortie, AB est écrasé par les valeurs générées pendant la
!>          réduction en forme tridiagonale. Si UPLO = 'U', la première
!>          surdiagonale et la diagonale de la matrice tridiagonale T sont
!>          retournées dans les lignes KD et KD+1 de AB, et si UPLO = 'L',
!>          la diagonale et la première sous-diagonale de T sont retournées
!>          dans les deux premières lignes de AB.
!
>

LDA

B 
!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KD + 1.
!
>

W 
!>          W est un tableau REAL, dimension (N)
!>          Si INFO = 0, les valeurs propres dans l'ordre croissant.
!
>

Z 
!>          Z est un tableau COMPLEX, dimension (LDZ, N)
!>          Si JOBZ = 'V', alors si INFO = 0, Z contient les vecteurs propres
!>          orthonormaux de la matrice A, la i-ième colonne de Z contenant le
!>          vecteur propre associé à W(i).
!>          Si JOBZ = 'N', alors Z n'est pas référencé.
!
>

LD

Z 
!>          LDZ est INTEGER
!>          La dimension principale du tableau Z.  LDZ >= 1, et si
!>          JOBZ = 'V', LDZ >= max(1,N).
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie le LWORK optimal.
!
>

LWOR

K 
!>          LWORK est INTEGER
!>          La longueur du tableau WORK. LWORK >= 1, quand N <= 1 ;
!>          sinon
!>          Si JOBZ = 'N' et N > 1, LWORK doit être interrogé.
!>                                   LWORK = MAX(1, dimension) où
!>                                   dimension = (2KD+1)*N + KD*NTHREADS
!>                                   où KD est la taille de la bande.
!>                                   NTHREADS est le nombre de threads utilisés quand
!>                                   la compilation openMP est activée, sinon =1.
!>          Si JOBZ = 'V' et N > 1, LWORK doit être interrogé. Pas encore disponible.
!>
!>          Si LWORK = -1, alors une interrogation de l'espace de travail est
!>          supposée ; la routine calcule uniquement les tailles optimales des
!>          tableaux WORK, RWORK et IWORK, renvoie ces valeurs comme les
!>          premières entrées des tableaux WORK, RWORK et IWORK, et aucun
!>          message d'erreur lié à LWORK ou LRWORK ou LIWORK n'est émis par
!>          XERBLA.
!
>

RWOR

K 
!>          RWORK est un tableau REAL,
!>                                         dimension (LRWORK)
!>          À la sortie, si INFO = 0, RWORK(1) renvoie le LRWORK optimal.
!
>

LRWOR

K 
!>          LRWORK est INTEGER
!>          La dimension du tableau RWORK.
!>          Si N <= 1,               LRWORK doit être au moins 1.
!>          Si JOBZ = 'N' et N > 1, LRWORK doit être au moins N.
!>          Si JOBZ = 'V' et N > 1, LRWORK doit être au moins
!>                        1 + 5*N + 2*N**2.
!>
!>          Si LRWORK = -1, alors une interrogation de l'espace de travail est
!>          supposée ; la routine calcule uniquement les tailles optimales des
!>          tableaux WORK, RWORK et IWORK, renvoie ces valeurs comme les
!>          premières entrées des tableaux WORK, RWORK et IWORK, et aucun
!>          message d'erreur lié à LWORK ou LRWORK ou LIWORK n'est émis par
!>          XERBLA.
!
>

IWOR

K 
!>          IWORK est un tableau INTEGER, dimension (MAX(1,LIWORK))
!>          À la sortie, si INFO = 0, IWORK(1) renvoie le LIWORK optimal.
!
>

LIWOR

K 
!>          LIWORK est INTEGER
!>          La dimension du tableau IWORK.
!>          Si JOBZ = 'N' ou N <= 1, LIWORK doit être au moins 1.
!>          Si JOBZ = 'V' et N > 1, LIWORK doit être au moins 3 + 5*N .
!>
!>          Si LIWORK = -1, alors une interrogation de l'espace de travail est
!>          supposée ; la routine calcule uniquement les tailles optimales des
!>          tableaux WORK, RWORK et IWORK, renvoie ces valeurs comme les
!>          premières entrées des tableaux WORK, RWORK et IWORK, et aucun
!>          message d'erreur lié à LWORK ou LRWORK ou LIWORK n'est émis par
!>          XERBLA.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie.
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale.
!>          > 0 :  si INFO = i, l'algorithme n'a pas convergé ; i
!>                 éléments hors diagonale d'une forme tridiagonale
!>                 intermédiaire n'ont pas convergé vers zéro.
!
>
