subroutine chpr2 (character uplo, integer n, complex alpha, complex, dimension(*) x, integer incx, complex, dimension(*) y, integer incy, complex, dimension(*) ap)

CHPR2

But :

!>
!> CHPR2 effectue l'opération hermitienne de rang 2
!>
!>    A := alpha*x*y**H + conjg( alpha )*y*x**H + A,
!>
!> où alpha est un scalaire, x et y sont des vecteurs de n éléments et A est une
!> matrice hermitienne de taille n par n, fournie sous forme compactée.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>           À l'entrée, UPLO spécifie si la partie triangulaire supérieure ou inférieure
!>           de la matrice A est fournie dans le tableau compacté
!>           AP comme suit :
!>
!>              UPLO = 'U' ou 'u'   La partie triangulaire supérieure de A est
!>                                  fournie dans AP.
!>
!>              UPLO = 'L' ou 'l'   La partie triangulaire inférieure de A est
!>                                  fournie dans AP.
!>

N

!>          N est INTEGER
!>           À l'entrée, N spécifie l'ordre de la matrice A.
!>           N doit être au moins égal à zéro.
!>

ALPHA

!>          ALPHA est COMPLEX
!>           À l'entrée, ALPHA spécifie le scalaire alpha.
!>

X

!>          X est un tableau COMPLEX, de dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           Avant l'entrée, le tableau incrémenté X doit contenir le vecteur x
!>           de n éléments.
!>

INCX

!>          INCX est INTEGER
!>           À l'entrée, INCX spécifie l'incrément pour les éléments de
!>           X. INCX ne doit pas être zéro.
!>

Y

!>          Y est un tableau COMPLEX, de dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           Avant l'entrée, le tableau incrémenté Y doit contenir le vecteur y
!>           de n éléments.
!>

INCY

!>          INCY est INTEGER
!>           À l'entrée, INCY spécifie l'incrément pour les éléments de
!>           Y. INCY ne doit pas être zéro.
!>

AP

!>          AP est un tableau COMPLEX, de dimension au moins
!>           ( ( n*( n + 1 ) )/2 ).
!>           Avant l'entrée avec UPLO = 'U' ou 'u', le tableau AP doit
!>           contenir la partie triangulaire supérieure de la matrice hermitienne
!>           compactée séquentiellement, colonne par colonne, de sorte que AP( 1 )
!>           contient a( 1, 1 ), AP( 2 ) et AP( 3 ) contiennent a( 1, 2 )
!>           et a( 2, 2 ) respectivement, et ainsi de suite. À la sortie, le tableau
!>           AP est écrasé par la partie triangulaire supérieure de la
!>           matrice mise à jour.
!>           Avant l'entrée avec UPLO = 'L' ou 'l', le tableau AP doit
!>           contenir la partie triangulaire inférieure de la matrice hermitienne
!>           compactée séquentiellement, colonne par colonne, de sorte que AP( 1 )
!>           contient a( 1, 1 ), AP( 2 ) et AP( 3 ) contiennent a( 2, 1 )
!>           et a( 3, 1 ) respectivement, et ainsi de suite. À la sortie, le tableau
!>           AP est écrasé par la partie triangulaire inférieure de la
!>           matrice mise à jour.
!>           Notez que les parties imaginaires des éléments diagonaux n'ont pas
!>           besoin d'être définies, elles sont supposées être égales à zéro, et à la sortie elles
!>           sont mises à zéro.
!>
