sous-routine chbtrd (character vect, character uplo, integer n, integer kd, complex, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) d, real, dimension( * ) e, complex, dimension( ldq, * ) q, integer ldq, complex, dimension( * ) work, integer

info)

CHBTR

D

But

: 
!>
!> CHBTRD réduit une matrice hermitienne par bandes A à une forme
!> tridiagonale symétrique réelle T par une transformation de similarité
!> unitaire :
!> Q**H * A * Q = T.
!
>

Paramètre

s
VECT 
!>          VECT est CHARACTER*1
!>          = 'N' : ne pas former Q ;
!>          = 'V' : former Q ;
!>          = 'U' : mettre à jour une matrice X en formant X*Q.
!
>

UPL

O 
!>          UPLO est CHARACTER*1
!>          = 'U' : le triangle supérieur de A est stocké ;
!>          = 'L' : le triangle inférieur de A est stocké.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

K

D 
!>          KD est INTEGER
!>          Le nombre de surdiagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'.  KD >= 0.
!
>

A

B 
!>          AB est un tableau COMPLEX, dimension (LDAB,N)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice
!>          hermitienne par bandes A, stocké dans les premières KD+1
!>          lignes du tableau.  La j-ième colonne de A est stockée dans
!>          la j-ième colonne du tableau AB comme suit :
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) pour max(1,j-kd)<=i<=j ;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+kd).
!>          À la sortie, les éléments diagonaux de AB sont écrasés par
!>          les éléments diagonaux de la matrice tridiagonale T ; si
!>          KD > 0, les éléments de la première surdiagonale (si UPLO =
!>          'U') ou de la première sous-diagonale (si UPLO = 'L') sont
!>          écrasés par les éléments hors diagonale de T ; le reste de
!>          AB est écrasé par des valeurs générées pendant la réduction.
!
>

LDA

B 
!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KD+1.
!
>

D 
!>          D est un tableau REAL, dimension (N)
!>          Les éléments diagonaux de la matrice tridiagonale T.
!
>

E 
!>          E est un tableau REAL, dimension (N-1)
!>          Les éléments hors diagonale de la matrice tridiagonale T :
!>          E(i) = T(i,i+1) si UPLO = 'U' ; E(i) = T(i+1,i) si UPLO = 'L'.
!
>

Q 
!>          Q est un tableau COMPLEX, dimension (LDQ,N)
!>          À l'entrée, si VECT = 'U', alors Q doit contenir une matrice
!>          X de taille N par N ; si VECT = 'N' ou 'V', Q n'a pas besoin
!>          d'être défini.
!>
!>          À la sortie :
!>          si VECT = 'V', Q contient la matrice unitaire Q de taille N par N ;
!>          si VECT = 'U', Q contient le produit X*Q ;
!>          si VECT = 'N', le tableau Q n'est pas référencé.
!
>

LD

Q 
!>          LDQ est INTEGER
!>          La dimension principale du tableau Q.
!>          LDQ >= 1, et LDQ >= N si VECT = 'V' ou 'U'.
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (N)
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!
>
