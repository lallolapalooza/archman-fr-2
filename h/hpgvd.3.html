subroutine chpgvd (integer itype, character jobz, character uplo, integer n, complex, dimension( * ) ap, complex, dimension( * ) bp, real, dimension( * ) w, complex, dimension( ldz, * ) z, integer ldz, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, integer lrwork, integer, dimension( * ) iwork, integer liwork, integer info)

CHPGVD

But :

!>
!> CHPGVD calcule toutes les valeurs propres et, optionnellement, les vecteurs propres
!> d'un problème aux valeurs propres généralisé hermitien-défini positif, de la forme
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  ou B*A*x=(lambda)*x.  Ici A et
!> B sont supposés hermitiens, stockés au format compact, et B est également
!> défini positif.
!> Si les vecteurs propres sont souhaités, il utilise un algorithme de type divide and conquer.
!>
!>

Paramètres

ITYPE 
!>          ITYPE est INTEGER
!>          Spécifie le type de problème à résoudre :
!>          = 1 :  A*x = (lambda)*B*x
!>          = 2 :  A*B*x = (lambda)*x
!>          = 3 :  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ est CHARACTER*1
!>          = 'N' :  Calculer uniquement les valeurs propres ;
!>          = 'V' :  Calculer les valeurs propres et les vecteurs propres.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U' :  Les triangles supérieurs de A et B sont stockés ;
!>          = 'L' :  Les triangles inférieurs de A et B sont stockés.
!>

N

!>          N est INTEGER
!>          L'ordre des matrices A et B.  N >= 0.
!>

AP

!>          AP est un tableau COMPLEX, dimension (N*(N+1)/2)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice hermitienne
!>          A, compacté colonne par colonne dans un tableau linéaire.  La j-ième colonne de A
!>          est stockée dans le tableau AP comme suit :
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L', AP(i + (j-1)*(2*n-j)/2) = A(i,j) pour j<=i<=n.
!>
!>          À la sortie, le contenu de AP est détruit.
!>

BP

!>          BP est un tableau COMPLEX, dimension (N*(N+1)/2)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice hermitienne
!>          B, compacté colonne par colonne dans un tableau linéaire.  La j-ième colonne de B
!>          est stockée dans le tableau BP comme suit :
!>          si UPLO = 'U', BP(i + (j-1)*j/2) = B(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L', BP(i + (j-1)*(2*n-j)/2) = B(i,j) pour j<=i<=n.
!>
!>          À la sortie, le facteur triangulaire U ou L de la factorisation
!>          de Cholesky B = U**H*U ou B = L*L**H, dans le même format de stockage
!>          que B.
!>

W

!>          W est un tableau REAL, dimension (N)
!>          Si INFO = 0, les valeurs propres dans l'ordre croissant.
!>

Z

!>          Z est un tableau COMPLEX, dimension (LDZ, N)
!>          Si JOBZ = 'V', alors si INFO = 0, Z contient la matrice Z des
!>          vecteurs propres.  Les vecteurs propres sont normalisés comme suit :
!>          si ITYPE = 1 ou 2, Z**H*B*Z = I ;
!>          si ITYPE = 3, Z**H*inv(B)*Z = I.
!>          Si JOBZ = 'N', alors Z n'est pas référencé.
!>

LDZ

!>          LDZ est INTEGER
!>          La dimension principale du tableau Z.  LDZ >= 1, et si
!>          JOBZ = 'V', LDZ >= max(1,N).
!>

WORK

!>          WORK est un tableau COMPLEX, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie le LWORK requis.
!>

LWORK

!>          LWORK est INTEGER
!>          La dimension du tableau WORK.
!>          Si N <= 1,               LWORK >= 1.
!>          Si JOBZ = 'N' et N > 1, LWORK >= N.
!>          Si JOBZ = 'V' et N > 1, LWORK >= 2*N.
!>
!>          Si LWORK = -1, alors une interrogation d'espace de travail est supposée ; la routine
!>          calcule uniquement les tailles requises des tableaux WORK, RWORK et
!>          IWORK, renvoie ces valeurs comme les premières entrées des
!>          tableaux WORK, RWORK et IWORK, et aucun message d'erreur
!>          lié à LWORK ou LRWORK ou LIWORK n'est émis par XERBLA.
!>

RWORK

!>          RWORK est un tableau REAL, dimension (MAX(1,LRWORK))
!>          À la sortie, si INFO = 0, RWORK(1) renvoie le LRWORK requis.
!>

LRWORK

!>          LRWORK est INTEGER
!>          La dimension du tableau RWORK.
!>          Si N <= 1,               LRWORK >= 1.
!>          Si JOBZ = 'N' et N > 1, LRWORK >= N.
!>          Si JOBZ = 'V' et N > 1, LRWORK >= 1 + 5*N + 2*N**2.
!>
!>          Si LRWORK = -1, alors une interrogation d'espace de travail est supposée ; la
!>          routine calcule uniquement les tailles requises des tableaux WORK, RWORK
!>          et IWORK, renvoie ces valeurs comme les premières entrées
!>          des tableaux WORK, RWORK et IWORK, et aucun message d'erreur
!>          lié à LWORK ou LRWORK ou LIWORK n'est émis par XERBLA.
!>

IWORK

!>          IWORK est un tableau INTEGER, dimension (MAX(1,LIWORK))
!>          À la sortie, si INFO = 0, IWORK(1) renvoie le LIWORK requis.
!>

LIWORK

!>          LIWORK est INTEGER
!>          La dimension du tableau IWORK.
!>          Si JOBZ  = 'N' ou N <= 1, LIWORK >= 1.
!>          Si JOBZ  = 'V' et N > 1, LIWORK >= 3 + 5*N.
!>
!>          Si LIWORK = -1, alors une interrogation d'espace de travail est supposée ; la
!>          routine calcule uniquement les tailles requises des tableaux WORK, RWORK
!>          et IWORK, renvoie ces valeurs comme les premières entrées
!>          des tableaux WORK, RWORK et IWORK, et aucun message d'erreur
!>          lié à LWORK ou LRWORK ou LIWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 :  CPPTRF ou CHPEVD a renvoyé un code d'erreur :
!>             <= N :  si INFO = i, CHPEVD n'a pas réussi à converger ;
!>                    i éléments hors diagonale d'une forme
!>                    tridiagonale intermédiaire n'ont pas convergé vers zéro ;
!>             > N :   si INFO = N + i, pour 1 <= i <= n, alors le mineur
!>                    principal de tête d'ordre i de B n'est pas positif.
!>                    La factorisation de B n'a pas pu être complétée et
!>                    aucune valeur propre ou vecteur propre n'a été calculé.
!>

!>          stockage que B.

W

!>          W est un tableau DOUBLE PRECISION, dimension (N)
!>          Si INFO = 0, les valeurs propres dans l'ordre croissant.

!>          Z est un tableau COMPLEX*16, dimension (LDZ, N)
!>          Si JOBZ = 'V', alors si INFO = 0, Z contient la matrice Z des
!>          vecteurs propres.  Les vecteurs propres sont normalisés comme
!>          suit :
!>          si ITYPE = 1 ou 2, Z**H*B*Z = I ;
!>          si ITYPE = 3, Z**H*inv(B)*Z = I.

!>          LDZ est INTEGER
!>          La dimension principale du tableau Z.  LDZ >= 1, et si
!>          JOBZ = 'V', LDZ >= max(1,N).

!>          WORK est un tableau COMPLEX*16, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie le LWORK requis.
!>

LWORK

!>          LWORK est INTEGER
!>          La dimension du tableau WORK.
!>          Si N <= 1,               LWORK >= 1.
!>          Si JOBZ = 'N' et N > 1, LWORK >= N.
!>          Si JOBZ = 'V' et N > 1, LWORK >= 2*N.

!>          Si LWORK = -1, alors une interrogation d'espace de travail est
!>          supposée ; la routine ne calcule que les tailles requises des

!>          premières entrées des tableaux WORK, RWORK et IWORK, et aucun
!>          message d'erreur lié à LWORK ou LRWORK ou LIWORK n'est émis par
!>          XERBLA.
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, dimension (MAX(1,LRWORK))
!>          À la sortie, si INFO = 0, RWORK(1) renvoie le LRWORK requis.
!>

LRWORK

!>          LRWORK est INTEGER
!>          La dimension du tableau RWORK.
!>          Si N <= 1,               LRWORK >= 1.
!>          Si JOBZ = 'N' et N > 1, LRWORK >= N.
!>          Si JOBZ = 'V' et N > 1, LRWORK >= 1 + 5*N + 2*N**2.
!>
!>          Si LRWORK = -1, alors une interrogation d'espace de travail est
!>          supposée ; la routine ne calcule que les tailles requises des
!>          tableaux WORK, RWORK et IWORK, renvoie ces valeurs comme
!>          premières entrées des tableaux WORK, RWORK et IWORK, et aucun
!>          message d'erreur lié à LWORK ou LRWORK ou LIWORK n'est émis par
!>          XERBLA.

IWORK

!>          IWORK est un tableau INTEGER, dimension (MAX(1,LIWORK))
!>          À la sortie, si INFO = 0, IWORK(1) renvoie le LIWORK requis.

LIWORK

!>          LIWORK est INTEGER
!>          La dimension du tableau IWORK.
!>          Si JOBZ  = 'N' ou N <= 1, LIWORK >= 1.
!>          Si JOBZ  = 'V' et N > 1, LIWORK >= 3 + 5*N.
!>
!>          Si LIWORK = -1, alors une interrogation d'espace de travail est
!>          supposée ; la routine ne calcule que les tailles requises des
!>          tableaux WORK, RWORK et IWORK, renvoie ces valeurs comme
!>          premières entrées des tableaux WORK, RWORK et IWORK, et aucun
!>          message d'erreur lié à LWORK ou LRWORK ou LIWORK n'est émis par
!>          XERBLA.

!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0:  ZPPTRF ou ZHPEVD a renvoyé un code d'erreur :
