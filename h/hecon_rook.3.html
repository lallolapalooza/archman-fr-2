subroutine checon_rook (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real anorm, real rcond, complex, dimension( * ) work, integer

info)

CHECON_ROOK estime l'inverse du conditionnement pour les matrices HE en utilisant la factorisation obtenue avec l'une des méthodes de pivotement diagonal borné (max 2

échanges)

But

: 
!>
!> CHECON_ROOK estime l'inverse du conditionnement d'une matrice
!> hermitienne complexe A en utilisant la factorisation A = U*D*U**H ou
!> A = L*D*L**H calculée par CHETRF_ROOK.
!>
!> Une estimation de norm(inv(A)) est obtenue, et l'inverse du
!> conditionnement est calculé comme RCOND = 1 / (ANORM * norm(inv(A))).
!
>

Paramètre

s
UPLO 
!>          UPLO est CHARACTER*1
!>          Indique si les détails de la factorisation sont stockés
!>          sous forme de matrice triangulaire supérieure ou inférieure.
!>          = 'U' :  triangulaire supérieur, la forme est A = U*D*U**H ;
!>          = 'L' :  triangulaire inférieur, la forme est A = L*D*L**H.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          La matrice diagonale par blocs D et les multiplicateurs utilisés
!>          pour obtenir le facteur U ou L tels que calculés par CHETRF_ROOK.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!
>

IPI

V 
!>          IPIV est un tableau INTEGER, dimension (N)
!>          Détails des échanges et de la structure par blocs de D
!>          tels que déterminés par CHETRF_ROOK.
!
>

ANOR

M 
!>          ANORM est REAL
!>          La 1-norme de la matrice originale A.
!
>

RCON

D 
!>          RCOND est REAL
!>          L'inverse du conditionnement de la matrice A,
!>          calculé comme RCOND = 1/(ANORM * AINVNM), où AINVNM est une
!>          estimation de la 1-norme de inv(A) calculée dans cette routine.
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (2*N)
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!
>
