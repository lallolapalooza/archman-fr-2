subroutine cheevd_2stage (character jobz, character uplo, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) w, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, integer lrwork, integer, dimension( * ) iwork, integer liwork, integer

info)

CHEEVD_2STAGE calcule les valeurs propres et, en option, les vecteurs propres gauche et/ou droit pour les matrices

HE

But

: 
!>
!> CHEEVD_2STAGE calcule toutes les valeurs propres et, en option, les vecteurs propres d'une
!> matrice hermitienne complexe A en utilisant la technique à 2 étages pour
!> la réduction en tridiagonale. Si les vecteurs propres sont souhaités, un algorithme
!> divide and conquer est utilisé.
!>
!
>

Paramètre

s
JOBZ 
!>          JOBZ est CHARACTER*1
!>          = 'N' :  calcule uniquement les valeurs propres ;
!>          = 'V' :  calcule les valeurs propres et les vecteurs propres.
!>                   Non disponible dans cette version.
!
>

UPL

O 
!>          UPLO est CHARACTER*1
!>          = 'U' :  le triangle supérieur de A est stocké ;
!>          = 'L' :  le triangle inférieur de A est stocké.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA, N)
!>          À l'entrée, la matrice hermitienne A.  Si UPLO = 'U', la
!>          partie triangulaire supérieure N×N de A contient la
!>          partie triangulaire supérieure de la matrice A.  Si UPLO = 'L',
!>          la partie triangulaire inférieure N×N de A contient
!>          la partie triangulaire inférieure de la matrice A.
!>          À la sortie, si JOBZ = 'V', alors si INFO = 0, A contient les
!>          vecteurs propres orthonormaux de la matrice A.
!>          Si JOBZ = 'N', alors à la sortie le triangle inférieur (si UPLO='L')
!>          ou le triangle supérieur (si UPLO='U') de A, y compris la
!>          diagonale, est détruit.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!
>

W 
!>          W est un tableau REAL, dimension (N)
!>          Si INFO = 0, les valeurs propres dans l'ordre croissant.
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie le LWORK optimal.
!
>

LWOR

K 
!>          LWORK est INTEGER
!>          La dimension du tableau WORK.
!>          Si N <= 1,               LWORK doit être au moins 1.
!>          Si JOBZ = 'N' et N > 1, LWORK doit être interrogé.
!>                                   LWORK = MAX(1, dimension) où
!>                                   dimension = max(stage1,stage2) + (KD+1)*N + N+1
!>                                             = N*KD + N*max(KD+1,FACTOPTNB)
!>                                               + max(2*KD*KD, KD*NTHREADS)
!>                                               + (KD+1)*N + N+1
!>                                   où KD est la taille de bloc de la réduction,
!>                                   FACTOPTNB est le bloc utilisé par l'algorithme QR ou LQ,
!>                                   généralement FACTOPTNB=128 est un bon choix
!>                                   NTHREADS est le nombre de threads utilisé lorsque
!>                                   la compilation openMP est activée, sinon =1.
!>          Si JOBZ = 'V' et N > 1, LWORK doit être au moins 2*N + N**2
!>
!>          Si LWORK = -1, alors une interrogation d'espace de travail est supposée ; la routine
!>          calcule uniquement les tailles optimales des tableaux WORK, RWORK et
!>          IWORK, renvoie ces valeurs comme premières entrées des
!>          tableaux WORK, RWORK et IWORK, et aucun message d'erreur
!>          lié à LWORK ou LRWORK ou LIWORK n'est émis par XERBLA.
!
>

RWOR

K 
!>          RWORK est un tableau REAL,
!>                                         dimension (LRWORK)
!>          À la sortie, si INFO = 0, RWORK(1) renvoie le LRWORK optimal.
!
>

K 
!>          LRWORK est INTEGER
!>          La dimension du tableau RWORK.
!>          Si N <= 1,                LRWORK doit être au moins 1.
!>          Si JOBZ  = 'N' et N > 1, LRWORK doit être au moins N.
!>          Si JOBZ  = 'V' et N > 1, LRWORK doit être au moins
!>                         1 + 5*N + 2*N**2.
!>
!>          Si LRWORK = -1, alors une interrogation d'espace de travail est supposée ; la
!>          routine calcule uniquement les tailles optimales des tableaux WORK, RWORK
!>          et IWORK, renvoie ces valeurs comme premières entrées
!>          des tableaux WORK, RWORK et IWORK, et aucun message d'erreur
!>          lié à LWORK ou LRWORK ou LIWORK n'est émis par XERBLA.
!
>

IWOR

K 
!>          IWORK est un tableau INTEGER, dimension (MAX(1,LIWORK))
!>          À la sortie, si INFO = 0, IWORK(1) renvoie le LIWORK optimal.
!
>

LIWOR

K 
!>          LIWORK est INTEGER
!>          La dimension du tableau IWORK.
!>          Si N <= 1,                LIWORK doit être au moins 1.
!>          Si JOBZ  = 'N' et N > 1, LIWORK doit être au moins 1.
!>          Si JOBZ  = 'V' et N > 1, LIWORK doit être au moins 3 + 5*N.
!>
!>          Si LIWORK = -1, alors une interrogation d'espace de travail est supposée ; la
!>          routine calcule uniquement les tailles optimales des tableaux WORK, RWORK
!>          et IWORK, renvoie ces valeurs comme premières entrées
!>          des tableaux WORK, RWORK et IWORK, et aucun message d'erreur
!>          lié à LWORK ou LRWORK ou LIWORK n'est émis par XERBLA.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 :  si INFO = i et JOBZ = 'N', alors l'algorithme a échoué
!>                 à converger ; i éléments hors diagonale d'une forme
!>                 tridiagonale intermédiaire n'ont pas convergé vers zéro ;

!>                 mod(INFO,N+1).
!

!> matrice symétrique réelle A en utilisant la technique en 2 étapes pour
!> la réduction en tridiagonale. Si des vecteurs propres sont souhaités, il utilise un

!>          Si JOBZ = 'N', alors à la sortie le triangle inférieur (si UPLO='L')c==============================================================================c
c                                                                              c
c ZHEEVD_2STAGE Computes the eigenvalues and, optionally, the                  c
c                  left and/or right eigenvectors for HE matrices             c
c                                                                              c
c==============================================================================c
c
c Purpose:
!>          ZHEEVD_2STAGE computes all eigenvalues and, optionally, eigenvectors of a
!>          complex Hermitian matrix A using the 2stage technique for
!>          the reduction to tridiagonal.  If eigenvectors are desired, it uses a
!>          divide and conquer algorithm.
!>
!> Parameters
!> ----------
!>
!> JOBZ    (CHARACTER*1)
!>         = 'N':  Compute eigenvalues only;
!>         = 'V':  Compute eigenvalues and eigenvectors.
!>                 Not available in this release.
!>
!> UPLO    (CHARACTER*1)
!>         = 'U':  Upper triangle of A is stored;
!>         = 'L':  Lower triangle of A is stored.
!>
!> N       (INTEGER)
!>         The order of the matrix A.  N >= 0.
!>
!> A       (COMPLEX*16 array, dimension (LDA, N))
!>         On entry, the Hermitian matrix A.  If UPLO = 'U', the
!>         leading N-by-N upper triangular part of A contains the
!>         upper triangular part of the matrix A.  If UPLO = 'L',
!>         the leading N-by-N lower triangular part of A contains
!>         the lower triangular part of the matrix A.
!>         On exit, if JOBZ = 'V', then if INFO = 0, A contains the
!>         orthonormal eigenvectors of the matrix A.
!>         If JOBZ = 'N', then on exit the lower triangle (if UPLO='L')
!>         or the upper triangle (if UPLO='U') of A, including the
!>         diagonal, is destroyed.
!>
!> LDA     (INTEGER)
!>         The leading dimension of the array A.  LDA >= max(1,N).
!>
!> W       (DOUBLE PRECISION array, dimension (N))
!>         If INFO = 0, the eigenvalues in ascending order.
!>
!> WORK    (COMPLEX*16 array, dimension (MAX(1,LWORK)))
!>         On exit, if INFO = 0, WORK(1) returns the optimal LWORK.
!>
!> LWORK   (INTEGER)
!>         The dimension of the array WORK.
!>         If N <= 1,               LWORK must be at least 1.
!>         If JOBZ = 'N' and N > 1, LWORK must be queried.
!>                                  LWORK = MAX(1, dimension) where
!>                                  dimension = max(stage1,stage2) + (KD+1)*N + N+1
!>                                            = N*KD + N*max(KD+1,FACTOPTNB)
!>                                              + max(2*KD*KD, KD*NTHREADS)
!>                                              + (KD+1)*N + N+1
!>                                  where KD is the blocking size of the reduction,
!>                                  FACTOPTNB is the blocking used by the QR or LQ
!>                                  algorithm, usually FACTOPTNB=128 is a good choice
!>                                  NTHREADS is the number of threads used when
!>                                  openMP compilation is enabled, otherwise =1.
!>         If JOBZ = 'V' and N > 1, LWORK must be at least 2*N + N**2
!>
!>         If LWORK = -1, then a workspace query is assumed; the routine
!>         only calculates the optimal sizes of the WORK, RWORK and
!>         IWORK arrays, returns these values as the first entries of
!>         the WORK, RWORK and IWORK arrays, and no error message
!>         related to LWORK or LRWORK or LIWORK is issued by XERBLA.
!>
!> RWORK   (DOUBLE PRECISION array, dimension (LRWORK))
!>         On exit, if INFO = 0, RWORK(1) returns the optimal LRWORK.
!>
!> LRWORK  (INTEGER)
!>         The dimension of the array RWORK.
!>         If N <= 1,                LRWORK must be at least 1.
!>         If JOBZ  = 'N' and N > 1, LRWORK must be at least N.
!>         If JOBZ  = 'V' and N > 1, LRWORK must be at least
!>                        1 + 5*N + 2*N**2.
!>
!>         If LRWORK = -1, then a workspace query is assumed; the
!>         routine only calculates the optimal sizes of the WORK, RWORK
!>         and IWORK arrays, returns these values as the first entries
!>         of the WORK, RWORK and IWORK arrays, and no error message
!>         related to LWORK or LRWORK or LIWORK is issued by XERBLA.
!>
!> IWORK   (INTEGER array, dimension (MAX(1,LIWORK)))
!>         On exit, if INFO = 0, IWORK(1) returns the optimal LIWORK.
!>
!> LIWORK  (INTEGER)
!>         The dimension of the array IWORK.
!>         If N <= 1,                LIWORK must be at least 1.
!>         If JOBZ  = 'N' and N > 1, LIWORK must be at least 1.
!>         If JOBZ  = 'V' and N > 1, LIWORK must be at least 3 + 5*N.
!>
!>         If LIWORK = -1, then a workspace query is assumed; the
!>         routine only calculates the optimal sizes of the WORK and
!>         IWORK arrays, returns these values as the first entries of
!>         the WORK and IWORK arrays, and no error message related to
!>         LWORK or LIWORK is issued by XERBLA.
!>
!> INFO    (INTEGER)
!>         = 0:  successful exit
!>         < 0:  if INFO = -i, the i-th argument had an illegal value
!>         > 0:  if INFO = i and JOBZ = 'N', then the algorithm failed
!>               to converge; i off-diagonal elements of an intermediate
!>               tridiagonal form did not converge to zero;
!>               if INFO = i and JOBZ = 'V', then the algorithm failed
!>               to compute an eigenvalue while working on the submatrix
!>               lying in rows and columns INFO/(N+1) through
!>               mod(INFO,N+1).
!>
      SUBROUTINE ZHEEVD_2STAGE(JOBZ, UPLO, N, A, LDA, W, WORK, LWORK, RWORK,
     $                         LRWORK, IWORK, LIWORK, INFO)
C
C     .. Scalar Arguments ..
      CHARACTER*1 JOBZ, UPLO
      INTEGER N, LDA, LWORK, LRWORK, LIWORK, INFO

      END IF

         INFO = -2
      END IF
      IF (N .LT. 0) THEN
         INFO = -3
      END IF

         IF (JOBZ .EQ. 'N') THEN
            LWKOPT = 1
         ELSE IF (JOBZ .EQ. 'V') THEN
            LWKOPT = 2*N + N**2
         END IF
         IF (LWORK .LT. LWKOPT) THEN
           INFO = -9
         END IF
      END IF
      IF (N .GT. 1 .AND. LRWORK .LT. 1) THEN
         IF (JOBZ .EQ. 'N') THEN
            LRWKOPT = N
         ELSE IF (JOBZ .EQ. 'V') THEN
            LRWKOPT = 1 + 5*N + 2*N**2
         END IF
         IF (LRWORK .LT. LRWKOPT) THEN
           INFO = -11
         END IF
      END IF

            LIWKOPT = 1
         ELSE IF (JOBZ .EQ. 'V') THEN

           INFO = -13
         END IF

            INFO = -11
         END IF

            INFO = -13
         END IF

            INFO = -9
         END IF

            RWORK(1) = DBLE(1 + 5*N + 2*N**2)
            INFO = -11
         END IF

            IWORK(1) = 3 + 5*N
            INFO = -13
         END IF
      END IF

C     If workspace query is satisfied, return optimal sizes.
C
      IF (INFO .LT. 0) THEN
         CALL XERBLA('ZHEEVD_2STAGE', -INFO)
         RETURN
      END IF
C
C     Call the implementation routine.
C
      CALL ZHEEVD_2STAGE_IMPL(JOBZ, UPLO, N, A, LDA, W, WORK, LWORK, RWORK,
