subroutine chemv (character uplo, integer n, complex alpha, complex, dimension(lda,*) a, integer lda, complex, dimension(*) x, integer incx, complex beta, complex, dimension(*) y, integer

incy)

CHEM

V

But

: 
!>
!> CHEMV effectue l'opération matrice-vecteur
!>
!>    y := alpha*A*x + beta*y,
!>
!> où alpha et beta sont des scalaires, x et y sont des vecteurs à n éléments et
!> A est une matrice hermitienne de dimension n par n.
!
>

Paramètre

s
UPLO 
!>          UPLO est de type CHARACTER*1
!>           À l'entrée, UPLO spécifie si la partie triangulaire supérieure ou
!>           inférieure du tableau A doit être référencée comme suit :
!>
!>              UPLO = 'U' ou 'u'   Seule la partie triangulaire supérieure de A
!>                                  doit être référencée.
!>
!>              UPLO = 'L' ou 'l'   Seule la partie triangulaire inférieure de A
!>                                  doit être référencée.
!
>

N 
!>          N est de type INTEGER
!>           À l'entrée, N spécifie l'ordre de la matrice A.
!>           N doit être au moins égal à zéro.
!
>

ALPH

A 
!>          ALPHA est de type COMPLEX
!>           À l'entrée, ALPHA spécifie le scalaire alpha.
!
>

A 
!>          A est un tableau COMPLEX, dimension ( LDA, N )
!>           Avant l'entrée avec UPLO = 'U' ou 'u', la partie triangulaire
!>           supérieure de dimension n par n du tableau A doit contenir la partie
!>           triangulaire supérieure de la matrice hermitienne et la partie
!>           triangulaire strictement inférieure de A n'est pas référencée.
!>           Avant l'entrée avec UPLO = 'L' ou 'l', la partie triangulaire
!>           inférieure de dimension n par n du tableau A doit contenir la partie
!>           triangulaire inférieure de la matrice hermitienne et la partie
!>           triangulaire strictement supérieure de A n'est pas référencée.
!>           Notez que les parties imaginaires des éléments diagonaux n'ont pas
!>           besoin d'être définies et sont supposées être égales à zéro.
!
>

LD

A 
!>          LDA est de type INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle que
!>           déclarée dans le (sous-)programme appelant. LDA doit être au moins
!>           max( 1, n ).
!
>

X 
!>          X est un tableau COMPLEX, dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           Avant l'entrée, le tableau incrémenté X doit contenir le vecteur x
!>           à n éléments.
!
>

INC

X 
!>          INCX est de type INTEGER
!>           À l'entrée, INCX spécifie l'incrément pour les éléments de X.
!>           INCX ne doit pas être égal à zéro.
!
>

BET

A 
!>          BETA est de type COMPLEX
!>           À l'entrée, BETA spécifie le scalaire beta. Lorsque BETA est fourni
!>           comme zéro, Y n'a pas besoin d'être défini en entrée.
!
>

Y 
!>          Y est un tableau COMPLEX, dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           Avant l'entrée, le tableau incrémenté Y doit contenir le vecteur y
!>           à n éléments. À la sortie, Y est écrasé par le vecteur y mis à jour.
!
>

INC

Y 
!>          INCY est de type INTEGER
!>           À l'entrée, INCY spécifie l'incrément pour les éléments de Y.
!>           INCY ne doit pas être égal à zéro.
!
>
