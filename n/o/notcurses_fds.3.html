NOM

notcurses_fds - déverser des descripteurs de fichiers et des sous-processus dans des plans

SYNOPSIS

#include <notcurses/notcurses.h>

struct ncplane;
struct ncfdplane;
struct ncsubproc;
typedef struct ncfdplane_options {
  void* curry; // paramètre fourni aux callbacks
  bool follow; // continuer à lire après avoir atteint la fin ?
} ncfdplane_options;
typedef struct ncsubproc_options {
  void* curry; // paramètre fourni aux callbacks
  uint64_t restart_period;  // redémarrer après la sortie
} ncsubproc_options;

typedef int(ncfdplane_callback)(struct ncfdplane n, const void* buf, size_t s, void* curry);

typedef int(ncfdplane_done_cb)(struct ncfdplane n, int fderrno, void* curry);

struct ncfdplane* ncfdplane_create(struct ncplane* n, const ncfdplane_options* opts, int fd, ncfdplane_callback cbfxn, ncfdplane_done_cb donecbfxn);

struct ncplane* ncfdplane_plane(struct ncfdplane* n);

int ncfdplane_destroy(struct ncfdplane* n);

struct ncsubproc* ncsubproc_createv(struct ncplane* n, const ncsubproc_options* opts, const char* bin, char* const arg[], ncfdplane_callback cbfxn, ncfdplane_done_cb donecbfxn);

struct ncsubproc* ncsubproc_createvp(struct ncplane* n, const ncsubproc_options* opts, const char* bin, char* const arg[], ncfdplane_callback cbfxn, ncfdplane_done_cb donecbfxn);

struct ncsubproc* ncsubproc_createvpe(struct ncplane* n, const ncsubproc_options* opts, const char* bin, char* const arg[], char* const env[], ncfdplane_callback cbfxn, ncfdplane_done_cb donecbfxn);

struct ncplane* ncsubproc_plane(struct ncsubproc* n);

int ncsubproc_destroy(struct ncsubproc* n);

DESCRIPTION

Ces widgets provoquent la lecture d'un descripteur de fichier jusqu'à EOF, et le déversement dans un ncplane déroulant. La lecture aura lieu dans un contexte géré par notcurses (les détails de ce contexte ne sont pas définis et ne doivent pas être surdépendus), qui invoquera les callbacks fournis avec les données lues. Essentiellement, ce sont simplement des interfaces portables pour la lecture asynchrone, ncsubproc fournissant également la gestion des sous-processus.

Si ncsubproc_destroy est appelé avant que le sous-processus ne soit sorti, il recevra un SIGKILL. Si ncsubproc_destroy ou ncfdplane_destroy est appelé pendant qu'un callback est en cours d'invocation, la fonction de destruction bloquera jusqu'à ce que le callback soit terminé. Si un callback utilisateur retourne une valeur non nulle, l'objet appelant se détruira lui-même. Si un callback utilisateur appelle lui-même la fonction de destruction pertinente, le thread se terminera comme si une valeur non nulle avait été retournée, et les ressources du ncsubproc seront alors récupérées.

Il est essentiel que la fonction de destruction soit appelée une seule fois, que ce soit depuis le contexte du thread ou de l'extérieur de ce contexte.

NOTES

ncsubproc utilise des pidfds et pidfd_send_signal(2), et utilise ainsi de manière fiable les signaux (il ne ciblera jamais un processus autre que le sous-processus réel).
