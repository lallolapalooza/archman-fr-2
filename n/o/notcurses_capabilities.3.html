NOM

notcurses_capabilities - détection des capacités d'exécution

SYNOPSIS

#include <notcurses/notcurses.h>

const char* notcurses_detected_terminal(const struct notcurses* nc);

unsigned notcurses_supported_styles(const struct notcurses* nc);

unsigned notcurses_palette_size(const struct notcurses* nc);

bool notcurses_cantruecolor(const struct notcurses* nc);

bool notcurses_canfade(const struct notcurses* nc);

bool notcurses_canchangecolor(const struct notcurses* nc);

bool notcurses_canopen_images(const struct notcurses* nc);

bool notcurses_canopen_videos(const struct notcurses* nc);

bool notcurses_canutf8(const struct notcurses* nc);

bool notcurses_canhalfblock(const struct notcurses* nc);

bool notcurses_canquadrant(const struct notcurses* nc);

bool notcurses_cansextant(const struct notcurses* nc);

bool notcurses_canoctant(const struct notcurses* nc);

bool notcurses_canbraille(const struct notcurses* nc);

bool notcurses_canpixel(const struct notcurses* nc);

typedef enum {
  NCPIXEL_NONE = 0,
  NCPIXEL_SIXEL,           // sixel
  NCPIXEL_LINUXFB,         // linux framebuffer
  NCPIXEL_ITERM2,          // iTerm2
  NCPIXEL_KITTY_STATIC,    // kitty pre-0.20.0
  NCPIXEL_KITTY_ANIMATED,  // kitty pre-0.22.0
  NCPIXEL_KITTY_SELFREF,   // kitty 0.22.0+, wezterm
} ncpixelimpl_e;

ncpixelimpl_e notcurses_check_pixel_support(struct notcurses* nc);

DESCRIPTION

notcurses_detected_terminal renvoie une chaîne de caractères libre décrivant le terminal détecté. La détection de terminal prend en compte toute base de données de terminaux spécifiée (voir notcurses_init(3)), les variables d'environnement TERM, TERM_PROGRAM et TERM_PROGRAM_VERSION, la réponse à une séquence de contrôle XTGETTCAP[TN], la réponse aux séquences de contrôle d'attributs d'envoi d'appareil primaires, secondaires et tertiaires, et la phase de la lune. Vous ne devriez pas baser la logique sur cette réponse ; toutes les propriétés pertinentes du terminal devraient être abstraites par Notcurses. Ceci n'est disponible que pour le diagnostic.

notcurses_supported_styles renvoie un masque de bits représentant les styles pour lesquels le terminal annonce un support.

notcurses_palette_size renvoie la taille de la palette du terminal, utilisée pour la couleur indexée par palette. Elle renverra toujours au moins 1. Ceci est indépendant du support de la couleur TrueColor RGB. Aucun terminal n'est connu pour supporter plus de 256 couleurs indexées.

notcurses_cantruecolor renvoie true si le terminal annonce un support pour la couleur TrueColor RGB. Notez que les API RGB de Notcurses peuvent être utilisées même en l'absence de support RGB du terminal (Notcurses mappera les valeurs RGB à la palette).

notcurses_canfade renvoie true si Notcurses dispose d'un moyen d'effectuer des fondus.

notcurses_canchangecolor renvoie true si le terminal annonce un support pour modifier ses entrées de palette.

notcurses_canopen_images renvoie true si Notcurses a été compilé avec le support multimédia.

notcurses_canopen_video renvoie true si Notcurses a été compilé avec le support multimédia capable de décoder des vidéos.

notcurses_canutf8 renvoie true si la locale configurée utilise l'encodage UTF-8, et si la locale a été chargée avec succès.

notcurses_canoctant renvoie true si les heuristiques suggèrent que le terminal peut correctement afficher les 16 octants Unicode. notcurses_cansextant renvoie true si les heuristiques suggèrent que le terminal peut correctement afficher les 13 sextants Unicode, et notcurses_canquadrant et notcurses_canhalfblock renvoient true si les heuristiques suggèrent que le terminal peut correctement afficher les quadrants et les demi-blocs Unicode, respectivement. notcurses_canbraille renvoie true si le Braille Unicode est censé fonctionner sur le terminal. Aucune de ces fonctions ne renvoie true à moins que l'encodage UTF-8 ne soit utilisé.

notcurses_check_pixel_support renvoie une implémentation de pixel non nulle si le support des bitmaps (via un quelconque mécanisme) a été détecté, et sinon 0 (NCPIXEL_NONE). notcurses_canpixel réduit cela à un booléen.

NOTES

Certains terminaux annoncent un support pour TrueColor, mais réduisent ensuite la qualité ou dégradent autrement le RGB fourni. Dans ce cas, notcurses_cantruecolor renverra true, mais le spectre complet ne sera pas disponible.
