Paquet: libnbd

nbd_opt_list(3) LIBNBD nbd_opt_list(3)

NOM

nbd_opt_list - demande au serveur de lister toutes les exportations pendant la négociation

SYNTAXE

#include <libnbd.h>
typedef struct {
  int (*callback) (void *user_data, const char *name,
                   const char *description);
  void *user_data;
  void (*free) (void *user_data);
} nbd_list_callback;
int nbd_opt_list (
      struct nbd_handle *h,
      nbd_list_callback list_callback
    );

DESCRIPTION

Demande au serveur de lister toutes les exportations qu'il prend en charge. Ceci ne peut être utilisé que si nbd_set_opt_mode(3) a activé le mode option.

La fonction "list" est appelée une fois par exportation annoncée, avec toute donnée "user_data" passée à cette fonction, et avec "name" et "description" fournis par le serveur. De nombreux serveurs omettent les descriptions, auquel cas "description" sera une chaîne vide. N'oubliez pas qu'il n'est pas sûr d'appeler nbd_set_export_name(3) depuis le contexte de la fonction de rappel ; votre code doit plutôt copier tout "name" nécessaire pour une utilisation ultérieure après la fin de cette fonction. Actuellement, la valeur de retour du rappel est ignorée, bien qu'un retour de -1 doive être évité.

Par souci de commodité, lorsque cette fonction réussit, elle renvoie le nombre d'exportations annoncées par le serveur.

Tous les serveurs ne comprennent pas cette requête et, même lorsqu'elle est comprise, le serveur peut intentionnellement envoyer une liste vide pour éviter d'être une fuite d'informations, peut rencontrer une erreur après avoir livré des résultats partiels, ou peut refuser de répondre à plus d'une requête par connexion pour éviter des négociations qui ne se résolvent pas. Ainsi, cette fonction peut réussir même lorsqu'aucune exportation n'est signalée, ou peut échouer mais avoir une liste non vide. De même, le protocole NBD ne spécifie pas de limite supérieure au nombre d'exportations qui pourraient être annoncées, donc le code client doit être conscient qu'un serveur peut envoyer une longue liste.

Pour nbd-server(1), vous devrez autoriser les clients à faire des requêtes de liste en ajoutant "allowlist=true" à la section "[generic]" de /etc/nbd-server/config. Pour qemu-nbd(8), une description est définie avec -D.

VALEUR DE RETOUR

Cet appel renvoie un entier ≥ 0.

ERREURS

En cas d'erreur, -1 est renvoyé.

Consultez "GESTION DES ERREURS" dans libnbd(3) pour obtenir plus de détails sur l'erreur.

Les paramètres suivants ne doivent pas être NULL : "h". Pour plus d'informations, voir "Paramètres non NULL" dans libnbd(3).

ÉTAT DU HANDLE

nbd_opt_list peut être appelé lorsque le handle est dans l'état suivant :

┌─────────────────────────────────────┬─────────────────────────┐
│ Handle créé, avant connexion      │ ❌ erreur               │
│ Connexion en cours                    │ ❌ erreur               │
│ Connexion & handshake (opt_mode) │ ✅ autorisé             │
│ Connecté au serveur               │ ❌ erreur               │
│ Connexion fermée                  │ ❌ erreur               │
│ Handle mort                       │ ❌ erreur               │
└─────────────────────────────────────┴─────────────────────────┘

VERSION

Cette fonction est apparue pour la première fois dans libnbd 1.4.

Si vous avez besoin de tester si cette fonction est disponible au moment de la compilation, vérifiez si la macro suivante est définie :

#define LIBNBD_HAVE_NBD_OPT_LIST 1

EXEMPLE

Cet exemple est également disponible sous le nom examples/list-exports.c dans le code source de libnbd.

/* Cet exemple montre comment lister les exportations NBD.
 *
 * Pour tester cela avec qemu-nbd :
 *   $ qemu-nbd -x "hello" -t -k /tmp/sock disk.img
 *   $ ./run examples/list-exports /tmp/sock
 *   [0] hello
 *   Quelle exportation connecter (-1 pour quitter) ? 0
 *   Connexion à hello ...
 *   /tmp/sock: hello: taille = 2048 octets
 *
 * Pour tester cela avec nbdkit (nécessite 1.22) :
 *   $ nbdkit -U /tmp/sock sh - <<\EOF
 *   case $1 in
 *     list_exports) echo NAMES; echo foo; echo foobar ;;
 *     open) echo "$3" ;;
 *     get_size) echo "$2" | wc -c ;;
 *     pread) echo "$2" | dd bs=1 skip=$4 count=$3 ;;
 *     *) exit 2 ;;
 *   esac
 *   EOF
 *   $ ./run examples/list-exports /tmp/sock
 *   [0] foo
 *   [1] foobar
 *   Quelle exportation connecter (-1 pour quitter) ? 1
 *   Connexion à foobar ...
 *   /tmp/sock: foobar: taille = 7 octets
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <stdint.h>
#include <string.h>
#include <inttypes.h>
#include <errno.h>
#include <libnbd.h>
struct export_list {
  int i;
  char **names;
};
/* Fonction de rappel pour nbd_opt_list */
static int
list_one (void *opaque, const char *name,
          const char *description)
{
  struct export_list *l = opaque;
  char **names;
  printf ("[%d] %s\n", l->i, name);
  if (*description)
    printf ("  (%s)\n", description);
  names = realloc (l->names,
                   (l->i + 1) * sizeof *names);
  if (!names) {
    perror ("realloc");
    exit (EXIT_FAILURE);
  }
  names[l->i] = strdup (name);
  if (!names[l->i]) {
    perror ("strdup");
    exit (EXIT_FAILURE);
  }
  l->names = names;
  l->i++;
  return 0;
}
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  int i;
  const char *name;
  int64_t size;
  struct export_list list = { 0 };
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* Créer le handle libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Définir le mode opt. */
  nbd_set_opt_mode (nbd, true);
  /* Se connecter au serveur NBD via un
   * socket de domaine Unix. Si nous ne
   * parvenons pas à passer en mode option,
   * une liste n'est pas possible.
   */
  if (nbd_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  if (!nbd_aio_is_negotiating (nbd)) {
    fprintf (stderr, "Le serveur ne prend pas en charge "
             "la liste des exportations.\n");
    exit (EXIT_FAILURE);
  }
  /* Afficher la liste des exportations. */
  if (nbd_opt_list (nbd,
                    (nbd_list_callback) {
                      .callback = list_one,
                      .user_data = &list, }) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Afficher la liste des exportations. */
  printf ("Quelle exportation connecter ? ");
  if (scanf ("%d", &i) != 1) exit (EXIT_FAILURE);
  if (i == -1) {
    if (nbd_opt_abort (nbd) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
    nbd_close (nbd);
    exit (EXIT_SUCCESS);
  }
  if (i < 0 || i >= list.i) {
    fprintf (stderr, "indice %d hors limites", i);
    exit (EXIT_FAILURE);
  }
  name = list.names[i];
  printf ("Connexion à %s ...\n", name);
  /* Reprendre la connexion à l'exportation choisie. */
  if (nbd_set_export_name (nbd, name) == -1 ||
      nbd_opt_go (nbd) == -1) {
    fprintf (stderr, "%s\n"
;, nbd_get_error ()); exit (EXIT_FAILURE); } if (!nbd_aio_is_ready (nbd)) { fprintf (stderr, "le serveur s'est fermé prématurément\n"); exit (EXIT_FAILURE); } /* Lire la taille en octets et l'afficher. */ size = nbd_get_size (nbd); if (size == -1) { fprintf (stderr, "%s\n", nbd_get_error ()); exit (EXIT_FAILURE); } printf ("%s: %s: taille = %" PRIi64 " octets\n", argv[1], name, size); /* Fermer le handle libnbd. */ nbd_close (nbd); for (i = 0; i < list.i; i++) free (list.names[i]); free (list.names); exit (EXIT_SUCCESS); }
