Paquet: libnbd

Licences: LGPL-2.1-or-later

nbd_set_opt_mode(3) LIBNBD nbd_set_opt_mode(3)

NOM

nbd_set_opt_mode - contrôler le mode d'options, pour la mise en pause pendant la négociation des options

SYNTAXE

#include <libnbd.h>
int nbd_set_opt_mode (
      struct nbd_handle *h, bool enable
    );

DESCRIPTION

Définir ce drapeau sur vrai afin de demander qu'une commande de connexion "nbd_connect_*" fasse une pause pour la négociation des options plutôt que de procéder jusqu'à l'état prêt, lors de la communication avec un serveur nouveau style. Ce réglage n'a aucun effet lors de la connexion à un serveur ancien style.

Notez que libnbd tente par défaut "NBD_OPT_STARTTLS", "NBD_OPT_EXTENDED_HEADERS", et "NBD_OPT_STRUCTURED_REPLY" avant de vous laisser contrôler les étapes de négociation restantes ; si vous avez besoin de contrôle sur ces étapes également, définissez d'abord nbd_set_tls(3) sur "LIBNBD_TLS_DISABLE", et nbd_set_request_extended_headers(3) ou nbd_set_request_structured_replies(3) sur faux, avant de commencer la tentative de connexion.

Lorsque le mode option est activé, vous avez un contrôle précis sur les options qui sont négociées, par rapport à la valeur par défaut où le serveur négocie tout en votre nom en utilisant les réglages effectués avant de commencer la connexion. Pour quitter le mode et passer à l'état prêt, vous devez utiliser nbd_opt_go(3) avec succès ; un échec de nbd_opt_go(3) retourne à l'état de négociation pour permettre un changement de nom d'exportation avant de réessayer. Vous pouvez également utiliser nbd_opt_abort(3) ou nbd_shutdown(3) pour terminer la connexion sans finaliser la négociation.

VALEUR DE RETOUR

Si l'appel réussit, la fonction retourne 0.

ERREURS

En cas d'erreur, -1 est retourné.

Reportez-vous à « GESTION DES ERREURS » dans libnbd(3) pour savoir comment obtenir plus de détails sur l'erreur.

Les paramètres suivants ne doivent pas être NULL : "h". Pour plus d'informations, voir « Paramètres non NULL » dans libnbd(3).

ETAT DU HANDLE

nbd_set_opt_mode peut être appelé lorsque le handle est dans l'état suivant :

┌─────────────────────────────────────┬─────────────────────────┐
│ Handle créé, avant la connexion   │ ✅ autorisé             │
│ Connexion en cours                  │ ❌ erreur               │
│ Connexion & handshake (opt_mode) │ ❌ erreur               │
│ Connecté au serveur             │ ❌ erreur               │
│ Connexion fermée                │ ❌ erreur               │
│ Handle mort                       │ ❌ erreur               │
└─────────────────────────────────────┴─────────────────────────┘

VERSION

Cette fonction est apparue pour la première fois dans libnbd 1.4.

Si vous devez tester si cette fonction est disponible à la compilation, vérifiez si la macro suivante est définie :

#define LIBNBD_HAVE_NBD_SET_OPT_MODE 1

EXEMPLE

Cet exemple est également disponible sous le nom examples/list-exports.c dans le code source de libnbd.

/* Cet exemple montre comment lister les exports NBD.
 *
 * Pour tester cela avec qemu-nbd :
 *   $ qemu-nbd -x "hello" -t -k /tmp/sock disk.img
 *   $ ./run examples/list-exports /tmp/sock
 *   [0] hello
 *   Quel export connecter (-1 pour quitter) ? 0
 *   Connexion à hello ...
 *   /tmp/sock: hello: taille = 2048 octets
 *
 * Pour tester cela avec nbdkit (nécessite 1.22) :
 *   $ nbdkit -U /tmp/sock sh - <<\EOF
 *   case $1 in
 *     list_exports) echo NAMES; echo foo; echo foobar ;;
 *     open) echo "$3" ;;
 *     get_size) echo "$2" | wc -c ;;
 *     pread) echo "$2" | dd bs=1 skip=$4 count=$3 ;;
 *     *) exit 2 ;;
 *   esac
 *   EOF
 *   $ ./run examples/list-exports /tmp/sock
 *   [0] foo
 *   [1] foobar
 *   Quel export connecter (-1 pour quitter) ? 1
 *   Connexion à foobar ...
 *   /tmp/sock: foobar: taille = 7 octets
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <stdint.h>
#include <string.h>
#include <inttypes.h>
#include <errno.h>
#include <libnbd.h>
struct export_list {
  int i;
  char **names;
};
/* Fonction de rappel pour nbd_opt_list */
static int
list_one (void *opaque, const char *name,
          const char *description)
{
  struct export_list *l = opaque;
  char **names;
  printf ("[%d] %s\n", l->i, name);
  if (*description)
    printf ("  (%s)\n", description);
  names = realloc (l->names,
                   (l->i + 1) * sizeof *names);
  if (!names) {
    perror ("realloc");
    exit (EXIT_FAILURE);
  }
  names[l->i] = strdup (name);
  if (!names[l->i]) {
    perror ("strdup");
    exit (EXIT_FAILURE);
  }
  l->names = names;
  l->i++;
  return 0;
}
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  int i;
  const char *name;
  int64_t size;
  struct export_list list = { 0 };
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* Créer le handle libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Définir le mode opt. */
  nbd_set_opt_mode (nbd, true);
  /* Se connecter au serveur NBD via un
   * socket de domaine Unix.  Si nous ne
   * sommes pas entrés en mode option,
   * une liste n'est pas possible.
   */
  if (nbd_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  if (!nbd_aio_is_negotiating (nbd)) {
    fprintf (stderr, "Le serveur ne prend pas en charge "
             "la liste des exports.\n");
    exit (EXIT_FAILURE);
  }
  /* Afficher la liste des exports. */
  if (nbd_opt_list (nbd,
                    (nbd_list_callback) {
                      .callback = list_one,
                      .user_data = &list, }) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Afficher la liste des exports. */
  printf ("Quel export connecter ? ");
  if (scanf ("%d", &i) != 1) exit (EXIT_FAILURE);
  if (i == -1) {
    if (nbd_opt_abort (nbd) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
    nbd_close (nbd);
    exit (EXIT_SUCCESS);
  }
  if (i < 0 || i >= list.i) {
    fprintf (stderr, "l'index %d est hors limites", i);
    exit (EXIT_FAILURE);
  }
  name = list.names[i];
  printf ("Connexion à %s ...\n", name);
  /* Reprendre la connexion à l'exportation choisie. */
  if (nbd_set_export_name (nbd, name) == -1 ||
      nbd_opt_go (nbd) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  if (!nbd_aio_is_ready (nbd)) {
    fprintf (stderr, "le serveur s'est fermé prématurément\n");
    exit (EXIT_FAILURE);
  }
  /* Lire la taille en octets et l'afficher. */
  size = nbd_get_size (nbd);
  if (size == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  printf ("%s: %s: taille = %" PRIi64 " octets\n",
          argv[1], name, size);
  /* Fermer le handle libnbd. */
  nbd_close (nbd);
  for (i = 0; i < list.i; i++)
    free (list.names[i]);
  free (list.names);
  exit
(EXIT_SUCCESS); }
