nbd_opt_go(3) LIBNBD nbd_opt_go(3)

NOM

nbd_opt_go - termine la négociation et passe à l'utilisation d'une exportation

SYNOPSIS

#include <libnbd.h>
int nbd_opt_go (
      struct nbd_handle *h
    );

DESCRIPTION

Demande au serveur de terminer la négociation et de passer à la diffusion de l'exportation spécifiée précédemment par le dernier appel à nbd_set_export_name(3) ou nbd_connect_uri(3). Ceci ne peut être utilisé que si nbd_set_opt_mode(3) a activé le mode option.

Par défaut, libnbd demandera automatiquement tous les contextes méta enregistrés par nbd_add_meta_context(3) dans le cadre de cet appel ; mais cela peut être supprimé avec nbd_set_request_meta_context(3), particulièrement si nbd_opt_set_meta_context(3) a été utilisé plus tôt dans la séquence de négociation.

Si cela échoue, le serveur peut toujours être en cours de négociation, où il est possible de tenter une autre option comme un nom d'exportation différent ; bien que les anciens serveurs aient plutôt terminé la connexion.

VALEUR DE RETOUR

Si l'appel réussit, la fonction retourne 0.

ERREURS

En cas d'erreur, -1 est retourné.

Voir "GESTION DES ERREURS" dans libnbd(3) pour obtenir plus de détails sur l'erreur.

Les paramètres suivants ne doivent pas être NULL : "h". Pour plus d'informations, voir "Paramètres non-NULL" dans libnbd(3).

ETAT DU HANDLE

nbd_opt_go peut être appelé lorsque le handle est dans l'état suivant :

┌─────────────────────────────────────┬─────────────────────────┐
│ Handle créé, avant connexion      │ ❌ erreur              │
│ Connexion en cours                  │ ❌ erreur              │
│ Connexion & handshake (opt_mode)│ ✅ autorisé            │
│ Connecté au serveur             │ ❌ erreur              │
│ Connexion fermée                │ ❌ erreur              │
│ Handle mort                       │ ❌ erreur              │
└─────────────────────────────────────┴─────────────────────────┘

VERSION

Cette fonction est apparue pour la première fois dans libnbd 1.4.

Si vous avez besoin de tester si cette fonction est disponible au moment de la compilation, vérifiez si la macro suivante est définie :

#define LIBNBD_HAVE_NBD_OPT_GO 1

EXEMPLE

Cet exemple est également disponible sous le nom examples/list-exports.c dans le code source de libnbd.

/* Cet exemple montre comment lister les exportations NBD.
 *
 * Pour tester ceci avec qemu-nbd :
 *   $ qemu-nbd -x "hello" -t -k /tmp/sock disk.img
 *   $ ./run examples/list-exports /tmp/sock
 *   [0] hello
 *   Quelle exportation choisir (-1 pour quitter) ? 0
 *   Connexion à hello ...
 *   /tmp/sock: hello: taille = 2048 octets
 *
 * Pour tester ceci avec nbdkit (nécessite 1.22) :
 *   $ nbdkit -U /tmp/sock sh - <<\EOF
 *   case $1 in
 *     list_exports) echo NAMES; echo foo; echo foobar ;;
 *     open) echo "$3" ;;
 *     get_size) echo "$2" | wc -c ;;
 *     pread) echo "$2" | dd bs=1 skip=$4 count=$3 ;;
 *     *) exit 2 ;;
 *   esac
 *   EOF
 *   $ ./run examples/list-exports /tmp/sock
 *   [0] foo
 *   [1] foobar
 *   Quelle exportation choisir (-1 pour quitter) ? 1
 *   Connexion à foobar ...
 *   /tmp/sock: foobar: taille = 7 octets
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <stdint.h>
#include <string.h>
#include <inttypes.h>
#include <errno.h>
#include <libnbd.h>
struct export_list {
  int i;
  char **names;
};
/* Fonction de rappel pour nbd_opt_list */
static int
list_one (void *opaque, const char *name,
          const char *description)
{
  struct export_list *l = opaque;
  char **names;
  printf ("[%d] %s\n", l->i, name);
  if (*description)
    printf ("  (%s)\n", description);
  names = realloc (l->names,
                   (l->i + 1) * sizeof *names);
  if (!names) {
    perror ("realloc");
    exit (EXIT_FAILURE);
  }
  names[l->i] = strdup (name);
  if (!names[l->i]) {
    perror ("strdup");
    exit (EXIT_FAILURE);
  }
  l->names = names;
  l->i++;
  return 0;
}
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  int i;
  const char *name;
  int64_t size;
  struct export_list list = { 0 };
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* Créer le handle libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Définir le mode opt. */
  nbd_set_opt_mode (nbd, true);
  /* Se connecter au serveur NBD via un
   * socket de domaine Unix. Si nous ne
   * sommes pas entrés en mode option, alors
   * une liste n'est pas possible.
   */
  if (nbd_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  if (!nbd_aio_is_negotiating (nbd)) {
    fprintf (stderr, "Le serveur ne supporte pas "
             "la liste des exportations.\n");
    exit (EXIT_FAILURE);
  }
  /* Afficher la liste des exportations. */
  if (nbd_opt_list (nbd,
                    (nbd_list_callback) {
                      .callback = list_one,
                      .user_data = &list, }) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Afficher la liste des exportations. */
  printf ("Quelle exportation choisir ? ");
  if (scanf ("%d", &i) != 1) exit (EXIT_FAILURE);
  if (i == -1) {
    if (nbd_opt_abort (nbd) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
    nbd_close (nbd);
    exit (EXIT_SUCCESS);
  }
  if (i < 0 || i >= list.i) {
    fprintf (stderr, "l'index %d est hors limites", i);
    exit (EXIT_FAILURE);
  }
  name = list.names[i];
  printf ("Connexion à %s ...\n", name);
  /* Reprendre la connexion à l'exportation choisie. */
  if (nbd_set_export_name (nbd, name) == -1 ||
      nbd_opt_go (nbd) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  if (!nbd_aio_is_ready (nbd)) {
    fprintf (stderr, "le serveur s'est fermé prématurément\n");
    exit (EXIT_FAILURE);
  }
  /* Lire la taille en octets et l'afficher. */
  size = nbd_get_size (nbd);
  if (size == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  printf ("%s: %s: taille = %" PRIi64 " octets\n",
          argv[1], name, size);
  /* Fermer le handle libnbd. */
  nbd_close (nbd);
  for (i = 0; i < list.i; i++)
    free (list.names[i]);
  free (list.names);
  exit (EXIT_SUCCESS);
}
