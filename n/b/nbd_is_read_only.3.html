nbd_is_read_only(3)

Paquet: libnbd

nbd_is_read_only(3) LIBNBD nbd_is_read_only(3)

NOM

nbd_is_read_only - l'export NBD est-il en lecture seule ?

SYNTAXE

#include <libnbd.h>
int nbd_is_read_only (
      struct nbd_handle *h
    );

DESCRIPTION

Retourne vrai si l'export NBD est en lecture seule ; les écritures et les opérations similaires à l'écriture échoueront.

Cet appel ne bloque pas, car il retourne des données qui sont sauvegardées dans le handle depuis la négociation du protocole NBD.

VALEUR DE RETOUR

Cet appel retourne une valeur booléenne.

ERREURS

En cas d'erreur, -1 est retourné.

Se référer à « GESTION DES ERREURS » dans libnbd(3) pour obtenir plus de détails sur l'erreur.

Les paramètres suivants ne doivent pas être NULL : "h". Pour plus d'informations, voir « Paramètres non NULL » dans libnbd(3).

ÉTAT DU HANDLE

nbd_is_read_only peut être appelé lorsque le handle est dans les états suivants :

┌─────────────────────────────────────┬─────────────────────────┐
│ Handle créé, avant connexion       │ ❌ erreur              │
│ Connexion en cours                  │ ❌ erreur              │
│ Connexion & négociation (opt_mode)│ ✅ autorisé            │
│ Connecté au serveur               │ ✅ autorisé            │
│ Connexion fermée                  │ ✅ autorisé            │
│ Handle mort                       │ ❌ erreur              │
└─────────────────────────────────────┴─────────────────────────┘

VERSION

Cette fonction est apparue pour la première fois dans libnbd 1.0.

Si vous avez besoin de vérifier si cette fonction est disponible au moment de la compilation, vérifiez si la macro suivante est définie :

#define LIBNBD_HAVE_NBD_IS_READ_ONLY 1

EXEMPLE

Cet exemple est également disponible sous le nom examples/server-flags.c dans le code source de libnbd.

/* Cet exemple montre comment se connecter à un serveur NBD
 * et afficher les drapeaux d'exportation.
 *
 * Vous pouvez le tester avec nbdkit comme ceci :
 *
 * nbdkit -U - memory 1M \
 *   --run './server-flags $unixsocket'
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <libnbd.h>
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  char *str;
  int flag;
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* Créer le handle libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Demander des informations complètes. */
#if LIBNBD_HAVE_NBD_SET_FULL_INFO /* Ajouté en 1.4 */
  if (nbd_set_full_info (nbd, true) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
#endif
  /* Se connecter au serveur NBD via un
   * socket de domaine Unix.
   */
  if (nbd_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Voir si le serveur a fourni des détails supplémentaires,
   * en utilisant les fonctions ajoutées en 1.4
   */
#if LIBNBD_HAVE_NBD_GET_EXPORT_DESCRIPTION
  str = nbd_get_canonical_export_name (nbd);
  if (str)
    printf ("canonical_name = %s\n", str);
  free (str);
  str = nbd_get_export_description (nbd);
  if (str)
    printf ("description = %s\n", str);
  free (str);
#endif
  /* Lire et afficher les drapeaux. */
#define PRINT_FLAG(flag_fn)                     \
  flag = flag_fn (nbd);                         \
  if (flag == -1) {                             \
    fprintf (stderr, "%s\n", nbd_get_error ()); \
    exit (EXIT_FAILURE);                        \
  }                                             \
  printf (#flag_fn " = %s\n",                   \
          flag ? "true" : "false");
  PRINT_FLAG (nbd_can_cache);
  PRINT_FLAG (nbd_can_df);
  PRINT_FLAG (nbd_can_flush);
  PRINT_FLAG (nbd_can_fua);
  PRINT_FLAG (nbd_can_multi_conn);
  PRINT_FLAG (nbd_can_trim);
  PRINT_FLAG (nbd_can_zero);
#if LIBNBD_HAVE_NBD_CAN_FAST_ZERO
  /* Ajouté en 1.2 */
  PRINT_FLAG (nbd_can_fast_zero);
#endif
#if LIBNBD_HAVE_NBD_CAN_BLOCK_STATUS_PAYLOAD
  /* Ajouté en 1.18 */
  PRINT_FLAG (nbd_can_block_status_payload);
#endif
  PRINT_FLAG (nbd_is_read_only);
  PRINT_FLAG (nbd_is_rotational);
  /* Fermer le handle libnbd. */
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}

nbd_is_read_only(3)

nbd_is_read_only - Check if an NBD device is read-only

NAME

nbd_is_read_only - Check if an NBD device is read-only

SYNOPSIS

#include <libnbd.h>

int nbd_is_read_only(NbdHandle *h, int *is_read_only, NbdError **err);

DESCRIPTION

The nbd_is_read_only() function checks if the NBD device represented by the handle h is read-only.

If the device is read-only, the integer pointed to by is_read_only is set to 1. Otherwise, it is set to 0.

On success, the function returns 0. On failure, it returns -1 and sets the error object pointed to by err.

RETURN VALUE

Returns 0 on success, -1 on failure.

EXAMPLE

#include <libnbd.h>
#include <stdio.h>
#include <stdlib.h>

int main() {
    NbdHandle *h = NULL;
    NbdError *err = NULL;
    int is_read_only;

    h = nbd_connect_unix ("/var/run/nbd.sock", &err);
    if (h == NULL) {
        fprintf (stderr, "failed to connect to NBD server: %s\n",
                 nbd_err_message (err));
        nbd_free_error (err);
        exit (EXIT_FAILURE);
    }

    if (nbd_is_read_only (h, &is_read_only, &err) == -1) {
        fprintf (stderr, "failed to check read-only status: %s\n",
                 nbd_err_message (err));
        nbd_free_error (err);
        nbd_shutdown (h);
        exit (EXIT_FAILURE);
    }

    if (is_read_only) {
        printf ("The NBD device is read-only.\n");
    } else {
        printf ("The NBD device is read-write.\n");
    }

    nbd_shutdown (h);
    exit (EXIT_SUCCESS);
}
