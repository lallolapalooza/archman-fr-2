NOM

nbd_aio_in_flight - vérifie le nombre de commandes aio toujours en cours

SYNTAXE

#include <libnbd.h>
int nbd_aio_in_flight (
      struct nbd_handle *h
    );

DESCRIPTION

Renvoie le nombre de commandes aio en cours qui attendent encore une réponse du serveur avant de pouvoir être clôturées. Si cela renvoie une valeur non nulle lors de la demande de déconnexion du serveur (voir nbd_aio_disconnect(3) et nbd_shutdown(3)), libnbd n'essaie pas d'attendre la fin de ces commandes gracieusement ; si le serveur abandonne des commandes lors de l'arrêt, nbd_aio_command_completed(3) signalera ces commandes comme échouées avec un statut de "ENOTCONN".

VALEUR DE RETOUR

Cet appel renvoie un entier ≥ 0.

ERREURS

En cas d'erreur, -1 est renvoyé.

Reportez-vous à "GESTION DES ERREURS" dans libnbd(3) pour obtenir plus de détails sur l'erreur.

Les paramètres suivants ne doivent pas être NULL : "h". Pour plus d'informations, voir "Paramètres non-NULL" dans libnbd(3).

ÉTAT DU HANDLE

nbd_aio_in_flight peut être appelé lorsque le handle se trouve dans les états suivants :

┌─────────────────────────────────────┬─────────────────────────┐
│ Handle créé, avant la connexion      │ ❌ erreur                │
│ Connexion en cours                    │ ❌ erreur                │
│ Connexion & handshake (opt_mode)    │ ❌ erreur                │
│ Connecté au serveur               │ ✅ autorisé            │
│ Connexion arrêtée                  │ ✅ autorisé            │
│ Handle mort                         │ ✅ autorisé            │
└─────────────────────────────────────┴─────────────────────────┘

VERSION

Cette fonction est apparue pour la première fois dans libnbd 1.0.

Si vous avez besoin de tester la disponibilité de cette fonction au moment de la compilation, vérifiez si la macro suivante est définie :

#define LIBNBD_HAVE_NBD_AIO_IN_FLIGHT 1

EXEMPLE

Cet exemple est également disponible sous le nom examples/aio-connect-read.c dans le code source de libnbd.

/* Cet exemple montre comment utiliser l'API bas niveau AIO (asynchrone)
 * pour se connecter à un serveur et lire le disque.
 *
 * Voici quelques façons d'essayer cet exemple :
 *
 * nbdkit -U - linuxdisk . \
 *   --run './aio-connect-read $unixsocket'
 *
 * nbdkit -U - floppy . \
 *   --run './aio-connect-read $unixsocket'
 *
 * nbdkit -U - pattern size=1M \
 *   --run './aio-connect-read $unixsocket'
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <inttypes.h>
#include <errno.h>
#include <assert.h>
#include <libnbd.h>
#define NR_SECTORS 32
#define SECTOR_SIZE 512
struct data {
  uint64_t offset;
  char sector[SECTOR_SIZE];
};
static int
hexdump (void *user_data, int *error)
{
  struct data *data = user_data;
  FILE *pp;
  if (*error) {
    errno = *error;
    perror ("échec de la lecture");
    exit (EXIT_FAILURE);
  }
  printf ("secteur à l'offset 0x%" PRIx64 ":\n",
          data->offset);
  pp = popen ("hexdump -C", "w");
  if (pp == NULL) {
    perror ("popen: hexdump");
    exit (EXIT_FAILURE);
  }
  fwrite (data->sector, SECTOR_SIZE, 1, pp);
  pclose (pp);
  printf ("\n");
  /* Retourner 1 à partir du callback retire automatiquement
   * la commande.
   */
  return 1;
}
static struct data data[NR_SECTORS];
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  size_t i;
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* Crée le handle libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Se connecte au serveur NBD via un socket de domaine Unix.
   * Ceci ne fait que démarrer la connexion.
   */
  if (nbd_aio_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Attend que la connexion soit terminée. L'utilisation de
   * nbd_poll ici n'est qu'un exemple. Vous pourriez aussi
   * intégrer ceci à poll(2), glib ou une autre boucle
   * principale. Lisez libnbd(3) et le fichier source
   * lib/poll.c.
   */
  while (!nbd_aio_is_ready (nbd)) {
    if (nbd_poll (nbd, -1) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  assert (nbd_get_size (nbd) >= NR_SECTORS * SECTOR_SIZE);
  /* Émet des commandes de lecture pour les NR premiers secteurs. */
  for (i = 0; i < NR_SECTORS; ++i) {
    data[i].offset = i * SECTOR_SIZE;
    /* Le callback (hexdump) est appelé lorsque la commande
     * se termine. Le tampon doit rester valide pendant que
     * la commande s'exécute.
     */
    if (nbd_aio_pread (nbd, data[i].sector, SECTOR_SIZE,
                       data[i].offset,
                       (nbd_completion_callback) {
                         .callback = hexdump,
                         .user_data = &data[i],
                       }, 0) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* Exécute la boucle principale jusqu'à ce que toutes les commandes soient
   * terminées et clôturées. Encore une fois, l'utilisation de nbd_poll
   * ici n'est qu'un exemple.
   */
  while (nbd_aio_in_flight (nbd) > 0) {
    if (nbd_poll (nbd, -1) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* Ferme le handle libnbd. */
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}

NAME

nbd_aio_in_flight - Vérifier si une opération asynchrone est en cours

SYNTAXE

#include <libnbd.h>

int
nbd_aio_in_flight (NBD_AIO *aio);

DESCRIPTION

La fonction nbd_aio_in_flight retourne une valeur non nulle si l'objet NBD_AIO spécifié par aio a une opération en cours. Elle retourne zéro si aucune opération n'est en cours.

VALEUR DE RETOUR

Une valeur non nulle si une opération est en cours, zéro sinon.

ERREURS

Aucune erreur n'est signalée.

ÉTAT DU HANDLE

Cette fonction peut être appelée sur n'importe quel objet NBD_AIO, y compris ceux qui sont fermés.

VERSION

Cette fonction est disponible depuis la version 1.0 de libnbd.

EXEMPLE

// Exemple d'utilisation (non fonctionnel sans contexte)
NBD_AIO *aio = nbd_aio_new(NULL, NULL); // Création simplifiée, nécessite une gestion des erreurs

// ... lancer une opération asynchrone ...

if (nbd_aio_in_flight(aio)) {
    printf("Opération asynchrone en cours.\n");
} else {
    printf("Aucune opération asynchrone en cours.\n");
}

// ... gérer la fin de l'opération ...

nbd_aio_free(aio);

VOIR AUSSI

nbd_aio_new(3), nbd_aio_destroy(3)
