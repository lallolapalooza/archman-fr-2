nbd_aio_pread(3) LIBNBD nbd_aio_pread(3)

NOM

nbd_aio_pread - lire depuis le serveur NBD

SYNTAXE

#include <libnbd.h>
typedef struct {
  int (*callback) (void *user_data, int *error);
  void *user_data;
  void (*free) (void *user_data);
} nbd_completion_callback;
int64_t nbd_aio_pread (
          struct nbd_handle *h, void *buf, size_t count,
          uint64_t offset,
          nbd_completion_callback completion_callback,
          uint32_t flags
        );

DESCRIPTION

Émet une commande de lecture vers le serveur NBD.

Pour vérifier si la commande est terminée, appelez nbd_aio_command_completed(3). Ou fournissez le "completion_callback" optionnel qui sera invoqué comme décrit dans "Callbacks d'achèvement" dans libnbd(3).

Notez que vous devez vous assurer que "buf" est valide jusqu'à ce que la commande soit terminée. De plus, si le paramètre "error" de "completion_callback" est défini ou si nbd_aio_command_completed(3) signale une erreur, et si nbd_get_pread_initialize(3) retourne vrai, alors libnbd a nettoyé "buf", mais il n'est pas spécifié si le contenu de "buf" sera lu comme zéro ou comme résultats partiels du serveur. Si nbd_get_pread_initialize(3) retourne faux, alors libnbd n'a pas nettoyé "buf", et le contenu est indéfini en cas d'échec.

Les autres paramètres se comportent comme documenté dans nbd_pread(3).

Par défaut, libnbd rejettera les tentatives d'utilisation de cette fonction avec des paramètres susceptibles de provoquer une erreur du serveur, comme la demande d'un indicateur de commande inconnu. La fonction nbd_set_strict_mode(3) peut être utilisée pour modifier les scénarios qui doivent attendre une réponse du serveur plutôt que d'échouer rapidement.

VALEUR DE RETOUR

Cet appel retourne le cookie 64 bits de la commande. Le cookie est ≥ 1. Les cookies sont uniques (par descripteur libnbd, pas globalement).

ERREURS

En cas d'erreur, -1 est retourné.

Reportez-vous à "GESTION DES ERREURS" dans libnbd(3) pour obtenir plus de détails sur l'erreur.

Les paramètres suivants ne doivent pas être NULL : "h", "buf". Pour plus d'informations, voir "Paramètres non-NULL" dans libnbd(3).

ÉTAT DU DESCRIPTEUR

nbd_aio_pread peut être appelé lorsque le descripteur est dans l'état suivant :

┌─────────────────────────────────────┬─────────────────────────┐
│ Descripteur créé, avant connexion     │ ❌ erreur                │
│ Connexion en cours                   │ ❌ erreur                │
│ Connexion & handshake (opt_mode) │ ❌ erreur                │
│ Connecté au serveur               │ ✅ autorisé             │
│ Connexion fermée                    │ ❌ erreur                │
│ Descripteur mort                    │ ❌ erreur                │
└─────────────────────────────────────┴─────────────────────────┘

VERSION

Cette fonction est apparue pour la première fois dans libnbd 1.0.

Si vous avez besoin de vérifier si cette fonction est disponible au moment de la compilation, vérifiez si le macro suivant est défini :

#define LIBNBD_HAVE_NBD_AIO_PREAD 1

EXEMPLE

Cet exemple est également disponible sous le nom examples/aio-connect-read.c dans le code source de libnbd.

/* Cet exemple montre comment utiliser l'API bas niveau AIO (asynchrone)
 * pour se connecter à un serveur et lire le disque.
 *
 * Voici quelques façons d'essayer cet exemple :
 *
 * nbdkit -U - linuxdisk . \
 *   --run './aio-connect-read $unixsocket'
 *
 * nbdkit -U - floppy . \
 *   --run './aio-connect-read $unixsocket'
 *
 * nbdkit -U - pattern size=1M \
 *   --run './aio-connect-read $unixsocket'
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <inttypes.h>
#include <errno.h>
#include <assert.h>
#include <libnbd.h>
#define NR_SECTORS 32
#define SECTOR_SIZE 512
struct data {
  uint64_t offset;
  char sector[SECTOR_SIZE];
};
static int
hexdump (void *user_data, int *error)
{
  struct data *data = user_data;
  FILE *pp;
  if (*error) {
    errno = *error;
    perror ("failed to read");
    exit (EXIT_FAILURE);
  }
  printf ("secteur à l'offset 0x%" PRIx64 ":\n",
          data->offset);
  pp = popen ("hexdump -C", "w");
  if (pp == NULL) {
    perror ("popen: hexdump");
    exit (EXIT_FAILURE);
  }
  fwrite (data->sector, SECTOR_SIZE, 1, pp);
  pclose (pp);
  printf ("\n");
  /* Retourner 1 depuis le callback retraite
   * automatiquement la commande.
   */
  return 1;
}
static struct data data[NR_SECTORS];
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  size_t i;
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* Créer le descripteur libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Se connecter au serveur NBD via un socket de domaine Unix.
   * Ceci ne fait que démarrer la connexion.
   */
  if (nbd_aio_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Attendre que la connexion soit terminée. L'utilisation de
   * nbd_poll ici est juste un exemple. Vous pourriez également
   * l'intégrer avec poll(2), glib ou une autre boucle principale.
   * Lire libnbd(3) et le fichier source lib/poll.c.
   */
  while (!nbd_aio_is_ready (nbd)) {
    if (nbd_poll (nbd, -1) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  assert (nbd_get_size (nbd) >= NR_SECTORS * SECTOR_SIZE);
  /* Émettre des commandes de lecture pour les NR premiers secteurs. */
  for (i = 0; i < NR_SECTORS; ++i) {
    data[i].offset = i * SECTOR_SIZE;
    /* Le callback (hexdump) est appelé lorsque la commande
     * se termine. Le tampon doit continuer d'exister pendant
     * que la commande est en cours d'exécution.
     */
    if (nbd_aio_pread (nbd, data[i].sector, SECTOR_SIZE,
                       data[i].offset,
                       (nbd_completion_callback) {
                         .callback = hexdump,
                         .user_data = &data[i],
                       }, 0) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* Exécuter la boucle principale jusqu'à ce que toutes les commandes soient
   * terminées et traitées. Encore une fois, l'utilisation de nbd_poll
   * ici est juste un exemple.
   */
  while (nbd_aio_in_flight (nbd) > 0) {
    if (nbd_poll (nbd, -1) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* Fermer le descripteur libnbd. */
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}
