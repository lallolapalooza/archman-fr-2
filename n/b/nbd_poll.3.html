nbd_poll(3)

Paquet: libnbd

Licences: LGPL-2.1-or-later

nbd_poll(3) LIBNBD nbd_poll(3)

NOM

nbd_poll - interroger le handle une fois

SYNTAXE

#include <libnbd.h>
int nbd_poll (
      struct nbd_handle *h, int timeout
    );

DESCRIPTION

Ceci est une implémentation simple de poll(2) qui est utilisée en interne par les appels API synchrones. En cas de succès, elle retourne 0 si le "timeout" (en millisecondes) se produit, ou 1 si l'interrogation s'est terminée et que la machine d'état a progressé. Réglez "timeout" sur -1 pour bloquer indéfiniment (mais faites attention à ce qu'une action éventuelle soit réellement attendue - par exemple, si la connexion est établie mais qu'il n'y a pas de commandes en cours, utiliser un timeout infini bloquera de façon permanente).

Cette fonction est principalement utile comme exemple de la manière d'intégrer libnbd à votre propre boucle principale, plutôt que d'être destinée à être utilisée telle quelle.

VALEUR DE RETOUR

Cet appel retourne un entier ≥ 0.

ERREURS

En cas d'erreur, -1 est retourné.

Voir "GESTION DES ERREURS" dans libnbd(3) pour obtenir plus de détails sur l'erreur.

Les paramètres suivants ne doivent pas être NULL : "h". Pour plus d'informations, voir "Paramètres non NULL" dans libnbd(3).

VERSION

Cette fonction est apparue pour la première fois dans libnbd 1.0.

Si vous avez besoin de tester si cette fonction est disponible au moment de la compilation, vérifiez si la macro suivante est définie :

#define LIBNBD_HAVE_NBD_POLL 1

EXEMPLE

Cet exemple est également disponible sous le nom examples/aio-connect-read.c dans le code source de libnbd.

/* Cet exemple montre comment utiliser l'API de bas niveau AIO (asynchrone)
 * pour se connecter à un serveur et lire le disque.
 *
 * Voici quelques façons d'essayer cet exemple :
 *
 * nbdkit -U - linuxdisk . \
 *   --run './aio-connect-read $unixsocket'
 *
 * nbdkit -U - floppy . \
 *   --run './aio-connect-read $unixsocket'
 *
 * nbdkit -U - pattern size=1M \
 *   --run './aio-connect-read $unixsocket'
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <inttypes.h>
#include <errno.h>
#include <assert.h>
#include <libnbd.h>
#define NR_SECTORS 32
#define SECTOR_SIZE 512
struct data {
  uint64_t offset;
  char sector[SECTOR_SIZE];
};
static int
hexdump (void *user_data, int *error)
{
  struct data *data = user_data;
  FILE *pp;
  if (*error) {
    errno = *error;
    perror ("échec de la lecture");
    exit (EXIT_FAILURE);
  }
  printf ("secteur à l'offset 0x%" PRIx64 ":\n",
          data->offset);
  pp = popen ("hexdump -C", "w");
  if (pp == NULL) {
    perror ("popen: hexdump");
    exit (EXIT_FAILURE);
  }
  fwrite (data->sector, SECTOR_SIZE, 1, pp);
  pclose (pp);
  printf ("\n");
  /* Retourner 1 à partir de la fonction de rappel retire
   * automatiquement la commande.
   */
  return 1;
}
static struct data data[NR_SECTORS];
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  size_t i;
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* Créer le handle libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Se connecter au serveur NBD via un socket de domaine Unix.
   * Ceci ne fait que démarrer la connexion.
   */
  if (nbd_aio_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Attendre que la connexion soit terminée. L'utilisation de
   * nbd_poll ici n'est qu'un exemple. Vous pourriez également
   * l'intégrer avec poll(2), glib ou une autre boucle principale.
   * Lire libnbd(3) et le fichier source lib/poll.c.
   */
  while (!nbd_aio_is_ready (nbd)) {
    if (nbd_poll (nbd, -1) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  assert (nbd_get_size (nbd) >= NR_SECTORS * SECTOR_SIZE);
  /* Émettre des commandes de lecture pour les NR premiers secteurs. */
  for (i = 0; i < NR_SECTORS; ++i) {
    data[i].offset = i * SECTOR_SIZE;
    /* La fonction de rappel (hexdump) est appelée lorsque la commande
     * est terminée. Le tampon doit rester valide tant que
     * la commande est en cours d'exécution.
     */
    if (nbd_aio_pread (nbd, data[i].sector, SECTOR_SIZE,
                       data[i].offset,
                       (nbd_completion_callback) {
                         .callback = hexdump,
                         .user_data = &data[i],
                       }, 0) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* Exécuter la boucle principale jusqu'à ce que toutes les commandes soient
   * terminées et retirées. Encore une fois, l'utilisation de nbd_poll
   * ici n'est qu'un exemple.
   */
  while (nbd_aio_in_flight (nbd) > 0) {
    if (nbd_poll (nbd, -1) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* Fermer le handle libnbd. */
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}
