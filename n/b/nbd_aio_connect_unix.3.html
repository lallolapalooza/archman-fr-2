nbd_aio_connect_unix(3)

nbd_aio_connect_unix(3) LIBNBD nbd_aio_connect_unix(3)

NOM

nbd_aio_connect_unix - se connecter au serveur NBD via un socket de domaine Unix

SYNTAXE

#include <libnbd.h>
int nbd_aio_connect_unix (
      struct nbd_handle *h, const char *unixsocket
    );

DESCRIPTION

Commence la connexion au serveur NBD via un socket de domaine Unix ("unixsocket"). Les paramètres se comportent comme décrit dans nbd_connect_unix(3).

Vous pouvez vérifier si la tentative de connexion est toujours en cours en appelant nbd_aio_is_connecting(3). Si nbd_set_opt_mode(3) est activé, la connexion est prête pour la négociation manuelle des options une fois que nbd_aio_is_negotiating(3) renvoie vrai ; sinon, la tentative de connexion inclura la poignée de main NBD, et sera prête à être utilisée une fois que nbd_aio_is_ready(3) renverra vrai.

VALEUR DE RETOUR

Si l'appel réussit, la fonction renvoie 0.

ERREURS

En cas d'erreur, -1 est renvoyé.

Se référer à "Gestion des erreurs" dans libnbd(3) pour obtenir plus de détails sur l'erreur.

Les paramètres suivants ne doivent pas être NULL : "h", "unixsocket". Pour plus d'informations, voir "Paramètres non NULL" dans libnbd(3).

ETAT DU HANDLE

nbd_aio_connect_unix peut être appelé lorsque le handle est dans l'état suivant :

┌─────────────────────────────────────┬─────────────────────────┐
│ Handle créé, avant connexion     │ ✅ autorisé              │
│ Connexion en cours                │ ❌ erreur                │
│ Connexion & handshake (opt_mode) │ ❌ erreur                │
│ Connecté au serveur               │ ❌ erreur                │
│ Connexion fermée                  │ ❌ erreur                │
│ Handle mort                       │ ❌ erreur                │
└────────────────────────────────────────┴─────────────────────────┘

VERSION

Cette fonction est apparue pour la première fois dans libnbd 1.0.

Si vous avez besoin de tester si cette fonction est disponible au moment de la compilation, vérifiez si la macro suivante est définie :

#define LIBNBD_HAVE_NBD_AIO_CONNECT_UNIX 1

EXEMPLE

Cet exemple est également disponible sous le nom examples/aio-connect-read.c dans le code source de libnbd.

/* Cet exemple montre comment utiliser l'API bas niveau AIO (asynchrone)
 * pour se connecter à un serveur et lire le disque.
 *
 * Voici quelques façons d'essayer cet exemple :
 *
 * nbdkit -U - linuxdisk . \
 *   --run './aio-connect-read $unixsocket'
 *
 * nbdkit -U - floppy . \
 *   --run './aio-connect-read $unixsocket'
 *
 * nbdkit -U - pattern size=1M \
 *   --run './aio-connect-read $unixsocket'
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <inttypes.h>
#include <errno.h>
#include <assert.h>
#include <libnbd.h>
#define NR_SECTORS 32
#define SECTOR_SIZE 512
struct data {
  uint64_t offset;
  char sector[SECTOR_SIZE];
};
static int
hexdump (void *user_data, int *error)
{
  struct data *data = user_data;
  FILE *pp;
  if (*error) {
    errno = *error;
    perror ("échec de la lecture");
    exit (EXIT_FAILURE);
  }
  printf ("secteur à l'offset 0x%" PRIx64 ":\n",
          data->offset);
  pp = popen ("hexdump -C", "w");
  if (pp == NULL) {
    perror ("popen: hexdump");
    exit (EXIT_FAILURE);
  }
  fwrite (data->sector, SECTOR_SIZE, 1, pp);
  pclose (pp);
  printf ("\n");
  /* Retourner 1 à partir du rappel retire automatiquement
   * la commande.
   */
  return 1;
}
static struct data data[NR_SECTORS];
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  size_t i;
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* Crée le handle libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Se connecte au serveur NBD via un socket de domaine Unix.
   * Cela démarre seulement la connexion.
   */
  if (nbd_aio_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Attend que la connexion soit terminée. L'utilisation de
   * nbd_poll ici est uniquement un exemple. Vous pourriez
   * également intégrer cela avec poll(2), glib ou une autre
   * boucle principale. Lire libnbd(3) et le fichier source
   * lib/poll.c.
   */
  while (!nbd_aio_is_ready (nbd)) {
    if (nbd_poll (nbd, -1) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  assert (nbd_get_size (nbd) >= NR_SECTORS * SECTOR_SIZE);
  /* Émet des commandes de lecture pour les NR premiers secteurs. */
  for (i = 0; i < NR_SECTORS; ++i) {
    data[i].offset = i * SECTOR_SIZE;
    /* Le rappel (hexdump) est appelé lorsque la commande
     * est terminée. Le tampon doit continuer d'exister pendant
     * que la commande s'exécute.
     */
    if (nbd_aio_pread (nbd, data[i].sector, SECTOR_SIZE,
                       data[i].offset,
                       (nbd_completion_callback) {
                         .callback = hexdump,
                         .user_data = &data[i],
                       }, 0) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* Exécute la boucle principale jusqu'à ce que toutes les commandes
   * soient terminées et retirées. Encore une fois, l'utilisation de
   * nbd_poll ici est uniquement un exemple.
   */
  while (nbd_aio_in_flight (nbd) > 0) {
    if (nbd_poll (nbd, -1) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* Ferme le handle libnbd. */
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}
