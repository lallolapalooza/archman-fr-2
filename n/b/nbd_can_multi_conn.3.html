NOM

nbd_can_multi_conn - le serveur supporte-t-il multi-conn ?

SYNTAXE

#include <libnbd.h>
int nbd_can_multi_conn (
      struct nbd_handle *h
    );

DESCRIPTION

Retourne vrai si le serveur supporte multi-conn. Retourne faux si le serveur ne le supporte pas.

Il n'est pas sûr d'ouvrir plusieurs descripteurs connectés au même serveur si vous allez écrire sur le serveur et que le serveur ne publie pas le support de multi-conn. La manière sûre de vérifier cela est d'ouvrir une connexion, de vérifier que ce drapeau est vrai, puis d'ouvrir les connexions supplémentaires nécessaires.

Cet appel ne bloque pas, car il renvoie des données qui sont enregistrées dans le descripteur lors de la poignée de main du protocole NBD.

VALEUR DE RETOUR

Cet appel renvoie une valeur booléenne.

ERREURS

En cas d'erreur, -1 est retourné.

Référez-vous à "ERROR HANDLING" dans libnbd(3) pour savoir comment obtenir plus de détails sur l'erreur.

Les paramètres suivants ne doivent pas être NULL : "h". Pour plus d'informations, voir "Non-NULL parameters" dans libnbd(3).

ETAT DU DESCRIPTEUR

nbd_can_multi_conn peut être appelé lorsque le descripteur est dans les états suivants :

┌─────────────────────────────────────┬─────────────────────────┐
│ Descripteur créé, avant connexion   │ ❌ erreur                │
│ Connexion en cours                    │ ❌ erreur                │
│ Connexion & poignée de main (opt_mode) │ ✅ autorisé             │
│ Connecté au serveur             │ ✅ autorisé             │
│ Connexion fermée                │ ✅ autorisé             │
│ Descripteur mort                       │ ❌ erreur                │
└─────────────────────────────────────┴─────────────────────────┘

VERSION

Cette fonction est apparue pour la première fois dans libnbd 1.0.

Si vous avez besoin de tester si cette fonction est disponible au moment de la compilation, vérifiez si la macro suivante est définie :

#define LIBNBD_HAVE_NBD_CAN_MULTI_CONN 1

EXEMPLE

Cet exemple est également disponible sous le nom examples/server-flags.c dans le code source de libnbd.

/* Cet exemple montre comment se connecter à un serveur NBD
 * et afficher les drapeaux d'exportation.
 *
 * Vous pouvez le tester avec nbdkit comme suit :
 *
 * nbdkit -U - memory 1M \
 *   --run './server-flags $unixsocket'
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <libnbd.h>
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  char *str;
  int flag;
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* Créer le descripteur libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Demander des informations complètes. */
#if LIBNBD_HAVE_NBD_SET_FULL_INFO /* Ajouté en 1.4 */
  if (nbd_set_full_info (nbd, true) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
#endif
  /* Se connecter au serveur NBD via un
   * socket de domaine Unix.
   */
  if (nbd_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Voir si le serveur a fourni des détails supplémentaires,
   * en utilisant les fonctions ajoutées en 1.4
   */
#if LIBNBD_HAVE_NBD_GET_EXPORT_DESCRIPTION
  str = nbd_get_canonical_export_name (nbd);
  if (str)
    printf ("canonical_name = %s\n", str);
  free (str);
  str = nbd_get_export_description (nbd);
  if (str)
    printf ("description = %s\n", str);
  free (str);
#endif
  /* Lire et afficher les drapeaux. */
#define PRINT_FLAG(flag_fn)                     \
  flag = flag_fn (nbd);                         \
  if (flag == -1) {                             \
    fprintf (stderr, "%s\n", nbd_get_error ()); \
    exit (EXIT_FAILURE);                        \
  }                                             \
  printf (#flag_fn " = %s\n",                   \
          flag ? "true" : "false");
  PRINT_FLAG (nbd_can_cache);
  PRINT_FLAG (nbd_can_df);
  PRINT_FLAG (nbd_can_flush);
  PRINT_FLAG (nbd_can_fua);
  PRINT_FLAG (nbd_can_multi_conn);
  PRINT_FLAG (nbd_can_trim);
  PRINT_FLAG (nbd_can_zero);
#if LIBNBD_HAVE_NBD_CAN_FAST_ZERO
  /* Ajouté en 1.2 */
  PRINT_FLAG (nbd_can_fast_zero);
#endif
#if LIBNBD_HAVE_NBD_CAN_BLOCK_STATUS_PAYLOAD
  /* Ajouté en 1.18 */
  PRINT_FLAG (nbd_can_block_status_payload);
#endif
  PRINT_FLAG (nbd_is_read_only);
  PRINT_FLAG (nbd_is_rotational);
  /* Fermer le descripteur libnbd. */
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}

NOM

nbd_can_multi_conn - Vérifie la capacité de plusieurs connexions

SYNOPIS

#include <lib Nbd.h>

int nbd_can_multi_conn(struct nbd_handle *handle);

DESCRIPTION

La fonction nbd_can_multi_conn() renvoie un entier qui indique si le serveur NBD connecté à handle prend en charge plusieurs connexions simultanées (ce qui est nécessaire pour certaines opérations comme la copie de données via nbd_copy_from() et nbd_copy_to()).

VALEUR DE RETOUR

Renvoie 1 si plusieurs connexions sont prises en charge, 0 sinon.

ERREURS

Aucune erreur spécifique n'est retournée par cette fonction. Si handle est NULL, le comportement est indéfini.

GESTION_ÉTAT

Cette fonction peut être appelée à tout moment après l'établissement d'une connexion NBD.

VERSION

Cette fonction est disponible depuis la version 1.0.0 de lib Nbd.

EXEMPLE

#include <stdio.h>
#include <lib Nbd.h>

int main() {
    struct nbd_handle *handle = nbd_connect("127.0.0.1", "10809");
    if (!handle) {
        fprintf(stderr, "Erreur de connexion\n");
        return 1;
    }

    if (nbd_can_multi_conn(handle)) {
        printf("Le serveur NBD prend en charge plusieurs connexions.\n");
    } else {
        printf("Le serveur NBD ne prend pas en charge plusieurs connexions.\n");
    }

    nbd_disconnect(handle);
    return 0;
}

VOIR AUSSI

nbd_connect(3), nbd_disconnect(3), nbd_copy_from(3), nbd_copy_to(3)
