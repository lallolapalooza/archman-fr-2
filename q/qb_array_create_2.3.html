NOM

qb_array_create_2 - Crée un tableau avec des éléments de taille fixe.

SYNTAXE

#include <qb/qbarray.h>

qb_array_t * qb_array_create_2(
    size_t  max_elements,       /* nombre maximal initial d'éléments */
    size_t  element_size,       /* taille de chaque élément */
    size_t  autogrow_elements   /* le nombre d'éléments par lequel il faut croître automatiquement. */
);

PARAMETRES

max_elements nombre maximal initial d'éléments

element_size taille de chaque élément

autogrow_elements le nombre d'éléments par lequel il faut croître automatiquement.

DESCRIPTION

VALEUR DE RETOUR

instance de tableau
