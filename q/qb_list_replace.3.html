NOM

qb_list_replace - Remplace une entrée existante par une nouvelle.

SYNTAXE

#include <qb/qblist.h>

static void qb_list_replace(
    struct qb_list_head   *old_one,  /* l'élément à remplacer */
    struct qb_list_head   *new_one   /* le nouvel élément à insérer */
);

PARAMETRES

old_one l'élément à remplacer

new_one le nouvel élément à insérer

DESCRIPTION

STRUCTURES

    struct   next;

    struct   prev;
};
