QB_LOG_FILTER_CTL(3) Manuel du programmeur libqb QB_LOG_FILTER_CTL(3)

NOM

qb_log_filter_ctl - Ceci vous permet de modifier les champs 'tags' et 'targets' du site d'appel à l'exécution.

SYNOPSIS

#include <qb/qblog.h>

int32_t qb_log_filter_ctl(
    int32_t                  value,
    enum qb_log_filter_conf  c,
    enum qb_log_filter_type  type,
    const char              *text,
    uint8_t                  low_priority
);

DESCRIPTION

STRUCTURES


enum qb_log_filter_conf {


      QB_LOG_FILTER_ADD;

      QB_LOG_FILTER_REMOVE;

      QB_LOG_FILTER_CLEAR_ALL;

      QB_LOG_TAG_SET;

      QB_LOG_TAG_CLEAR;

      QB_LOG_TAG_CLEAR_ALL;
};


enum qb_log_filter_type {


      QB_LOG_FILTER_FILE;

      QB_LOG_FILTER_FUNCTION;

      QB_LOG_FILTER_FORMAT;

      QB_LOG_FILTER_FILE_REGEX;

      QB_LOG_FILTER_FUNCTION_REGEX;

      QB_LOG_FILTER_FORMAT_REGEX;
};
