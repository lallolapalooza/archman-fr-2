NOM

qb_log_ctl - Fonction principale de contrôle de la journalisation.

SYNTAXE

#include <qb/qblog.h>

int32_t qb_log_ctl(
    /*
     * QB_LOG_SYSLOG, QB_LOG_STDERR ou résultat de qb_log_file_open()
     */
    int32_t           target,
    /*
     * directive de configuration ("quoi configurer") qui accepte un argument int32_t
     * déterminant la nouvelle valeur sauf si elle est ignorée pour une directive particulière
     * dans son ensemble (directives incompatibles : QB_LOG_CONF_IDENT)
     */
    enum qb_log_conf  conf_type,
    /*
     * la nouvelle valeur pour une directive de configuration modifiant l'état, ignorée
     * sinon
     */
    int32_t           arg
);

PARAMÈTRES

target QB_LOG_SYSLOG, QB_LOG_STDERR ou résultat de qb_log_file_open()

conf_type directive de configuration ("quoi configurer") qui accepte un argument int32_t déterminant la nouvelle valeur sauf si elle est ignorée pour une directive particulière dans son ensemble (directives incompatibles : QB_LOG_CONF_IDENT)

arg la nouvelle valeur pour une directive de configuration modifiant l'état, ignorée sinon

STRUCTURES


enum qb_log_conf {


      QB_LOG_CONF_ENABLED;

      QB_LOG_CONF_FACILITY;

      QB_LOG_CONF_DEBUG;

      QB_LOG_CONF_SIZE;

      QB_LOG_CONF_THREADED;

      QB_LOG_CONF_PRIORITY_BUMP;

      QB_LOG_CONF_STATE_GET;

      QB_LOG_CONF_FILE_SYNC;

      QB_LOG_CONF_EXTENDED;

      QB_LOG_CONF_IDENT;

      QB_LOG_CONF_MAX_LINE_LEN;

      QB_LOG_CONF_ELLIPSIS;

      QB_LOG_CONF_USE_JOURNAL;
};

VALEUR DE RETOUR

-errno en cas d'erreur

0 en cas de succès
