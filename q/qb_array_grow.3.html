NOM

qb_array_grow - Agrandir le tableau.

SYNOPSIS

#include <qb/qbarray.h>

int32_t qb_array_grow(
    qb_array_t   *a,             /* instance du tableau */
    size_t        max_elements   /* la nouvelle taille maximale du tableau */
);

PARAMETRES

a instance du tableau

max_elements la nouvelle taille maximale du tableau

DESCRIPTION

VALEUR DE RETOUR

(0 == succès, sinon -errno)
