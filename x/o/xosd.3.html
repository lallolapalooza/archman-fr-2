XOSD(3) Manuel des fonctions de bibliothèque XOSD(3)

NOM

xosd - bibliothèque d'affichage à l'écran X

SYNOPSIS

#include <xosd.h>

xosd* xosd_create(int nombre_lignes);

int xosd_uninit (xosd* osd);

int xosd_display (xosd* osd, int ligne,
                  xosd_command commande, ...);

int xosd_is_onscreen(xosd* osd);

int xosd_wait_until_no_display(xosd* osd);

int xosd_hide (xosd* osd);

int xosd_show (xosd* osd);

int xosd_set_pos (xosd* osd, xosd_pos pos);

int xosd_set_vertical_offset (xosd* osd, int decalage);

int xosd_set_horizontal_offset (xosd* osd, int decalage);

int xosd_set_align (xosd* osd, xosd_align align)

int xosd_set_shadow_offset (xosd* osd, int decalage_ombre);

int xosd_set_timeout (xosd* osd, int delai);

int xosd_set_font (xosd* osd, char* police);

int xosd_set_colour (xosd* osd, char* couleur);

int xosd_get_colour (xosd* osd,
                     int* rouge, int* vert, int* bleu);

int xosd_scroll (xosd* osd, int lignes);

int xosd_get_number_lines ( xosd* osd);

DESCRIPTION

xosd est une bibliothèque pour afficher un affichage à l'écran (comme celui de nombreuses télévisions) sur votre écran X. Chaque fonction devrait avoir sa propre page de manuel.

La dernière version peut être trouvée sur http://www.ignavus.net/software.html

Elle est distribuée sous la Licence Publique Générale GNU.

BOGUES

Aucun bogue connu pour le moment. Il y a probablement des fonctions qui ne sont pas listées ici.

Les rapports de bugs peuvent être envoyés à tnw13@cosc.canterbury.ac.nz

EXEMPLE

#include <xosd.h>
int main (int argc, char *argv[])
{
	xosd *osd;
	osd = xosd_create (1);
	xosd_set_font(osd, "fixed");
	xosd_set_colour(osd, "LawnGreen");
	xosd_set_timeout(osd, 3);
	xosd_set_shadow_offset(osd, 1);

	xosd_display (osd, 0, XOSD_string, "Exemple de sortie XOSD");
	xosd_wait_until_no_display(osd);
	xosd_uninit (osd);
	return EXIT_SUCCESS;
}
