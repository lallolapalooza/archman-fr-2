SYNOPSIS

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_rdf* ldns_rr_dnskey_flags(const ldns_rr *r);

bool ldns_rr_dnskey_set_flags(ldns_rr *r, ldns_rdf *f);

ldns_rdf* ldns_rr_dnskey_protocol(const ldns_rr *r);

bool ldns_rr_dnskey_set_protocol(ldns_rr *r, ldns_rdf *f);

ldns_rdf* ldns_rr_dnskey_algorithm(const ldns_rr *r);

bool ldns_rr_dnskey_set_algorithm(ldns_rr *r, ldns_rdf *f);

ldns_rdf* ldns_rr_dnskey_key(const ldns_rr *r);

bool ldns_rr_dnskey_set_key(ldns_rr *r, ldns_rdf *f);

DESCRIPTION

ldns_rr_dnskey_flags() renvoie les indicateurs d'un enregistrement LDNS_RR_TYPE_DNSKEY
r : l'enregistrement de ressources
Retourne un ldns_rdf* avec les indicateurs ou NULL en cas d'échec

ldns_rr_dnskey_set_flags() définit les indicateurs d'un enregistrement LDNS_RR_TYPE_DNSKEY
r : l'enregistrement de ressources à utiliser
f : les indicateurs à définir
Retourne true en cas de succès, false sinon

ldns_rr_dnskey_protocol() renvoie le protocole d'un enregistrement LDNS_RR_TYPE_DNSKEY
r : l'enregistrement de ressources
Retourne un ldns_rdf* avec le protocole ou NULL en cas d'échec

ldns_rr_dnskey_set_protocol() définit le protocole d'un enregistrement LDNS_RR_TYPE_DNSKEY
r : l'enregistrement de ressources à utiliser
f : le protocole à définir
Retourne true en cas de succès, false sinon

ldns_rr_dnskey_algorithm() renvoie l'algorithme d'un enregistrement LDNS_RR_TYPE_DNSKEY
r : l'enregistrement de ressources
Retourne un ldns_rdf* avec l'algorithme ou NULL en cas d'échec

ldns_rr_dnskey_set_algorithm() définit l'algorithme d'un enregistrement LDNS_RR_TYPE_DNSKEY
r : l'enregistrement de ressources à utiliser
f : l'algorithme à définir
Retourne true en cas de succès, false sinon

ldns_rr_dnskey_key() renvoie les données de la clé d'un enregistrement LDNS_RR_TYPE_DNSKEY
r : l'enregistrement de ressources
Retourne un ldns_rdf* avec les données de la clé ou NULL en cas d'échec

ldns_rr_dnskey_set_key() définit les données de la clé d'un enregistrement LDNS_RR_TYPE_DNSKEY
r : l'enregistrement de ressources à utiliser
f : les données de la clé à définir
Retourne true en cas de succès, false sinon
