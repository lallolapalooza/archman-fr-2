#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

size_t ldns_key_list_key_count(const ldns_key_list *key_list);

ldns_key* ldns_key_list_key(const ldns_key_list *key, size_t nr);

RSA* ldns_key_rsa_key(const ldns_key *k);

DESCRIPTION

ldns_key_list_key_count() retourne le nombre de clés dans la liste de clés
key_list : la liste de clés
Retourne le nombre de clés dans la liste

ldns_key_list_key() retourne un pointeur vers la clé dans la liste à la position donnée
key : la clé
nr : la position dans la liste
Retourne la clé

ldns_key_rsa_key() retourne la structure RSA (openssl) contenue dans la clé
k : la clé à examiner
Retourne la structure RSA * dans la clé

ldns_key_dsa_key() retourne la structure DSA (openssl) contenue dans la clé

ldns_key_algorithm() retourne l'algorithme de signature de la clé
k : la clé
Retourne l'algorithme

ldns_key_hmac_key() retourne les données de la clé hmac
k : la clé
Retourne les données de la clé hmac

ldns_key_origttl() retourne le ttl d'origine de la clé
k : la clé
Retourne le ttl d'origine

ldns_key_inception() retourne la date de prise d'effet de la clé
k : la clé
Retourne la date de prise d'effet

ldns_key_expiration() retourne la date d'expiration de la clé
k : la clé
Retourne la date d'expiration

ldns_key_keytag() retourne le keytag
k : la clé
Retourne le keytag

ldns_key_pubkey_owner() retourne le propriétaire de la clé publique
k : la clé
Retourne le propriétaire

ldns_key_flags() retourne le drapeau de la clé
k : la clé
Retourne le drapeau
