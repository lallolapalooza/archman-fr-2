SYNTAXE

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_rdf* ldns_native2rdf_int8(ldns_rdf_type type, uint8_t value);

ldns_rdf* ldns_native2rdf_int16(ldns_rdf_type type, uint16_t value);

ldns_rdf* ldns_native2rdf_int32(ldns_rdf_type type, uint32_t value);

ldns_rdf* ldns_native2rdf_int16_data(size_t size, uint8_t *data);

uint8_t ldns_rdf2native_int8(const ldns_rdf *rd);

uint16_t ldns_rdf2native_int16(const ldns_rdf *rd);

uint32_t ldns_rdf2native_int32(const ldns_rdf *rd);

DESCRIPTION

ldns_native2rdf_int8() renvoie le rdf contenant la représentation native uint8_t.
type : le type ldns_rdf à utiliser
value : l'uint8_t à utiliser
Renvoie un ldns_rdf* avec la valeur convertie

ldns_native2rdf_int16() renvoie le rdf contenant la représentation native uint16_t.
type : le type ldns_rdf à utiliser
value : l'uint16_t à utiliser
Renvoie un ldns_rdf* avec la valeur convertie

ldns_native2rdf_int32() renvoie un rdf qui contient la valeur int32 donnée.

Étant donné que plusieurs types de rdf peuvent contenir un int32, le type doit être spécifié
type : le type ldns_rdf à utiliser
value : l'uint32_t à utiliser
Renvoie un ldns_rdf* avec la valeur convertie

ldns_native2rdf_int16_data() renvoie un rdf int16_data qui contient les données du tableau donné, précédées d'un int16 spécifiant la longueur.

La mémoire est copiée et un LDNS_RDF_TYPE_INT16DATA est renvoyé
size : la taille des données
*data : pointeur vers les données réelles

Renvoie ldns_rd* le rdf avec les données

ldns_rdf2native_int8() renvoie la représentation native uint8_t du rdf.
rd : le ldns_rdf sur lequel opérer
Renvoie l'uint8_t, la valeur extraite

ldns_rdf2native_int16() renvoie la représentation native uint16_t du rdf.
rd : le ldns_rdf sur lequel opérer
Renvoie l'uint16_t, la valeur extraite

ldns_rdf2native_int32() renvoie la représentation native uint32_t du rdf.
rd : le ldns_rdf sur lequel opérer
Renvoie l'uint32_t, la valeur extraite

ldns_rdf2native_sockaddr_storage()

ldns_rdf2native_time_t() renvoie la représentation native time_t du rdf.
rd : le ldns_rdf sur lequel opérer
Renvoie l'heure_t, la valeur extraite (actuellement 32 bits)
