NOM

ldns_rdf, ldns_rdf_type - type de champ rdata

SYNTAXE

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

DESCRIPTION

ldns_rdf
Champ de données d'enregistrement de ressources (Resource Record Data field).

Les données sont un tableau d'octets ordonnés en réseau, dont la taille est spécifiée par
le champ (16 bits) size. Pour le parser correctement, utilisez le type
spécifié dans le champ (16 bits) type avec une valeur issue de \ref ldns_rdf_type.
struct ldns_struct_rdf
{
La taille des données (en octets) :
size_t _size;
Le type des données :
ldns_rdf_type _type;
Pointeur vers les données (octets bruts) :
void *_data;
};
typedef struct ldns_struct_rdf ldns_rdf;

ldns_rdf_type
Les différents types de champs RDATA.
enum ldns_enum_rdf_type
{
aucun :
LDNS_RDF_TYPE_NONE,
nom de domaine :
LDNS_RDF_TYPE_DNAME,
8 bits :
LDNS_RDF_TYPE_INT8,
16 bits :
LDNS_RDF_TYPE_INT16,
32 bits :
LDNS_RDF_TYPE_INT32,
enregistrement A :
LDNS_RDF_TYPE_A,
enregistrement AAAA :
LDNS_RDF_TYPE_AAAA,
chaîne txt :
LDNS_RDF_TYPE_STR,
données apl :
LDNS_RDF_TYPE_APL,
chaîne b32 :
LDNS_RDF_TYPE_B32_EXT,
chaîne b64 :
LDNS_RDF_TYPE_B64,
chaîne hex :
LDNS_RDF_TYPE_HEX,
codes de type nsec :
LDNS_RDF_TYPE_NSEC,
type d'enregistrement a :
LDNS_RDF_TYPE_TYPE,
classe a :
LDNS_RDF_TYPE_CLASS,
algorithme de certificat :
LDNS_RDF_TYPE_CERT_ALG,
algorithme a :
LDNS_RDF_TYPE_ALG,
types inconnus :
LDNS_RDF_TYPE_UNKNOWN,
temps (32 bits) :
LDNS_RDF_TYPE_TIME,
période :
LDNS_RDF_TYPE_PERIOD,
temps tsig 48 bits :
LDNS_RDF_TYPE_TSIGTIME,
/** Représente l'algorithme de clé publique, les champs HIT et clé publique
pour les types d'enregistrement HIP. Un type rdf spécifique à HIP est utilisé en raison
de la disposition inhabituelle en format filaire (voir RFC 5205 Section 5) */
LDNS_RDF_TYPE_HIP,
/** rdata de type quelconque de longueur variable où la longueur
est spécifiée par les 2 premiers octets */
LDNS_RDF_TYPE_INT16_DATA,
protocoles et bitmaps de port :
LDNS_RDF_TYPE_SERVICE,
données de localisation :
LDNS_RDF_TYPE_LOC,
services bien connus :
LDNS_RDF_TYPE_WKS,
NSAP :
LDNS_RDF_TYPE_NSAP,
ATMA :
LDNS_RDF_TYPE_ATMA,
IPSECKEY :
LDNS_RDF_TYPE_IPSECKEY,
sel de hachage nsec3 :
LDNS_RDF_TYPE_NSEC3_SALT,
chaîne base32 nsec3 (avec octet de longueur sur le fil :
LDNS_RDF_TYPE_NSEC3_NEXT_OWNER,

/** 4 short représentés par 4 nombres hexadécimaux de 16 bits
* séparés par des deux-points. Pour NID et L64.
*/
LDNS_RDF_TYPE_ILNP64,

6 nombres hexadécimaux de 8 bits séparés par des tirets. Pour EUI48. :
LDNS_RDF_TYPE_EUI48,
8 nombres hexadécimaux de 8 bits séparés par des tirets. Pour EUI64. :
LDNS_RDF_TYPE_EUI64,

/** Une séquence non nulle de lettres et de chiffres US-ASCII en minuscules.
* Pour CAA.
*/
LDNS_RDF_TYPE_TAG,

/** Un <chaîne-de-caractères> encodant le champ valeur comme spécifié
* dans [RFC1035], Section 5.1., encodé comme rdata restant.
* Pour CAA.
*/
LDNS_RDF_TYPE_LONG_STR,

/** Depuis RFC7218 les enregistrements TLSA peuvent être donnés avec des mnémoniques,
* d'où ces types de champs rdata. Mais comme pour les DNSKEYs, la sortie
* est toujours numérique.
*/
LDNS_RDF_TYPE_CERTIFICATE_USAGE,
LDNS_RDF_TYPE_SELECTOR,
LDNS_RDF_TYPE_MATCHING_TYPE,

draft-ietf-mboned-driad-amt-discovery *:
LDNS_RDF_TYPE_AMTRELAY,

draft-ietf-dnsop-svcb-https *:
LDNS_RDF_TYPE_SVCPARAMS,

/* Alias */
LDNS_RDF_TYPE_BITMAP = LDNS_RDF_TYPE_NSEC
};
typedef enum ldns_enum_rdf_type ldns_rdf_type;
