ldns_dnssec_name
Structure contenant tous les enregistrements de ressources pour un nom de domaine
Y compris le NSEC3 dérivé, s'il est présent
struct ldns_struct_dnssec_name
{
/**
* pointeur vers un dname contenant le nom.
* Pointe généralement vers le nom propriétaire du premier RR du premier RRset
*/
ldns_rdf *name;
/**
* Habituellement, le nom est un pointeur vers le nom propriétaire du premier rr pour
* ce nom, mais parfois il n'y a pas de données réelles à pointer,
* par exemple dans
* les noms représentant des terminaux vides. Si c'est le cas, mettre name_alloced à vrai pour
* indiquer que ces données doivent également être libérées lorsque le nom est libéré
*/
bool name_alloced;
/**
* Les rrsets pour ce nom
*/
ldns_dnssec_rrsets *rrsets;
/**
* NSEC pointant vers le nom suivant (ou NSEC3 pointant vers le NSEC3 suivant)
*/
ldns_rr *nsec;
/**
* signatures pour l'enregistrement NSEC
*/
ldns_dnssec_rrs *nsec_signatures;
/**
* Contrairement à ce que suggère le nom is_glue, ce champ est mis à vrai par
* ldns_dnssec_zone_mark_glue() ou ldns_dnssec_zone_mark_and_get_glue()
* lorsque le nom, que cette structure dnssec_name représente, est occlus.
* Les noms qui contiennent d'autres rrsets occlus et des enregistrements avec glue sur
* le point de délégation N'AURONT PAS ce booléen mis à vrai.
* Ce champ ne doit PAS être lu directement, mais uniquement via la
* fonction ldns_dnssec_name_is_glue() !
*/
bool is_glue;
/**
* pointeur pour stocker le nom haché (utilisé uniquement dans une zone NSEC3
*/
ldns_rdf *hashed_name;
};

typedef struct ldns_struct_dnssec_name ldns_dnssec_name;

struct ldns_struct_dnssec_rrs

};

typedef struct ldns_struct_dnssec_rrs ldns_dnssec_rrs;

ldns_dnssec_rrsets

struct ldns_struct_dnssec_rrsets
{

};

typedef struct ldns_struct_dnssec_rrsets ldns_dnssec_rrsets;
