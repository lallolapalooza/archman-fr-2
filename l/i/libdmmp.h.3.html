NOM

libdmmp.h - API de Device Mapper Multipath.

SYNTAXE

#include <libdmmp/libdmmp.h>

DESCRIPTION

Toutes les fonctions publiques de libdmmp ont leurs propres pages de manuel. Utilisez 'man 3 <nom_fonction>' pour vérifier les détails d'utilisation.

UTILISATION

Pour utiliser libdmmp dans votre projet, nous vous suggérons d'utiliser la méthode 'pkg-config' :


* Ajoutez cette ligne dans votre configure.ac :


PKG_CHECK_MODULES([LIBDMMP], [libdmmp])


* Ajoutez ces lignes dans votre Makefile.am :


foo_LDFLAGS += $(LIBDMMP_LIBS)
foo_CFLAGS += $(LIBDMMP_CFLAGS)

GESTION DES LOGS

La fonction de gestion des logs peut être définie via 'dmmp_context_log_func_set()'. La priorité des logs peut être définie via 'dmmp_context_log_priority_set()'.

Par défaut, la priorité des logs est 'DMMP_LOG_PRIORITY_WARNING'. Par défaut, la fonction de gestion des logs affiche les logs sur la sortie d'erreur standard (STDERR), et son code est listé ci-dessous au cas où vous voudriez créer votre propre fonction de gestion des logs.


static int _DMMP_LOG_STRERR_ALIGN_WIDTH = 80;


static void _log_stderr(struct dmmp_context *ctx,
enum dmmp_log_priority priority,
const char *file, int line,
const char *func_name,
const char *format, va_list args)
{
int printed_bytes = 0;


printed_bytes += fprintf(stderr, "libdmmp %s: ",
dmmp_log_priority_str(priority));
printed_bytes += vfprintf(stderr, format, args);
userdata = dmmp_context_userdata_get(ctx);
if (userdata != NULL)
fprintf(stderr, "(avec données utilisateur à l'adresse mémoire %p)",
userdata);


if (printed_bytes < _DMMP_LOG_STRERR_ALIGN_WIDTH) {
fprintf(stderr, "%*s # %s:%s():%d0,
_DMMP_LOG_STRERR_ALIGN_WIDTH - printed_bytes, "", file,
func_name, line);
} else {
fprintf(stderr, " # %s:%s():%d0, file, func_name, line);
}
}

EXEMPLE DE CODE


#include <libdmmp/libdmmp.h>


int main(int argc, char *argv[]) {
struct dmmp_context *ctx = NULL;
struct dmmp_mpath **dmmp_mps = NULL;
struct dmmp_path_group **dmmp_pgs = NULL;
struct dmmp_path **dmmp_ps = NULL;
uint32_t dmmp_mp_count = 0;
uint32_t dmmp_pg_count = 0;
uint32_t dmmp_p_count = 0;
const char *name = NULL;
const char *wwid = NULL;
uint32_t i = 0;
int rc = DMMP_OK;


ctx = dmmp_context_new();
dmmp_context_log_priority_set(ctx, DMMP_LOG_PRIORITY_DEBUG);
// Par défaut, les logs seront affichés sur la sortie d'erreur standard, vous pouvez
// changer cela via dmmp_context_log_func_set()
rc = dmmp_mpath_array_get(ctx, &dmmp_mps, &dmmp_mp_count);
if (rc != DMMP_OK) {
printf("dmmp_mpath_array_get() a échoué avec %d : %s", rc,
dmmp_strerror(rc));
goto out;
}
for (i = 0; i < dmmp_mp_count; ++i) {
name = dmmp_mpath_name_get(dmmp_mps[i]);
wwid = dmmp_mpath_wwid_get(dmmp_mps[i]);
printf("dmmp_mpath_array_get() : J'ai trouvé mpath : %s %s0, name,
wwid);
// Vous pouvez utiliser dmmp_path_group_array_get() pour récupérer
// des informations sur les groupes de chemins, puis appeler dmmp_path_array_get()
// pour des informations sur les chemins.
}


out:
dmmp_context_free(ctx);
dmmp_mpath_array_free(dmmp_mps, dmmp_mp_count);
if (rc != DMMP_OK)
exit(1);
exit(0);
}
