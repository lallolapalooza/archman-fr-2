NOM

libunwind-ptrace -- support de ptrace() dans libunwind

SYNTAXE

#include <libunwind-ptrace.h>

unw_accessors_t _UPT_accessors;

void *_UPT_create(pid_t);
void _UPT_destroy(void *);

int _UPT_find_proc_info(unw_addr_space_t, unw_word_t, unw_proc_info_t *, int, void *);
void _UPT_put_unwind_info(unw_addr_space_t, unw_proc_info_t *, void *);
int _UPT_get_dyn_info_list_addr(unw_addr_space_t, unw_word_t *, void *);
int _UPT_access_mem(unw_addr_space_t, unw_word_t, unw_word_t *, int, void *);
int _UPT_access_reg(unw_addr_space_t, unw_regnum_t, unw_word_t *, int, void *);
int _UPT_access_fpreg(unw_addr_space_t, unw_regnum_t, unw_fpreg_t *, int, void *);
int _UPT_get_proc_name(unw_addr_space_t, unw_word_t, char *, size_t, unw_word_t *, void *);
int _UPT_resume(unw_addr_space_t, unw_cursor_t *, void *);

DESCRIPTION

L'appel système ptrace(2) permet à un processus d'accéder à l'état de la machine et à la mémoire virtuelle d'un *autre* processus. Avec le bon ensemble de routines de rappel, il est donc possible de connecter libunwind à un autre processus via ptrace(2). Bien que ce ne soit pas très difficile à faire directement, libunwind facilite encore cette tâche en fournissant des routines de rappel prêtes à l'emploi à cet effet. Les routines et variables implémentant cette fonctionnalité utilisent le préfixe de nom _UPT, qui signifie « unwind via ptrace ».

Une application qui souhaite utiliser le support distant de ptrace de libunwind doit suivre les étapes suivantes.

1.
Créez un nouvel espace d'adressage libunwind qui représente le processus cible. Ceci est fait en appelant unw_create_addr_space(). Dans de nombreux cas, l'application voudra simplement passer l'adresse de _UPT_accessors comme premier argument à cette routine. Faire ainsi garantira que libunwind pourra correctement dérouler le processus cible.
2.
Activez le mode ptrace sur le processus cible, soit en créant un nouveau processus, en invoquant PTRACE_TRACEME, puis en démarrant le programme cible (via execve(2)), soit en vous attachant directement à un processus déjà en cours d'exécution (via PTRACE_ATTACH).
3.
Une fois que l'ID de processus (pid) du processus cible est connu, une structure d'informations UPT peut être créée en appelant _UPT_create(), en passant le pid du processus cible comme seul argument.
4.
Le pointeur opaque retourné doit ensuite être passé comme pointeur « argument » (troisième argument) à unw_init_remote().

Dans des circonstances spéciales, une application peut préférer n'utiliser que certaines parties du support distant de ptrace de libunwind. Pour cette raison, les routines de rappel individuelles (_UPT_find_proc_info(), _UPT_put_unwind_info(), etc.) sont également disponibles pour une utilisation directe. Bien sûr, les adresses de ces routines pourraient également être récupérées à partir de _UPT_accessors, mais cela empêcherait l'initialisation statique. De plus, lors de l'utilisation de _UPT_accessors, *toutes* les routines de rappel seront liées à l'application, même si elles ne sont jamais réellement appelées.

La routine _UPT_resume() peut être utilisée pour reprendre l'exécution du processus cible. Elle invoque simplement ptrace(2) avec une valeur de commande PTRACE_CONT.

Lorsque l'application a terminé d'utiliser libunwind sur le processus cible, _UPT_destroy() doit être appelée, en lui passant le pointeur opaque qui a été retourné par l'appel à _UPT_create(). Cela garantit que toute la mémoire et les autres ressources sont libérées.

DISPONIBILITE

Étant donné que ptrace(2) fonctionne uniquement au sein d'une seule machine, le support distant de ptrace de libunwind n'est par définition pas disponible dans les versions de libunwind configurées pour l'enroulement croisé.

SECURITE DES THREADS

Le support distant de ptrace de libunwind suppose qu'une seule structure d'informations UPT n'est jamais partagée entre les threads. Pour cette raison, aucun verrouillage explicite n'est utilisé. Tant qu'un seul thread utilise une structure d'informations UPT à un moment donné, cette fonctionnalité est thread-safe.

VALEUR DE RETOUR

_UPT_create() peut retourner un pointeur NULL s'il ne parvient pas à créer la structure d'informations UPT pour quelque raison que ce soit. Pour l'implémentation actuelle, la seule raison pour laquelle cet appel peut échouer est lorsque le système manque de mémoire.

EXEMPLE

    #include <libunwind-ptrace.h>
    #include <stdio.h>
    #include <stdlib.h>
    int
    main (int argc, char **argv)
    {
      if (argc != 2) {
        fprintf (stderr, "usage: %s PID\n", argv[0]);
        exit (EXIT_FAILURE);
      }
      char *endptr;
      pid_t target_pid = strtoul (argv[1], &endptr, 10);
      if (target_pid == 0 && argv[1] == endptr) {
        fprintf (stderr, "usage: %s PID\n", argv[0]);
        exit (EXIT_FAILURE);
      }
      unw_addr_space_t as = unw_create_addr_space (&_UPT_accessors, 0);
      if (!as) {
        fprintf (stderr, "unw_create_addr_space() failed");
        exit (EXIT_FAILURE);
      }
      void *ui = _UPT_create (target_pid);
      if (!ui) {
        fprintf (stderr, "_UPT_create() failed");
        exit (EXIT_FAILURE);
      }
      unw_cursor_t cursor;
      int ret = unw_init_remote (&cursor, as, ui);
      if (ret < 0) {
        fprintf (stderr, "unw_init_remote() failed: ret=%d\n", ret);
        exit (EXIT_FAILURE);
      }
      do {
        unw_proc_info_t pi;
        ret = unw_get_proc_info (&cursor, &pi);
        if (ret == -UNW_ENOINFO) {
          fprintf (stdout, "no info\n");
        } else if (ret >= 0) {
          printf ("\tproc=%#016lx-%#016lx\thandler=%#016lx lsda=%#016lx",
                  (long) pi.start_ip, (long) pi.end_ip,
                  (long) pi.handler, (long) pi.lsda);
        }
        ret = unw_step (&cursor);
      } while (ret > 0);
      if (ret < 0) {
        fprintf (stderr, "unwind failed with ret=%d\n", ret);
        exit (EXIT_FAILURE);
      }
      _UPT_destroy (ui);
      unw_destroy_addr_space (as);
      exit (EXIT_SUCCESS);
    }

NOM

libunwind-ptrace - Interface de débogage pour libunwind

SYNOPIS

#include <libunwind.h>

unw_cursor_t cursor;

unw_context_t context;

DESCRIPTION

La bibliothèque libunwind fournit une interface standard pour obtenir des informations sur la pile d'appels d'un processus. L'interface libunwind-ptrace est une interface de bas niveau qui permet d'inspecter la pile d'appels d'un processus différent à l'aide de l'appel système ptrace.

Cette interface est destinée à être utilisée par les débogueurs et autres outils de diagnostic qui ont besoin d'analyser la pile d'appels d'un processus en cours d'exécution.

FONCTIONS

  • unw_cursor_t : Type de données représentant un curseur de pile.
  • unw_context_t : Type de données représentant le contexte d'un thread.
  • unw_init_cursor(unw_cursor_t *cursor, unw_context_t *context) : Initialise un curseur de pile avec le contexte donné.
  • unw_step(unw_cursor_t *cursor) : Avance le curseur à l'appelant suivant dans la pile.
  • unw_get_reg(unw_cursor_t *cursor, unw_regnum_t regnum, unw_word_t *val) : Obtient la valeur d'un registre spécifié.
  • unw_get_proc_name(unw_cursor_t *cursor, char *buf, size_t len, unw_word_t *offset) : Obtient le nom de la procédure en cours.

EXEMPLE

L'exemple suivant montre comment utiliser libunwind-ptrace pour imprimer la pile d'appels d'un processus spécifié par son PID. 


#include <stdio.h>
#include <stdlib.h>
#include <libunwind.h>

int main(int argc, char **argv) {
    if (argc != 2) {
        fprintf(stderr, "Usage: %s <pid>\n", argv[0]);
        return 1;
    }

    pid_t pid = atoi(argv[1]);
    unw_cursor_t cursor;
    unw_context_t context;
    char buf[1024];
    unw_word_t offset;

    // Attacher au processus cible
    if (unw_attach_pid(pid, &context) < 0) {
        perror("unw_attach_pid");
        return 1;
    }

    // Initialiser le curseur
    if (unw_init_cursor(&cursor, &context) < 0) {
        perror("unw_init_cursor");
        return 1;
    }

    // Parcourir la pile d'appels
    while (unw_step(&cursor) > 0) {
        unw_get_proc_name(&cursor, buf, sizeof(buf), &offset);
        printf("0x%llx in %s (+ 0x%llx)\n", (unsigned long long)unw_get_reg(&cursor, UNW_REG_IP), buf, (unsigned long long)offset);
    }

    // Détacher du processus cible
    unw_detach(pid);

    return 0;
}
