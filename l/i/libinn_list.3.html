NOM

list - routines de liste

SYNTAXE

#include <inn/list.h>
struct node {
    struct node *succ;
    struct node *pred;
};
struct list {
    struct node *head;
    struct node *tail;
    struct node *tailpred;
};
void list_new(struct list *list);
struct node *list_addhead(struct list *list, struct node *node);
struct node *list_addtail(struct list *list, struct node *node);
struct node *list_head(struct list *list);
struct node *list_tail(struct list *list);
struct node *list_succ(struct node *node);
struct node *list_pred(struct node *node);
struct node *list_remhead(struct list *list);
struct node *list_remtail(struct list *list);
struct node *list_remove(struct node *node);
struct node *list_insert(struct list *list, struct node *node,
                         struct node *pred);
bool list_isempty(struct list *list);

DESCRIPTION

list_new initialise l'en-tête de liste list afin de créer une liste vide.

list_addhead ajoute node à la tête de list, en retournant le nœud qui vient d'être ajouté.

list_addtail ajoute node à la queue de list, en retournant le nœud qui vient d'être ajouté.

list_head retourne un pointeur vers le nœud en tête de list ou NULL si la liste est vide.

list_tail retourne un pointeur vers le nœud en queue de list ou NULL si la liste est vide.

list_succ retourne le nœud suivant (successeur) sur la liste après node ou NULL si node était le dernier nœud.

list_pred retourne le nœud précédent sur la liste avant node ou NULL si node était le premier nœud.

list_remhead retire le premier nœud de list et le retourne à l'appelant. Si la liste est vide, NULL est retourné.

list_remtail retire le dernier nœud de list et le retourne à l'appelant. Si la liste est vide, NULL est retourné.

list_remove retire node de la liste sur laquelle il se trouve et le retourne à l'appelant.

list_insert insère node dans list après le nœud pred. Si pred est NULL, alors node est ajouté à la tête de list.
