NOM

libcaca-font - Le format de police de libcaca (version 1)

Tous les types sont en big endian.

struct
{
magic:
   uint8_t caca_header[2];    // "\xCA\xCA"
   uint8_t caca_file_type[2]; // "FT"
font_header:
   uint32_t control_size;     // Taille du contrôle (font_data - font_header)
   uint32_t data_size;        // Taille des données (EOF - font_data)
   uint16_t version;          // Version du format de police
                              //  bit 0: mis à 1 si la police est compatible
                              //         avec la version 1 du format
                              //  bits 1-15: non utilisés pour le moment, doivent être à 0
   uint16_t blocks;           // Nombre de blocs dans la police
   uint32_t glyphs;           // Nombre total de glyphes dans la police
   uint16_t bpp;              // Bits par pixel pour les données de glyphes (valeurs
                              // valides : 1, 2, 4 et 8)
   uint16_t width;            // Largeur standard du glyphe
   uint16_t height;           // Hauteur standard du glyphe
   uint16_t maxwidth;         // Largeur maximale du glyphe
   uint16_t maxheight;        // Hauteur maximale du glyphe
   uint16_t flags;            // Indicateurs de fonctionnalités
                              //  bit 0: mis à 1 si la police est à largeur fixe
                              //  bits 1-15: non utilisés pour le moment, doivent être à 0
block_info:
   struct
   {
      uint32_t start;         // Index Unicode du premier glyphe
      uint32_t stop;          // Index Unicode du dernier glyphe + 1
      uint32_t index;         // Index d'information du glyphe du premier glyphe
   }
   block_list[blocks];
glyph_info:
   struct
   {
      uint16_t width;         // Largeur du glyphe en pixels
      uint16_t height;        // Hauteur du glyphe en pixels
      uint32_t data_offset;   // Décalage (à partir des données) vers les données
                              // pour le premier caractère
   }
   glyph_list[glyphs];
control_extension_1:
control_extension_2:
   ...
control_extension_N:
   ...                        // réservé pour usage futur
font_data:
   uint8_t data[data_size];   // données des glyphes
data_extension_1:
data_extension_2:
   ...
data_extension_N:
   ...                        // réservé pour usage futur
};
