libmtp - libusb-glue.h

NAME

libusb-glue - glue code between libusb and libmtp

SYNOPSIS

#include <libmtp.h>

Detailed Description

This is the glue code between libusb and libmtp. It is intended to make it easier for people to use libmtp without having to worry about libusb directly.

The constants defined below are used to create flags used by the devices.

#define LIBMTP_USB_DATA( buffer, length, base)

Value:

  do { \
    if ((LIBMTP_debug & LIBMTP_DEBUG_DATA) != 0) \
      data_dump_ascii (stdout, buffer, length, base); \
  } while (0)

#define LIBMTP_USB_DEBUG( format, args...)

Value:

  do { \
    if ((LIBMTP_debug & LIBMTP_DEBUG_USB) != 0) \
      fprintf(stdout, "LIBMTP %s[%d]: " format, __FUNCTION__, __LINE__, ##args); \
  } while (0)
