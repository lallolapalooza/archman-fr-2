NOM

libopenjp2 - une bibliothèque pour lire et écrire des fichiers image JPEG2000.

SYNTAXE

#include <openjpeg.h>

FORMATS DE CONVERSION

PGX: imagetopgx() / pgxtoimage()

PXM: imagetopnm() / pnmtoimage()

BMP: imagetobmp() / bmptoimage()

TIF: imagetotif() / tiftoimage()

RAW: imagetoraw() / rawtoimage()

TGA: imagetotga() / tgatoimage()

PNG: imagetopng() / pngtoimage()

YUV: imagetoyuv() / yuvtoimage() (MJ2)

PARAMETRES DE DECOMPRESSION

typedef struct opj_dparameters
{
/*
Définit le nombre de niveaux de résolution les plus élevés à rejeter.
La résolution de l'image est effectivement divisée par 2 à la puissance
du nombre de niveaux rejetés.
Le facteur de réduction est limité par le plus petit nombre total de
niveaux de décomposition parmi les tuiles.
si != 0, alors la dimension originale divisée par 2^(réduction);
si == 0 ou non utilisé, l'image est décodée à pleine résolution
*/
int cp_reduce;
/*
Définit le nombre maximum de couches de qualité à décoder.
S'il y a moins de couches de qualité que le nombre spécifié,
toutes les couches de qualité sont décodées.
si != 0, alors seules les "couches" premières sont décodées;
si == 0 ou non utilisé, toutes les couches de qualité sont décodées
*/
int cp_layer;


/* paramètres de l'encodeur en ligne de commande (non utilisés dans la bibliothèque) */
/* nom du fichier d'entrée */
char infile[OPJ_PATH_LEN];
/* nom du fichier de sortie */
char outfile[OPJ_PATH_LEN];
/* format du fichier d'entrée : voir OPJ_CODEC_FORMAT */
int decod_format;
/* format du fichier de sortie */
int cod_format;


/* paramètres de décodage JPWL */
/* active les capacités de correction JPWL */
bool jpwl_correct;
/* nombre de composantes attendu */
int jpwl_exp_comps;
/* nombre maximum de tuiles */
int jpwl_max_tiles;


/*
Spécifie si le décodage doit être effectué sur l'intégralité de
la souche de code, ou être limité à l'en-tête principal
Limiter le décodage à l'en-tête principal permet d'extraire
les caractéristiques de la souche de code
si == NO_LIMITATION, la souche de code entière est décodée;
si == LIMIT_TO_MAIN_HEADER, seul l'en-tête principal est décodé;
*/
OPJ_LIMIT_DECODING cp_limit_decoding;
} opj_dparameters_t;
