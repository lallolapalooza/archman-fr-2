libucontext(3) Manuel des fonctions de bibliothèque libucontext(3)

NOM

libucontext - une bibliothèque pour l'échange de contextes en espace utilisateur

SYNOPSIS

#include <libucontext/libucontext.h>

typedef struct {
  /* dépend de l'architecture cible */
} libucontext_mcontext_t;
typedef struct {
  void *ss_sp;
  int ss_flags;
  size_t ss_size;
} libucontext_stack_t;
typedef struct libucontext_ucontext {
  unsigned int uc_flags;
  struct libucontext_ucontext *uc_link;
  libucontext_stack_t uc_stack;
  libucontext_mcontext_t uc_mcontext;
} libucontext_ucontext_t;

int libucontext_getcontext(libucontext_ucontext_t *ucp);

int libucontext_setcontext(const libucontext_ucontext_t *ucp);

void libucontext_makecontext(libucontext_ucontext_t *ucp, void (*func)(), int argc, ...);

int libucontext_swapcontext(libucontext_ucontext_t *oucp, const libucontext_ucontext_t *ucp);

DESCRIPTION

La bibliothèque libucontext fournit une implémentation des fonctions ucontext SysV. Celles-ci sont traditionnellement utilisées pour implémenter l'échange de contextes en espace utilisateur. Ceci est réalisé en utilisant les fonctions libucontext_getcontext, libucontext_setcontext, libucontext_makecontext et libucontext_swapcontext selon ce qui convient.

La fonction libucontext_getcontext initialise une structure pointée par ucp avec le contexte utilisateur actuel.

La fonction libucontext_setcontext définit le contexte utilisateur actuel à la structure pointée par ucp. Elle ignore le contexte utilisateur actuel.

La fonction libucontext_swapcontext sauvegarde le contexte utilisateur actuel dans une structure pointée par oucp puis définit le contexte utilisateur actuel au nouveau contexte dans une structure pointée par ucp.

La fonction libucontext_makecontext modifie un contexte utilisateur dans une structure pointée par ucp pour exécuter une fonction pointée par func et met en place une liste d'arguments de argc valeurs.

CAVEATS

Dans SysV, les fonctions ucontext sauvegardent et restaurent les masques de signaux. Cependant, la bibliothèque libucontext ne le fait pas. En pratique, cela n'a généralement pas d'importance, car les utilisateurs de ces fonctions modifient rarement le masque de signaux entre les contextes.

D'autres implémentations peuvent ou non sauvegarder et restaurer des registres de processeur supplémentaires que cette implémentation ne sauvegarde pas. La bibliothèque libucontext ne sauvegarde et ne restaure que les registres à usage général. En pratique, cela s'est avéré suffisant.

EXEMPLE

Un exemple pratique montrant le multitraitement coopératif. Ce programme est destiné à des fins d'illustration uniquement et a été écrit d'une manière privilégiant la simplicité par rapport à la performance et à la robustesse :

#include <assert.h>
#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>
#include <libucontext/libucontext.h>
libucontext_ucontext_t mainctx = {};
libucontext_ucontext_t *curthr = &mainctx;
libucontext_ucontext_t *threads = NULL;
size_t thrcount = 0;
void
yieldto(libucontext_ucontext_t *target)
{
	libucontext_ucontext_t *oldthr = curthr;
	curthr = target;
	libucontext_swapcontext(oldthr, curthr);
}
void
yield(void)
{
	libucontext_ucontext_t *newthr;
	/* nous définissons uc_flags à une valeur non nulle pour signaler l'achèvement du thread. */
	do
		newthr = &threads[random() % thrcount];
	while (newthr == curthr || newthr->uc_flags);
	srandom(time(NULL));
	yieldto(newthr);
}
void
worker(size_t multiple)
{
	size_t accum = 1;
	for (size_t i = 0; i < 10; i++)
	{
		accum += (multiple * i);
		printf("[%p] accumulé %zun", curthr, accum);
		yield();
	}
	/* marquer le thread comme terminé, afin de ne pas revenir ici */
	curthr->uc_flags = 1;
}
void
create(size_t multiple)
{
	libucontext_ucontext_t *cursor;
	thrcount += 1;
	threads = realloc(threads, sizeof(*threads) * thrcount);
	cursor = &threads[thrcount - 1];
	memset(cursor, '0', sizeof *cursor);
	/* initialiser les valeurs du nouveau thread à notre contexte actuel */
	libucontext_getcontext(cursor);
	/* configurer uc_link */
	cursor->uc_link = thrcount > 1 ? &threads[thrcount - 2] : &mainctx;
	/* configurer une pile */
	cursor->uc_stack.ss_size = 8192;
	cursor->uc_stack.ss_sp = calloc(1, cursor->uc_stack.ss_size);
	/* configurer l'appel de fonction */
	libucontext_makecontext(cursor, worker, 1, multiple);
}
int
main(int argc, const char *argv[])
{
	srandom(time(NULL));
	libucontext_getcontext(&mainctx);
	for (size_t i = 1; i < 4; i++)
		create(i);
	/* lancer les threads en cédant au dernier */
	yieldto(&threads[thrcount - 1]);
	return EXIT_SUCCESS;
}
