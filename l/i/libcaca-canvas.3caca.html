NOM

libcaca-canvas - Le format de canevas libcaca (version 1)

Tous les types sont en big endian.

struct
{
magic:
   uint8_t caca_header[2];    // "\xCA\xCA"
   uint8_t caca_file_type[2]; // "CV"
canvas_header:
   uint32_t control_size;     // Taille du contrôle (canvas_data - canvas_header)
   uint32_t data_size;        // Taille des données (EOF - canvas_data)
   uint16_t version;          // Version du format de canevas
                              //  bit 0: mis à 1 si le canevas est compatible
                              //         avec la version 1 du format
                              //  bits 1-15: inutilisés pour l'instant, doivent être 0
   uint32_t frames;           // Nombre d'images
   uint16_t flags;            // Indicateurs de fonctionnalités
                              //  bits 0-15: inutilisés pour l'instant, doivent être 0
frame_info:
   struct
   {
      uint32_t width;         // Largeur de l'image
      uint32_t height;        // Hauteur de l'image
      uint32_t duration;      // Durée de l'image en millisecondes, 0 pour
                              // ne pas spécifier de durée
      uint32_t attr;          // Attribut du contexte graphique
      int32_t cursor_x;       // Coordonnée X du curseur
      int32_t cursor_y;       // Coordonnée Y du curseur
      int32_t handle_x;       // Coordonnée X du point de manipulation
      int32_t handle_y;       // Coordonnée Y du point de manipulation
   }
   frame_list[frames];
control_extension_1:
control_extension_2:
   ...
control_extension_N:
   ...                        // réservé pour utilisation future
canvas_data:
   uint8_t data[data_size];   // données du canevas
data_extension_1:
data_extension_2:
   ...
data_extension_N:
   ...                        // réservé pour utilisation future
};
