N

ILO

!>          ILO est ENTIER

IHI

!>

NSHIFTS

!>          NSHIFTS est ENTIER
!>          Le nombre désiré de shifts à utiliser
!>

NBLOCK_DESIRED

!>          NBLOCK_DESIRED est ENTIER
!>          La taille désirée des fenêtres de calcul
!>

SR

!>          SR est un tableau de DOUBLE PRÉCISION. SR contient
!>          les parties réelles des shifts à utiliser.
!>

SI

!>          SI est un tableau de DOUBLE PRÉCISION. SI contient

!>

!>

!>

B

!>

LDB

!>          La dimension principale du tableau B.  LDB >= max( 1, N ).
!>

Q

!>

LDQ

!>          LDQ est ENTIER
!>

Z

!>

LDZ

!>          LDZ est ENTIER
!>

QC

!>          QC est un tableau de DOUBLE PRÉCISION, dimension (LDQC, NBLOCK_DESIRED)
!>

LDQC

!>          LDQC est ENTIER
!>

ZC

!>          ZC est un tableau de DOUBLE PRÉCISION, dimension (LDZC, NBLOCK_DESIRED)
!>

LDZC

!>          LDZ est ENTIER
!>

WORK

!>          WORK est un tableau de DOUBLE PRÉCISION, dimension (MAX(1,LWORK))

LWORK

!>          LWORK est ENTIER
!>          La dimension du tableau WORK.  LWORK >= max(1,N).
