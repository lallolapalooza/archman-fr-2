subroutine cgbrfs (character trans, integer n, integer kl, integer ku, integer nrhs, complex, dimension( ldab, * ) ab, integer ldab, complex, dimension( ldafb, * ) afb, integer ldafb, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, complex, dimension( * ) work, real, dimension( * ) rwork, integer

info)

CGBRF

S

Objectif

: 
!>
!> CGBRFS améliore la solution calculée d'un système d'équations
!> linéaires lorsque la matrice des coefficients est à bande, et fournit
!> des bornes d'erreur et des estimations d'erreur arrière pour la solution.
!
>

Paramètre

s
TRANS 
!>          TRANS est CHARACTER*1
!>          Spécifie la forme du système d'équations :
!>          = 'N' :  A * X = B     (pas de transposition)
!>          = 'T' :  A**T * X = B  (transposition)
!>          = 'C' :  A**H * X = B  (transposition conjuguée)
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

K

L 
!>          KL est INTEGER
!>          Le nombre de sous-diagonales dans la bande de A.  KL >= 0.
!
>

K

U 
!>          KU est INTEGER
!>          Le nombre de sur-diagonales dans la bande de A.  KU >= 0.
!
>

NRH

S 
!>          NRHS est INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de colonnes
!>          des matrices B et X.  NRHS >= 0.
!
>

A

B 
!>          AB est un tableau COMPLEX, dimension (LDAB,N)
!>          La matrice à bande originale A, stockée dans les lignes 1 à KL+KU+1.
!>          La j-ième colonne de A est stockée dans la j-ième colonne du
!>          tableau AB comme suit :
!>          AB(ku+1+i-j,j) = A(i,j) pour max(1,j-ku)<=i<=min(n,j+kl).
!
>

LDA

B 
!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KL+KU+1.
!
>

AF

B 
!>          AFB est un tableau COMPLEX, dimension (LDAFB,N)
!>          Détails de la factorisation LU de la matrice à bande A, telle
!>          que calculée par CGBTRF.  U est stockée comme une matrice à bande
!>          triangulaire supérieure avec KL+KU sur-diagonales dans les lignes 1 à KL+KU+1, et
!>          les multiplicateurs utilisés pendant la factorisation sont stockés dans
!>          les lignes KL+KU+2 à 2*KL+KU+1.
!
>

LDAF

B 
!>          LDAFB est INTEGER
!>          La dimension principale du tableau AFB.  LDAFB >= 2*KL*KU+1.
!
>

IPI

V 
!>          IPIV est un tableau INTEGER, dimension (N)
!>          Les indices de pivot de CGBTRF ; pour 1<=i<=N, la ligne i de la
!>          matrice a été échangée avec la ligne IPIV(i).
!
>

B 
!>          B est un tableau COMPLEX, dimension (LDB,NRHS)
!>          La matrice du second membre B.
!
>

LD

B 
!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!
>

X 
!>          X est un tableau COMPLEX, dimension (LDX,NRHS)
!>          À l'entrée, la matrice de solution X, telle que calculée par CGBTRS.
!>          À la sortie, la matrice de solution améliorée X.
!
>

LD

X 
!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!
>

FER

R 
!>          FERR est un tableau REAL, dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur de solution
!>          X(j) (la j-ième colonne de la matrice de solution X).
!>          Si XTRUE est la vraie solution correspondant à X(j), FERR(j)
!>          est une borne supérieure estimée pour la magnitude du plus grand
!>          élément dans (X(j) - XTRUE) divisée par la magnitude du
!>          plus grand élément dans X(j).  L'estimation est aussi fiable que
!>          l'estimation pour RCOND, et est presque toujours une légère
!>          surestimation de la vraie erreur.
!
>

BER

R 
!>          BERR est un tableau REAL, dimension (NRHS)
!>          L'erreur arrière relative composante par composante de chaque vecteur
!>          de solution X(j) (c'est-à-dire le plus petit changement relatif dans
!>          n'importe quel élément de A ou B qui rend X(j) une solution exacte).
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (2*N)
!
>

RWOR

K 
!>          RWORK est un tableau REAL, dimension (N)
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!
>

Paramètres internes

: 
!>  ITMAX est le nombre maximal d'étapes de raffinement itératif.
!
>
