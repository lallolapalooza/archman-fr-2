subroutine cgbcon (character norm, integer n, integer kl, integer ku, complex, dimension( ldab, * ) ab, integer ldab, integer, dimension( * ) ipiv, real anorm, real rcond, complex, dimension( * ) work, real, dimension( * ) rwork, integer

info)

CGBCO

N

But

: 
!>
!> CGBCON estime l'inverse du nombre de condition d'une matrice
!> complexe par bande générale A, dans la norme 1 ou la norme
!> infinie, en utilisant la factorisation LU calculée par CGBTRF.
!>
!> Une estimation de norm(inv(A)) est obtenue, et l'inverse du
!> nombre de condition est calculé comme suit :
!>    RCOND = 1 / ( norm(A) * norm(inv(A)) ).
!
>

Paramètre

s
NORM 
!>          NORM est CHARACTER*1
!>          Indique si le nombre de condition en norme 1 ou le
!>          nombre de condition en norme infinie est requis :
!>          = '1' ou 'O' : norme 1 ;
!>          = 'I' :        norme infinie.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

K

L 
!>          KL est INTEGER
!>          Le nombre de sous-diagonales dans la bande de A.  KL >= 0.
!
>

K

U 
!>          KU est INTEGER
!>          Le nombre de sur-diagonales dans la bande de A.  KU >= 0.
!
>

A

B 
!>          AB est un tableau COMPLEX, dimension (LDAB,N)
!>          Détails de la factorisation LU de la matrice par bande A,
!>          telle que calculée par CGBTRF.  U est stockée comme une
!>          matrice par bande triangulaire supérieure avec KL+KU
!>          sur-diagonales dans les lignes 1 à KL+KU+1, et les
!>          multiplicateurs utilisés pendant la factorisation sont
!>          stockés dans les lignes KL+KU+2 à 2*KL+KU+1.
!
>

LDA

B 
!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= 2*KL+KU+1.
!
>

IPI

V 
!>          IPIV est un tableau INTEGER, dimension (N)
!>          Les indices de pivot ; pour 1 <= i <= N, la ligne i de la
!>          matrice a été échangée avec la ligne IPIV(i).
!
>

ANOR

M 
!>          ANORM est REAL
!>          Si NORM = '1' ou 'O', la norme 1 de la matrice originale A.
!>          Si NORM = 'I', la norme infinie de la matrice originale A.
!
>

RCON

D 
!>          RCOND est REAL
!>          L'inverse du nombre de condition de la matrice A,
!>          calculé comme RCOND = 1/(norm(A) * norm(inv(A))).
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (2*N)
!
>

RWOR

K 
!>          RWORK est un tableau REAL, dimension (N)
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur
!>                illégale
!
>
