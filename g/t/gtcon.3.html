subroutine cgtcon (character norm, integer n, complex, dimension( * ) dl, complex, dimension( * ) d, complex, dimension( * ) du, complex, dimension( * ) du2, integer, dimension( * ) ipiv, real anorm, real rcond, complex, dimension( * ) work, integer

info)

CGTCO

N

But

: 
!>
!> CGTCON estime l'inverse du conditionnement d'une matrice tridiagonale
!> complexe A en utilisant la factorisation LU telle que calculée par
!> CGTTRF.
!>
!> Une estimation de norm(inv(A)) est obtenue, et l'inverse du
!> conditionnement est calculé comme RCOND = 1 / (ANORM * norm(inv(A))).
!
>

Paramètre

s
NORM 
!>          NORM est CHARACTER*1
!>          Indique si le conditionnement en norme 1 ou le
!>          conditionnement en norme infini est requis :
!>          = '1' ou 'O' : norme 1 ;
!>          = 'I' :        norme infinie.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

D

L 
!>          DL est un tableau COMPLEX, dimension (N-1)
!>          Les (n-1) multiplicateurs qui définissent la matrice L de la
!>          factorisation LU de A telle que calculée par CGTTRF.
!
>

D 
!>          D est un tableau COMPLEX, dimension (N)
!>          Les n éléments diagonaux de la matrice triangulaire supérieure U de la
!>          factorisation LU de A.
!
>

D

U 
!>          DU est un tableau COMPLEX, dimension (N-1)
!>          Les (n-1) éléments de la première surdiagonale de U.
!
>

DU

2 
!>          DU2 est un tableau COMPLEX, dimension (N-2)
!>          Les (n-2) éléments de la deuxième surdiagonale de U.
!
>

IPI

V 
!>          IPIV est un tableau INTEGER, dimension (N)
!>          Les indices de pivot ; pour 1 <= i <= n, la ligne i de la matrice a été
!>          échangée avec la ligne IPIV(i).  IPIV(i) sera toujours soit
!>          i soit i+1 ; IPIV(i) = i indique qu'un échange de lignes n'a pas été
!>          requis.
!
>

ANOR

M 
!>          ANORM est REAL
!>          Si NORM = '1' ou 'O', la norme 1 de la matrice originale A.
!>          Si NORM = 'I', la norme infinie de la matrice originale A.
!
>

RCON

D 
!>          RCOND est REAL
!>          L'inverse du conditionnement de la matrice A,
!>          calculé comme RCOND = 1/(ANORM * AINVNM), où AINVNM est une
!>          estimation de la norme 1 de inv(A) calculée dans cette routine.
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (2*N)
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!
>
