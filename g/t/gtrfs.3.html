subroutine cgtrfs (character trans, integer n, integer nrhs, complex, dimension( * ) dl, complex, dimension( * ) d, complex, dimension( * ) du, complex, dimension( * ) dlf, complex, dimension( * ) df, complex, dimension( * ) duf, complex, dimension( * ) du2, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, complex, dimension( * ) work, real, dimension( * ) rwork, integer

info)

CGTRF

S

But

: 
!>
!> CGTRFS améliore la solution calculée d'un système d'équations
!> linéaires lorsque la matrice des coefficients est tridiagonale, et
!> fournit des bornes d'erreur et des estimations d'erreur arrière
!> pour la solution.
!
>

Paramètre

s
TRANS 
!>          TRANS est de type CHARACTER*1
!>          Spécifie la forme du système d'équations :
!>          = 'N' :  A * X = B     (Pas de transposition)
!>          = 'T' :  A**T * X = B  (Transposition)
!>          = 'C' :  A**H * X = B  (Transposition conjuguée)
!
>

N 
!>          N est de type INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

NRH

S 
!>          NRHS est de type INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de
!>          colonnes de la matrice B.  NRHS >= 0.
!
>

D

L 
!>          DL est un tableau COMPLEX de dimension (N-1)
!>          Les (n-1) éléments sous-diagonaux de A.
!
>

D 
!>          D est un tableau COMPLEX de dimension (N)
!>          Les éléments diagonaux de A.
!
>

D

U 
!>          DU est un tableau COMPLEX de dimension (N-1)
!>          Les (n-1) éléments super-diagonaux de A.
!
>

DL

F 
!>          DLF est un tableau COMPLEX de dimension (N-1)
!>          Les (n-1) multiplicateurs qui définissent la matrice L
!>          issue de la factorisation LU de A telle que calculée par
!>          CGTTRF.
!
>

D

F 
!>          DF est un tableau COMPLEX de dimension (N)
!>          Les n éléments diagonaux de la matrice triangulaire
!>          supérieure U issue de la factorisation LU de A.
!
>

DU

F 
!>          DUF est un tableau COMPLEX de dimension (N-1)
!>          Les (n-1) éléments de la première super-diagonale de U.
!
>

DU

2 
!>          DU2 est un tableau COMPLEX de dimension (N-2)
!>          Les (n-2) éléments de la deuxième super-diagonale de U.
!
>

IPI

V 
!>          IPIV est un tableau INTEGER de dimension (N)
!>          Les indices de pivot ; pour 1 <= i <= n, la ligne i de la
!>          matrice a été échangée avec la ligne IPIV(i).  IPIV(i)
!>          sera toujours soit i soit i+1 ; IPIV(i) = i indique qu'un
!>          échange de lignes n'a pas été nécessaire.
!
>

B 
!>          B est un tableau COMPLEX de dimension (LDB,NRHS)
!>          La matrice second membre B.
!
>

LD

B 
!>          LDB est de type INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!
>

X 
!>          X est un tableau COMPLEX de dimension (LDX,NRHS)
!>          À l'entrée, la matrice solution X, telle que calculée par
!>          CGTTRS.
!>          À la sortie, la matrice solution améliorée X.
!
>

LD

X 
!>          LDX est de type INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!
>

FER

R 
!>          FERR est un tableau REAL de dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur
!>          solution X(j) (la j-ième colonne de la matrice solution X).
!>          Si XTRUE est la solution réelle correspondant à X(j),
!>          FERR(j) est une borne supérieure estimée pour la grandeur
!>          du plus grand élément de (X(j) - XTRUE) divisée par la
!>          grandeur du plus grand élément de X(j).  L'estimation est
!>          aussi fiable que celle de RCOND et est presque toujours
!>          une légère surestimation de l'erreur réelle.
!
>

BER

R 
!>          BERR est un tableau REAL de dimension (NRHS)
!>          L'erreur arrière relative composante par composante de
!>          chaque vecteur solution X(j) (c'est-à-dire la plus petite
!>          variation relative de n'importe quel élément de A ou B qui
!>          rend X(j) une solution exacte).
!
>

WOR

K 
!>          WORK est un tableau COMPLEX de dimension (2*N)
!
>

RWOR

K 
!>          RWORK est un tableau REAL de dimension (N)
!
>

INF

O 
!>          INFO est de type INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur
!>          illégale
!
>

Paramètres internes

: 
!>  ITMAX est le nombre maximal d'étapes de raffinement itératif.
!
