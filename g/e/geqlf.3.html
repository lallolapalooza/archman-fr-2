subroutine cgeqlf (integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( * ) work, integer lwork, integer

info)

CGEQL

F

But

: 
!>
!> CGEQLF calcule une factorisation QL d'une matrice complexe A de dimensions M par N :
!> A = Q * L.
!
>

Paramètre

s
M 
!>          M est INTEGER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!
>

N 
!>          N est INTEGER
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          À l'entrée, la matrice A de dimensions M par N.
!>          À la sortie,
!>          si m >= n, le triangle inférieur du sous-tableau
!>          A(m-n+1:m,1:n) contient la matrice triangulaire inférieure L de dimensions N par N ;
!>          si m <= n, les éléments sur et sous la (n-m)-ième
!>          surdiagonale contiennent la matrice trapézoïdale inférieure L de dimensions M par N ;
!>          les éléments restants, avec le tableau TAU, représentent la
!>          matrice unitaire Q sous forme de produit de réflecteurs élémentaires
!>          (voir Détails supplémentaires).
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!
>

TA

U 
!>          TAU est un tableau COMPLEX, dimension (min(M,N))
!>          Les facteurs scalaires des réflecteurs élémentaires (voir Détails
!>          supplémentaires).
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie le LWORK optimal.
!
>

LWOR

K 
!>          LWORK est INTEGER
!>          La dimension du tableau WORK.
!>          LWORK >= 1 si MIN(M,N) = 0, et LWORK >= N sinon.
!>          Pour des performances optimales, LWORK >= N*NB, où NB est
!>          la taille de bloc optimale.
!>
!>          Si LWORK = -1, une interrogation d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!
>

Détails supplémentaires

: 
!>
!>  La matrice Q est représentée comme un produit de réflecteurs élémentaires
!>
!>     Q = H(k) . . . H(2) H(1), où k = min(m,n).
!>
!>  Chaque H(i) a la forme
!>
!>     H(i) = I - tau * v * v**H
!>
!>  où tau est un scalaire complexe, et v est un vecteur complexe avec
!>  v(m-k+i+1:m) = 0 et v(m-k+i) = 1 ; v(1:m-k+i-1) est stocké à la sortie dans
!>  A(1:m-k+i-1,n-k+i), et tau dans TAU(i).
!
>
