subroutine cgemmtr (character uplo, character transa, character transb, integer n, integer k, complex alpha, complex, dimension(lda,*) a, integer lda, complex, dimension(ldb,*) b, integer ldb, complex beta, complex, dimension(ldc,*) c, integer

ldc)

CGEMMT

R

But

: 
!>
!> CGEMMTR effectue l'une des opérations matrice-matrice
!>
!>    C := alpha*op( A )*op( B ) + beta*C,
!>
!> où op( X ) est l'une des formes
!>
!>    op( X ) = X   ou   op( X ) = X**T,
!>
!> alpha et beta sont des scalaires, et A, B et C sont des matrices, avec op( A )
!> une matrice n par k, op( B ) une matrice k par n et C une matrice n par n.
!> Ainsi, la routine n'accède et ne met à jour que la partie triangulaire
!> supérieure ou inférieure de la matrice résultat C. Ce comportement peut être
!> utilisé si la matrice résultante C est connue comme étant hermitienne ou
!> symétrique.
!
>

Paramètre

s
UPLO 
!>          UPLO est CHARACTER*1
!>           À l'entrée, UPLO spécifie si la partie triangulaire inférieure ou
!>           supérieure de C est accédée et mise à jour.
!>
!>              UPLO = 'L' ou 'l', la partie triangulaire inférieure de C est utilisée.
!>
!>              UPLO = 'U' ou 'u', la partie triangulaire supérieure de C est utilisée.
!
>

TRANS

A 
!>          TRANSA est CHARACTER*1
!>           À l'entrée, TRANSA spécifie la forme de op( A ) à utiliser dans la
!>           multiplication matrice-matrice comme suit :
!>
!>              TRANSA = 'N' ou 'n',  op( A ) = A.
!>
!>              TRANSA = 'T' ou 't',  op( A ) = A**T.
!>
!>              TRANSA = 'C' ou 'c',  op( A ) = A**H.
!
>

TRANS

B 
!>          TRANSB est CHARACTER*1
!>           À l'entrée, TRANSB spécifie la forme de op( B ) à utiliser dans la
!>           multiplication matrice-matrice comme suit :
!>
!>              TRANSB = 'N' ou 'n',  op( B ) = B.
!>
!>              TRANSB = 'T' ou 't',  op( B ) = B**T.
!>
!>              TRANSB = 'C' ou 'c',  op( B ) = B**H.
!
>

N 
!>          N est INTEGER
!>           À l'entrée, N spécifie le nombre de lignes et de colonnes de la
!>           matrice C, le nombre de colonnes de op(B) et le nombre de lignes
!>           de op(A). N doit être au moins zéro.
!
>

K 
!>          K est INTEGER
!>           À l'entrée, K spécifie le nombre de colonnes de la matrice
!>           op( A ) et le nombre de lignes de la matrice op( B ). K doit
!>           être au moins zéro.
!
>

ALPH

A 
!>          ALPHA est COMPLEX.
!>           À l'entrée, ALPHA spécifie le scalaire alpha.
!
>

A 
!>          A est un tableau COMPLEX, dimension ( LDA, ka ), où ka est
!>           k lorsque TRANSA = 'N' ou 'n', et n sinon.
!>           Avant l'entrée avec TRANSA = 'N' ou 'n', la partie n par k
!>           principale du tableau A doit contenir la matrice A, sinon la
!>           partie k par m principale du tableau A doit contenir la matrice A.
!
>

LD

A 
!>          LDA est INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle que
!>           déclarée dans le (sous-)programme appelant. Lorsque TRANSA = 'N'
!>           ou 'n', LDA doit être au moins max( 1, n ), sinon LDA doit être
!>           au moins max( 1, k ).
!
>

B 
!>          B est un tableau COMPLEX, dimension ( LDB, kb ), où kb est
!>           n lorsque TRANSB = 'N' ou 'n', et k sinon.
!>           Avant l'entrée avec TRANSB = 'N' ou 'n', la partie k par n
!>           principale du tableau B doit contenir la matrice B, sinon la
!>           partie n par k principale du tableau B doit contenir la matrice B.
!
>

LD

B 
!>          LDB est INTEGER
!>           À l'entrée, LDB spécifie la première dimension de B telle que
!>           déclarée dans le (sous-)programme appelant. Lorsque TRANSB = 'N'
!>           ou 'n', LDB doit être au moins max( 1, k ), sinon LDB doit être
!>           au moins max( 1, n ).
!
>

BET

A 
!>          BETA est COMPLEX.
!>           À l'entrée, BETA spécifie le scalaire beta. Lorsque BETA est
!>           fourni comme zéro, C n'a pas besoin d'être défini à l'entrée.
!
>

C 
!>          C est un tableau COMPLEX, dimension ( LDC, N )
!>           Avant l'entrée, la partie n par n principale du tableau C doit
!>           contenir la matrice C, sauf lorsque beta est zéro, auquel cas C
!>           n'a pas besoin d'être défini à l'entrée.
!>           À la sortie, la partie triangulaire supérieure ou inférieure de la
!>           matrice C est écrasée par la matrice n par n
!>           ( alpha*op( A )*op( B ) + beta*C ).
!
>

LD

C 
!>          LDC est INTEGER
!>           À l'entrée, LDC spécifie la première dimension de C telle que
!>           déclarée dans le (sous-)programme appelant. LDC doit être au
!>           moins max( 1, n ).
!
>
