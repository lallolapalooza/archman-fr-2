gearman_return_t gearman_client_do_high_background(gearman_client_st *client, const char *function_name, const char *unique, const void *workload, size_t workload_size, gearman_job_handle_t job_handle)
gearman_return_t gearman_client_do_low_background(gearman_client_st *client, const char *function_name, const char *unique, const void *workload, size_t workload_size, gearman_job_handle_t job_handle)

DESCRIPTION

gearman_client_do_background() exécute une requête unique vers le serveur gearmand et retourne le statut via gearman_return_t.

gearman_client_add_task_high_background() et gearman_client_add_task_low_background() sont identiques à gearman_client_do_background(), elles définissent seulement la priorité gearman_priority_t à haute ou basse.

Si job_handle n'est pas NULL, il sera rempli avec le nom du job_handle pour la tâche créée. Le job handle doit avoir la taille de GEARMAN_JOB_HANDLE_SIZE. Veuillez consulter gearman_job_handle_t pour plus d'informations.

VALEUR DE RETOUR

gearman_return_t

#include <string.h>
#include <stdlib.h>
#include <stdio.h>
#include <libgearman-1.0/gearman.h>
int main(void)
{
  gearman_client_st *client= gearman_client_create(NULL);
  gearman_return_t ret= gearman_client_add_server(client, "localhost", 0);
  if (gearman_failed(ret))
  {
    return EXIT_FAILURE;
  }
  gearman_job_handle_t job_handle;
  gearman_return_t rc= gearman_client_do_background(client,
                                                    "reverse_function",
                                                    "unique_value",
                                                    "my string to reverse", strlen("my string to reverse"),
                                                    job_handle);
  if (gearman_success(rc))
  {
    // Utiliser la valeur
    printf("%s\n", job_handle);
  }
  gearman_client_free(client);
  return 0;
}
