subroutine cgetrf (integer m, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, integer info)

CGETRF

But :

!>
!> CGETRF calcule une factorisation LU d'une matrice générale M par N A
!> en utilisant le pivotage partiel avec échange de lignes.
!>
!> La factorisation a la forme
!>    A = P * L * U
!> où P est une matrice de permutation, L est triangulaire inférieure avec des éléments
!> diagonaux unitaires (trapézoïdale inférieure si m > n), et U est triangulaire
!> supérieure (trapézoïdale supérieure si m < n).
!>
!> Il s'agit de la version de niveau 3 BLAS de l'algorithme regardant vers la droite.
!>

Paramètres

M 
!>          M est INTEGER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

A

!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          À l'entrée, la matrice M par N à factoriser.
!>          À la sortie, les facteurs L et U de la factorisation
!>          A = P*L*U ; les éléments diagonaux unitaires de L ne sont pas stockés.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (min(M,N))
!>          Les indices de pivot ; pour 1 <= i <= min(M,N), la ligne i de la
!>          matrice a été échangée avec la ligne IPIV(i).
!>

INFO

!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 :  si INFO = i, U(i,i) est exactement nul. La factorisation
!>                a été complétée, mais le facteur U est exactement
!>                singulier, et une division par zéro se produira s'il est utilisé
!>                pour résoudre un système d'équations.
!>
