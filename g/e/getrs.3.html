subroutine cgetrs (character trans, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, integer

info)

CGETR

S

But

: 
!>
!> CGETRS résout un système d'équations linéaires
!>    A * X = B,  A**T * X = B,  ou  A**H * X = B
!> avec une matrice générale N par N A en utilisant la factorisation LU calculée
!> par CGETRF.
!
>

Paramètre

s
TRANS 
!>          TRANS est CHARACTER*1
!>          Spécifie la forme du système d'équations :
!>          = 'N' :  A * X = B     (Pas de transposition)
!>          = 'T' :  A**T * X = B  (Transposition)
!>          = 'C' :  A**H * X = B  (Transposition conjuguée)
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

NRH

S 
!>          NRHS est INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          Les facteurs L et U de la factorisation A = P*L*U
!>          telle que calculée par CGETRF.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!
>

IPI

V 
!>          IPIV est un tableau INTEGER, dimension (N)
!>          Les indices de pivot de CGETRF ; pour 1<=i<=N, la ligne i de la
!>          matrice a été échangée avec la ligne IPIV(i).
!
>

B 
!>          B est un tableau COMPLEX, dimension (LDB,NRHS)
!>          À l'entrée, la matrice du second membre B.
!>          À la sortie, la matrice solution X.
!
>

LD

B 
!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!
>
