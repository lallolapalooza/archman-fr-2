subroutine cgecon (character norm, integer n, complex, dimension( lda, * ) a, integer lda, real anorm, real rcond, complex, dimension( * ) work, real, dimension( * ) rwork, integer

info)

CGECO

N

But

: 
!>
!> CGECON estime l'inverse du nombre de condition d'une matrice
!> complexe générale A, soit en 1-norme soit en norme d'infini, en
!> utilisant la factorisation LU calculée par CGETRF.
!>
!> Une estimation de norm(inv(A)) est obtenue, et l'inverse du
!> nombre de condition est calculé comme suit :
!>    RCOND = 1 / ( norm(A) * norm(inv(A)) ).
!
>

Paramètre

s
NORM 
!>          NORM est CHARACTER*1
!>          Indique si le nombre de condition en 1-norme ou le
!>          nombre de condition en norme d'infini est requis :
!>          = '1' ou 'O' :  1-norme ;
!>          = 'I' :         norme d'infini.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          Les facteurs L et U de la factorisation A = P*L*U
!>          tels que calculés par CGETRF.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!
>

ANOR

M 
!>          ANORM est REAL
!>          Si NORM = '1' ou 'O', la 1-norme de la matrice originale A.
!>          Si NORM = 'I', la norme d'infini de la matrice originale A.
!
>

RCON

D 
!>          RCOND est REAL
!>          L'inverse du nombre de condition de la matrice A,
!>          calculé comme RCOND = 1/(norm(A) * norm(inv(A))).
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (2*N)
!
>

RWOR

K 
!>          RWORK est un tableau REAL, dimension (2*N)
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale.
!>                Les NaN sont des valeurs illégales pour ANORM, et ils se propagent vers
!>                le paramètre de sortie RCOND.
!>                L'infini est illégal pour ANORM, et il se propage vers le paramètre de sortie
!>                RCOND sous forme de 0.
!>          = 1 :  si RCOND = NaN, ou
!>                   RCOND = Inf, ou
!>                   la norme calculée de l'inverse de A est 0.
!>                Dans ce dernier cas, RCOND = 0 est retourné.
!
>
