sous-routine cgemm (character transa, character transb, integer m, integer n, integer k, complex alpha, complex, dimension(lda,*) a, integer lda, complex, dimension(ldb,*) b, integer ldb, complex beta, complex, dimension(ldc,*) c, integer

ldc)

CGEM

M

But

: 
!>
!> CGEMM effectue l'une des opérations matrice-matrice suivantes
!>
!>    C := alpha*op( A )*op( B ) + beta*C,
!>
!> où op( X ) est l'une des formes suivantes
!>
!>    op( X ) = X   ou   op( X ) = X**T   ou   op( X ) = X**H,
!>
!> alpha et beta sont des scalaires, et A, B et C sont des matrices, où op( A )
!> est une matrice m par k, op( B ) est une matrice k par n et C est une matrice m par n.
!
>

Paramètre

s
TRANSA 
!>          TRANSA est de type CHARACTER*1
!>           À l'entrée, TRANSA spécifie la forme de op( A ) à utiliser dans
!>           la multiplication de matrices comme suit :
!>
!>              TRANSA = 'N' ou 'n',  op( A ) = A.
!>
!>              TRANSA = 'T' ou 't',  op( A ) = A**T.
!>
!>              TRANSA = 'C' ou 'c',  op( A ) = A**H.
!
>

TRANS

B 
!>          TRANSB est de type CHARACTER*1
!>           À l'entrée, TRANSB spécifie la forme de op( B ) à utiliser dans
!>           la multiplication de matrices comme suit :
!>
!>              TRANSB = 'N' ou 'n',  op( B ) = B.
!>
!>              TRANSB = 'T' ou 't',  op( B ) = B**T.
!>
!>              TRANSB = 'C' ou 'c',  op( B ) = B**H.
!
>

M 
!>          M est de type INTEGER
!>           À l'entrée, M spécifie le nombre de lignes de la matrice
!>           op( A ) et de la matrice C. M doit être au moins zéro.
!
>

N 
!>          N est de type INTEGER
!>           À l'entrée, N spécifie le nombre de colonnes de la matrice
!>           op( B ) et le nombre de colonnes de la matrice C. N doit être
!>           au moins zéro.
!
>

K 
!>          K est de type INTEGER
!>           À l'entrée, K spécifie le nombre de colonnes de la matrice
!>           op( A ) et le nombre de lignes de la matrice op( B ). K doit
!>           être au moins zéro.
!
>

ALPH

A 
!>          ALPHA est de type COMPLEX
!>           À l'entrée, ALPHA spécifie le scalaire alpha.
!
>

A 
!>          A est un tableau COMPLEX, de dimension ( LDA, ka ), où ka est
!>           k lorsque TRANSA = 'N' ou 'n', et m sinon.
!>           Avant l'appel avec TRANSA = 'N' ou 'n', la partie m par k
!>           principale du tableau A doit contenir la matrice A, sinon
!>           la partie k par m principale du tableau A doit contenir la
!>           matrice A.
!
>

LD

A 
!>          LDA est de type INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle que déclarée
!>           dans le (sous-)programme appelant. Lorsque TRANSA = 'N' ou 'n', LDA
!>           doit être au moins max( 1, m ), sinon LDA doit être au
!>           moins max( 1, k ).
!
>

B 
!>          B est un tableau COMPLEX, de dimension ( LDB, kb ), où kb est
!>           n lorsque TRANSB = 'N' ou 'n', et k sinon.
!>           Avant l'appel avec TRANSB = 'N' ou 'n', la partie k par n
!>           principale du tableau B doit contenir la matrice B, sinon
!>           la partie n par k principale du tableau B doit contenir la
!>           matrice B.
!
>

LD

B 
!>          LDB est de type INTEGER
!>           À l'entrée, LDB spécifie la première dimension de B telle que déclarée
!>           dans le (sous-)programme appelant. Lorsque TRANSB = 'N' ou 'n', LDB
!>           doit être au moins max( 1, k ), sinon LDB doit être au
!>           moins max( 1, n ).
!
>

BET

A 
!>          BETA est de type COMPLEX
!>           À l'entrée, BETA spécifie le scalaire beta. Lorsque BETA est
!>           fourni comme zéro, C n'a pas besoin d'être initialisé à l'entrée.
!
>

C 
!>          C est un tableau COMPLEX, de dimension ( LDC, N )
!>           Avant l'appel, la partie m par n principale du tableau C doit
!>           contenir la matrice C, sauf lorsque beta est zéro, auquel cas
!>           C n'a pas besoin d'être initialisé à l'entrée.
!>           À la sortie, le tableau C est écrasé par la matrice m par n
!>           ( alpha*op( A )*op( B ) + beta*C ).
!
>

LD

C 
!>          LDC est de type INTEGER
!>           À l'entrée, LDC spécifie la première dimension de C telle que déclarée
!>           dans le (sous-)programme appelant. LDC doit être au
!>           moins max( 1, m ).
!
>
