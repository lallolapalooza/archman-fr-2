subroutine cgesc2 (integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) rhs, integer, dimension( * ) ipiv, integer, dimension( * ) jpiv, real

scale)

CGESC2 résout un système d'équations linéaires en utilisant la factorisation LU avec pivotage complet calculée par

sgetc2.

But

: 
!>
!> CGESC2 résout un système d'équations linéaires
!>
!>           A * X = scale* RHS
!>
!> avec une matrice générale N par N A en utilisant la factorisation LU avec
!> pivotage complet calculée par CGETC2.
!>
!
>

Paramètre

s
N 
!>          N est INTEGER
!>          Le nombre de colonnes de la matrice A.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA, N)
!>          À l'entrée, la partie LU de la factorisation de la matrice n par n
!>          A calculée par CGETC2 :  A = P * L * U * Q
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1, N).
!
>

RH

S 
!>          RHS est un tableau COMPLEX, dimension N.
!>          À l'entrée, le vecteur second membre b.
!>          À la sortie, le vecteur solution X.
!
>

IPI

V 
!>          IPIV est un tableau INTEGER, dimension (N).
!>          Les indices de pivot ; pour 1 <= i <= N, la ligne i de la
!>          matrice a été échangée avec la ligne IPIV(i).
!
>

JPI

V 
!>          JPIV est un tableau INTEGER, dimension (N).
!>          Les indices de pivot ; pour 1 <= j <= N, la colonne j de la
!>          matrice a été échangée avec la colonne JPIV(j).
!
>

SCAL

E 
!>          SCALE est REAL
!>           À la sortie, SCALE contient le facteur d'échelle. SCALE est choisi
!>           0 <= SCALE <= 1 pour prévenir le débordement dans la solution.
!
>
