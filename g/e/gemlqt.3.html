sous-routine cgemlqt (character side, character trans, integer m, integer n, integer k, integer mb, complex, dimension( ldv, * ) v, integer ldv, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer info)

CGEMLQT

But :

!>
!> CGEMLQT écrase la matrice complexe générale M par N, C, avec
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q C            C Q
!> TRANS = 'C':   Q**H C            C Q**H
!>
!> où Q est une matrice unitaire complexe définie comme le produit de K
!> réflecteurs élémentaires :
!>
!>       Q = H(1) H(2) . . . H(K) = I - V T V**H
!>
!> générée en utilisant la représentation compacte WY telle que renvoyée par CGELQT.
!>
!> Q est d'ordre M si SIDE = 'L' et d'ordre N si SIDE = 'R'.
!>

Paramètres

SIDE 
!>          SIDE est CHARACTER*1
!>          = 'L': appliquer Q ou Q**H à Gauche ;
!>          = 'R': appliquer Q ou Q**H à Droite.
!>

TRANS

!>          TRANS est CHARACTER*1
!>          = 'N': Pas de transposée, appliquer Q ;
!>          = 'C': Transposée conjuguée, appliquer Q**H.
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice C. M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice C. N >= 0.
!>

K

!>          K est INTEGER
!>          Le nombre de réflecteurs élémentaires dont le produit définit
!>          la matrice Q.
!>          Si SIDE = 'L', M >= K >= 0 ;
!>          si SIDE = 'R', N >= K >= 0.
!>

MB

!>          MB est INTEGER
!>          La taille de bloc utilisée pour le stockage de T. K >= MB >= 1.
!>          Cela doit être la même valeur de MB utilisée pour générer T
!>          dans CGELQT.
!>

V

!>          V est un tableau COMPLEXE, de dimension
!>                               (LDV,M) si SIDE = 'L',
!>                               (LDV,N) si SIDE = 'R'
!>          La i-ème ligne doit contenir le vecteur qui définit le
!>          réflecteur élémentaire H(i), pour i = 1,2,...,k, tel que renvoyé par
!>          CGELQT dans les K premières lignes de son argument de tableau A.
!>

LDV

!>          LDV est INTEGER
!>          La dimension principale du tableau V. LDV >= max(1,K).
!>

T

!>          T est un tableau COMPLEXE, de dimension (LDT,K)
!>          Les facteurs triangulaires supérieurs des réflecteurs de bloc
!>          tels que renvoyés par CGELQT, stockés sous forme de matrice MB par K.
!>

LDT

!>          LDT est INTEGER
!>          La dimension principale du tableau T. LDT >= MB.
!>

C

!>          C est un tableau COMPLEXE, de dimension (LDC,N)
!>          À l'entrée, la matrice C de M par N.
!>          À la sortie, C est écrasé par Q C, Q**H C, C Q**H ou C Q.
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale du tableau C. LDC >= max(1,M).
!>

WORK

!>          WORK est un tableau COMPLEXE. La dimension de
!>          WORK est N*MB si SIDE = 'L', ou M*MB si SIDE = 'R'.
!>

INFO

!>          INFO est INTEGER
!>          = 0: sortie réussie
!>          < 0: si INFO = -i, le i-ème argument était invalide
!>
