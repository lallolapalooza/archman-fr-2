subroutine cgetc2 (integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, integer, dimension( * ) jpiv, integer

info)

CGETC2 calcule la factorisation LU avec pivotage complet de la matrice générale n par

n.

But

: 
!>
!> CGETC2 calcule une factorisation LU, en utilisant le pivotage complet, de la
!> matrice A de taille n par n. La factorisation a la forme A = P * L * U * Q,
!> où P et Q sont des matrices de permutation, L est triangulaire inférieure avec
!> des éléments diagonaux unitaires et U est triangulaire supérieure.
!>
!> Il s'agit d'une version de l'algorithme BLAS de niveau 1.
!
>

Paramètre

s
N 
!>          N est INTEGER
!>          L'ordre de la matrice A. N >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA, N)
!>          À l'entrée, la matrice n par n à factoriser.
!>          À la sortie, les facteurs L et U provenant de la factorisation
!>          A = P*L*U*Q ; les éléments diagonaux unitaires de L ne sont pas stockés.
!>          Si U(k, k) semble être inférieur à SMIN, U(k, k) reçoit la
!>          valeur de SMIN, donnant un système perturbé non singulier.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1, N).
!
>

IPI

V 
!>          IPIV est un tableau INTEGER, dimension (N).
!>          Les indices de pivot ; pour 1 <= i <= N, la ligne i de la
!>          matrice a été échangée avec la ligne IPIV(i).
!
>

JPI

V 
!>          JPIV est un tableau INTEGER, dimension (N).
!>          Les indices de pivot ; pour 1 <= j <= N, la colonne j de la
!>          matrice a été échangée avec la colonne JPIV(j).
!
>

INF

O 
!>          INFO est INTEGER
!>           = 0 : sortie réussie
!>           > 0 : si INFO = k, U(k, k) est susceptible de provoquer un débordement si
!>                l'on essaie de résoudre x dans Ax = b. U est donc perturbé
!>                pour éviter le débordement.
!
>
