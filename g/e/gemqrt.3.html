subroutine cgemqrt (character side, character trans, integer m, integer n, integer k, integer nb, complex, dimension( ldv, * ) v, integer ldv, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer info)

CGEMQRT

But :

!>
!> CGEMQRT remplace la matrice générale complexe M par N C par
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q C            C Q
!> TRANS = 'C':    Q**H C            C Q**H
!>
!> où Q est une matrice orthogonale complexe définie comme le produit de K
!> réflecteurs élémentaires :
!>
!>       Q = H(1) H(2) . . . H(K) = I - V T V**H
!>
!> générée à l'aide de la représentation WY compacte renvoyée par CGEQRT.
!>
!> Q est d'ordre M si SIDE = 'L' et d'ordre N si SIDE = 'R'.
!>

Paramètres

SIDE 
!>          SIDE est CHARACTER*1
!>          = 'L' : applique Q ou Q**H à gauche ;
!>          = 'R' : applique Q ou Q**H à droite.
!>

TRANS

!>          TRANS est CHARACTER*1
!>          = 'N' : pas de transposition, applique Q ;
!>          = 'C' : transposition conjuguée, applique Q**H.
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice C. M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice C. N >= 0.
!>

K

!>          K est INTEGER
!>          Le nombre de réflecteurs élémentaires dont le produit définit
!>          la matrice Q.
!>          Si SIDE = 'L', M >= K >= 0 ;
!>          si SIDE = 'R', N >= K >= 0.
!>

NB

!>          NB est INTEGER
!>          La taille de bloc utilisée pour le stockage de T. K >= NB >= 1.
!>          Cela doit correspondre à la même valeur de NB utilisée pour générer T
!>          dans CGEQRT.
!>

V

!>          V est un tableau COMPLEX de dimension (LDV,K)
!>          La i-ième colonne doit contenir le vecteur qui définit le
!>          réflecteur élémentaire H(i), pour i = 1,2,...,k, tel que renvoyé par
!>          CGEQRT dans les K premières colonnes de son argument de tableau A.
!>

LDV

!>          LDV est INTEGER
!>          La dimension principale du tableau V.
!>          Si SIDE = 'L', LDA >= max(1,M) ;
!>          si SIDE = 'R', LDA >= max(1,N).
!>

T

!>          T est un tableau COMPLEX de dimension (LDT,K)
!>          Les facteurs triangulaires supérieurs des réflecteurs de bloc
!>          tels que renvoyés par CGEQRT, stockés sous forme d'une matrice NB par N.
!>

LDT

!>          LDT est INTEGER
!>          La dimension principale du tableau T. LDT >= NB.
!>

C

!>          C est un tableau COMPLEX de dimension (LDC,N)
!>          À l'entrée, la matrice C de M par N.
!>          À la sortie, C est remplacée par Q C, Q**H C, C Q**H ou C Q.
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale du tableau C. LDC >= max(1,M).
!>

WORK

!>          WORK est un tableau COMPLEX. La dimension de WORK est
!>           N*NB si SIDE = 'L', ou M*NB si SIDE = 'R'.
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ième argument avait une valeur illégale
!>
