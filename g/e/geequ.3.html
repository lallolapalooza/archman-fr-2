subroutine cgeequ (integer m, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) r, real, dimension( * ) c, real rowcnd, real colcnd, real amax, integer

info)

CGEEQ

U

But

: 
!>
!> CGEEQU calcule des facteurs d'échelonnage de lignes et de colonnes destinés à équilibrer une
!> matrice A de dimensions M par N et à réduire son conditionnement. R renvoie les facteurs
!> d'échelonnage des lignes et C les facteurs d'échelonnage des colonnes, choisis de manière à
!> tenter de rendre la valeur absolue du plus grand élément de chaque ligne et de chaque colonne
!> de la matrice B dont les éléments sont B(i,j)=R(i)*A(i,j)*C(j) égale à 1.
!>
!> R(i) et C(j) sont limités à être compris entre SMLNUM = plus petit nombre sûr
!> et BIGNUM = plus grand nombre sûr. L'utilisation de ces facteurs
!> d'échelonnage n'est pas garantie de réduire le conditionnement de A mais
!> fonctionne bien en pratique.
!
>

Paramètre

s
M 
!>          M est INTEGER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!
>

N 
!>          N est INTEGER
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!
>

A 
!>          A est un tableau COMPLEX de dimension (LDA,N)
!>          La matrice A de dimensions M par N dont les facteurs
!>          d'équilibrage doivent être calculés.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!
>

R 
!>          R est un tableau REAL de dimension (M)
!>          Si INFO = 0 ou INFO > M, R contient les facteurs
!>          d'échelonnage des lignes de A.
!
>

C 
!>          C est un tableau REAL de dimension (N)
!>          Si INFO = 0, C contient les facteurs d'échelonnage des colonnes de A.
!
>

ROWCN

D 
!>          ROWCND est REAL
!>          Si INFO = 0 ou INFO > M, ROWCND contient le rapport du
!>          plus petit R(i) au plus grand R(i).  Si ROWCND >= 0.1 et
!>          AMAX n'est ni trop grand ni trop petit, il n'est pas utile
!>          d'échelonner par R.
!
>

COLCN

D 
!>          COLCND est REAL
!>          Si INFO = 0, COLCND contient le rapport du plus petit
!>          C(i) au plus grand C(i).  Si COLCND >= 0.1, il n'est pas
!>          utile d'échelonner par C.
!
>

AMA

X 
!>          AMAX est REAL
!>          Valeur absolue du plus grand élément de la matrice.  Si AMAX est très
!>          proche du débordement ou très proche du sous-échelonnement, la matrice
!>          devrait être mise à l'échelle.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 :  si INFO = i, et i est
!>                <= M :  la i-ème ligne de A est exactement nulle
!>                >  M :  la (i-M)-ème colonne de A est exactement nulle
!
>
