subroutine cgebak (character job, character side, integer n, integer ilo, integer ihi, real, dimension( * ) scale, integer m, complex, dimension( ldv, * ) v, integer ldv, integer

info)

CGEBA

K

But

: 
!>
!> CGEBAK forme les vecteurs propres droits ou gauches d'une matrice
!> générale complexe par transformation inverse appliquée aux vecteurs
!> propres calculés de la matrice équilibrée produite par CGEBAL.
!
>

Paramètre

s
JOB 
!>          JOB est CHARACTER*1
!>          Spécifie le type de transformation inverse requise :
!>          = 'N' : ne rien faire, retour immédiat ;
!>          = 'P' : effectuer la transformation inverse pour la permutation uniquement ;
!>          = 'S' : effectuer la transformation inverse pour la mise à l'échelle uniquement ;
!>          = 'B' : effectuer les transformations inverses pour la permutation et
!>                 la mise à l'échelle.
!>          JOB doit être identique à l'argument JOB fourni à CGEBAL.
!
>

SID

E 
!>          SIDE est CHARACTER*1
!>          = 'R' :  V contient les vecteurs propres droits ;
!>          = 'L' :  V contient les vecteurs propres gauches.
!
>

N 
!>          N est INTEGER
!>          Le nombre de lignes de la matrice V.  N >= 0.
!
>

IL

O 
!>          ILO est INTEGER
!
>

IH

I 
!>          IHI est INTEGER
!>          Les entiers ILO et IHI déterminés par CGEBAL.
!>          1 <= ILO <= IHI <= N, si N > 0 ; ILO=1 et IHI=0, si N=0.
!
>

SCAL

E 
!>          SCALE est un tableau REAL, dimension (N)
!>          Détails des facteurs de permutation et de mise à l'échelle, tels que
!>          renvoyés par CGEBAL.
!
>

M 
!>          M est INTEGER
!>          Le nombre de colonnes de la matrice V.  M >= 0.
!
>

V 
!>          V est un tableau COMPLEX, dimension (LDV,M)
!>          À l'entrée, la matrice des vecteurs propres droits ou gauches à
!>          transformer, telle que renvoyée par CHSEIN ou CTREVC.
!>          À la sortie, V est écrasée par les vecteurs propres transformés.
!
>

LD

V 
!>          LDV est INTEGER
!>          La dimension principale du tableau V. LDV >= max(1,N).
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale.
!
>
