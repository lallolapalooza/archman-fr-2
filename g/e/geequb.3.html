subroutine cgeequb (integer m, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) r, real, dimension( * ) c, real rowcnd, real colcnd, real amax, integer

info)

CGEEQU

B

But

: 
!>
!> CGEEQUB calcule des facteurs de mise à l'échelle de lignes et de
!> colonnes destinés à équilibrer une matrice A de dimensions M par N
!> et à réduire son conditionnement. R retourne les facteurs de mise à
!> l'échelle des lignes et C ceux des colonnes, choisis de manière à
!> tenter de rendre l'élément le plus grand de chaque ligne et de
!> chaque colonne de la matrice B, dont les éléments sont
!> B(i,j)=R(i)*A(i,j)*C(j), de valeur absolue au plus égale à la
!> base.
!>
!> R(i) et C(j) sont limités à être une puissance de la base comprise
!> entre SMLNUM = plus petit nombre sûr et BIGNUM = plus grand nombre
!> sûr. L'utilisation de ces facteurs de mise à l'échelle n'est pas
!> garantie de réduire le conditionnement de A, mais fonctionne bien en
!> pratique.
!>
!> Cette routine diffère de CGEEQU en ce qu'elle limite les facteurs de
!> mise à l'échelle à une puissance de la base. En l'absence de
!> débordement et de sous-dépassement, la mise à l'échelle par ces
!> facteurs n'introduit pas d'erreurs d'arrondi supplémentaires.
!> Cependant, les ordres de grandeur des entrées mises à l'échelle ne
!> sont plus approximativement égaux à 1, mais se situent entre
!> sqrt(radix) et 1/sqrt(radix).
!
>

Paramètre

s
M 
!>          M est INTEGER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!
>

N 
!>          N est INTEGER
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!
>

A 
!>          A est un tableau COMPLEX, de dimension (LDA,N)
!>          La matrice M par N dont les facteurs d'équilibrage sont
!>          à calculer.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!
>

R 
!>          R est un tableau REAL, de dimension (M)
!>          Si INFO = 0 ou INFO > M, R contient les facteurs de mise à
!>          l'échelle des lignes pour A.
!
>

C 
!>          C est un tableau REAL, de dimension (N)
!>          Si INFO = 0, C contient les facteurs de mise à l'échelle
!>          des colonnes pour A.
!
>

ROWCN

D 
!>          ROWCND est REAL
!>          Si INFO = 0 ou INFO > M, ROWCND contient le rapport du
!>          plus petit R(i) au plus grand R(i). Si ROWCND >= 0.1 et
!>          AMAX n'est ni trop grand ni trop petit, il n'est pas
!>          utile de mettre à l'échelle par R.
!
>

COLCN

D 
!>          COLCND est REAL
!>          Si INFO = 0, COLCND contient le rapport du plus petit
!>          C(i) au plus grand C(i). Si COLCND >= 0.1, il n'est pas
!>          utile de mettre à l'échelle par C.
!
>

AMA

X 
!>          AMAX est REAL
!>          Valeur absolue du plus grand élément de la matrice. Si
!>          AMAX est très proche du débordement ou très proche du
!>          sous-dépassement, la matrice devrait être mise à
!>          l'échelle.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur
!>                illégale
!>          > 0 : si INFO = i, et i est
!>                <= M : la i-ème ligne de A est exactement nulle
!>                >  M : la (i-M)-ème colonne de A est exactement
!>                       nulle
!
>
