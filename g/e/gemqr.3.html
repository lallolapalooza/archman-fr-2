subroutine cgemqr (character side, character trans, integer m, integer n, integer k, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) t, integer tsize, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer lwork, integer

info)

CGEMQ

R

But

: 
!>
!> CGEMQR remplace la matrice réelle générale C de taille M par N par
!>
!>                      SIDE = 'L'     SIDE = 'R'
!>      TRANS = 'N':      Q * C          C * Q
!>      TRANS = 'T':      Q**H * C       C * Q**H
!>
!> où Q est une matrice complexe unitaire définie comme le produit
!> de réflecteurs élémentaires par blocs calculés par la factorisation
!> QR tall skinny (CGEQR)
!>
!
>

Paramètre

s
SIDE 
!>          SIDE est CHARACTER*1
!>          = 'L' : applique Q ou Q**H à gauche ;
!>          = 'R' : applique Q ou Q**H à droite.
!
>

TRAN

S 
!>          TRANS est CHARACTER*1
!>          = 'N' :  pas de transposition, applique Q ;
!>          = 'C' :  transposition conjuguée, applique Q**H.
!
>

M 
!>          M est INTEGER
!>          Le nombre de lignes de la matrice A.  M >=0.
!
>

N 
!>          N est INTEGER
!>          Le nombre de colonnes de la matrice C. N >= 0.
!
>

K 
!>          K est INTEGER
!>          Le nombre de réflecteurs élémentaires dont le produit définit
!>          la matrice Q.
!>          Si SIDE = 'L', M >= K >= 0 ;
!>          si SIDE = 'R', N >= K >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA,K)
!>          Partie de la structure de données représentant Q telle que retournée par CGEQR.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.
!>          Si SIDE = 'L', LDA >= max(1,M) ;
!>          si SIDE = 'R', LDA >= max(1,N).
!
>

T 
!>          T est un tableau COMPLEX, dimension (MAX(5,TSIZE)).
!>          Partie de la structure de données représentant Q telle que retournée par CGEQR.
!
>

TSIZ

E 
!>          TSIZE est INTEGER
!>          La dimension du tableau T. TSIZE >= 5.
!
>

C 
!>          C est un tableau COMPLEX, dimension (LDC,N)
!>          À l'entrée, la matrice C de taille M par N.
!>          À la sortie, C est remplacée par Q*C ou Q**H*C ou C*Q**H ou C*Q.
!
>

LD

C 
!>          LDC est INTEGER
!>          La dimension principale du tableau C. LDC >= max(1,M).
!
>

WOR

K 
!>          (espace de travail) tableau COMPLEX, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) retourne le LWORK minimal.
!
>

LWOR

K 
!>          LWORK est INTEGER
!>          La dimension du tableau WORK. LWORK >= 1.
!>          Si LWORK = -1, une interrogation sur l'espace de travail est supposée. Le sous-programme
!>          calcule uniquement la taille du tableau WORK, retourne cette
!>          valeur sous la forme de WORK(1), et aucun message d'erreur lié à WORK
!>          n'est émis par XERBLA.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!
>
