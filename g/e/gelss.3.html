subroutine cgelss (integer m, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, real, dimension( * ) s, real rcond, integer rank, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, integer

info)

CGELSS résout des systèmes surdéterminés ou sous-déterminés pour les matrices

GE

But

: 
!>
!> CGELSS calcule la solution de norme minimale d'un problème de
!> moindres carrés linéaire complexe :
!>
!> Minimiser 2-norme(| b - A*x |).
!>
!> en utilisant la décomposition en valeurs singulières (SVD) de A. A est une
!> matrice M par N qui peut être de rang déficient.
!>
!> Plusieurs vecteurs de second membre b et vecteurs solution x peuvent être
!> traités dans un seul appel ; ils sont stockés en tant que colonnes de la
!> matrice de second membre B de taille M par NRHS et de la matrice solution X
!> de taille N par NRHS.
!>
!> Le rang effectif de A est déterminé en traitant comme nulles les valeurs
!> singulières inférieures à RCOND fois la plus grande valeur singulière.
!
>

Paramètre

s
M 
!>          M est INTEGER
!>          Le nombre de lignes de la matrice A. M >= 0.
!
>

N 
!>          N est INTEGER
!>          Le nombre de colonnes de la matrice A. N >= 0.
!
>

NRH

S 
!>          NRHS est INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de colonnes
!>          des matrices B et X. NRHS >= 0.
!
>

A 
!>          A est un tableau COMPLEX de dimension (LDA,N)
!>          À l'entrée, la matrice A de taille M par N.
!>          À la sortie, les premières min(m,n) lignes de A sont écrasées par
!>          ses vecteurs singuliers droits, stockés par ligne.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,M).
!
>

B 
!>          B est un tableau COMPLEX de dimension (LDB,NRHS)
!>          À l'entrée, la matrice de second membre B de taille M par NRHS.
!>          À la sortie, B est écrasée par la matrice solution X de taille N
!>          par NRHS.
!>          Si m >= n et RANK = n, la somme des carrés des résidus pour la
!>          solution dans la i-ème colonne est donnée par la somme des carrés
!>          des modules des éléments n+1:m dans cette colonne.
!
>

LD

B 
!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,M,N).
!
>

S 
!>          S est un tableau REAL de dimension (min(M,N))
!>          Les valeurs singulières de A en ordre décroissant.
!>          Le conditionnement de A dans la 2-norme = S(1)/S(min(m,n)).
!
>

RCON

D 
!>          RCOND est REAL
!>          RCOND est utilisé pour déterminer le rang effectif de A.
!>          Les valeurs singulières S(i) <= RCOND*S(1) sont traitées comme nulles.
!>          Si RCOND < 0, la précision machine est utilisée à la place.
!
>

RAN

K 
!>          RANK est INTEGER
!>          Le rang effectif de A, c'est-à-dire le nombre de valeurs singulières
!>          supérieures à RCOND*S(1).
!
>

WOR

K 
!>          WORK est un tableau COMPLEX de dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie le LWORK optimal.
!
>

LWOR

K 
!>          LWORK est INTEGER
!>          La dimension du tableau WORK. LWORK >= 1, et aussi :
!>          LWORK >=  2*min(M,N) + max(M,N,NRHS)
!>          Pour de bonnes performances, LWORK devrait généralement être plus grand.
!>
!>          Si LWORK = -1, une interrogation d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message
!>          d'erreur lié à LWORK n'est émis par XERBLA.
!
>

RWOR

K 
!>          RWORK est un tableau REAL de dimension (5*min(M,N))
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale.
!>          > 0 :  l'algorithme de calcul de la SVD n'a pas convergé ;
!>                 si INFO = i, i éléments hors diagonale d'une forme
!>                 bidiagonale intermédiaire n'ont pas convergé vers zéro.
!
>
