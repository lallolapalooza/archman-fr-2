LIBGVC(3) Manuel des fonctions de bibliothèque LIBGVC(3)

NOM

libgvc - Bibliothèque de contexte Graphviz

SYNTAXE

#include <graphviz/gvc.h>
/* initialiser un contexte graphviz */
extern GVC_t *gvNEWcontext(const lt_symlist_t *builtins, int demand_loading);
/*  initialiser un contexte graphviz — alternative */
/*     (encapsule les deux fonctions ci-dessus en utilisant les informations intégrées dans libgvc) */
extern GVC_t *gvContext(void);
/* analyser les arguments de la ligne de commande — argv[0] définit minimalement le moteur de disposition */
extern int gvParseArgs(GVC_t *gvc, int argc, char **argv);
extern graph_t *gvNextInputGraph(GVC_t *gvc);
/* calculer une disposition en utilisant un moteur spécifié */
extern int gvLayout(GVC_t *gvc, graph_t *g, char *engine);
/* calculer une disposition en utilisant le moteur de disposition des arguments de ligne de commande */
extern int gvLayoutJobs(GVC_t *gvc, graph_t *g);
/* vérifier si une disposition a été effectuée */
extern bool gvLayoutDone(graph_t *g);
/* rendre la disposition dans les attributs de chaîne du graphe */
extern void attach_attrs(graph_t *g);
/* analyser une chaîne html */
extern char *agstrdup_html(char *s);
extern int aghtmlstr(char *s);
/* rendre la disposition dans un format spécifié vers un fichier FILE ouvert */
extern int gvRender(GVC_t *gvc, graph_t *g, char *format, FILE *out);
/* rendre la disposition dans un format spécifié vers un fichier FILE ouvert */
extern int gvRenderFilename(GVC_t *gvc, graph_t *g, char *format, char *filename);
/* rendre la disposition selon les options -T et -o trouvées par gvParseArgs */
extern int gvRenderJobs(GVC_t *gvc, graph_t *g);
/* nettoyer les structures de données de disposition — les dispositions ne sont pas imbricables (pour l'instant) */
extern int gvFreeLayout(GVC_t *gvc, graph_t *g);
/* nettoyer le contexte graphviz */
extern int gvFreeContext(GVC_t *gvc);
/* s'informer sur les plugins disponibles */
/* voir le commentaire dans gvc.h            */
extern char** gvPluginList(GVC_t *gvc, char* kind, int* cnt, char*);

DESCRIPTION

libgvc fournit un contexte pour les applications souhaitant manipuler et rendre des graphes. Elle fournit l'analyse des lignes de commande, du code de rendu commun et un mécanisme de plugin pour les rendu.

VOIR AUSSI

dot(1), neato(1), cdt(3), cgraph(3)
