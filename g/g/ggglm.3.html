subroutine cggglm (integer n, integer m, integer p, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) d, complex, dimension( * ) x, complex, dimension( * ) y, complex, dimension( * ) work, integer lwork, integer

info)

CGGGL

M

But

: 
!>
!> CGGGLM résout un problème général de modèle linéaire de Gauss-Markov (GLM) :
!>
!>         minimiser || y ||_2   sous la contrainte   d = A*x + B*y
!>             x
!>
!> où A est une matrice N×M, B est une matrice N×P, et d est un
!> vecteur N donné. On suppose que M <= N <= M+P, et
!>
!>            rank(A) = M    et    rank( A B ) = N.
!>
!> Sous ces hypothèses, l'équation contrainte est toujours
!> cohérente, et il existe une solution x unique et une solution y
!> de norme 2 minimale, obtenue à l'aide d'une factorisation QR
!> généralisée des matrices (A, B) donnée par
!>
!>    A = Q*(R),   B = Q*T*Z.
!>          (0)
!>
!> En particulier, si la matrice B est carrée non singulière, alors le problème
!> GLM est équivalent au problème suivant de moindres carrés linéaires pondérées
!>
!>              minimiser || inv(B)*(d-A*x) ||_2
!>                  x
!>
!> où inv(B) désigne l'inverse de B.
!>
!> Les appelants de cette sous-routine doivent noter que les vérifications de
!> singularité/déficit de rang implémentées dans cette sous-routine sont
!> rudimentaires. La sous-routine CTRTRS appelée par cette sous-routine ne signale
!> un échec dû à la singularité que si le problème est exactement singulier.
!>
!> Il est concevable qu'un (ou plusieurs) des facteurs impliqués dans la
!> factorisation QR généralisée de la paire (A, B) soit proche de la singularité
!> à un niveau sous-normale sans que cette sous-routine ne signale d'erreur. Les
!> solutions calculées pour de tels problèmes quasi-déficients en rang peuvent
!> être moins précises en raison d'une perte de précision numérique.
!>
!
>

Paramètre

s
N 
!>          N est INTEGER
!>          Le nombre de lignes des matrices A et B.  N >= 0.
!
>

M 
!>          M est INTEGER
!>          Le nombre de colonnes de la matrice A.  0 <= M <= N.
!
>

P 
!>          P est INTEGER
!>          Le nombre de colonnes de la matrice B.  P >= N-M.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA,M)
!>          À l'entrée, la matrice A de dimension N×M.
!>          À la sortie, la partie triangulaire supérieure du tableau A contient
!>          la matrice triangulaire supérieure R de dimension M×M.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,N).
!
>

B 
!>          B est un tableau COMPLEX, dimension (LDB,P)
!>          À l'entrée, la matrice B de dimension N×P.
!>          À la sortie, si N <= P, le triangle supérieur du sous-tableau
!>          B(1:N,P-N+1:P) contient la matrice triangulaire supérieure T de
!>          dimension N×N ;
!>          si N > P, les éléments sur et au-dessus de la (N-P)-ème sous-diagonale
!>          contiennent la matrice trapézoïdale supérieure T de dimension N×P.
!
>

LD

B 
!>          LDB est INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,N).
!
>

D 
!>          D est un tableau COMPLEX, dimension (N)
!>          À l'entrée, D est le membre de gauche de l'équation GLM.
!>          À la sortie, D est détruit.
!
>

X 
!>          X est un tableau COMPLEX, dimension (M)
!
>

Y 
!>          Y est un tableau COMPLEX, dimension (P)
!>
!>          À la sortie, X et Y sont les solutions du problème GLM.
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie la valeur optimale de LWORK.
!
>

LWOR

K 
!>          LWORK est INTEGER
!>          La dimension du tableau WORK. LWORK >= max(1,N+M+P).
!>          Pour des performances optimales, LWORK >= M+min(N,P)+max(N,P)*NB,
!>          où NB est une borne supérieure pour les tailles de blocs optimales de
!>          CGEQRF, CGERQF, CUNMQR et CUNMRQ.
!>
!>          Si LWORK = -1, alors une interrogation d'espace de travail est
!>          supposée ; la routine calcule uniquement la taille optimale du tableau
!>          WORK, renvoie cette valeur comme premier élément du tableau WORK, et
!>          aucun message d'erreur lié à LWORK n'est émis par XERBLA.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie.
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale.
!>          = 1 :  le facteur triangulaire supérieur R associé à A dans la
!>                 factorisation QR généralisée de la paire (A, B) est exactement
!>                 singulier, de sorte que rank(A) < M ; la solution des moindres
!>                 carrés n'a pas pu être calculée.
!>          = 2 :  la partie inférieure (N-M) par (N-M) du facteur trapézoïdal
!>                 supérieur T associé à B dans la factorisation QR généralisée
!>                 de la paire (A, B) est exactement singulier, de sorte que
!>                 rank( A B ) < N ; la solution des moindres carrés n'a pas pu
!>                 être calculée.
!
>
