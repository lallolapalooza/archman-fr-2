subroutine cggrqf (integer m, integer p, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) taua, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) taub, complex, dimension( * ) work, integer lwork, integer

CGGRQ

F

But

: 
!>
!> CGGRQF calcule une factorisation RQ généralisée d'une matrice A de
!> dimensions M par N et d'une matrice B de dimensions P par N :
!>
!>             A = R*Q,        B = Z*T*Q,
!>
!> où Q est une matrice unitaire N par N, Z est une matrice unitaire
!> P par P, et R et T prennent l'une des formes suivantes :
!>
!> si M <= N,  R = ( 0  R12 ) M,   ou si M > N,  R = ( R11 ) M-N,
!>                  N-M  M                           ( R21 ) N
!>                                                      N
!>
!> où R12 ou R21 est triangulaire supérieur, et
!>
!> si P >= N,  T = ( T11 ) N  ,   ou si P < N,  T = ( T11  T12 ) P,
!>                 (  0  ) P-N                         P   N-P
!>                    N
!>
!> où T11 est triangulaire supérieur.
!>
!> En particulier, si B est carrée et non singulière, la factorisation
!> GRQ de A et B donne implicitement la factorisation RQ de A*inv(B) :
!>
!>              A*inv(B) = (R*inv(T))*Z**H
!>
!> où inv(B) désigne l'inverse de la matrice B, et Z**H désigne la
!> transposée conjuguée de la matrice Z.
!
>

Paramètre

s
M 
!>          M est INTEGER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!
>

!>          P est INTEGER
!>          Le nombre de lignes de la matrice B.  P >= 0.
!
>

!>          N est INTEGER
!>          Le nombre de colonnes des matrices A et B. N >= 0.
!
>

!>          À la sortie, si M <= N, le triangle supérieur du sous-tableau
!>          A(1:M,N-M+1:N) contient la matrice triangulaire supérieure R de
!>          dimensions M par M ;
!>          si M > N, les éléments sur et au-dessus de la (M-N)-ème
!>          sous-diagonale contiennent la matrice trapézoïdale supérieure R
!>          de dimensions M par N ; les éléments restants, avec le tableau
!>          TAUA, représentent la matrice unitaire Q sous forme de produit
!>          de réflecteurs élémentaires (voir Détails supplémentaires).
!
>

LD

!

!>          dimensions min(P,N) par N (T est triangulaire supérieure si
!>          P >= N) ; les éléments sous la diagonale, avec le tableau TAUB,

!>  La matrice Q est représentée comme un produit de réflecteurs élémentaires
!>
!>     Q = H(1) H(2) . . . H(k), où k = min(m,n).
!>
!>  Chaque H(i) a la forme
!>
!>     H(i) = I - taua * v * v**H

!>  où taua est un scalaire complexe, et v est un vecteur complexe avec
!>  v(n-k+i+1:n) = 0 et v(n-k+i) = 1 ; v(1:n-k+i-1) est stocké à la sortie dans

!>  Pour former Q explicitement, utilisez la sous-routine LAPACK CUNGRQ.
!>  Pour utiliser Q afin de mettre à jour une autre matrice, utilisez la
!>  sous-routine LAPACK CUNMRQ.
!>
!>  La matrice Z est représentée comme un produit de réflecteurs élémentaires
!>
!>     Z = H(1) H(2) . . . H(k), où k = min(p,n).
!>
!>  Chaque H(i) a la forme
!>
!>     H(i) = I - taub * v * v**H
!>
!>  où taub est un scalaire complexe, et v est un vecteur complexe avec
!>  v(1:i-1) = 0 et v(i) = 1 ; v(i+1:p) est stocké à la sortie dans B(i+1:p,i),
!>  et taub dans TAUB(i).
!>  Pour former Z explicitement, utilisez la sous-routine LAPACK CUNGQR.
!>  Pour utiliser Z afin de mettre à jour une autre matrice, utilisez la
!>  sous-routine LAPACK CUNMQR.
!

     INTEGER M, N, K, LDA, LDB, LWORK, INFO
     COMPLEX A( LDA, * ), WORK( * )
     COMPLEX TAUA( * ), TAUB( * )

 SUBROUTINE ZGGRQ( M, N, K, A, LDA, TAUA, B, LDB, TAUB, WORK, LWORK, INFO )

     INTEGER M, N, K, LDA, LDB, LWORK, INFO
     DOUBLE COMPLEX A( LDA, * ), WORK( * )
     DOUBLE COMPLEX TAUA( * ), TAUB( * )








DESCRIPTION


La routine SGGRQ calcule la factorisation QR
    généralisée de deux matrices A et B:


        A = R * Q,    B = T * Q,



où Q est une matrice unitaire N x N, R est une matrice M x N
    supérieure, et T est une matrice K x N supérieure. Q est représentée comme
    un produit de réflecteurs élémentaires.


La routine ZGGRQF calcule la factorisation RQ
    généralisée des matrices A et B:


        A = R*Q,        B = Z*T*Q,



où Q est une matrice unitaire N x N, Z est une matrice unitaire P x P,
    R et T ont des formes spécifiques dépendant des dimensions.




subroutine SGGRQF (integer M, integer P, integer N, real, dimension( LDA, * ) A, integer LDA, real, dimension( * ) TAUA, real, dimension( LDB, * ) B, integer LDB, real, dimension( * ) TAUB, real, dimension( * ) WORK, integer LWORK, integer INFO)


SGGRQF

!> et d'une matrice P-par-N B:
!>
!>             A = R*Q,        B = Z*T*Q,

!>                  N-M  M                           ( R21 ) N
!>                                                      N
!>
