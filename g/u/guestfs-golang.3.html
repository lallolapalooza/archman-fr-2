NOM

guestfs-golang - Comment utiliser libguestfs depuis Go

SYNTAXE

import "libguestfs.org/guestfs"

g, errno := guestfs.Create ()
if errno != nil {
    panic (fmt.Sprintf ("impossible de créer le handle : %s", errno))
}
defer g.Close ()
if err := g.Add_drive ("test.img"); err != nil {
    panic (err)
}
if err := g.Launch (); err != nil {
    panic (err)
}
if err := g.Shutdown (); err != nil {
    panic (err)
}

DESCRIPTION

Cette page de manuel documente comment appeler libguestfs depuis le langage de programmation Go. Cette page documente uniquement les différences par rapport à l'API C et donne quelques exemples. Si vous n'êtes pas familier avec l'utilisation de libguestfs, vous devez également lire guestfs(3).

IMPORTATION DU MODULE

Le module s'appelle "guestfs". Le nom complet du package à importer est "libguestfs.org/guestfs".

CRÉATION ET FERMETURE DU HANDLE

Utilisez soit "guestfs.Create" soit "guestfs.Create_flags" pour créer le handle. Le handle est fermé implicitement s'il est collecté par le garbage collector. Cependant, il est probablement judicieux de le fermer explicitement, soit en appelant "g.Close ()", soit en différant le même appel.

ERREURS

"guestfs.Create" et "guestfs.Create_flags" retournent une simple valeur de type *error, qui est en réalité juste un "errno" C encapsulé dans la structure golang appropriée.

Tous les autres appels retournent une valeur de type *GuestfsError qui, si elle n'est pas nulle, est une structure plus riche contenant la chaîne d'erreur de libguestfs, l'errno (si disponible) et l'opération qui a échoué. Ceci peut également être converti en une chaîne de caractères pour l'affichage.

LIMITATIONS

  • Pas de support pour les événements (voir "EVENTS" dans guestfs(3)).
  • Les UUID ne sont pas retournés dans les structures.

EXEMPLE 1 : CRÉER UNE IMAGE DISQUE

/* Exemple montrant comment créer une image disque. */

package main

import (
       "fmt"
       "libguestfs.org/guestfs"
)

func main() {
       output := "disk.img"

       g, errno := guestfs.Create ()
       if errno != nil {
               panic (errno)
       }
       defer g.Close ()

       /* Crée une image disque sparse au format brut, d'une taille de 512 Mo. */
       if err := g.Disk_create (output, "raw", 512 * 1024 * 1024); err != nil {
               panic (err)
       }

       /* Définit le drapeau de traçage afin que nous puissions voir chaque appel à libguestfs. */
       g.Set_trace (true)

       /* Attache l'image disque à libguestfs. */
       optargs := guestfs.OptargsAdd_drive{
               Format_is_set: true,
               Format: "raw",
               Readonly_is_set: true,
               Readonly: false,
       }
       if err := g.Add_drive (output, &optargs); err != nil {
               panic (err)
       }

       /* Lance le backend libguestfs. */
       if err := g.Launch (); err != nil {
               panic (err)
       }

       /* Obtient la liste des périphériques. Comme nous n'avons ajouté qu'un seul disque
        * ci-dessus, nous nous attendons à ce que cette liste contienne un seul
        * élément.
        */
       devices, err := g.List_devices ()
       if err != nil {
               panic (err)
       }
       if len(devices) != 1 {
               panic ("périphérique unique attendu depuis list-devices")
       }

       /* Partionne le disque en une seule partition MBR. */
       err = g.Part_disk (devices[0], "mbr")
       if err != nil {
               panic (err)
       }

       /* Obtient la liste des partitions. Nous nous attendons à un seul élément, qui
        * est la partition que nous venons de créer.
        */
       partitions, err := g.List_partitions ()
       if err != nil {
               panic (err)
       }
       if len(partitions) != 1 {
               panic ("partition unique attendue depuis list-partitions")
       }

       /* Crée un système de fichiers sur la partition. */
       err = g.Mkfs ("ext4", partitions[0], nil)
       if err != nil {
               panic (err)
       }

       /* Monte maintenant le système de fichiers afin que nous puissions ajouter des fichiers. */
       err = g.Mount (partitions[0], "/")
       if err != nil {
               panic (err)
       }

       /* Crée quelques fichiers et répertoires. */
       err = g.Touch ("/empty")
       if err != nil {
               panic (err)
       }
       message := []byte("Bonjour, le monde\n")
       err = g.Write ("/hello", message)
       if err != nil {
               panic (err)
       }
       err = g.Mkdir ("/foo")
       if err != nil {
               panic (err)
       }

       /* Ceci téléverse le fichier local /etc/resolv.conf dans
        * l'image disque.
        */
       err = g.Upload ("/etc/resolv.conf", "/foo/resolv.conf")
       if err != nil {
               panic (err)
       }

       /* Parce que nous avons écrit sur le disque et que nous voulons détecter les erreurs d'écriture,
        * appelons g:shutdown. Vous n'avez pas besoin de faire cela :
        * g.Close le fera implicitement.
        */
       if err = g.Shutdown (); err != nil {
               panic (fmt.Sprintf ("écriture sur le disque échouée : %s", err))
       }
}

EXEMPLE 2 : INSPECTION D'UNE IMAGE DISQUE DE MACHINE VIRTUELLE

/* Exemple montrant comment inspecter un disque de machine virtuelle. */

package main

import (
       "fmt"
       "os"
       "libguestfs.org/guestfs"
)

func main() {
       if len(os.Args) < 2 {
               panic ("usage: inspect-vm disk.img")
       }
       disk := os.Args[1]

       g, errno := guestfs.Create ()
       if errno != nil {
               panic (fmt.Sprintf ("impossible de créer le handle : %s", errno))
       }

       /* Attache l'image disque en lecture seule à libguestfs. */
       optargs := guestfs.OptargsAdd_drive{
               Format_is_set: true,
               Format: "raw",
               Readonly_is_set: true,
               Readonly: true,
       }
       if err := g.Add_drive (disk, &optargs); err != nil {
               panic (err)
       }

       /* Lance le backend libguestfs. */
       if err := g.Launch (); err != nil {
               panic (err)
       }

       /* Demande à libguestfs d'inspecter les systèmes d'exploitation. */
       roots, err := g.Inspect_os ()
       if err != nil {
               panic (err)
       }
       if len(roots) == 0 {
               panic ("inspect-vm : aucun système d'exploitation trouvé")
       }

       for _, root := range roots {
               fmt.Printf ("Périphérique racine : %s\n", root)

               /* Affiche les informations de base sur le système d'exploitation. */
               s, _ := g.Inspect_get_product_name (root)
               fmt.Printf ("  Nom du produit : %s\n", s)
               major, _ := g.Inspect_get_major_version (root)
               minor, _ := g.Inspect_get_minor_version (root)
               fmt.Printf ("  Version :      %d.%d\n", major, minor)
               s, _ = g.Inspect_get_type (root)
               fmt.Printf ("  Type :         %s\n", s)
               s, _ = g.Inspect_get_distro (root)
               fmt.Printf ("  Distribution : %s\n", s)

               /* XXX Exemple incomplet. Trier les clés par longueur
                * est inutilement difficile en golang.
                */
       }
}
