DESCRIPTIO

N

Cette page manuelle documente comment appeler libguestfs depuis le langage de programmation Perl. Cette page documente uniquement les différences avec l'API C et donne quelques exemples. Si vous n'êtes pas familier avec l'utilisation de libguestfs, vous devez aussi lire guestfs(3). Pour lire l'API Perl complète, voir Sys::Guestf

s(3).

EXEMPLE 1 : CRÉER UNE IMAGE DE

DISQUE 
#!/usr/bin/env perl

# Exemple montrant comment créer une image de disque.

use strict;
use warnings;
use Sys::Guestfs;

my $output = "disk.img";

my $g = new Sys::Guestfs ();

# Créer une image de disque creuse au format brut, de 512 Mo.
$g->disk_create ($output, "raw", 512 * 1024 * 1024);

# Définir l'indicateur de trace afin de pouvoir voir chaque appel à libguestfs.
$g->set_trace (1);

# Attacher l'image de disque à libguestfs.
$g->add_drive_opts ($output, format => "raw", readonly => 0);

# Exécuter le moteur de libguestfs.
$g->launch ();

# Obtenir la liste des périphériques. Comme nous n'avons ajouté qu'un seul
# disque ci-dessus, nous nous attendons à ce que cette liste contienne un
# seul élément.
my @devices = $g->list_devices ();
if (@devices != 1) {
    die "error: expected a single device from list-devices";
}

# Partitionner le disque en une seule partition MBR.
$g->part_disk ($devices[0], "mbr");

# Obtenir la liste des partitions. Nous nous attendons à un seul élément,
# qui est la partition que nous venons de créer.
my @partitions = $g->list_partitions ();
if (@partitions != 1) {
    die "error: expected a single partition from list-partitions";
}

# Créer un système de fichiers sur la partition.
$g->mkfs ("ext4", $partitions[0]);

# Maintenant, monter le système de fichiers pour pouvoir y ajouter des fichiers.
$g->mount ($partitions[0], "/");

# Créer quelques fichiers et répertoires.
$g->touch ("/empty");
my $message = "Hello, world\n";
$g->write ("/hello", $message);
$g->mkdir ("/foo");

# Celui-ci téléverse le fichier local /etc/resolv.conf dans
# l'image de disque.
$g->upload ("/etc/resolv.conf", "/foo/resolv.conf");

# Parce que nous avons écrit sur le disque et que nous voulons détecter les
# erreurs d'écriture, appelez $g->shutdown. Vous n'êtes pas obligé de le
# faire : $g->close le fera implicitement.
$g->shutdown ();

# Notez également que les descripteurs sont automatiquement fermés s'ils
# sont libérés par comptage de références. Vous n'avez besoin d'appeler close
# que si vous souhaitez fermer le descripteur immédiatement.
$g->close
()
;

EXEMPLE 2 : INSPECTER UNE IMAGE DE DISQUE DE MACHINE

VIRTUELLE 
#!/usr/bin/env perl

# Exemple montrant comment inspecter un disque de machine virtuelle.

use strict;
use warnings;
use Sys::Guestfs;

if (@ARGV < 1) {
    die "usage: inspect_vm disk.img"
}

my $disk = $ARGV[0];

my $g = new Sys::Guestfs ();

# Attacher l'image de disque en lecture seule à libguestfs.
# Vous pouvez aussi ajouter ici un argument optionnel format => ... C'est
# recommandé car la détection automatique du format est peu sécurisée.
$g->add_drive_opts ($disk, readonly => 1);

# Exécuter le moteur de libguestfs.
$g->launch ();

# Demander à libguestfs d'inspecter les systèmes d'exploitation.
my @roots = $g->inspect_os ();
if (@roots == 0) {
    die "inspect_vm: no operating systems found";
}

for my $root (@roots) {
    printf "Root device: %s\n", $root;

    # Afficher les informations de base sur le système d'exploitation.
    printf "  Product name: %s\n", $g->inspect_get_product_name ($root);
    printf "  Version:      %d.%d\n",
        $g->inspect_get_major_version ($root),
        $g->inspect_get_minor_version ($root);
    printf "  Type:         %s\n", $g->inspect_get_type ($root);
    printf "  Distro:       %s\n", $g->inspect_get_distro ($root);

    # Monter les disques, comme guestfish -i.
    #
    # Trier les clés par longueur, la plus courte d'abord, afin de monter
    # les systèmes de fichiers dans le bon ordre.
    my %mps = $g->inspect_get_mountpoints ($root);
    my @mps = sort { length $a <=> length $b } (keys %mps);
    for my $mp (@mps) {
        eval { $g->mount_ro ($mps{$mp}, $mp) };
        if ($@) {
            print "$@ (ignored)\n"
        }
    }

    # Si le fichier /etc/issue.net existe, afficher jusqu'à 3 lignes.
    my $filename = "/etc/issue.net";
    if ($g->is_file ($filename)) {
        printf "--- %s ---\n", $filename;
        my @lines = $g->head_n (3, $filename);
        print "$_\n" foreach @lines;
    }

    # Démonter tout.
    $g->umount_all ()
}
