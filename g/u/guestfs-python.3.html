N

Cette page manuel documente comment appeler libguestfs depuis le langage de programmation Python. Cette page documente uniquement les différences par rapport à l'API C et donne quelques exemples. Si vous n'êtes pas familier avec l'utilisation de libguestfs, vous devez également lire guestfs

(3).

EXEMPLE 1 : CRÉER UNE IMAGE DE

DISQUE 
# Exemple montrant comment créer une image de disque.

import guestfs

output = "disk.img"

# Tout nouveau code Python doit passer python_return_dict=True
# au constructeur. Cela indique que votre programme souhaite
# recevoir des dictionnaires Python pour les méthodes de l'API qui retournent
# des tables de hachage.
g = guestfs.GuestFS(python_return_dict=True)

# Créer une image de disque creuse au format brut, de 512 Mo.
g.disk_create(output, "raw", 512 * 1024 * 1024)

# Activer l'indicateur de trace pour pouvoir voir chaque appel à libguestfs.
g.set_trace(1)

# Attacher l'image de disque à libguestfs.
g.add_drive_opts(output, format="raw", readonly=0)

# Exécuter le moteur de libgue
stfs. g.launch() # Obtenir la liste des périphériques. Comme nous n'avons ajouté qu'un seul # disque ci-dessus, nous nous attendons à ce que cette liste contienne un # seul élément. devices = g.list_devices() assert (len(devices) == 1) # Partitionner le disque en une seule partition MBR. g.part_disk(devices[0], "mbr") # Obtenir la liste des partitions. Nous nous attendons à un seul élément, qui # est la partition que nous venons de créer. partitions = g.list_partitions() assert (len(partitions) == 1) # Créer un système de fichiers sur la partition. g.mkfs("ext4", partitions[0]) # Maintenant monter le système de fichiers pour pouvoir ajouter des fichiers. g.mount(partitions[0], "/
") # Créer quelques fichiers et répertoires. g.touch("/empty") message = "Hello, world\n" g.write("/hello", message) g.mkdir("/foo") # Celui-ci téléverse le fichier local /etc/resolv.conf dans # l'image de disque. g.upload("/etc/resolv.conf", "/foo/resolv.conf") # Parce que nous avons écrit sur le disque et que nous voulons détecter les # erreurs d'écriture, appelez g.shutdown. Vous n'avez pas besoin de faire # cela : g.close le fera implicitement. g.shutdown() # Notez également que les descripteurs sont automatiquement fermés s'ils sont # libérés par le comptage de références. Vous n'avez besoin d'appeler close # que si vous souhaitez fermer le descripteur immédiatement. g.close()

EXEMPLE 2 : INSPECTER UNE IMAGE DE DISQUE DE MACHINE

VIRTUELLE 
# Exemple montrant comment inspecter un disque de machine virtuelle.

import sys
import guestfs

if len(sys.argv) < 2:
    print("inspect_vm: missing disk image to inspect", file=sys.stderr)
    sys.exit(1)
disk = sys.argv[1]

# Tout nouveau code Python doit passer python_return_dict=True
# au constructeur. Cela indique que votre programme souhaite
# recevoir des dictionnaires Python pour les méthodes de l'API qui retournent
# des tables de hachage.
g = guestfs.GuestFS(python_return_dict=True)

# Attacher l'image de disque en lecture seule à libguestfs.
g.add_drive_opts(disk, readonly=1)

# Exécuter le moteur de libguestfs.
g.launch()

# Demander à libguestf
s d'inspecter les systèmes d'exploitation. roots = g.inspect_os() if len(roots) == 0: print("inspect_vm: no operating systems found", file=sys.stderr) sys.exit(1) for root in roots: print("Root device: %s" % root) # Afficher les informations de base sur le système d'exploitation. print(" Product name: %s" % (g.inspect_get_product_name(root))) print(" Version: %d.%d" % (g.inspect_get_major_version(root), g.inspect_get_minor_version(root))) print(" Type: %s" % (g.inspect_get_type(root))) print(" Distro: %s" % (g.inspect_get_distro(root))) # Monter les disques, comme guestfish -
i. # # Trier les clés par longueur, la plus courte d'abord, afin de monter # les systèmes de fichiers dans le bon ordre. mps = g.inspect_get_mountpoints(root) for device, mp in sorted(mps.items(), key=lambda k: len(k[0])): try: g.mount_ro(mp, device) except RuntimeError as msg: print("%s (ignored)" % msg) # Si le fichier /etc/issue.net existe, afficher jusqu'à 3 lignes. filename = "/etc/issue.net" if g.is_file(filename): print("--- %s ---" % filename) lines = g.head_n(3, filename) for line in lines: print(line) # Tout démonter. g.umount_all()
