NOM

guestfs-ocaml - Comment utiliser libguestfs depuis OCaml

SYNOPSIS

Style module :

let g = Guestfs.create () in
Guestfs.add_drive_opts g ~format:"raw" ~readonly:true "disk.img";
Guestfs.launch g;

Style orienté objet :

 let g = new Guestfs.guestfs () in
 g#add_drive_opts ~format:"raw" ~readonly:true "disk.img";
 g#launch ();
 ocamlfind opt prog.ml -package guestfs -linkpkg -o prog
ou :
 ocamlopt -I +guestfs mlguestfs.cmxa prog.ml -o prog

DESCRIPTION

Cette page manuel documente comment appeler libguestfs depuis le langage de programmation OCaml. Cette page documente uniquement les différences avec l'API C et donne quelques exemples. Si vous n'êtes pas familier avec l'utilisation de libguestfs, vous devez aussi lire guestfs(3).

STYLES DE PROGRAMMATION

Les liaisons OCaml prennent en charge deux styles de programmation différents. Vous pouvez utiliser un style module, avec chaque fonction C mappée à une fonction OCaml :

int guestfs_set_verbose (guestfs_h *g, int flag);

devient :

val Guestfs.set_verbose : Guestfs.t -> bool -> unit

Alternativement, vous pouvez utiliser un style orienté objet, en appelant des méthodes sur la classe "Guestfs.guestfs":

method set_verbose : bool -> unit

Le style orienté objet est généralement plus concis, et la légère pénalité de performance n'est pas perceptible dans la surcharge générale de l'exécution des fonctions libguestfs.

FERMETURE DU POIGNÉE

Le poignée est fermé lorsqu'il est récupéré par le ramasse-miettes. Comme les poignées libguestfs incluent beaucoup d'état, il est aussi possible de les fermer (et donc de les libérer) explicitement en appelant "Guestfs.close" ou la méthode "#close" .

EXCEPTIONS

Les erreurs des fonctions libguestfs sont mappées dans l'exception "Guestfs.Error" . Celle-ci a un paramètre unique qui est le message d'erreur (une chaîne).

Appeler n'importe quelle fonction/méthode sur un poignée fermé lève "Guestfs.Handle_closed". Le paramètre unique est le nom de la fonction que vous avez appelée.

EXEMPLE : CRÉER UNE IMAGE DE DISQUE

(* Exemple montrant comment créer une image de disque. *)

open Unix
open Printf

let output = "disk.img"

let () =
  let g = new Guestfs.guestfs () in

  (* Créer une image de disque creuse au format brut, de 512 Mo. *)
  g#disk_create output "raw" (Int64.of_int (512 * 1024 * 1024));

  (* Définir l'indicateur de trace pour que nous puissions voir chaque appel libguestfs. *)
  g#set_trace true;

  (* Attacher l'image de disque à libguestfs. *)
  g#add_drive_opts ~format:"raw" ~readonly:false output;

  (* Exécuter le back-end libguestfs. *)
  g#launch ();

  (* Obtenir la liste des périphériques. Comme nous n'avons ajouté qu'un seul
   * disque ci-dessus, nous nous attendons à ce que cette liste contienne un
   * seul élément.
   *)
  let devices = g#list_devices () in
  if Array.length devices <> 1 then
    failwith "error: expected a single device from list-devices";

  (* Partitionner le disque en une seule partition MBR. *)
  g#part_disk devices.(0) "mbr";

  (* Obtenir la liste des partitions. Nous nous attendons à un seul élément, qui
   * est la partition que nous venons de créer.
   *)
  let partitions = g#list_partitions () in
  if Array.length partitions <> 1 then
    failwith "error: expected a single partition from list-partitions";

  (* Créer un système de fichiers sur la partition. *)
  g#mkfs "ext4" partitions.(0);

  (* Maintenant monter le système de fichiers pour que nous puissions ajouter des fichiers. *)
  g#mount partitions.(0) "/";

  (* Créer quelques fichiers et répertoires. *)
  g#touch "/empty";
  let message = "Hello, world\n" in
  g#write "/hello" message;
  g#mkdir "/foo";

  (* Celui-ci téléverse le fichier local /etc/resolv.conf dans
   * l'image de disque.
   *)
  g#upload "/etc/resolv.conf" "/foo/resolv.conf";

  (* Comme nous avons écrit sur le disque et que nous voulons détecter les
   * erreurs d'écriture, appelez g#shutdown. Vous n'avez pas besoin de faire
   * cela : g#close le fera implicitement.
   *)
  g#shutdown ();

  (* Notez aussi que les poignées sont automatiquement fermés s'ils sont
   * récupérés par le ramasse-miettes. Vous n'avez besoin d'appeler close
   * que si vous voulez fermer le poignée immédiatement.
   *)
  g#close ()

EXEMPLE : INSPECTER UNE IMAGE DE DISQUE DE MACHINE VIRTUELLE

(* Exemple montrant comment inspecter un disque de machine virtuelle. *)

open Printf

let disk =
  if Array.length Sys.argv = 2 then
    Sys.argv.(1)
  else
    failwith "usage: inspect_vm disk.img"

let () =
  let g = new Guestfs.guestfs () in

  (* Attacher l'image de disque en lecture seule à libguestfs. *)
  g#add_drive_opts (*~format:"raw"*) ~readonly:true disk;

  (* Exécuter le back-end libguestfs. *)
  g#launch ();

  (* Demander à libguestfs d'inspecter les systèmes d'exploitation. *)
  let roots = g#inspect_os () in
  if Array.length roots = 0 then
    failwith "inspect_vm: no operating systems found";

  Array.iter (
    fun root ->
      printf "Root device: %s\n" root;

      (* Afficher des informations de base sur le système d'exploitation. *)
      printf "  Product name: %s\n" (g#inspect_get_product_name root);
      printf "  Version:      %d.%d\n"
        (g#inspect_get_major_version root)
        (g#inspect_get_minor_version root);
      printf "  Type:         %s\n" (g#inspect_get_type root);
      printf "  Distro:       %s\n" (g#inspect_get_distro root);

      (* Monter les disques, comme guestfish -i.
       *
       * Trier les clés par longueur, la plus courte d'abord, afin que nous
       * montions les systèmes de fichiers dans le bon ordre.
       *)
      let mps = g#inspect_get_mountpoints root in
      let cmp (a,_) (b,_) =
        compare (String.length a) (String.length b) in
      let mps = List.sort cmp mps in
      List.iter (
        fun (mp, dev) ->
          try g#mount_ro dev mp
          with Guestfs.Error msg -> eprintf "%s (ignored)\n" msg
      ) mps;

      (* Si le fichier /etc/issue.net existe, afficher jusqu'à 3 lignes. *)
      let filename = "/etc/issue.net" in
      if g#is_file filename then (
        printf "--- %s ---\n" filename;
        let lines = g#head_n 3 filename in
        Array.iter print_endline lines
      );

      (* Tout démonter. *)
      g#umount_all ()
  ) roots

EXEMPLE : ACTIVER LE DÉBOGAGE ET LA JOURNALISATION

(* Exemple montrant comment activer le débogage et le capturer dans n'importe
 * quel système de journalisation personnalisé.
 *)

(* Les événements qui nous intéressent. Ce masque de bits couvre tous les
 * messages de trace et de débogage.
 *)
let event_bitmask = [
  Guestfs.EVENT_LIBRARY;
  Guestfs.EVENT_WARNING;
  Guestfs.EVENT_APPLIANCE;
  Guestfs.EVENT_TRACE
]

let rec main () =
  let g = new Guestfs.guestfs () in

  (* Par défaut, les informations de débogage sont imprimées sur stderr. Pour
   * les capturer ailleurs, vous devez configurer un gestionnaire d'événements
   * qui sera rappelé à mesure que les messages de débogage sont générés. Pour
   * cela, utilisez l'API d'événements.
   *
   * Pour plus d'informations, voir EVENTS dans guestfs(3).
   *)
  ignore (g#set_event_callback message_callback event_bitmask);

  (* Voici comment le débogage est activé :
   *
   * Définir l'indicateur 'trace' dans le poignée signifie que chaque appel
   * libguestfs est journalisé (nom, paramètres, retour). Cet indicateur est
   * utile pour voir comment libguestfs est utilisé par un programme.
   *
   * Définir l'indicateur 'verbose' active une grande quantité de débogage
   * supplémentaire dans tout le système. C'est utile s'il y a une erreur
   * libguestfs que vous ne comprenez pas.
   *
   * Notez que vous devriez définir les indicateurs tôt après la création du
   * poignée. En particulier, si vous définissez l'indicateur verbose après
   * le lancement, vous ne verrez pas tous les messages.
   *
   * Pour plus d'informations, voir :
   * http://libguestfs.org/guestfs-faq.1.html#debugging-libguestfs
   *
   * Les messages d'erreur levés par les API ne sont *pas* des informations de
   * débogage, et ils ne sont pas affectés par tout cela. Vous devrez peut-être
   * les journaliser séparément.
   *)
  g#set_trace true;
  g#set_verbose true;

  (* Effectuer quelques opérations qui généreront de nombreux messages de
   * trace et de débogage.
   *)
  g#add_drive "/dev/null";
  g#launch ();
  g#close ()

(*
 * This example simply redirects trace messages to stdout.
 *)

open Guestfs

(*
 * Callback function for guestfs events.
 *)
let message_callback event event_handle buf array =
  (*
   * When a specific event like `Trace` or `Debug` is received, the `array`
   * and `array_len` parameters passed to this callback are not meaningful.
   * Only `buf` and `buf_len` will contain the trace or debug message.
   *
   * In this example, we redirect these messages to stdout, but you could
   * obviously do something more advanced.
   *)
  if String.length buf > 0 then (
    let event_name = Guestfs.event_to_string [event] in
    Printf.printf "[%s] %S\n%!" event_name buf
  )

(*
 * The main function.
 *)
let main () =
  (* Create a new libguestfs handle. *)
  let g = Guestfs.create () in

  (* Register the message callback. *)
  Guestfs.set_event_callback g message_callback;

  (* Enable trace and debug messages. *)
  Guestfs.set_log_level g `trace;
  Guestfs.set_log_channel g `stdio;

  (*
   * In a real program, you would call other guestfs functions here.
   * For example, to inspect a disk image:
   *
   * let () =
   *   match Sys.argv with
   *   | [|_; disk_image_filename|] ->
   *       let () = Guestfs.add_drive g disk_image_filename in
   *       let () = Guestfs.launch g in
   *       let root_device =
   *         match Guestfs.inspect g disk_image_filename with
   *         | Some device -> device
   *         | None -> failwith ("Could not inspect " ^ disk_image_filename)
   *       in
   *       Printf.printf "Root device: %s\n" root_device
   *   | _ ->
   *       Printf.eprintf "Usage: %s disk_image_filename\n" Sys.argv.(0);
   *       exit 1
   * in
   *)

  (* For this example, we just launch libguestfs without doing anything
   * else, so that the trace messages are printed. *)
  let () = Guestfs.launch g in

  (* Destroy the handle. *)
  Guestfs.destroy g
;;

(* Call the main function. *)
let () = main ()

Ceci est un exemple sur la manière de gérer les messages de débogage et de journalisation de libguestfs.

 *

 *
 * Ce programme est un logiciel libre ; vous pouvez le redistribuer et/ou le modifier
 * sous les termes de la Licence Publique Générale GNU telle que publiée par la
 * Free Software Foundation ; soit la version 2 de la Licence, soit (à votre

 *
 * Ce programme est distribué dans l'espoir qu'il sera utile, mais
 * SANS AUCUNE GARANTIE ; sans même la garantie implicite de
 * QUALITÉ MARCHANDE ou d'ADÉQUATION À UN USAGE PARTICULIER. Voir la Licence Publique Générale GNU pour plus de détails.
 *
 * Vous devriez avoir reçu une copie de la Licence Publique Générale GNU avec ce programme ;
 * sinon, écrivez à la Free Software Foundation, Inc., 51 Franklin Street, Fifth Floor,
 * Boston, MA  02110-1301 USA
 *)

(*
 * Cet exemple redirige simplement les messages de trace vers la sortie standard.
 *)

open Guestfs

(*
 * Fonction de rappel pour les événements guestfs.
 *)
let message_callback event event_handle buf array =
  (*
   * Lorsqu'un événement spécifique comme `Trace` ou `Debug` est reçu, les paramètres `array`
   * et `array_len` passés à cette fonction de rappel ne sont pas significatifs.
   * Seuls `buf` et `buf_len` contiendront le message de trace ou de débogage.
   *
   * Dans cet exemple, nous redirigeons ces messages vers la sortie standard, mais vous pourriez
   * évidemment faire quelque chose de plus avancé.
   *)
  if String.length buf > 0 then (
    let event_name = Guestfs.event_to_string [event] in
    Printf.printf "[%s] %S\n%!" event_name buf
  )

(*
 * La fonction principale.
 *)
let main () =
  (* Crée un nouveau descripteur libguestfs. *)
  let g = Guestfs.create () in

  (* Enregistre la fonction de rappel des messages. *)
  Guestfs.set_event_callback g message_callback;

  (* Active les messages de trace et de débogage. *)
  Guestfs.set_log_level g `trace;
  Guestfs.set_log_channel g `stdio;

  (*
   * Dans un vrai programme, vous appelleriez d'autres fonctions guestfs ici.
   * Par exemple, pour inspecter une image disque :
   *
   * let () =
   *   match Sys.argv with
   *   | [|_; disk_image_filename|] ->
   *       let () = Guestfs.add_drive g disk_image_filename in
   *       let () = Guestfs.launch g in
   *       let root_device =
   *         match Guestfs.inspect g disk_image_filename with
   *         | Some device -> device
   *         | None -> failwith ("Impossible d'inspecter " ^ disk_image_filename)
   *       in
   *       Printf.printf "Périphérique racine : %s\n" root_device
   *   | _ ->
   *       Printf.eprintf "Usage : %s nom_fichier_image_disque\n" Sys.argv.(0);
   *       exit 1
   * in
   *)

  (* Pour cet exemple, nous lançons simplement libguestfs sans rien faire d'autre,
   * afin que les messages de trace soient affichés. *)
  let () = Guestfs.launch g in

  (* Détruit le descripteur. *)
  Guestfs.destroy g
;;

(* Appelle la fonction principale. *)
let () = main ()
