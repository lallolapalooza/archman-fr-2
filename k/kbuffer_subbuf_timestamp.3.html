Paquet: libtraceevent

LIBTRACEEVENT(3) Manuel libtraceevent LIBTRACEEVENT(3)

NOM

kbuffer_timestamp, kbuffer_subbuf_timestamp - Fonctions pour lire diverses données d'un descripteur kbuffer

SYNTAXE

#include <kbuffer.h>
unsigned long long kbuffer_timestamp(struct kbuffer *kbuf);
unsigned long long kbuffer_subbuf_timestamp(struct kbuffer *kbuf, void *subbuf);

DESCRIPTION

La fonction kbuffer_timestamp() renvoie l'horodatage de l'événement courant de kbuf.

La fonction kbuffer_subbuf_timestamp() renvoie l'horodatage pour le sous-buffer qui a été chargé dans kbuf. Il s'agit généralement (mais sans garantie) de l'horodatage du premier événement du sous-buffer.

La fonction kbuffer_start_of_data() renvoie l'offset où la delta

VALEUR DE RETOUR

kbuffer_read_event() renvoie l'événement auquel le descripteur kbuf est actuellement positionné, ou NULL si le dernier événement a été dépassé (par kbuffer_next_event()).

kbuffer_next_event() renvoie l'événement suivant après l'événement courant, ou NULL s'il n'y a plus d'événements.

kbuffer_read_at_offset() renvoie l'événement à un offset donné à partir du début du sous-buffer stocké dans kbuf, ou NULL s'il n'existe pas d'événement. Notez que l'offset doit seulement être un offset qui tombe sur l'enregistrement, ou est à son début. Il n'a pas besoin d'être exactement au début de l'enregistrement.

kbuffer_missed_events() renvoie 0 s'il n'y a pas eu d'événements manqués avant le sous-buffer chargé. Renvoie -1 s'il y a eu un nombre inconnu d'événements manqués, ou si le nombre d'événements manqués est connu, ce nombre sera renvoyé.

kbuffer_event_size() renvoie la taille de la charge utile de données de l'événement courant de kbuf.

kbuffer_curr_size() renvoie la taille de l'enregistrement entier de l'événement courant de kbuf. Ceci inclut la taille des métadonnées de cet enregistrement.

kbuf_curr_offset() renvoie l'offset de l'enregistrement courant depuis le début du sous-buffer kbuf.

kbuf_curr_index() renvoie l'index de l'enregistrement courant depuis le début de la section de données kbuf.

EXEMPLE

#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/stat.h>
#include <kbuffer.h>
int main (int argc, char **argv)
{
        unsigned long long ts;
        struct kbuffer *kbuf;
        struct stat st;
        char *buf;
        void *event;
        int save_offset = -1;
        int record_size;
        int offset;
        int index;
        int size;
        int ret;
        int fd;
        int i = 0;
        if (argc < 2) {
                printf("usage: %s page-brute-de-sous-buffer\n", argv[0]);
                printf(" Essayez : dd count=1 bs=4096 if=/sys/kernel/tracing/per_cpu/cpu0/trace_pipe_raw of=/tmp/file\n");
                exit(0);
        }
        if (stat(argv[1], &st) < 0) {
                perror("stat");
                exit(-1);
        }
        buf = malloc(st.st_size);
        if (!buf) {
                perror("Allocation du buffer");
                exit(-1);
        }
        fd = open(argv[1], O_RDONLY);
        if (fd < 0) {
                perror(argv[1]);
                exit(-1);
        }
        ret = read(fd, buf, st.st_size);
        if (ret < 0) {
                perror("Lecture du buffer");
                exit(-1);
        }
        close(fd);
        kbuf = kbuffer_alloc(KBUFFER_ENDIAN_SAME_AS_HOST,
                             KBUFFER_LSIZE_SAME_AS_HOST);
        if (!kbuf) {
                perror("Création de kbuffer");
                exit(-1);
        }
        ret = kbuffer_load_subbuffer(kbuf, buf);
        if (ret < 0) {
                perror("Chargement du sous-buffer");
                exit(-1);
        }
        if (kbuffer_subbuffer_size(kbuf) > st.st_size) {
                fprintf(stderr, "kbuffer est plus grand que la taille brute %d > %ld\n",
                        kbuffer_subbuffer_size(kbuf), st.st_size);
                exit(-1);
        }
        ret = kbuffer_missed_events(kbuf);
        if (ret) {
                if (ret > 0)
                        printf("Manqué %d événements avant ce buffer\n", ret);
                else
                        printf("Manqué un nombre inconnu d'événements avant ce buffer\n");
        }
        do {
                event = kbuffer_read_event(kbuf, &ts);
                if (event) {
                        record_size = kbuffer_curr_size(kbuf);
                        offset = kbuffer_curr_offset(kbuf);
                        index = kbuffer_curr_index(kbuf);
                        size = kbuffer_event_size(kbuf);
                        if (i == 20)
                                save_offset = offset;
                        printf(" événement %3d ts:%lld\ttaille_enregistrement:%d taille:%d\tindex:%d offset:%d\n",
                               i++, ts, record_size, size, index, offset);
                        event = kbuffer_next_event(kbuf, NULL);
                }
        } while (event);
        if (!event)
                printf("Fin du sous-buffer\n");
        if (save_offset > 0) {
                event = kbuffer_read_at_offset(kbuf, save_offset, &ts);
                if (!event) {
                        fprintf(stderr, "Étrange, impossible de trouver l'événement 20 à l'offset %d\n", save_offset);
                        exit(-1);
                }
                record_size = kbuffer_curr_size(kbuf);
                offset = kbuffer_curr_offset(kbuf);
                index = kbuffer_curr_index(kbuf);
                size = kbuffer_event_size(kbuf);
                printf("\n événement 20 sauvegardé ts:%lld\ttaille_enregistrement:%d taille:%d\tindex:%d offset:%d\n\n",
                       ts, record_size, size, index, offset);
        }
        kbuffer_free(kbuf);
        return 0;
}
