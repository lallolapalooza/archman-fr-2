ts_read(3)

Paquet: tslib

Licences: LGPL-2.1-only

TS_READ(3) tslib TS_READ(3)

NOM

ts_read, ts_read_raw, ts_read_mt, ts_read_raw_mt - lire des échantillons tactiles tslib

SYNTAXE

#include <tslib.h>

int ts_read(struct tsdev *dev, struct ts_sample *samp, int nr);

int ts_read_raw(struct tsdev *dev, struct ts_sample *samp, int nr);

int ts_read_mt(struct tsdev *dev, struct ts_sample_mt **samp, int slots, int nr);

int ts_read_raw_mt(struct tsdev *dev, struct ts_sample_mt **samp, int slots, int nr);

DESCRIPTION

ts_read() lit nr échantillons d'entrée avec les filtres de tslib appliqués. struct ts_sample est définie comme suit :

struct ts_sample {
        int             x;
        int             y;
        unsigned int    pressure;
        struct timeval  tv;
};

ts_read_mt() lit nr * slots échantillons d'entrée avec les filtres de tslib appliqués. struct ts_sample_mt est définie comme suit :

struct ts_sample_mt {
        /* codes d'événements ABS_MT_* les plus récents.
         * voir linux/input.h pour les descriptions */
        int             x;
        int             y;
        unsigned int    pressure;
        int             slot;
        int             tracking_id;
        int             tool_type;
        int             tool_x;
        int             tool_y;
        unsigned int    touch_major;
        unsigned int    width_major;
        unsigned int    touch_minor;
        unsigned int    width_minor;
        int             orientation;
        int             distance;
        int             blob_id;
        struct timeval  tv;
        /* état BTN_TOUCH */
        short           pen_down;
        /* le bit TSLIB_MT_VALID est défini dans valid si cet échantillon
         * contient de nouvelles données ;
	 * valid est mis à 0 sinon */
        short           valid;
};

L'utilisateur doit fournir la quantité de mémoire décrite dans nr et slots pour les contenir.

ts_read_raw() et ts_read_raw_mt() font la même chose sans les filtres de tslib appliqués.

VALEUR DE RETOUR

Le nombre d'échantillons effectivement lus est retourné. En particulier en mode non bloquant, voir ts_setup(), cela peut être inférieur à ce qui a été demandé lors de l'appel. En cas d'échec, un numéro d'erreur négatif est retourné.

EXEMPLE

Le programme suivant lit en continu les échantillons d'entrée multitouch de tslib et imprime les valeurs de slot et de position sur la sortie standard lorsque l'écran tactile est touché.

#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <tslib.h>
#define READ_SAMPLES 1
#define MAX_SLOTS 5
int main(int argc, char **argv)
{
        struct tsdev *ts;
        struct ts_sample_mt **samp_mt = NULL;
        int i, j;
        int ret;
        ts = ts_setup(NULL, 0);
        if (!ts)
                return -1;
        samp_mt = malloc(READ_SAMPLES * sizeof(struct ts_sample_mt **));
        if (!samp_mt)
                return -1;
        for (i = 0; i < READ_SAMPLES; i++) {
                samp_mt[i] = calloc(MAX_SLOTS, sizeof(struct ts_sample_mt));
                if (!samp_mt[i])
                        return -1;
        }
        while(1) {
                ret = ts_read_mt(ts, samp_mt, MAX_SLOTS, READ_SAMPLES);
                for (i = 0; i < ret; i++) {
                        printf("échantillon nr %d0, i);
                        for (j = 0; i < MAX_SLOTS; j++) {
                                if (!(samp_mt[i][j].valid & TSLIB_MT_VALID))
                                        continue;
                                printf("slot %d: X:%d Y: %d0,
                                       samp_mt[i][j].slot,
                                       samp_mt[i][j].x,
                                       samp_mt[i][j].y);
                        }
                }
        }
}
