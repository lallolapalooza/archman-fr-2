NOM

tep_filter_alloc, tep_filter_free, tep_filter_reset, tep_filter_make_string, tep_filter_copy, tep_filter_compare, tep_filter_match, tep_event_filtered, tep_filter_remove_event, tep_filter_strerror, tep_filter_add_filter_str - API liées aux filtres d'événements.

SYNOPIS

#include <event-parse.h>
struct tep_event_filter *tep_filter_alloc(struct tep_handle *tep);
void tep_filter_free(struct tep_event_filter *filter);
void tep_filter_reset(struct tep_event_filter *filter);
enum tep_errno tep_filter_add_filter_str(struct tep_event_filter *filter, const char *filter_str);
int tep_event_filtered(struct tep_event_filter *filter, int event_id);
int tep_filter_remove_event(struct tep_event_filter *filter, int event_id);
enum tep_errno tep_filter_match(struct tep_event_filter *filter, struct tep_record *record);
int tep_filter_copy(struct tep_event_filter *dest, struct tep_event_filter *source);
int tep_filter_compare(struct tep_event_filter *filter1, struct tep_event_filter *filter2);
char *tep_filter_make_string(struct tep_event_filter *filter, int event_id);
int tep_filter_strerror(struct tep_event_filter *filter, enum tep_errno err, char *buf, size_t buflen);

DESCRIPTION

Des filtres peuvent être attachés aux événements tracés. Ils peuvent être utilisés pour filtrer divers événements lors de leur sortie. Chaque événement peut être filtré en fonction de ses paramètres, décrits dans le fichier de format de l'événement. Cet ensemble de fonctions peut être utilisé pour créer, supprimer, modifier et attacher des filtres d'événements.

La fonction tep_filter_alloc() crée un nouveau filtre d'événements. L'argument tep est le contexte de l'analyseur d'événements de trace.

La fonction tep_filter_free() libère un filtre d'événements et toutes les ressources qu'il avait utilisées.

La fonction tep_filter_reset() supprime toutes les règles d'un filtre d'événements et le réinitialise.

La fonction tep_filter_add_filter_str() ajoute une nouvelle règle au filter. L'argument filter_str est la chaîne de filtre, qui contient la règle.

La fonction tep_event_filtered() vérifie si l'événement avec event_id a un filter.

La fonction tep_filter_remove_event() supprime un filter pour un événement avec event_id.

La fonction tep_filter_match() teste si un record correspond au filter donné.

La fonction tep_filter_copy() copie un filtre source dans un filtre dest.

La fonction tep_filter_compare() compare deux filtres - filter1 et filter2.

La fonction tep_filter_make_string() construit une chaîne, affichant le contenu du filter pour l'event_id donné.

La fonction tep_filter_strerror() copie le tampon d'erreur du filter dans le buf donné avec la taille buflen. Si le tampon d'erreur est vide, une chaîne décrivant l'erreur err est copiée dans buf.

VALEUR RETOURNÉE

La fonction tep_filter_alloc() retourne un pointeur vers le filtre d'événements nouvellement créé, ou NULL en cas d'erreur.

La fonction tep_filter_add_filter_str() retourne 0 si la règle a été ajoutée avec succès ou un code d'erreur négatif. Utilisez tep_filter_strerror() pour voir le message d'erreur réel en cas d'erreur.

La fonction tep_event_filtered() retourne 1 si le filtre est trouvé pour l'événement donné, ou 0 sinon.

La fonction tep_filter_remove_event() retourne 1 si l'événement a été supprimé, ou 0 si l'événement n'a pas été trouvé.

La fonction tep_filter_match() retourne tep_errno, selon le résultat :

TEP_ERRNO__FILTER_MATCH        - filtre trouvé pour l'événement, l'enregistrement correspond.
TEP_ERRNO__FILTER_MISS         - filtre trouvé pour l'événement, l'enregistrement ne correspond pas.
TEP_ERRNO__FILTER_NOT_FOUND    - aucun filtre trouvé pour l'événement de l'enregistrement.
TEP_ERRNO__NO_FILTER           - aucune règle dans le filtre.

ou tout autre tep_errno, si une erreur s'est produite pendant le test.

La fonction tep_filter_copy() retourne 0 en cas de succès ou -1 si toutes les règles n'ont pas été copiées.

La fonction tep_filter_compare() retourne 1 si les deux filtres contiennent le même contenu, ou 0 s'ils sont différents.

La fonction tep_filter_make_string() retourne une chaîne, qui doit être libérée avec free(), ou NULL en cas d'erreur.

La fonction tep_filter_strerror() retourne 0 si le message a été rempli avec succès, ou -1 en cas d'erreur.

void process_record(struct tep_handle *tep, struct tep_record *record)
{
        struct tep_event *event;
        char *fstring;
        event = tep_find_event_by_record(tep, record);
        if (tep_event_filtered(filter, event->id) == 1) {
                /* L'événement a un filtre */
                fstring = tep_filter_make_string(filter, event->id);
                if (fstring != NULL) {
                        /* Le filtre pour l'événement est dans fstring */
                        free(fstring);
                }
        }
        switch (tep_filter_match(filter, record)) {
        case TEP_ERRNO__FILTER_MATCH:
                /* Le filtre correspond à l'enregistrement */
                break;
        case TEP_ERRNO__FILTER_MISS:
                /* Le filtre ne correspond pas à l'enregistrement */
                break;
        case TEP_ERRNO__FILTER_NOT_FOUND:
                /* Aucun filtre trouvé pour l'événement de l'enregistrement */
                break;
        case TEP_ERRNO__NO_FILTER:
                /* Il n'y a pas de règles dans le filtre */
                break
        default:
                /* Une erreur s'est produite pendant le test */
                break;
        }
        if (tep_filter_remove_event(filter, event->id) == 1) {
                /* L'événement a été supprimé du filtre */
        }
}

tep_filter_compare(3)

NAME

tep_filter_compare - Compare two TEP filters.

SYNOPSIS

#include <trace-event-parse.h>

int tep_filter_compare(
    const struct tep_filter *filter1,
    const struct tep_filter *filter2
);

DESCRIPTION

The tep_filter_compare() function compares two TEP filters, filter1 and filter2.

It returns 0 if the filters are equal, a negative value if filter1 is less than filter2, and a positive value if filter1 is greater than filter2.

The comparison is done lexicographically based on the filter's expression string.

RETURN VALUE

Returns 0 if the filters are equal, a negative value if filter1 is less than filter2, and a positive value if filter1 is greater than filter2.

EXAMPLE

#include <stdio.h>
#include <stdlib.h>
#include <trace-event-parse.h>

int main() {
    struct tep_handle *handle;
    struct tep_filter *filter1;
    struct tep_filter *filter2;
    int ret;

    handle = tep_handle_create(0);
    if (handle == NULL) {
        perror("tep_handle_create");
        return 1;
    }

    filter1 = tep_filter_alloc(handle, "sched_switch:comm=*");
    filter2 = tep_filter_alloc(handle, "sched_switch:comm=bash");

    if (filter1 == NULL || filter2 == NULL) {
        perror("tep_filter_alloc");
        tep_handle_destroy(handle);
        return 1;
    }

    ret = tep_filter_compare(filter1, filter2);

    if (ret < 0) {
        printf("filter1 is less than filter2\n");
    } else if (ret == 0) {
        printf("filter1 is equal to filter2\n");
    } else {
        printf("filter1 is greater than filter2\n");
    }

    tep_filter_free(filter1);
    tep_filter_free(filter2);
    tep_handle_destroy(handle);

    return 0;
}
