NOM

tracefs_iterate_raw_events, tracefs_iterate_stop, tracefs_follow_event, tracefs_follow_missed_events, tracefs_follow_event_clear, tracefs_follow_missed_events_clear, tracefs_iterate_snapshot_events - Itérer sur les événements dans le tampon circulaire

SYNTAXE

#include <tracefs.h>
int tracefs_iterate_raw_events(struct tep_handle *tep, struct tracefs_instance *instance,
                                 cpu_set_t *cpus, int cpu_size,
                                 int (*callback)(struct tep_event *, struct tep_record *, int, void *),
                                 void *callback_context);
void tracefs_iterate_stop(struct tracefs_instance *instance);
int tracefs_follow_event(struct tep_handle *tep, struct tracefs_instance *instance,
                          const char *system, const char *event_name,
                          int (*callback)(struct tep_event *,
                                          struct tep_record *,
                                          int, void *),
                          void *callback_data);
int tracefs_follow_missed_events(struct tracefs_instance *instance,
                          int (*callback)(struct tep_event *,
                                          struct tep_record *,
                                          int, void *),
                          void *callback_data);
int tracefs_follow_event_clear(struct tracefs_instance *instance,
                          const char *system, const char *event_name);
int tracefs_follow_missed_events_clear(struct tracefs_instance *instance);
int tracefs_iterate_snapshot_events(struct tep_handle *tep, struct tracefs_instance *instance,
                                 cpu_set_t *cpus, int cpu_size,
                                 int (*callback)(struct tep_event *, struct tep_record *, int, void *),
                                 void *callback_context);

DESCRIPTION

Itérateur de trace sur les événements bruts.

La fonction tracefs_iterate_raw_events() lit les tampons de données brutes de tracefs et appelle la fonction callback spécifiée pour chaque événement rencontré. Les événements sont itérés dans l'ordre trié : le plus ancien en premier. Un gestionnaire tep initialisé est requis (Voir tracefs_local_events(3)). Si instance est NULL, alors le tampon tracefs de plus haut niveau est utilisé, sinon le tampon correspondant à l'instance est lu. Pour filtrer uniquement sur un sous-ensemble de CPU, cpus et cpu_size peuvent être définis pour appeler callback uniquement avec les événements qui se sont produits sur les CPU spécifiés, sinon si cpus est NULL, alors la fonction callback sera appelée pour tous les événements, et cpu_size est ignoré. La fonction callback sera appelée avec les paramètres suivants : Un pointeur vers une structure tep_event correspondant au type d'événement de l'enregistrement ; L'enregistrement représentant l'événement ; Le CPU sur lequel l'événement s'est produit ; et un pointeur vers le callback_context spécifié par l'utilisateur. Si le callback retourne une valeur non nulle, l'itération s'arrête.

La fonction tracefs_iterate_snapshot_events() fonctionne de la même manière que tracefs_iterate_raw_events() sauf qu'elle fonctionne sur le tampon snapshot.

Utilisez tracefs_iterate_stop() pour forcer une exécution de tracefs_iterate_raw_events() à s'arrêter. Ceci peut être appelé depuis un callback appelé par l'itérateur (même si un retour non nul l'arrêtera), ou depuis un autre thread.

La fonction tracefs_follow_event() est utilisée avec tracefs_iterate_raw_events() mais au lieu que le callback soit appelé pour chaque événement, il n'est appelé que pour le system / event_name spécifié donné à la fonction. Le callback est le même que pour tracefs_iterate_raw_events(), et le callback_context passé sera également passé au callback. Notez que s'il retourne quelque chose d'autre que 0, il arrêtera la boucle avant que le callback de tracefs_iterate_raw_events() ne soit appelé.

La fonction tracefs_follow_missed_events() appellera le callback lorsque des événements manqués sont détectés. Elle définira le paramètre record du callback sur l'enregistrement qui suit les événements manqués et event sera du type de l'événement que record représente. cpu sera défini sur le CPU qui a manqué les événements, et callback_data sera le contenu passé à la fonction.

La fonction tracefs_follow_event_clear() supprimera les followers de instance qui correspondent à system et event_name. Si system et event_name sont tous deux NULL, alors elle supprimera tous les followers d'événements associés à instance. Si seul system est NULL, alors elle supprimera tous les followers qui suivent des événements correspondant à event_name. Si seule event_name est NULL, alors elle supprimera tous les followers qui sont attachés à des événements faisant partie d'un système correspondant à system.

La fonction tracefs_follow_missed_events_clear() supprimera tous les followers pour les événements manqués.

VALEUR DE RETOUR

La fonction tracefs_iterate_raw_events() retourne -1 en cas d'erreur ou 0 sinon.

Les deux fonctions tracefs_follow_event_clear() et tracefs_follow_missed_events_clear() retournent 0 en cas de succès et -1 en cas d'erreur, ou si elles n'ont trouvé aucun follower correspondant à supprimer.

EXEMPLE

#include <unistd.h>
#include <tracefs.h>
#include <stdbool.h>
#include <signal.h>
struct my_struct {
        bool            stopped;
};
#define MAX_COUNT 500000
static int counter;
static int callback(struct tep_event *event, struct tep_record *record,
                    int cpu, void *data)
{
        struct my_struct *my_data = data;
        static struct trace_seq seq;
        if (counter++ > MAX_COUNT) {
                my_data->stopped = true;
                return 1;
        }
        if (!seq.buffer)
                trace_seq_init(&seq);
        tep_print_event(event->tep, &seq, record, "%16s-%-5d [%03d] %6.1000d %s: %s\n",
                        TEP_PRINT_COMM, TEP_PRINT_PID, TEP_PRINT_CPU,
                        TEP_PRINT_TIME, TEP_PRINT_NAME, TEP_PRINT_INFO);
        trace_seq_terminate(&seq);
        trace_seq_do_printf(&seq);
        trace_seq_reset(&seq);
        return 0;
}
static int sched_callback(struct tep_event *event, struct tep_record *record,
                          int cpu, void *data)
{
        static struct tep_format_field *prev_pid;
        static struct tep_format_field *next_pid;
        unsigned long long pid;
        int this_pid = *(int *)data;
        if (!prev_pid) {
                prev_pid = tep_find_field(event, "prev_pid");
                next_pid = tep_find_field(event, "next_pid");
                if (!prev_pid || !next_pid) {
                        fprintf(stderr, "No pid fields??\n");
                        return -1;
                }
        }
        tep_read_number_field(prev_pid, record->data, &pid);
        if (pid == this_pid)
                printf("WE ARE LEAVING!\n");
        tep_read_number_field(next_pid, record->data, &pid);
        if (pid == this_pid)
                printf("WE ARE ARRIVING!\n");
        return 0;
}
static int missed_callback(struct tep_event *event, struct tep_record *record,
                           int cpu, void *data)
{
        printf("OOPS! cpu %d dropped ", cpu);
        if (record->missed_events > 0)
                printf("%lld ", record->missed_events);
        printf("events\n");
        return 0;
}
static struct tracefs_instance *instance;
static struct my_struct my_data;
static void sig(int s)
{
        tracefs_iterate_stop(instance);
        my_data.stopped = true;
}
int main (int argc, char **argv, char **env)
{
        struct tep_handle *tep;
        int this_pid = getpid();
        instance = tracefs_instance_create("my-buffer");
        if (!instance)
                return -1;
        signal(SIGINT, sig);
        tracefs_event_enable(instance, NULL, NULL);
        sleep(1);
        tracefs_event_disable(instance, NULL, NULL);
        tep = tracefs_local_events(NULL);
        tep_load_plugins(tep);
        tracefs_follow_missed_events(instance, missed_callback, NULL);
        tracefs_follow_event(tep, instance, "sched", "sched_switch", sched_callback, &this_pid);
        tracefs_iterate_raw_events(tep, instance, NULL, 0, callback, &my_data);
        /* Note, the clear here is to show how to clear all followers
         * in case tracefs_iterate_raw_events() is called again, but
         * does not want to include the followers.


        tracefs_follow_event_clear(instance, "sched", "sched_switch");
        tracefs_follow_missed_events_clear(instance);
        return 0;
}
