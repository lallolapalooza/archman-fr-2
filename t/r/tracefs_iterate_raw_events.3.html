LIBTRACEFS(3) Manuel libtracefs LIBTRACEFS(3)

NOM

tracefs_iterate_raw_events, tracefs_iterate_stop, tracefs_follow_event, tracefs_follow_missed_events, tracefs_follow_event_clear, tracefs_follow_missed_events_clear, tracefs_iterate_snapshot_events - Itérer sur les événements dans le tampon circulaire

SYNTAXE

#include <tracefs.h>
int tracefs_iterate_raw_events(struct tep_handle *tep, struct tracefs_instance *instance,
                                 cpu_set_t *cpus, int cpu_size,
                                 int (*callback)(struct tep_event *, struct tep_record *, int, void *),
                                 void *callback_context);
void tracefs_iterate_stop(struct tracefs_instance *instance);
int tracefs_follow_event(struct tep_handle *tep, struct tracefs_instance *instance,
                          const char *system, const char *event_name,
                          int (*callback)(struct tep_event *,
                                          struct tep_record *,
                                          int, void *),
                          void *callback_data);
int tracefs_follow_missed_events(struct tracefs_instance *instance,
                          int (*callback)(struct tep_event *,
                                          struct tep_record *,
                                          int, void *),
                          void *callback_data);
int tracefs_follow_event_clear(struct tracefs_instance *instance,
                          const char *system, const char *event_name);
int tracefs_follow_missed_events_clear(struct tracefs_instance *instance);
int tracefs_iterate_snapshot_events(struct tep_handle *tep, struct tracefs_instance *instance,
                                 cpu_set_t *cpus, int cpu_size,
                                 int (*callback)(struct tep_event *, struct tep_record *, int, void *),
                                 void *callback_context);

DESCRIPTION

Itérateur de trace sur les événements bruts.

La fonction tracefs_iterate_raw_events() lit les tampons de données brutes tracefs et appelle la fonction callback spécifiée pour chaque événement rencontré. Les événements sont itérés dans l'ordre trié : le plus ancien d'abord. Un gestionnaire tep initialisé est requis (voir tracefs_local_events(3)). Si instance est NULL, alors le tampon tracefs de niveau supérieur est utilisé, sinon le tampon pour l'instance correspondante est lu. Pour filtrer uniquement sur un sous-ensemble de CPU, cpus et cpu_size peuvent être définis pour appeler la fonction callback uniquement avec les événements qui se sont produits sur les CPU spécifiés, sinon si cpus est NULL, alors la fonction callback sera appelée pour tous les événements, et cpu_size sera ignoré. La fonction callback sera appelée avec les paramètres suivants : Un pointeur vers une struct tep_event correspondant au type d'événement de l'enregistrement ; L'enregistrement représentant l'événement ; Le CPU sur lequel l'événement s'est produit ; et un pointeur vers le callback_context spécifié par l'utilisateur. Si la fonction callback retourne une valeur non nulle, l'itération s'arrête.

La fonction tracefs_iterate_snapshot_events() fonctionne de la même manière que tracefs_iterate_raw_events(), sauf qu'elle opère sur le tampon de snapshot.

Utilisez tracefs_iterate_stop() pour forcer un appel en cours de tracefs_iterate_raw_events() à s'arrêter. Ceci peut être appelé depuis une fonction callback appelée par l'itérateur (bien qu'un retour non nul l'arrêtera), ou depuis un autre thread.

La fonction tracefs_follow_event() est utilisée avec tracefs_iterate_raw_events(), mais au lieu que la fonction callback soit appelée pour chaque événement, elle n'est appelée que pour le couple system / event_name spécifié à la fonction. La fonction callback est la même que pour tracefs_iterate_raw_events(), et le callback_context passé sera également passé à la fonction callback. Notez que si elle retourne autre chose que 0, cela arrêtera la boucle avant que la fonction callback de tracefs_iterate_raw_events() ne soit appelée.

La fonction tracefs_follow_missed_events() appellera la fonction callback lorsque des événements manqués seront détectés. Elle définira le paramètre record de la fonction callback sur l'enregistrement qui suit les événements manqués et event sera du type d'événement de l'enregistrement. cpu sera défini sur le CPU qui a manqué les événements, et callback_data sera le contenu qui a été passé à la fonction.

La fonction tracefs_follow_event_clear() supprimera les suiveurs de instance qui correspondent à system et event_name. Si system et event_name sont tous deux NULL, alors elle supprimera tous les suiveurs d'événements associés à instance. Si seul system est NULL, alors elle supprimera tous les suiveurs qui suivent des événements correspondant à event_name. Si seul event_name est NULL, alors elle supprimera tous les suiveurs qui sont attachés à des événements appartenant à un système correspondant à system.

La fonction tracefs_follow_missed_events_clear() supprimera tous les suiveurs pour les événements manqués.

VALEUR DE RETOUR

La fonction tracefs_iterate_raw_events() retourne -1 en cas d'erreur ou 0 sinon.

Les fonctions tracefs_follow_event_clear() et tracefs_follow_missed_events_clear() retournent 0 en cas de succès et -1 en cas d'erreur, ou si elles n'ont trouvé aucun suiveur correspondant à supprimer.

EXEMPLE

#include <unistd.h>
#include <tracefs.h>
#include <stdbool.h>
#include <signal.h>
struct my_struct {
        bool            stopped;
};
#define MAX_COUNT 500000
static int counter;
static int callback(struct tep_event *event, struct tep_record *record,
                    int cpu, void *data)
{
        struct my_struct *my_data = data;
        static struct trace_seq seq;
        if (counter++ > MAX_COUNT) {
                my_data->stopped = true;
                return 1;
        }
        if (!seq.buffer)
                trace_seq_init(&seq);
        tep_print_event(event->tep, &seq, record, "%16s-%-5d [%03d] %6.1000d %s: %s\n",
                        TEP_PRINT_COMM, TEP_PRINT_PID, TEP_PRINT_CPU,
                        TEP_PRINT_TIME, TEP_PRINT_NAME, TEP_PRINT_INFO);
        trace_seq_terminate(&seq);
        trace_seq_do_printf(&seq);
        trace_seq_reset(&seq);
        return 0;
}
static int sched_callback(struct tep_event *event, struct tep_record *record,
                          int cpu, void *data)
{
        static struct tep_format_field *prev_pid;
        static struct tep_format_field *next_pid;
        unsigned long long pid;
        int this_pid = *(int *)data;
        if (!prev_pid) {
                prev_pid = tep_find_field(event, "prev_pid");
                next_pid = tep_find_field(event, "next_pid");
                if (!prev_pid || !next_pid) {
                        fprintf(stderr, "Pas de champs pid??\n");
                        return -1;
                }
        }
        tep_read_number_field(prev_pid, record->data, &pid);
        if (pid == this_pid)
                printf("NOUS PARTONS !\n");
        tep_read_number_field(next_pid, record->data, &pid);
        if (pid == this_pid)
                printf("NOUS ARRIVONS !\n");
        return 0;
}
static int missed_callback(struct tep_event *event, struct tep_record *record,
                           int cpu, void *data)
{
        printf("OUPS ! cpu %d a manqué ", cpu);
        if (record->missed_events > 0)
                printf("%lld ", record->missed_events);
        printf("événements\n");
        return 0;
}
static struct tracefs_instance *instance;
static struct my_struct my_data;
static void sig(int s)
{
        tracefs_iterate_stop(instance);
        my_data.stopped = true;
}
int main (int argc, char **argv, char **env)
{
        struct tep_handle *tep;
        int this_pid = getpid();
        instance = tracefs_instance_create("mon-buffer");
        if (!instance)
                return -1;
        signal(SIGINT, sig);
        tracefs_event_enable(instance, NULL, NULL);
        sleep(1);
        tracefs_event_disable(instance, NULL, NULL);
        tep = tracefs_local_events(NULL);
        tep_load_plugins(tep);
        tracefs_follow_missed_events(instance, missed_callback, NULL);
        tracefs_follow_event(tep, instance, "sched", "sched_switch", sched_callback, &this_pid);
        tracefs_iterate_raw_events(tep, instance, NULL, 0, callback, &my_data);
        /* Note, le clear ici sert à montrer comment effacer tous les suiveurs
         * au cas où tracefs_iterate_raw_events() serait appelé à nouveau, mais
         * ne voudrait pas inclure les suiveurs précédents. */
        tracefs_follow_event_clear(instance, NULL, NULL);
        tracefs_follow_missed_events_clear(instance);
        return 0;
}

NOM

tracefs_iterate_raw_events - Itérer sur les événements bruts enregistrés

SYNOPIS

#include <tracefs.h>


int tracefs_iterate_raw_events(
    struct tracefs_instance *instance,
    tracefs_event_callback_t callback,
    void *data);

DESCRIPTION

La fonction tracefs_iterate_raw_events itère sur tous les événements bruts capturés dans le tampon de trace de l'instance instance. Pour chaque événement trouvé, elle appelle la fonction de rappel callback avec les données de l'événement et le pointeur de données utilisateur data.

La fonction de rappel callback doit avoir la signature suivante :

typedef int (*tracefs_event_callback_t)(
    struct tracefs_instance *instance,
    struct tracefs_raw_event *event,
    void *data);

Si la fonction de rappel retourne une valeur négative, l'itération s'arrête et cette valeur est retournée par tracefs_iterate_raw_events.

Si l'itération se termine sans que la fonction de rappel n'arrête le processus, tracefs_iterate_raw_events retourne 0.

Les événements bruts sont généralement des enregistrements de bas niveau tels qu'ils sont stockés dans le système de fichiers tracefs. Ils ne sont pas formatés ou analysés par libtracefs. Pour accéder aux événements formatés, utilisez tracefs_iterate_events(3).

VALEUR DE RETOUR

Retourne 0 en cas de succès, ou le code d'erreur retourné par la fonction de rappel si celle-ci a demandé l'arrêt prématuré. En cas d'erreur d'allocation de mémoire ou autre erreur interne, une valeur négative peut être retournée.

EXEMPLE

Voici un exemple simple d'utilisation de tracefs_iterate_raw_events :

#include <stdio.h>
#include <stdlib.h>
#include <signal.h>
#include <tracefs.h>

static struct tracefs_instance *instance;
static volatile int stopping = 0;
#define MAX_COUNT 1000

static int count_event_callback(struct tracefs_instance *instance,
                                struct tracefs_raw_event *event,
                                void *data)
{
        int *counter = data;

        (*counter)++;
        if (*counter > MAX_COUNT) {
                stopping = 1;
                return -1; /* Stop iteration */
        }
        return 0; /* Continue iteration */
}

static void sig_handler(int sig)
{
        stopping = 1;
}

int main(int argc, char **argv)
{
        int ret;
        int counter = 0;

        signal(SIGINT, sig_handler);
        signal(SIGTERM, sig_handler);

        /*
         * Crée une nouvelle instance tracefs et charge les événements.
         * Les détails de cette partie dépendent de votre cas d'utilisation.
         * Ici, nous supposons que nous utilisons les événements par défaut.
         */
        instance = tracefs_instance_new(NULL);
        if (!instance) {
                fprintf(stderr, "Failed to create tracefs instance\n");
                return 1;
        }

        /* Charge les événements par défaut */
        tracefs_load_default_events(instance);

        /* Lance l'enregistrement */
        ret = tracefs_start(instance);
        if (ret < 0) {
                fprintf(stderr, "Failed to start tracing: %d\n", ret);
                tracefs_instance_free(instance);
                return 1;
        }

        printf("Tracing started... Press Ctrl+C to stop.\n");

        /* Boucle d'attente et de traitement des événements */
        while (!stopping) {
                /* Attend que les événements arrivent */
                ret = tracefs_poll(instance, 100); /* Attend jusqu'à 100ms */
                if (ret < 0) {
                        fprintf(stderr, "Polling failed: %d\n", ret);
                        break;
                }

                /* Itère sur les événements bruts capturés */
                ret = tracefs_iterate_raw_events(instance, count_event_callback, &counter);
                if (ret < 0 && !stopping) {
                        fprintf(stderr, "Callback returned error: %d\n", ret);
                        break;
                }
        }

        /* Arrête l'enregistrement */
        tracefs_stop(instance);

        /*
         * Nettoie les événements déjà traités. Ce n'est pas nécessaire ici
         * car tracefs_instance_free() s'en chargera.
         */
        tracefs_follow_event_clear(instance, NULL, NULL);
        tracefs_follow_missed_events_clear(instance);
        tracefs_instance_destroy(instance);
        tracefs_instance_free(instance);
        if (my_data.stopped) {
                if (counter > MAX_COUNT)
                        printf("Finished max count\n");
                else
                        printf("Finished via signal\n");
        }
        return 0;
}
