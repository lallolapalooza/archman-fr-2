sous-routine ctrsyl (character trana, character tranb, integer isgn, integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldc, * ) c, integer ldc, real scale, integer info)

CTRSYL

But :

!>
!> CTRSYL résout l'équation matricielle complexe de Sylvester :
!>
!>    op(A)*X + X*op(B) = scale*C ou
!>    op(A)*X - X*op(B) = scale*C,
!>
!> où op(A) = A ou A**H, et A et B sont toutes deux triangulaires supérieures. A est
!> de taille M par M et B est de taille N par N ; le membre de droite C et la solution X sont
!> de taille M par N ; et scale est un facteur d'échelle de sortie, défini <= 1 pour éviter
!> le débordement dans X.
!>

Paramètres

TRANA 
!>          TRANA est CHARACTER*1
!>          Spécifie l'option op(A) :
!>          = 'N': op(A) = A    (Pas de transposée)
!>          = 'C': op(A) = A**H (Conjuguée transposée)
!>

TRANB

!>          TRANB est CHARACTER*1
!>          Spécifie l'option op(B) :
!>          = 'N': op(B) = B    (Pas de transposée)
!>          = 'C': op(B) = B**H (Conjuguée transposée)
!>

ISGN

!>          ISGN est INTEGER
!>          Spécifie le signe dans l'équation :
!>          = +1: résoudre op(A)*X + X*op(B) = scale*C
!>          = -1: résoudre op(A)*X - X*op(B) = scale*C
!>

M

!>          M est INTEGER
!>          L'ordre de la matrice A, et le nombre de lignes des
!>          matrices X et C. M >= 0.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice B, et le nombre de colonnes des
!>          matrices X et C. N >= 0.
!>

A

!>          A est un tableau COMPLEX, dimension (LDA,M)
!>          La matrice triangulaire supérieure A.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,M).
!>

B

!>          B est un tableau COMPLEX, dimension (LDB,N)
!>          La matrice triangulaire supérieure B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,N).
!>

C

!>          C est un tableau COMPLEX, dimension (LDC,N)
!>          En entrée, la matrice C de membre de droite M par N.
!>          En sortie, C est écrasé par la matrice solution X.
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale du tableau C. LDC >= max(1,M)
!>

SCALE

!>          SCALE est REAL
!>          Le facteur d'échelle, scale, défini <= 1 pour éviter le débordement dans X.
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument était invalide
!>          = 1 : A et B ont des valeurs propres communes ou très proches ; des valeurs perturbées
!>               ont été utilisées pour résoudre l'équation (mais les matrices
!>               A et B restent inchangées).
!>
