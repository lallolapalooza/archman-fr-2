tracefs_hist_continue(3)

Paquet: libtracefs

Licences: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) Manuel libtracefs LIBTRACEFS(3)

NOM

tracefs_hist_start, tracefs_hist_destroy, tracefs_hist_pause, tracefs_hist_continue, tracefs_hist_reset - Pause, continue ou effacer un histogramme existant

SYNTAXE

#include <tracefs.h>
int tracefs_hist_start(struct tracefs_instance *instance, struct tracefs_hist *hist);
int tracefs_hist_destroy(struct tracefs_instance *instance, struct tracefs_hist *hist);
int tracefs_hist_pause(struct tracefs_instance *instance, struct tracefs_hist *hist);
int tracefs_hist_continue(struct tracefs_instance *instance, struct tracefs_hist *hist);
int tracefs_hist_reset(struct tracefs_instance *instance, struct tracefs_hist *hist);

DESCRIPTION

tracefs_hist_start() est appelée pour réellement démarrer l'histogramme hist. L'instance est l'instance dans laquelle démarrer l'histogramme, NULL s'il doit démarrer au niveau supérieur.

tracefs_hist_pause() est appelée pour mettre en pause l'histogramme hist. L'instance est l'instance dans laquelle mettre en pause l'histogramme, NULL s'il est au niveau supérieur.

tracefs_hist_continue() est appelée pour reprendre un histogramme mis en pause hist. L'instance est l'instance pour laquelle reprendre l'histogramme, NULL s'il est au niveau supérieur.

tracefs_hist_reset() est appelée pour effacer / réinitialiser l'histogramme hist. L'instance est l'instance pour laquelle effacer l'histogramme, NULL s'il est au niveau supérieur.

tracefs_hist_destroy() est appelée pour supprimer l'histogramme afin qu'il n'existe plus. L'instance est l'instance à partir de laquelle supprimer l'histogramme, NULL s'il est au niveau supérieur.

VALEUR DE RETOUR

Toutes retournent zéro en cas de succès ou -1 en cas d'erreur.

EXEMPLE

#include <stdlib.h>
#include <tracefs.h>
enum commands {
        START,
        PAUSE,
        CONT,
        RESET,
        DELETE,
        SHOW,
};
int main (int argc, char **argv, char **env)
{
        struct tracefs_instance *instance;
        struct tracefs_hist *hist;
        struct tep_handle *tep;
        enum commands cmd;
        char *cmd_str;
        int ret;
        if (argc < 2) {
                fprintf(stderr, "usage: %s command\n", argv[0]);
                exit(-1);
        }
        cmd_str = argv[1];
        if (!strcmp(cmd_str, "start"))
                cmd = START;
        else if (!strcmp(cmd_str, "pause"))
                cmd = PAUSE;
        else if (!strcmp(cmd_str, "cont"))
                cmd = CONT;
        else if (!strcmp(cmd_str, "reset"))
                cmd = RESET;
        else if (!strcmp(cmd_str, "delete"))
                cmd = DELETE;
        else if (!strcmp(cmd_str, "show"))
                cmd = SHOW;
        else {
                fprintf(stderr, "Unknown command %s\n", cmd_str);
                exit(-1);
        }
        tep = tracefs_local_events(NULL);
        if (!tep) {
                perror("Reading tracefs");
                exit(-1);
        }
        instance = tracefs_instance_create("hist_test");
        if (!instance) {
                fprintf(stderr, "Failed instance create\n");
                exit(-1);
        }
        hist = tracefs_hist_alloc_2d(tep, "kmem", "kmalloc",
                                     "call_site",TRACEFS_HIST_KEY_SYM,
                                     "bytes_req", 0);
        if (!hist) {
                fprintf(stderr, "Failed hist create\n");
                exit(-1);
        }
        ret = tracefs_hist_add_value(hist, "bytes_alloc");
        ret |= tracefs_hist_add_sort_key(hist, "bytes_req");
        ret |= tracefs_hist_add_sort_key(hist, "bytes_alloc");
        ret |= tracefs_hist_sort_key_direction(hist, "bytes_alloc",
                                               TRACEFS_HIST_SORT_DESCENDING);
        if (ret) {
                fprintf(stderr, "Failed modifying histogram\n");
                exit(-1);
        }
        tracefs_error_clear(instance);
        switch (cmd) {
        case START:
                ret = tracefs_hist_start(instance, hist);
                if (ret) {
                        char *err = tracefs_error_last(instance);
                        if (err)
                                fprintf(stderr, "\n%s\n", err);
                }
                break;
        case PAUSE:
                ret = tracefs_hist_pause(instance, hist);
                break;
        case CONT:
                ret = tracefs_hist_continue(instance, hist);
                break;
        case RESET:
                ret = tracefs_hist_reset(instance, hist);
                break;
        case DELETE:
                ret = tracefs_hist_destroy(instance, hist);
                break;
        case SHOW: {
                char *content;
                content = tracefs_event_file_read(instance, "kmem", "kmalloc",
                                                  "hist", NULL);
                ret = content ? 0 : -1;
                if (content) {
                        printf("%s\n", content);
                        free(content);
                }
                break;
        }
        }
        if (ret)
                fprintf(stderr, "Failed: command\n");
        exit(ret);
}
