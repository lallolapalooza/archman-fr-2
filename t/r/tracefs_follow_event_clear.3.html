NOM

tracefs_iterate_raw_events, tracefs_iterate_stop, tracefs_follow_event, tracefs_follow_missed_events, tracefs_follow_event_clear, tracefs_follow_missed_events_clear, tracefs_iterate_snapshot_events - Itérer sur les événements du tampon circulaire

SYNOPIS

#include <tracefs.h>

                                 int (*callback)(struct tep_event *, struct tep_record *, int, void *),
                                 void *callback_context);
void tracefs_iterate_stop(struct tracefs_instance *instance);
int tracefs_follow_event(struct tep_handle *tep, struct tracefs_instance *instance,
                          const char *system, const char *event_name,
                          int (*callback)(struct tep_event *,
                                          struct tep_record *,
                                          int, void *),
                          void *callback_data);
int tracefs_follow_missed_events(struct tracefs_instance *instance,
                          int (*callback)(struct tep_event *,
                                          struct tep_record *,
                                          int, void *),
                          void *callback_data);
int tracefs_follow_event_clear(struct tracefs_instance *instance,
                          const char *system, const char *event_name);
int tracefs_follow_missed_events_clear(struct tracefs_instance *instance);
int tracefs_iterate_snapshot_events(struct tep_handle *tep, struct tracefs_instance *instance,
                                 cpu_set_t *cpus, int cpu_size,
                                 int (*callback)(struct tep_event *, struct tep_record *, int, void *),
                                 void *callback_context);

DESCRIPTION

Itérateur de trace sur les événements bruts.

La fonction tracefs_iterate_raw_events() lit les tampons de données brutes tracefs et appelle la fonction callback spécifiée pour chaque événement rencontré. Les événements sont itérés dans l'ordre trié : le plus ancien en premier. Un gestionnaire tep initialisé est requis (Voir tracefs_local_events(3)). Si instance est NULL, alors le tampon tracefs de niveau supérieur est utilisé, sinon le tampon correspondant à instance est lu. Pour filtrer uniquement sur un sous-ensemble de CPUs, cpus et cpu_size peuvent être définis pour appeler callback uniquement avec les événements qui se sont produits sur les CPUs spécifiés, sinon si cpus est NULL, la fonction callback sera appelée pour tous les événements, et cpu_size est ignoré. La fonction callback sera appelée avec les paramètres suivants : Un pointeur vers un struct tep_event qui correspond au type d'événement de l'enregistrement ; L'enregistrement représentant l'événement ; Le CPU sur lequel l'événement s'est produit ; et un pointeur vers le callback_context spécifié par l'utilisateur. Si le callback renvoie une valeur non nulle, l'itération s'arrête.

La fonction tracefs_iterate_snapshot_events() fonctionne de la même manière que tracefs_iterate_raw_events(), sauf qu'elle fonctionne sur le tampon snapshot.

Utilisez tracefs_iterate_stop() pour forcer l'arrêt d'une tracefs_iterate_raw_events() en cours d'exécution. Ceci peut être appelé depuis un callback appelé par l'itérateur (même si un retour de valeur non nulle l'arrête), ou depuis un autre thread.

La fonction tracefs_follow_event() est utilisée avec tracefs_iterate_raw_events() mais au lieu que le callback soit appelé pour chaque événement, il ne l'est que pour le system / event_name spécifié donné à la fonction. Le callback est le même que pour tracefs_iterate_raw_events(), et le callback_context passé sera également transmis au callback. Notez que s'il renvoie autre chose que 0, il arrêtera la boucle avant que le callback de tracefs_iterate_raw_events() ne soit appelé.

La fonction tracefs_follow_missed_events() appellera le callback lorsque des événements manqués seront détectés. Elle définira le paramètre record du callback à l'enregistrement qui suit les événements manqués et event sera du type d'événement de l'enregistrement. cpu sera défini au CPU qui a manqué les événements, et callback_data sera le contenu passé à la fonction.

La fonction tracefs_follow_event_clear() supprimera les "followers" de instance qui correspondent à system et event_name. Si system et event_name sont tous deux NULL, alors elle supprimera tous les "followers" d'événements associés à instance. Si seulement system est NULL, alors elle supprimera tous les "followers" qui suivent des événements correspondant à event_name. Si seulement event_name est NULL, alors elle supprimera tous les "followers" attachés à des événements faisant partie d'un système correspondant à system.

La fonction tracefs_follow_missed_events_clear() supprimera tous les "followers" pour les événements manqués.

VALEUR DE RETOUR

La fonction tracefs_iterate_raw_events() renvoie -1 en cas d'erreur ou 0 sinon.

Les fonctions tracefs_follow_event_clear() et tracefs_follow_missed_events_clear() renvoient 0 en cas de succès et -1 en cas d'erreur, ou si elles n'ont trouvé aucun "follower" correspondant à supprimer.

        struct my_struct *my_data = data;
        static struct trace_seq seq;
        if (counter++ > MAX_COUNT) {
                my_data->stopped = true;
                return 1;
        }
        if (!seq.buffer)
                trace_seq_init(&seq);
        tep_print_event(event->tep, &seq, record, "%16s-%-5d [%03d] %6.1000d %s: %s\n",

                        TEP_PRINT_TIME, TEP_PRINT_NAME, TEP_PRINT_INFO);
        trace_seq_terminate(&seq);
        trace_seq_do_printf(&seq);
        trace_seq_reset(&seq);

                        return -1;

}

tracefs_follow_event_clear(3) — libtracefs 1.8.2

NAME

tracefs_follow_event_clear, tracefs_follow_missed_events_clear, tracefs_instance_destroy, tracefs_instance_free, tracefs_instance_setup, tracefs_instance_start, tracefs_instance_stop - Helper functions for tracing

SYNOPSIS

#include <tracefs.h>

void tracefs_follow_event_clear(struct tracefs_instance *instance,
                                const char *event_name, const char *common_name);
void tracefs_follow_missed_events_clear(struct tracefs_instance *instance);
void tracefs_instance_destroy(struct tracefs_instance *instance);
void tracefs_instance_free(struct tracefs_instance *instance);
struct tracefs_instance *tracefs_instance_setup(const char *name);
int tracefs_instance_start(struct tracefs_instance *instance);
int tracefs_instance_stop(struct tracefs_instance *instance);

DESCRIPTION

The tracefs library provides functions for tracing events in the kernel via the perf_event framework and tracefs filesystem. This man page describes some helper functions that are part of the library.

tracefs_instance_setup() creates and initializes a tracing instance. The name parameter specifies the name of the instance, which corresponds to a directory name under the tracefs filesystem (e.g., /sys/kernel/tracing/instances/<name>).

tracefs_instance_start() starts the tracing instance, enabling the configured events.

tracefs_instance_stop() stops the tracing instance, disabling the configured events.

tracefs_follow_event_clear() clears a specific event from being followed. If event_name and common_name are NULL, it clears all followed events.

tracefs_follow_missed_events_clear() clears the missed events buffer for the instance.

tracefs_instance_destroy() destroys the tracing instance, but does not free the memory associated with it.

tracefs_instance_free() frees the memory associated with a tracing instance.

RETURN VALUE

tracefs_instance_setup() returns a pointer to a struct tracefs_instance on success, or NULL on failure.

tracefs_instance_start() and tracefs_instance_stop() return 0 on success, or a negative error code on failure.

tracefs_follow_event_clear(), tracefs_follow_missed_events_clear(), tracefs_instance_destroy(), and tracefs_instance_free() do not return a value.

EXAMPLE

#include <stdio.h>
#include <stdlib.h>
#include <signal.h>
#include <tracefs.h>

struct my_data {
        int stopped;
};

struct my_data my_data = { .stopped = 0 };
unsigned long counter = 0;
#define MAX_COUNT 1000000

void sig_handler(int sig)
{
        my_data.stopped = 1;
}

int main(void)
{
        struct tracefs_instance *instance;
        int ret;

        signal(SIGINT, sig_handler);

        instance = tracefs_instance_setup("my_instance");
        if (!instance) {
                fprintf(stderr, "Failed to setup tracefs instance\n");
                return 1;
        }

        /* Add events to trace */
        tracefs_instance_add_event(instance, "sched_switch", NULL);
        tracefs_instance_add_event(instance, "sched_wakeup", NULL);

        ret = tracefs_instance_start(instance);
        if (ret < 0) {
                fprintf(stderr, "Failed to start tracefs instance: %d\n", ret);
                tracefs_instance_destroy(instance);
                tracefs_instance_free(instance);
                return 1;
        }

        while (!my_data.stopped && counter < MAX_COUNT) {
                /* Do some work */
                counter++;
        }

        ret = tracefs_instance_stop(instance);
        if (ret < 0) {
                fprintf(stderr, "Failed to stop tracefs instance: %d\n", ret);
        }

        /*
         * The tracefs_follow_* functions clear events that are being followed.
         * It's not needed here because tracefs_instance_free() will clean them up.
         */
        tracefs_follow_event_clear(instance, NULL, NULL);
        tracefs_follow_missed_events_clear(instance);
        tracefs_instance_destroy(instance);
        tracefs_instance_free(instance);
        if (my_data.stopped) {
                if (counter > MAX_COUNT)
                        printf("Finished max count\n");
                else
                        printf("Finished via signal\n");
        }
        return 0;
}
