#include <tracefs.h>
int tracefs_synth_create(struct tracefs_synth *synth);
int tracefs_synth_destroy(struct tracefs_synth *synth);

int tracefs_synth_set_instance(struct tracefs_synth *synth, struct tracefs_instance *instance);
int tracefs_synth_trace(struct tracefs_synth *synth,
                        enum tracefs_synth_handler type, const char *var);
int tracefs_synth_snapshot(struct tracefs_synth *synth,
                           enum tracefs_synth_handler type, const char *var);
int tracefs_synth_save(struct tracefs_synth *synth,
                       enum tracefs_synth_handler type, const char *var,
                       char **save_fields);

DESCRIPTION

Les événements synthétiques sont des événements dynamiques créés en faisant correspondre deux autres événements, ce qui déclenche un événement synthétique. Un événement est l'événement de départ dont un champ est enregistré, et lorsque le second événement est exécuté, s'il possède un champ (ou des champs) correspondant au champ (ou aux champs) de l'événement de départ, il déclenche l'événement synthétique. Les valeurs des champs autres que les champs correspondants peuvent être transmises de l'événement de départ à l'événement final pour effectuer des calculs, ou simplement comme paramètre à l'événement synthétique.

Un cas d'utilisation courant consiste à utiliser "sched_waking" comme événement de départ. Cet événement est déclenché lorsqu'un processus est réveillé. Ensuite, utiliser "sched_switch" comme événement de fin. Cet événement est déclenché lorsqu'une nouvelle tâche est planifiée sur le CPU. En utilisant le "common_pid" des deux événements comme champs correspondants, le temps écoulé entre les deux événements est considéré comme la latence de réveil de ce processus. Utilisez TRACEFS_TIMESTAMP comme champ pour les deux événements afin de calculer la différence en nanosecondes, ou utilisez TRACEFS_TIMESTAMP_USECS comme champs de comparaison pour les deux événements afin de calculer la différence en microsecondes. Ceci est utilisé comme exemple ci-dessous.

tracefs_synth_create() crée l'événement synthétique dans le système. Par défaut, les déclencheurs d'histogramme sont créés dans l'instance de trace supérieure, car tout événement synthétique peut être utilisé globalement sur toutes les instances. Dans le cas où une application souhaite exclure les déclencheurs d'histogramme de l'instance de niveau supérieur, elle peut utiliser tracefs_synth_set_instance() pour que les histogrammes soient utilisés pour créer l'événement synthétique dans une instance autre que celle de niveau supérieur. Un descripteur d'événement synthétique doit être créé avec tracefs_synth_alloc(3) avant que cela puisse être utilisé pour le créer sur le système.

tracefs_synth_destroy() détruit l'événement synthétique. Il tentera d'arrêter son exécution dans son instance (celle du haut par défaut), mais s'il est exécuté dans une autre instance, cela peut échouer car elle est occupée.

tracefs_synth_complete() renvoie vrai si l'événement synthétique synth a à la fois un événement de départ et un événement de fin.

tracefs_synth_trace() Au lieu de simplement effectuer une trace lors de la correspondance des événements de début et de fin, effectue le gestionnaire type où TRACEFS_SYNTH_HANDLE_MAX effectuera une trace lorsque la variable donnée var atteint un nouveau maximum pour les clés correspondantes. Ou TRACEFS_SYNTH_HANDLE_CHANGE lorsqu'il s'agit d'un changement de var. var doit être l'un des éléments name utilisés dans tracefs_synth_add_end_field(3).

tracefs_synth_snapshot() Lorsque la variable donnée var est soit un nouveau maximum si handler est TRACEFS_SYNTH_HANDLE_MAX, soit simplement modifiée si TRACEFS_SYNTH_HANDLE_CHANGE, alors prendre un "instantané" du tampon. L'instantané déplace le tampon de "trace" normal vers un tampon "instantané", qui peut être accessible via le fichier "snapshot" dans le répertoire tracefs de niveau supérieur, ou l'une des instances. var change. var doit être l'un des éléments name utilisés dans tracefs_synth_add_end_field(3).

tracefs_synth_save() Lorsque la variable donnée var est soit un nouveau maximum si handler est TRACEFS_SYNTH_HANDLE_MAX, soit simplement modifiée si TRACEFS_SYNTH_HANDLE_CHANGE, alors enregistrer la liste save_fields donnée. Les champs seront stockés dans le fichier d'histogramme "hist" de l'événement qui peut être récupéré avec tracefs_event_file_read(3). var doit être l'un des éléments name utilisés dans tracefs_synth_add_end_field(3).

tracefs_synth_set_instance() Définit l'instance de trace, où les déclencheurs d'histogramme qui créent l'événement synthétique seront créés. Par défaut, l'instance supérieure est utilisée. Cette API doit être appelée avant l'appel à tracefs_synth_create(), afin d'utiliser la nouvelle instance lors de la création de l'événement. Notez que même si l'événement synthétique est créé dans une instance, il est toujours visible par toutes les autres instances, y compris celle de niveau supérieur. Autrement dit, d'autres instances peuvent activer l'événement synthétique créé et le tracer dans les tampons appartenant à l'instance qui l'a activé.

VALEUR DE RETOUR

Toutes les fonctions renvoient zéro en cas de succès ou -1 en cas d'erreur.

ERREURS

Les erreurs suivantes concernent tous les appels ci-dessus :

EPERM L'utilisateur n'est pas root lorsque cela est requis.

EINVAL Soit un paramètre n'est pas valide (NULL alors qu'il ne devrait pas l'être) soit un champ n'est pas compatible pour les calculs.

ENODEV Un événement ou l'un de ses champs n'est pas trouvé.

EBADE Les champs des événements de début et de fin ne sont pas compatibles pour la correspondance ou la comparaison.

ENOMEM pas assez de mémoire disponible.

Et d'autres erreurs peuvent s'être produites lors des appels système au système.

EXEMPLE

Voir tracefs_sql(3) pour une utilisation plus approfondie de ce code.

#include <stdlib.h>
#include <tracefs.h>
#define start_event "sched_waking"
#define start_field "pid"
#define end_event "sched_switch"
#define end_field "next_pid"
#define match_name "pid"
static struct tracefs_synth *synth;
static void make_event(void)
{
        struct tep_handle *tep;
        /* Charger tous les événements du système */
        tep = tracefs_local_events(NULL);
        /* Initialiser l'événement synthétique */
        synth = tracefs_synth_alloc(tep, "wakeup_lat",
                                    NULL, start_event,
                                    NULL, end_event,
                                    start_field, end_field,
                                    match_name);
        /* Le tep n'est plus nécessaire */
        tep_free(tep);
        /* Enregistrer le champ "prio" sous le nom "prio" de l'événement de départ */
        tracefs_synth_add_start_field(synth, "prio", NULL);
        /* Enregistrer le champ "next_comm" sous le nom "comm" de l'événement de fin */
        tracefs_synth_add_end_field(synth, "next_comm", "comm");
        /* Enregistrer le champ "prev_prio" sous le nom "prev_prio" de l'événement de fin */
        tracefs_synth_add_end_field(synth, "prev_prio", NULL);
        /*
         * Calculer la différence en microsecondes entre la fin et le début
         * et enregistrer sous le nom "delta"
         */
        tracefs_synth_add_compare_field(synth, TRACEFS_TIMESTAMP_USECS,
                                        TRACEFS_TIMESTAMP_USECS,
                                        TRACEFS_SYNTH_DELTA_END, "delta");
        /* Enregistrer uniquement si le "prio" de l'événement de départ est inférieur à 100 */
        tracefs_synth_append_start_filter(synth, TRACEFS_FILTER_COMPARE,
                                          "prio", TRACEFS_COMPARE_LT, "100");
        /*
         * Enregistrer uniquement si le "next_prio" de l'événement de fin est inférieur à 50
         * ou si la priorité de la tâche précédente n'était pas supérieure ou égale à 100.
         *   next_prio < 50 || !(prev_prio >= 100)
         */
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_COMPARE,
                                        "next_prio", TRACEFS_COMPARE_LT, "50");
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_OR, NULL, 0, NULL);
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_NOT, NULL, 0, NULL);
        /* ... le reste du filtre ... */
        /* Créer l'événement synthétique sur le système */
        tracefs_synth_create(synth);
}

DESCRIPTION

Les fonctions tracefs_synth_* permettent de créer et de manipuler des événements synthétiques pour le traçage du noyau Linux. Un événement synthétique est un événement de traçage personnalisé qui peut être défini par l'utilisateur. Il peut être utilisé pour capturer des informations spécifiques qui ne sont pas disponibles dans les événements de traçage standard.

Le flux de travail typique consiste à :

  1. Allouer un nouveau synthétiseur d'événement avec tracefs_synth_alloc.

  2. Définir les conditions de correspondance et de comparaison pour l'événement à l'aide de tracefs_synth_add_match_field, tracefs_synth_add_compare_field, tracefs_synth_add_start_field et tracefs_synth_add_end_field.

  3. Ajouter des filtres de début et de fin à l'aide de tracefs_synth_append_start_filter et tracefs_synth_append_end_filter.

  4. Finaliser la définition de l'événement avec tracefs_synth_complete.

  5. Afficher la commande pour créer l'événement avec tracefs_synth_echo_cmd, ou créer l'événement directement avec tracefs_synth_create.

  6. Libérer les ressources utilisées par le synthétiseur d'événement avec tracefs_synth_free.

Les fonctions suivantes sont également disponibles :

  • tracefs_synth_destroy: Supprime un événement synthétique existant.

  • tracefs_synth_get_event: Récupère le nom de l'événement synthétique.

  • tracefs_synth_get_start_hist: Récupère l'histogramme de début associé à l'événement synthétique.

  • tracefs_synth_get_name: Récupère le nom de l'événement synthétique.

  • tracefs_synth_raw_fmt: Permet de formater une chaîne de caractères à ajouter à la sortie de traçage.

  • tracefs_synth_show_event: Affiche la définition de l'événement synthétique.

  • tracefs_synth_show_start_hist: Affiche l'histogramme de début de l'événement synthétique.

  • tracefs_synth_show_end_hist: Affiche l'histogramme de fin de l'événement synthétique.

EXEMPLE

/* Crée un événement synthétique nommé "my_event".
 * L'événement correspondra lorsque le champ "next_prio" est inférieur à 50.
 * Il enregistrera également l'histogramme "next_prio" et "prev_prio".
 */
static void make_event(void)
{
        synth = tracefs_synth_alloc("my_event");
        if (!synth)
                return;

        /*
         * Définir le filtre de début :
         * "next_prio < 50"
         */
        tracefs_synth_append_start_filter(synth, TRACEFS_FILTER_COMPARE,
                                        "next_prio", TRACEFS_COMPARE_LT,
                                        "50");

        /*
         * Définir l'histogramme de début :
         * Nommer l'histogramme "next_prio", et ajouter les champs "next_prio" et "prev_prio".
         */
        tracefs_synth_add_start_field(synth, "next_prio");
        tracefs_synth_add_end_field(synth, "prev_prio");

        /*
         * Définir le filtre de fin :
         * "!(prev_prio >= 100)"
         * Note: le ')', fermant le filtre, sera ajouté par tracefs_synth_complete().
         */
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_OPEN_PAREN, NULL, 0, NULL);
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_COMPARE,
                                        "prev_prio", TRACEFS_COMPARE_GE, "100");
        /*
         * Note, ci-dessus, on a seulement ajouté : "next_prio < 50 || !(prev_prio >= 100"
         * C'est parce que, lorsque le synthétiseur sera exécuté, la parenthèse fermante
         * restante sera ajoutée. C'est-à-dire, la chaîne finira par être :
         * "next_prio < 50 || !(prev_prio >= 100)" lorsque tracefs_sync_create()
         * ou tracefs_sync_echo_cmd() sera exécuté.
         */
}
/* Affiche comment créer l'événement synthétique */
static void show_event(void)
{
        struct trace_seq s;
        trace_seq_init(&s);
        tracefs_synth_echo_cmd(&s, synth);
        trace_seq_terminate(&s);
        trace_seq_do_printf(&s);
        trace_seq_destroy(&s);
}
int main (int argc, char **argv)
{
        make_event();
        if (argc > 1) {
                if (!strcmp(argv[1], "create")) {
                        /* Créer l'événement synthétique */
                        tracefs_synth_create(synth);
                } else if (!strcmp(argv[1], "delete")) {
                        /* Supprimer l'événement synthétique */
                        tracefs_synth_destroy(synth);
                } else {
                        printf("usage: %s [create|delete]\n", argv[0]);
                        exit(-1);
                }
        } else
                show_event();
        tracefs_synth_free(synth);
        return 0;
}
