const char *tracefs_synth_show_event(struct tracefs_synth *synth);
const char *tracefs_synth_show_start_hist(struct tracefs_synth *synth);

struct tep_event *tracefs_synth_get_event(struct tep_handle *tep, struct tracefs_synth *synth);

DESCRIPTION

Les événements synthétiques sont des événements dynamiques créés en faisant correspondre deux autres événements, ce qui déclenche un événement synthétique. Un événement est l'événement de départ dont un champ est enregistré, et lorsque le second événement est exécuté, s'il possède un champ (ou des champs) qui correspond au champ (ou aux champs) de l'événement de départ, il déclenche l'événement synthétique. Les valeurs des champs autres que ceux correspondants peuvent être transmises de l'événement de départ à l'événement de fin pour effectuer des calculs, ou simplement passées en paramètre à l'événement synthétique.

Un cas d'utilisation courant est de définir "sched_waking" comme événement de départ. Cet événement est déclenché lorsqu'un processus est réveillé. Ensuite, définir "sched_switch" comme événement de fin. Cet événement est déclenché lorsqu'une nouvelle tâche est planifiée sur le CPU. En définissant le "common_pid" des deux événements comme champs de correspondance, le temps écoulé entre les deux événements est considéré comme la latence de réveil de ce processus. Utilisez TRACEFS_TIMESTAMP comme champ pour les deux événements afin de calculer la différence en nanosecondes, ou utilisez TRACEFS_TIMESTAMP_USECS comme champs de comparaison pour les deux événements afin de calculer la différence en microsecondes. Ceci est utilisé comme exemple ci-dessous.

Voir tracefs_synth_alloc(3) pour l'allocation des événements synthétiques, et tracefs_synth_create() pour la création de l'événement synthétique sur le système.

tracefs_synth_echo_cmd() agit comme tracefs_synth_create(), mais au lieu de créer l'événement synthétique dans le système, il écrira les commandes d'écho pour le créer manuellement dans le seq donné.

tracefs_synth_get_start_hist() renvoie un descripteur struct tracefs_hist décrivant l'histogramme utilisé pour créer l'événement synthétique.

enum tracefs_synth_handler {
        TRACEFS_SYNTH_HANDLE_MATCH,
        TRACEFS_SYNTH_HANDLE_MAX,
        TRACEFS_SYNTH_HANDLE_CHANGE,
};

tracefs_synth_get_name() renvoie le nom de l'événement synthétique ou NULL en cas d'erreur. La chaîne renvoyée appartient à l'objet d'événement synthétique et est libérée avec l'événement par tracefs_synth_free().

tracefs_synth_raw_fmt() écrit les chaînes de format brutes (événement dynamique et histogrammes) de l'événement synthétique dans le seq donné.

tracefs_synth_show_event() renvoie le format de l'événement dynamique utilisé par l'événement synthétique ou NULL en cas d'erreur. La chaîne renvoyée appartient à l'objet d'événement synthétique et est libérée avec l'événement par tracefs_synth_free().

tracefs_synth_show_start_hist() renvoie le format de l'histogramme de départ utilisé par l'événement synthétique ou NULL en cas d'erreur. La chaîne renvoyée appartient à l'objet d'événement synthétique et est libérée avec l'événement par tracefs_synth_free().

tracefs_synth_show_end_hist() renvoie le format de l'histogramme de fin utilisé par l'événement synthétique ou NULL en cas d'erreur. La chaîne renvoyée appartient à l'objet d'événement synthétique et est libérée avec l'événement par tracefs_synth_free().

La fonction tracefs_synth_get_event() renvoie un événement tep, décrivant l'événement synthétique donné. L'API détecte tous les événements dynamiques nouvellement créés ou supprimés. Le pointeur renvoyé vers l'événement tep est contrôlé par @tep et ne doit pas être libéré.

VALEUR DE RETOUR

tracefs_synth_get_name(), tracefs_synth_show_event(), tracefs_synth_show_start_hist() et tracefs_synth_show_end_hist() renvoient une chaîne appartenant à l'objet d'événement synthétique.

La fonction tracefs_synth_get_event() renvoie un pointeur vers un événement tep ou NULL en cas d'erreur ou si l'événement synthétique demandé est manquant. Le pointeur renvoyé vers l'événement tep est contrôlé par @tep et ne doit pas être libéré.

Toutes les autres fonctions renvoient zéro en cas de succès ou -1 en cas d'erreur.

ERREURS

Les erreurs suivantes s'appliquent à tous les appels ci-dessus :

EPERM Pas exécuté en tant qu'utilisateur root lorsque requis.

EINVAL Soit un paramètre n'est pas valide (NULL alors qu'il ne devrait pas l'être), soit un champ incompatible pour les calculs.

ENODEV Un événement ou l'un de ses champs n'est pas trouvé.

EBADE Les champs des événements de début et de fin sont incompatibles pour la mise en correspondance ou la comparaison.

ENOMEM Pas assez de mémoire disponible.

D'autres erreurs peuvent survenir à partir des appels système vers le système.

EXEMPLE

Voir tracefs_sql(3) pour une utilisation plus approfondie de ce code.

#include <stdlib.h>
#include <tracefs.h>
#define start_event "sched_waking"
#define start_field "pid"
#define end_event "sched_switch"
#define end_field "next_pid"
#define match_name "pid"
static struct tracefs_synth *synth;
static void make_event(void)
{
        struct tep_handle *tep;
        /* Charger tous les événements du système */
        tep = tracefs_local_events(NULL);
        /* Initialiser l'événement synthétique */
        synth = tracefs_synth_alloc(tep, "wakeup_lat",
                                    NULL, start_event,
                                    NULL, end_event,
                                    start_field, end_field,
                                    match_name);
        /* Le tep n'est plus nécessaire */
        tep_free(tep);
        /* Enregistrer le champ "prio" sous le nom "prio" à partir de l'événement de départ */
        tracefs_synth_add_start_field(synth, "prio", NULL);
        /* Enregistrer le champ "next_comm" sous le nom "comm" à partir de l'événement de fin */
        tracefs_synth_add_end_field(synth, "next_comm", "comm");
        /* Enregistrer le champ "prev_prio" sous le nom "prev_prio" à partir de l'événement de fin */
        tracefs_synth_add_end_field(synth, "prev_prio", NULL);
        /*
         * Prendre la différence de temps en microsecondes entre la fin et le début
         * et enregistrer sous le nom "delta"
         */
        tracefs_synth_add_compare_field(synth, TRACEFS_TIMESTAMP_USECS,
                                        TRACEFS_TIMESTAMP_USECS,
                                        TRACEFS_SYNTH_DELTA_END, "delta");
        /* Enregistrer uniquement si le champ "prio" de l'événement de départ est inférieur à 100 */
        tracefs_synth_append_start_filter(synth, TRACEFS_FILTER_COMPARE,
                                          "prio", TRACEFS_COMPARE_LT, "100");
        /*
         * Enregistrer uniquement si le champ "next_prio" de l'événement de fin est inférieur à 50
         * ou si la priorité de la tâche précédente n'était pas supérieure ou égale à 100.
         *   next_prio < 50 || !(prev_prio >= 100)
         */
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_COMPARE,
                                        "next_prio", TRACEFS_COMPARE_LT, "50");
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_OR, NULL, 0, NULL);
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_NOT, NULL, 0, NULL);
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_OPEN_PAREN, NULL, 0, NULL);
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_COMPARE,
                                        "prev_prio", TRACEFS_COMPARE_GE, "100");
        /*
         * Note, ce qui précède n'a ajouté que : "next_prio < 50 || !(prev_prio >= 100"
         * En effet, lorsque le synthé est exécuté, la parenthèse fermante restante
         * sera ajoutée. C'est-à-dire que la chaîne deviendra :
         * "next_prio < 50 || !(prev_prio >= 100)" lorsqu'on exécute tracefs_sync_create()
         * ou tracefs_sync_echo_cmd().
         */
}

DESCRIPTION

La bibliothèque libtracefs fournit une interface pour interagir avec le sous-système de traçage du noyau Linux, y compris la gestion des événements synthétiques. Les événements synthétiques sont des événements qui peuvent être créés et déclenchés par des programmes en espace utilisateur, contrairement aux événements de traçage générés par le noyau lui-même.

Ces fonctions permettent de créer, de supprimer et de gérer des événements synthétiques. Elles sont utilisées pour définir les champs d'un événement, les conditions de déclenchement, et pour écrire des données dans les traces.

tracefs_synth_alloc : Alloue et initialise une structure struct tracefs_synth. Cette structure représente un événement synthétique.

tracefs_synth_add_match_field : Ajoute un champ à l'événement synthétique qui sera utilisé pour faire correspondre des événements existants dans le système de traçage. Lorsque le champ correspond à la valeur spécifiée, l'événement synthétique est déclenché.

tracefs_synth_add_compare_field : Ajoute un champ à l'événement synthétique avec une opération de comparaison. Utile pour des conditions plus complexes.

tracefs_synth_add_start_field : Ajoute un champ qui sera enregistré au début de la trace d'un événement synthétique.

tracefs_synth_add_end_field : Ajoute un champ qui sera enregistré à la fin de la trace d'un événement synthétique.

tracefs_synth_append_start_filter : Ajoute un filtre pour les événements de début.

tracefs_synth_append_end_filter : Ajoute un filtre pour les événements de fin.

tracefs_synth_complete : Finalise la définition de l'événement synthétique.

tracefs_synth_create : Crée l'événement synthétique défini dans la structure synth dans le système de fichiers de traçage.

tracefs_synth_destroy : Supprime l'événement synthétique du système de fichiers de traçage.

tracefs_synth_trace : Écrit les données de l'événement synthétique dans la séquence de traçage.

tracefs_synth_snapshot : Prend un instantané de l'événement synthétique.

tracefs_synth_save : Sauvegarde la définition de l'événement synthétique dans un fichier.

tracefs_synth_free : Libère la mémoire allouée pour la structure struct tracefs_synth.

La fonction show_event dans l'exemple démontre comment utiliser ces fonctions pour afficher les détails d'un événement synthétique.

La fonction main gère les arguments de la ligne de commande pour créer ou supprimer un événement synthétique, ou pour afficher ses détails si aucun argument n'est fourni.

VALEUR RENVOYÉE

Les fonctions qui créent, détruisent ou modifient l'événement synthétique retournent 0 en cas de succès, et une valeur négative en cas d'erreur.

ERREURS

Les fonctions peuvent échouer pour diverses raisons, telles que des erreurs d'allocation de mémoire, des problèmes d'accès au système de fichiers de traçage, ou des arguments invalides.

EXEMPLE

static void show_event(void)
{
        struct trace_seq s;
        trace_seq_init(&s);
        tracefs_synth_echo_cmd(&s, synth);
        trace_seq_terminate(&s);
        trace_seq_do_printf(&s);
        trace_seq_destroy(&s);
}
int main (int argc, char **argv)
{
        make_event();
        if (argc > 1) {
                if (!strcmp(argv[1], "create")) {
                        /* Créer l'événement synthétique */
                        tracefs_synth_create(synth);
                } else if (!strcmp(argv[1], "delete")) {
                        /* Supprimer l'événement synthétique */
                        tracefs_synth_destroy(synth);
                } else {
                        printf("usage: %s [create|delete]\n", argv[0]);
                        exit(-1);
                }
        } else
                show_event();
        tracefs_synth_free(synth);
        return 0;
}
