const char *tracefs_synth_show_event(struct tracefs_synth *synth);
const char *tracefs_synth_show_start_hist(struct tracefs_synth *synth);
const char *tracefs_synth_show_end_hist(struct tracefs_synth *synth);
struct tep_event *tracefs_synth_get_event(struct tep_handle *tep, struct tracefs_synth *synth);

DESCRIPTION

Les événements synthétiques sont des événements dynamiques créés en faisant correspondre deux autres événements qui déclenchent un événement synthétique. Un événement est l'événement de départ dont un champ est enregistré, et lorsque le second événement est exécuté, s'il possède un champ (ou des champs) qui correspond au champ (ou aux champs) de l'événement de départ, il déclenchera l'événement synthétique. Les valeurs des champs autres que les champs correspondants peuvent être transmises de l'événement de départ à l'événement de fin pour effectuer des calculs, ou simplement pour les passer comme paramètre à l'événement synthétique.

Un cas d'utilisation courant consiste à utiliser "sched_waking" comme événement de départ. Cet événement est déclenché lorsqu'un processus est réveillé. Ensuite, "sched_switch" est utilisé comme événement de fin. Cet événement est déclenché lorsqu'une nouvelle tâche est planifiée sur le CPU. En utilisant le "common_pid" des deux événements comme champs de correspondance, le temps écoulé entre les deux événements est considéré comme la latence de réveil de ce processus. Utilisez TRACEFS_TIMESTAMP comme champ pour les deux événements afin de calculer la différence en nanosecondes, ou utilisez TRACEFS_TIMESTAMP_USECS comme champs de comparaison pour les deux événements afin de calculer la différence en microsecondes. Ceci est utilisé comme exemple ci-dessous.

Voir tracefs_synth_alloc(3) pour l'allocation des événements synthétiques, et tracefs_synth_create() pour la création de l'événement synthétique sur le système.

tracefs_synth_echo_cmd() fonctionne comme tracefs_synth_create(), mais au lieu de créer l'événement synthétique dans le système, il écrira les commandes d'écho pour le créer manuellement dans le seq donné.

tracefs_synth_get_start_hist() renvoie un descripteur struct tracefs_hist décrivant l'histogramme utilisé pour créer l'événement synthétique.

enum tracefs_synth_handler {
        TRACEFS_SYNTH_HANDLE_MATCH,
        TRACEFS_SYNTH_HANDLE_MAX,
        TRACEFS_SYNTH_HANDLE_CHANGE,
};

tracefs_synth_get_name() renvoie le nom de l'événement synthétique ou NULL en cas d'erreur. La chaîne renvoyée appartient à l'objet de l'événement synthétique et est libérée avec l'événement par tracefs_synth_free().

tracefs_synth_raw_fmt() écrit les chaînes de format brutes (événement dynamique et histogrammes) de l'événement synthétique dans le seq donné.

tracefs_synth_show_event() renvoie le format de l'événement dynamique utilisé par l'événement synthétique ou NULL en cas d'erreur. La chaîne renvoyée appartient à l'objet de l'événement synthétique et est libérée avec l'événement par tracefs_synth_free().

tracefs_synth_show_start_hist() renvoie le format de l'histogramme de début utilisé par l'événement synthétique ou NULL en cas d'erreur. La chaîne renvoyée appartient à l'objet de l'événement synthétique et est libérée avec l'événement par tracefs_synth_free().

tracefs_synth_show_end_hist() renvoie le format de l'histogramme de fin utilisé par l'événement synthétique ou NULL en cas d'erreur. La chaîne renvoyée appartient à l'objet de l'événement synthétique et est libérée avec l'événement par tracefs_synth_free().

La fonction tracefs_synth_get_event() renvoie un événement tep, décrivant l'événement synthétique donné. L'API détecte tous les événements dynamiques nouvellement créés ou supprimés. Le pointeur renvoyé vers l'événement tep est contrôlé par @tep et ne doit pas être libéré.

VALEUR DE RETOUR

tracefs_synth_get_name(), tracefs_synth_show_event(), tracefs_synth_show_start_hist() et tracefs_synth_show_end_hist() renvoient une chaîne appartenant à l'objet de l'événement synthétique.

La fonction tracefs_synth_get_event() renvoie un pointeur vers un événement tep ou NULL en cas d'erreur ou si l'événement synthétique demandé est manquant. Le pointeur renvoyé vers l'événement tep est contrôlé par @tep et ne doit pas être libéré.

Toutes les autres fonctions renvoient zéro en cas de succès et -1 en cas d'erreur.

ERREURS

Les erreurs suivantes s'appliquent à tous les appels ci-dessus :

EPERM Non exécuté en tant qu'utilisateur root lorsque cela est nécessaire.

EINVAL Soit un paramètre n'est pas valide (NULL alors qu'il ne devrait pas l'être), soit un champ incompatible avec les calculs.

ENODEV Un événement ou l'un de ses champs n'est pas trouvé.

EBADE Les champs des événements de début et de fin ne sont pas compatibles pour la correspondance ou la comparaison.

ENOMEM Pas assez de mémoire disponible.

D'autres erreurs peuvent s'être produites lors des appels système au système.

Voir tracefs_sql(3) pour une utilisation plus approfondie de certains de ces codes.

#include <stdlib.h>
#include <tracefs.h>
#define start_event "sched_waking"
#define start_field "pid"
#define end_event "sched_switch"
#define end_field "next_pid"
#define match_name "pid"
static struct tracefs_synth *synth;
static void make_event(void)
{
        struct tep_handle *tep;
        /* Charger tous les événements du système */
        tep = tracefs_local_events(NULL);
        /* Initialiser l'événement synthétique */
        synth = tracefs_synth_alloc(tep, "wakeup_lat",
                                    NULL, start_event,
                                    NULL, end_event,
                                    start_field, end_field,
                                    match_name);
        /* Le tep n'est plus nécessaire */
        tep_free(tep);
        /* Sauvegarder le champ "prio" sous le nom "prio" de l'événement de départ */
        tracefs_synth_add_start_field(synth, "prio", NULL);
        /* Sauvegarder le champ "next_comm" sous le nom "comm" de l'événement de fin */
        tracefs_synth_add_end_field(synth, "next_comm", "comm");
        /* Sauvegarder le champ "prev_prio" sous le nom "prev_prio" de l'événement de fin */
        tracefs_synth_add_end_field(synth, "prev_prio", NULL);
        /*
         * Prendre la différence de temps en microsecondes entre la fin et le début
         * et l'enregistrer sous le nom "delta"
         */
        tracefs_synth_add_compare_field(synth, TRACEFS_TIMESTAMP_USECS,
                                        TRACEFS_TIMESTAMP_USECS,
                                        TRACEFS_SYNTH_DELTA_END, "delta");
        /* Enregistrer uniquement si le champ "prio" de l'événement de départ est inférieur à 100 */
        tracefs_synth_append_start_filter(synth, TRACEFS_FILTER_COMPARE,
                                          "prio", TRACEFS_COMPARE_LT, "100");
        /*
         * Enregistrer uniquement si le champ "next_prio" de l'événement de fin est inférieur à 50
         * ou si la priorité de la tâche précédente n'était pas supérieure ou égale à 100.
         *   next_prio < 50 || !(prev_prio >= 100)
         */
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_COMPARE,
                                        "next_prio", TRACEFS_COMPARE_LT, "50");
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_OR, NULL, 0, NULL);
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_NOT, NULL, 0, NULL);
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_OPEN_PAREN, NULL, 0, NULL);
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_COMPARE,
                                        "prev_prio", TRACEFS_COMPARE_GE, "100");
        /*
         * Notez que ce qui précède n'a ajouté que : "next_prio < 50 || !(prev_prio >= 100"
         * C'est parce que, lorsque le synthé sera exécuté, la parenthèse fermante restante
         * sera ajoutée. C'est-à-dire que la chaîne se terminera par :
         * "next_prio < 50 || !(prev_prio >= 100)" lorsqu'une des fonctions tracefs_sync_create()
         * ou tracefs_sync_echo_cmd() sera exécutée.
         */
}
