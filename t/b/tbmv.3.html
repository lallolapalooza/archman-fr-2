sous-routine ctbmv (caractère uplo, caractère trans, caractère diag, entier n, entier k, complexe, dimension(lda,*) a, entier lda, complexe, dimension(*) x, entier incx)

CTBMV

Objectif :

!>
!> CTBMV  effectue l'une des opérations matrice-vecteur suivantes :
!>
!>    x := A*x,   ou   x := A**T*x,   ou   x := A**H*x,
!>
!> où x est un vecteur de n éléments et  A est une matrice bande
!> triangulaire supérieure ou inférieure de taille n par n, unitaire ou non,
!> avec ( k + 1 ) diagonales.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>           En entrée, UPLO spécifie si la matrice est une matrice triangulaire
!>           supérieure ou inférieure comme suit :
!>
!>              UPLO = 'U' ou 'u'   A est une matrice triangulaire supérieure.
!>
!>              UPLO = 'L' ou 'l'   A est une matrice triangulaire inférieure.
!>

TRANS

!>          TRANS est CHARACTER*1
!>           En entrée, TRANS spécifie l'opération à effectuer comme suit :
!>
!>              TRANS = 'N' ou 'n'   x := A*x.
!>
!>              TRANS = 'T' ou 't'   x := A**T*x.
!>
!>              TRANS = 'C' ou 'c'   x := A**H*x.
!>

DIAG

!>          DIAG est CHARACTER*1
!>           En entrée, DIAG spécifie si A est unitaire ou non :
!>
!>              DIAG = 'U' ou 'u'   A est supposée être unitaire.
!>
!>              DIAG = 'N' ou 'n'   A n'est pas supposée être unitaire.
!>

N

!>          N est INTEGER
!>           En entrée, N spécifie l'ordre de la matrice A.
!>           N doit être supérieur ou égal à zéro.
!>

K

!>          K est INTEGER
!>           En entrée, si UPLO = 'U' ou 'u', K spécifie le nombre de
!>           super-diagonales de la matrice A.
!>           En entrée, si UPLO = 'L' ou 'l', K spécifie le nombre de
!>           sous-diagonales de la matrice A.
!>           K doit satisfaire  0 .le. K.
!>

A

!>          A est un tableau COMPLEX, de dimension ( LDA, N ).
!>           Avant l'entrée si UPLO = 'U' ou 'u', la partie principale ( k + 1 )
!>           par n du tableau A doit contenir la partie triangulaire
!>           supérieure en bande de la matrice des coefficients, fournie colonne
!>           par colonne, avec la diagonale principale de la matrice dans la ligne
!>           ( k + 1 ) du tableau, la première super-diagonale commençant à la
!>           position 2 en ligne k, et ainsi de suite. Le triangle supérieur gauche
!>           de k par k du tableau A n'est pas référencé.
!>           Le segment de programme suivant transférera une matrice bande triangulaire
!>           supérieure depuis un stockage matriciel complet conventionnel vers un
!>           stockage en bande :
!>
!>                 DO 20, J = 1, N
!>                    M = K + 1 - J
!>                    DO 10, I = MAX( 1, J - K ), J
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           Avant l'entrée si UPLO = 'L' ou 'l', la partie principale ( k + 1 )
!>           par n du tableau A doit contenir la partie triangulaire
!>           inférieure en bande de la matrice des coefficients, fournie colonne
!>           par colonne, avec la diagonale principale de la matrice dans la ligne
!>           1 du tableau, la première sous-diagonale commençant à la position 1
!>           en ligne 2, et ainsi de suite. Le triangle inférieur droit de k par k
!>           du tableau A n'est pas référencé.
!>           Le segment de programme suivant transférera une matrice bande triangulaire
!>           inférieure depuis un stockage matriciel complet conventionnel vers un
!>           stockage en bande :
!>
!>                 DO 20, J = 1, N
!>                    M = 1 - J
!>                    DO 10, I = J, MIN( N, J + K )
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           Notez que lorsque DIAG = 'U' ou 'u', les éléments du tableau A
!>           correspondant aux éléments diagonaux de la matrice ne sont pas
!>           référencés, mais sont supposés être unitaires.
!>

LDA

!>          LDA est INTEGER
!>           En entrée, LDA spécifie la première dimension de A telle que déclarée
!>           dans le programme appelant (ou sous-programme). LDA doit être au moins
!>           ( k + 1 ).
!>

X

!>          X est un tableau COMPLEX, de dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           Avant l'entrée, le tableau incrémenté X doit contenir le vecteur
!>           x de n éléments. En sortie, X est écrasé par le vecteur x transformé.
!>

INCX

!>          INCX est INTEGER
!>           En entrée, INCX spécifie l'incrément pour les éléments de
!>           X. INCX ne doit pas être nul.
!>

SUBROUTINE ZTBMV(UPLO, TRANS, DIAG, N, K, A, LDA, X, INCX)

                        
      CHARACTER * 1 UPLO, TRANS, DIAG

                        
      INTEGER N, K, LDA, INCX

                        
      COMPLEX * 16 A( LDA, * ), X( * )

                        
      "

                        
      "  -- Translated by JULIA, 2012.

                        
      "  -- By default, the precision is double complex.

                        
      "

                        
      "     ZTBMV ...

                        
      "

                        
      "     Arguments

                        
      "

                        
      "  UPLO    - CHARACTER*1

                        
      "           On entry, UPLO specifies whether the matrix A is

                        
      "           upper or lower triangular as follows:

                        
      "              UPLO = 'U' or 'u'   A is upper triangular

                        
      "              UPLO = 'L' or 'l'   A is lower triangular

                        
      "

                        
      "  TRANS   - CHARACTER*1

                        
      "           On entry, TRANS specifies the operation to be performed:

                        
      "              TRANS = 'N' or 'n'   x := A*x

                        
      "              TRANS = 'T' or 't'   x := A**T*x

                        
      "              TRANS = 'C' or 'c'   x := A**H*x

                        
      "

                        
      "  DIAG    - CHARACTER*1

                        
      "           On entry, DIAG specifies whether or not A is unit

                        
      "           triangular as follows:

                        
      "              DIAG = 'U' or 'u'   A is assumed to be unit triangular.

                        
      "              DIAG = 'N' or 'n'   A is not assumed to be unit

                        
      "                                  triangular.

                        
      "

                        
      "  N       - INTEGER

                        
      "           On entry, N specifies the order of the matrix A.

                        
      "           N must be at least zero.

                        
      "

                        
      "  K       - INTEGER

                        
      "           On entry with UPLO = 'U' or 'u', K specifies the number of

                        
      "           super-diagonals of the matrix A.

                        
      "           On entry with UPLO = 'L' or 'l', K specifies the number of

                        
      "           sub-diagonals of the matrix A.

                        
      "           K must satisfy  0 .le. K.

                        
      "

                        
      "  A       - COMPLEX*16 array, dimension ( LDA, N ).

                        
      "           Before entry with UPLO = 'U' or 'u', the leading ( k + 1 )

                        
      "           by n part of the array A must contain the upper triangular

                        
      "           band part of the matrix of coefficients, supplied column by

                        
      "           column, with the leading diagonal of the matrix in row

                        
      "           ( k + 1 ) of the array, the first super-diagonal starting at

                        
      "           position 2 in row k, and so on. The top left k by k triangle

                        
      "           of the array A is not referenced.

                        
      "           The following program segment will transfer an upper

                        
      "           triangular band matrix from conventional full matrix storage

                        
      "           to band storage:

                        
      "

                        
      "                 DO 20, J = 1, N

                        
      "                    M = K + 1 - J

                        
      "                    DO 10, I = MAX( 1, J - K ), J

                        
      "                       A( M + I, J ) = matrix( I, J )

                        
      "              10    CONTINUE

                        
      "              20 CONTINUE

                        
      "

                        
      "           Before entry with UPLO = 'L' or 'l', the leading ( k + 1 )

                        
      "           by n part of the array A must contain the lower triangular

                        
      "           band part of the matrix of coefficients, supplied column by

                        
      "           column, with the leading diagonal of the matrix in row 1 of

                        
      "           the array, the first sub-diagonal starting at position 1 in

                        
      "           row 2, and so on. The bottom right k by k triangle of the

                        
      "           array A is not referenced.

                        
      "           The following program segment will transfer a lower

                        
      "           triangular band matrix from conventional full matrix storage

                        
      "           to band storage:

                        
      "

                        
      "                 DO 20, J = 1, N

                        
      "                    M = 1 - J

                        
      "                    DO 10, I = J, MIN( N, J + K )

                        
      "                       A( M + I, J ) = matrix( I, J )

                        
      "              10    CONTINUE

                        
      "              20 CONTINUE
