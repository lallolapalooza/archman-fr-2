RPC(3) Manuel des fonctions de bibliothèque RPC(3)

NOM

rpc — routines de bibliothèque pour les appels de procédure distante

SYNTAXE

#include <rpc/rpc.h>
#include <netconfig.h>

DESCRIPTION

Ces routines permettent aux programmes en langage C d'effectuer des appels de procédure sur d'autres machines à travers un réseau. D'abord, le client envoie une requête au serveur. À la réception de la requête, le serveur appelle une routine de dispatch pour effectuer le service demandé, puis renvoie une réponse.

Toutes les routines RPC nécessitent l'en-tête <rpc/rpc.h>. Les routines qui prennent une struct netconfig nécessitent également que <netconfig.h> soit incluse.

Nettype

Certaines des routines d'interface RPC de haut niveau prennent une chaîne nettype comme l'un des arguments (par exemple, clnt_create(), svc_create(), rpc_reg(), rpc_call()). Cette chaîne définit une classe de transports qui peuvent être utilisés pour une application particulière.

L'argument nettype peut être l'un des suivants :

netpath
Choisir parmi les transports indiqués par leurs noms de jetons dans la variable d'environnement NETPATH. Si NETPATH est non définie ou NULL, elle prend par défaut "visible". "netpath" est le nettype par défaut.
visible
Choisir les transports pour lesquels le drapeau visible (v) est défini dans le fichier /etc/netconfig.
circuit_v
Identique à "visible" sauf qu'il choisit uniquement les transports orientés connexion (sémantique "tpi_cots" ou "tpi_cots_ord") parmi les entrées du fichier /etc/netconfig.
datagram_v
Identique à "visible" sauf qu'il choisit uniquement les transports de datagrammes sans connexion (sémantique "tpi_clts") parmi les entrées du fichier /etc/netconfig.
circuit_n
Identique à "netpath" sauf qu'il choisit uniquement les transports de datagrammes orientés connexion (sémantique "tpi_cots" ou "tpi_cots_ord").
datagram_n
Identique à "netpath" sauf qu'il choisit uniquement les transports de datagrammes sans connexion (sémantique "tpi_clts").
udp
Fait référence à UDP Internet, versions 4 et 6.
tcp
Fait référence à TCP Internet, versions 4 et 6.

Si nettype est NULL, elle prend par défaut "netpath". Les transports sont essayés dans l'ordre de gauche à droite dans la variable NETPATH ou dans l'ordre de haut en bas dans le fichier /etc/netconfig.

Structures de données

Certaines des structures de données utilisées par le package RPC sont présentées ci-dessous.

La structure AUTH

/*
 * Informations d'authentification. Opaques pour le client.
 */
struct opaque_auth {
    enum_t    oa_flavor;    /* saveur d'authentification */
    caddr_t    oa_base;    /* adresse de données d'authentification supplémentaires */
    u_int    oa_length;    /* ne doit pas dépasser MAX_AUTH_BYTES */
};

/*
 * Handle d'authentification, interface pour les authentificateurs côté client.
 */
typedef struct {
    struct    opaque_auth    ah_cred;
    struct    opaque_auth    ah_verf;
    struct auth_ops {
        void    (*ah_nextverf)();
        int    (*ah_marshal)();    /* nextverf & sérialiser */
        int    (*ah_validate)();    /* valider le vérificateur */
        int    (*ah_refresh)();    /* rafraîchir les identifiants */
        void    (*ah_destroy)();    /* détruire cette structure */
    } *ah_ops;
    caddr_t ah_private;
} AUTH;

La structure CLIENT

/*
 * Handle RPC client.
 * Créé par des implémentations individuelles.
 * Le client est responsable de l'initialisation de l'authentification.
 */

typedef struct {
    AUTH    *cl_auth;    /* authentificateur */
    struct clnt_ops {
        enum clnt_stat    (*cl_call)();    /* appeler une procédure distante */
        void    (*cl_abort)();        /* aborder un appel */
        void    (*cl_geterr)();        /* obtenir un code d'erreur spécifique */
        bool_t    (*cl_freeres)();    /* libérer les résultats */
        void    (*cl_destroy)();    /* détruire cette structure */
        bool_t    (*cl_control)();    /* l'ioctl() de rpc */
    } *cl_ops;
    caddr_t    cl_private;    /* données privées */
    char    *cl_netid;    /* identifiant réseau */
    char    *cl_tp;        /* nom du périphérique */
} CLIENT;

La structure SVCXPRT

enum xprt_stat {
    XPRT_DIED,
    XPRT_MOREREQS,
    XPRT_IDLE
};

/*
 * Handle de transport côté serveur
 */
typedef struct {
    int    xp_fd;    /* descripteur de fichier pour le handle serveur */
    u_short    xp_port;    /* obsolète */
    const struct xp_ops {
        bool_t    (*xp_recv)();    /* recevoir les requêtes entrantes */
        enum xprt_stat    (*xp_stat)();    /* obtenir l'état du transport */
        bool_t    (*xp_getargs)();    /* obtenir les arguments */
        bool_t    (*xp_reply)();      /* envoyer la réponse */
        bool_t    (*xp_freeargs)(); /* libérer la mémoire allouée pour les arguments */
        void    (*xp_destroy)();    /* détruire cette structure */
    } *xp_ops;
    int    xp_addrlen;    /* longueur de l'adresse distante. Obsolète */
    struct sockaddr_in    xp_raddr; /* Obsolète */
    const struct xp_ops2 {
        bool_t    (*xp_control)();    /* fonction fourre-tout */
    } *xp_ops2;
    char    *xp_tp;    /* nom du périphérique fournisseur de transport */
    char    *xp_netid;    /* identifiant réseau */
    struct netbuf    xp_ltaddr;    /* adresse de transport locale */
    struct netbuf    xp_rtaddr;    /* adresse de transport distante */
    struct opaque_auth    xp_verf;    /* vérificateur de réponse brute */
    caddr_t    xp_p1;    /* privé : à utiliser par les ops svc */
    caddr_t    xp_p2;    /* privé : à utiliser par les ops svc */
    caddr_t    xp_p3;    /* privé : à utiliser par la lib svc */
    int    xp_type    /* type de transport */
} SVCXPRT;

La structure svc_reg

struct svc_req {
    rpcprog_t    rq_prog;    /* numéro de programme du service */
    rpcvers_t    rq_vers;    /* version du protocole du service */
    rpcproc_t    rq_proc;    /* la procédure souhaitée */
    struct opaque_auth    rq_cred;    /* identifiants bruts reçus du réseau */
    caddr_t    rq_clntcred;    /* identifiants formatés en lecture seule */
    SVCXPRT    *rq_xprt;    /* transport associé */
};

La structure XDR

/*
 * Opérations XDR.
 * XDR_ENCODE provoque l'encodage du type dans le flux.
 * XDR_DECODE provoque l'extraction du type du flux.
 * XDR_FREE peut être utilisé pour libérer l'espace alloué par une demande XDR_DECODE.
 */
enum xdr_op {
    XDR_ENCODE=0,
    XDR_DECODE=1,
    XDR_FREE=2
};
/*
 * C'est le nombre d'octets par unité de données externes.
 */
#define BYTES_PER_XDR_UNIT    (4)
#define RNDUP(x)  ((((x) + BYTES_PER_XDR_UNIT - 1) /
                   BYTES_PER_XDR_UNIT) \ * BYTES_PER_XDR_UNIT)

 * taille appropriée et le retourner.
