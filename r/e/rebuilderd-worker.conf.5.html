rebuilderd-worker.conf(5) Manuel des formats de fichiers rebuilderd-worker.conf(5)

NOM

rebuilderd-worker.conf - Fichiers de configuration de rebuilderd-worker

SYNTAXE

/etc/rebuilderd-worker.conf

DESCRIPTION

Ce fichier configure rebuilder-worker connect et spécifie à la fois le point de terminaison et le secret d'inscription. Les deux sont optionnels, mais le point de terminaison doit être explicitement fourni en ligne de commande s'il n'est pas configuré dans le fichier.

OPTIONS

endpoint=

Le serveur rebuilderd auquel se connecter.

signup_secret=

Le serveur autorisera notre clé ou exigera un secret d'inscription.

[build]

timeout=

Définit un délai d'attente en secondes après lequel diffoscope sera terminé (par défaut 24 heures).

max_bytes=

Définit une limite maximale en octets pour la sortie de diffoscope (par défaut : aucune). Lorsque cette limite est atteinte, le journal est tronqué mais le backend rebuilder n'est pas terminé.

silent=

Par défaut, la sortie de la compilation est transmise à stdout/stderr. Cela peut être désactivé en réglant cette option sur true.

[diffoscope]

enabled=

Génère et attache des différences avec diffoscope lors de la reconstruction.

args=

Passe des arguments supplémentaires à diffoscope. À utiliser avec prudence, certaines options peuvent ne pas fonctionner correctement.

timeout=

Définit un délai d'attente en secondes après lequel diffoscope sera terminé (par défaut 1 heure).

max_bytes=

Définit une limite maximale en octets pour la sortie de diffoscope (par défaut : aucune). Lorsque cette limite est atteinte, diffoscope est terminé et la sortie est tronquée.

EXEMPLE

## Le serveur rebuilderd auquel se connecter
endpoint = "http://127.0.0.1:8484"
## Le serveur autorisera notre clé ou exigera un secret d'inscription
#signup_secret = "votre_cle_inscription"
[build]
#timeout = 86400 # 24 heures
## Définit une limite maximale pour le journal de compilation en octets (par défaut : aucune).
## Lorsque cette limite est atteinte, le journal est tronqué mais le backend rebuilder N'EST PAS terminé.
max_bytes = 10485760 # 10 MiB
[diffoscope]
## Génère et attache des différences avec diffoscope lors de la reconstruction
enabled = false
## Passe des arguments supplémentaires à diffoscope. À utiliser avec prudence, certaines options peuvent ne pas fonctionner correctement.
#args = ["--max-container-depth", "2", "--fuzzy-threshold", "0"]
## Définit un délai d'attente en secondes après lequel diffoscope sera terminé (par défaut 3600)
#timeout = 600 # 10 minutes
## Définit une limite maximale pour la sortie de diffoscope en octets (par défaut : aucune).
## Lorsque cette limite est atteinte, diffoscope est terminé et la sortie est tronquée.
max_bytes = 41943040 # 40 MiB
