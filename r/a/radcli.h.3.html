struct send_data {
  uint8_t      code;          // code de paquet RADIUS. 
  uint8_t      seq_nbr;       // numéro de séquence du paquet. 
  char        *server;        // Nom/adresse du serveur RADIUS. 
  int          svc_port;      // port de destination du protocole RADIUS. 
  char        *secret;        // Secret partagé du serveur RADIUS. 
  int          timeout;       // délai d'attente de session en secondes. 
  int          retries;
  VALUE_PAIR  *send_pairs;    // Paires a/v supplémentaires à envoyer. 
  VALUE_PAIR  *receive_pairs; // Où placer les paires a/v reçues. 
};

Évitez d'utiliser cette structure directement, elle est incluse pour des raisons de compatibilité ascendante uniquement. Plusieurs de ses champs ont été dépréciés.


struct server {
  int      max;
  char    *name[RC_SERVER_MAX];
  uint16_t port[RC_SERVER_MAX];
  char    *secret[RC_SERVER_MAX];
  double   deadtime_ends[RC_SERVER_MAX]; // non utilisé 
};

Évitez d'utiliser cette structure directement. Utilisez les fonctions rc_avpair_get_.


struct rc_value_pair {
  char                   name[RC_NAME_LENGTH+1];      // nom de l'attribut si connu. 
  uint64_t               attribute; // valeur numérique de l'attribut de type rc_attr_id incluant le vendeur ; utilisez VENDOR() et ATTRID() pour séparer. 
  rc_attr_type           type;      // type de l'attribut. 
  uint32_t               lvalue;    // valeur de l'attribut si le type est PW_TYPE_INTEGER, PW_TYPE_DATE ou PW_TYPE_IPADDR. 
  char                   strvalue[AUTH_STRING_LEN+1];  // contient la valeur de l'attribut dans les autres cas. 
  struct rc_value_pair  *next;
  char                   pad[32];       // pad non utilisé 
};
