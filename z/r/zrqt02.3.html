subroutine zrqt02 (integer m, integer n, integer k, complex*16, dimension( lda, * ) a, complex*16, dimension( lda, * ) af, complex*16, dimension( lda, * ) q, complex*16, dimension( lda, * ) r, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( * ) result)

ZRQT02

Objectif :

!>
!> ZRQT02 teste ZUNGRQ, qui génère une matrice Q de taille m x n avec
!> des lignes orthonormées, définie comme le produit de k réflecteurs
!> élémentaires.
!>
!> Étant donnée la factorisation RQ d'une matrice A de taille m x n, ZRQT02 génère
!> la matrice orthogonale Q définie par la factorisation des k dernières lignes
!> de A ; elle compare R(m-k+1:m,n-m+1:n) avec
!> A(m-k+1:m,1:n)*Q(n-m+1:n,1:n)', et vérifie que les lignes de Q sont
!> orthonormées.
!>

Paramètres

M 
!>          M est un entier
!>          Le nombre de lignes de la matrice Q à générer.  M >= 0.
!>

N

!>          N est un entier
!>          Le nombre de colonnes de la matrice Q à générer.
!>          N >= M >= 0.
!>

K

!>          K est un entier
!>          Le nombre de réflecteurs élémentaires dont le produit définit la
!>          matrice Q. M >= K >= 0.
!>

A

!>          A est un tableau COMPLEX*16, de dimension (LDA,N)
!>          La matrice A de taille m x n qui a été factorisée par ZRQT01.
!>

AF

!>          AF est un tableau COMPLEX*16, de dimension (LDA,N)
!>          Détails de la factorisation RQ de A, tels que retournés par ZGERQF.
!>          Voir ZGERQF pour plus de détails.
!>

Q

!>          Q est un tableau COMPLEX*16, de dimension (LDA,N)
!>

R

!>          R est un tableau COMPLEX*16, de dimension (LDA,M)
!>

LDA

!>          LDA est un entier
!>          La dimension principale des tableaux A, AF, Q et L. LDA >= N.
!>

TAU

!>          TAU est un tableau COMPLEX*16, de dimension (M)
!>          Les facteurs scalaires des réflecteurs élémentaires correspondant
!>          à la factorisation RQ dans AF.
!>

WORK

!>          WORK est un tableau COMPLEX*16, de dimension (LWORK)
!>

LWORK

!>          LWORK est un entier
!>          La dimension du tableau WORK.
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, de dimension (M)
!>

RESULT

!>          RESULT est un tableau DOUBLE PRECISION, de dimension (2)
!>          Les ratios de test :
!>          RESULT(1) = norm( R - A*Q' ) / ( N * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q*Q' ) / ( N * EPS )
!>
