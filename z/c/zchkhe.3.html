sous-routine zchkhe (logique, dimension( * ) dotype, entier nn, entier, dimension( * ) nval, entier nnb, entier, dimension( * ) nbval, entier nns, entier, dimension( * ) nsval, double précision thresh, logique tsterr, entier nmax, complexe*16, dimension( * ) a, complexe*16, dimension( * ) afac, complexe*16, dimension( * ) ainv, complexe*16, dimension( * ) b, complexe*16, dimension( * ) x, complexe*16, dimension( * ) xact, complexe*16, dimension( * ) work, double précision, dimension( * ) rwork, entier, dimension( * ) iwork, entier nout)

ZCHKHE

Objectif :

!>
!> ZCHKHE teste ZHETRF, -TRI2, -TRS, -TRS2, -RFS, et -CON.
!>

Paramètres

DOTYPE 
!>          DOTYPE est un tableau LOGIQUE, de dimension (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .VRAI. ; si DOTYPE(j) = .FAUX., alors le type j n'est pas utilisé.
!>

NN

!>          NN est un ENTIER
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau d'ENTIERS, de dimension (NN)
!>          Les valeurs de la dimension de la matrice N.
!>

NNB

!>          NNB est un ENTIER
!>          Le nombre de valeurs de NB contenues dans le vecteur NBVAL.
!>

NBVAL

!>          NBVAL est un tableau d'ENTIERS, de dimension (NNB)
!>          Les valeurs de la taille de bloc NB.
!>

NNS

!>          NNS est un ENTIER
!>          Le nombre de valeurs de NRHS contenues dans le vecteur NSVAL.
!>

NSVAL

!>          NSVAL est un tableau d'ENTIERS, de dimension (NNS)
!>          Les valeurs du nombre de membres du second membre NRHS.
!>

THRESH

!>          THRESH est une DOUBLE PRÉCISION
!>          La valeur seuil pour les ratios de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour
!>          afficher chaque ratio de test, utilisez THRESH = 0.
!>

TSTERR

!>          TSTERR est une LOGIQUE
!>          Indicateur qui spécifie si les sorties d'erreur doivent être testées.
!>

NMAX

!>          NMAX est un ENTIER
!>          La valeur maximale permise pour N, utilisée pour dimensionner les
!>          tableaux de travail.
!>

A

!>          A est un tableau COMPLEXE*16, de dimension (NMAX*NMAX)
!>

AFAC

!>          AFAC est un tableau COMPLEXE*16, de dimension (NMAX*NMAX)
!>

AINV

!>          AINV est un tableau COMPLEXE*16, de dimension (NMAX*NMAX)
!>

B

!>          B est un tableau COMPLEXE*16, de dimension (NMAX*NSMAX)
!>          où NSMAX est la plus grande entrée dans NSVAL.
!>

X

!>          X est un tableau COMPLEXE*16, de dimension (NMAX*NSMAX)
!>

XACT

!>          XACT est un tableau COMPLEXE*16, de dimension (NMAX*NSMAX)
!>

WORK

!>          WORK est un tableau COMPLEXE*16, de dimension (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK est un tableau DOUBLE PRÉCISION, de dimension (max(NMAX,2*NSMAX))
!>

IWORK

!>          IWORK est un tableau ENTIER, de dimension (NMAX)
!>

NOUT

!>          NOUT est un ENTIER
!>          Le numéro de l'unité de sortie.
!>
