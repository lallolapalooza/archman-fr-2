subroutine zckglm (integer nn, integer, dimension( * ) nval, integer, dimension( * ) mval, integer, dimension( * ) pval, integer nmats, integer, dimension( 4 ) iseed, double precision thresh, integer nmax, complex*16, dimension( * ) a, complex*16, dimension( * ) af, complex*16, dimension( * ) b, complex*16, dimension( * ) bf, complex*16, dimension( * ) x, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer nin, integer nout, integer info)

ZCKGLM

But:

!>
!> ZCKGLM teste ZGGGLM - sousroutine pour la résolution du problème
!>                       du modèle linéaire généralisé.
!>

Paramètres

NN 
!>          NN est un entier
!>          Le nombre de valeurs de N, M et P contenues dans les vecteurs
!>          NVAL, MVAL et PVAL.
!>

NVAL

!>          NVAL est un tableau d'entiers, dimension (NN)
!>          Les valeurs de la dimension N des lignes de la matrice.
!>

MVAL

!>          MVAL est un tableau d'entiers, dimension (NN)
!>          Les valeurs de la dimension M des colonnes de la matrice.
!>

PVAL

!>          PVAL est un tableau d'entiers, dimension (NN)
!>          Les valeurs de la dimension P des colonnes de la matrice.
!>

NMATS

!>          NMATS est un entier
!>          Le nombre de types de matrices à tester pour chaque combinaison
!>          de dimensions matricielles. Si NMATS >= NTYPES (le nombre
!>          maximum de types de matrices), alors tous les types différents
!>          sont générés pour le test. Si NMATS < NTYPES, une autre ligne
!>          d'entrée est lue pour obtenir les numéros des types de matrices
!>          à utiliser.
!>

ISEED

!>          ISEED est un tableau d'entiers, dimension (4)
!>          À l'entrée, la graine du générateur de nombres aléatoires. Les
!>          éléments du tableau doivent être compris entre 0 et 4095,
!>          sinon ils seront réduits modulo 4096, et ISEED(4) doit être impair.
!>          À la sortie, la graine suivante dans la séquence de nombres
!>          aléatoires après la génération de toutes les matrices de test.
!>

THRESH

!>          THRESH est un double précision
!>          La valeur seuil pour les rapports de test. Un résultat est
!>          inclus dans le fichier de sortie si RESID >= THRESH. Pour
!>          imprimer chaque rapport de test, utilisez THRESH = 0.
!>

NMAX

!>          NMAX est un entier
!>          La valeur maximale autorisée pour M ou N, utilisée dans la
!>          dimensionnement des tableaux de travail.
!>

A

!>          A est un tableau COMPLEX*16, dimension (NMAX*NMAX)
!>

AF

!>          AF est un tableau COMPLEX*16, dimension (NMAX*NMAX)
!>

B

!>          B est un tableau COMPLEX*16, dimension (NMAX*NMAX)
!>

BF

!>          BF est un tableau COMPLEX*16, dimension (NMAX*NMAX)
!>

X

!>          X est un tableau COMPLEX*16, dimension (4*NMAX)
!>

RWORK

!>          RWORK est un tableau de double précision, dimension (NMAX)
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension (NMAX*NMAX)
!>

NIN

!>          NIN est un entier
!>          Le numéro d'unité pour l'entrée.
!>

NOUT

!>          NOUT est un entier
!>          Le numéro d'unité pour la sortie.
!>

INFO

!>          INFO est un entier
!>          = 0 : sortie réussie
!>          > 0 : Si ZLATMS renvoie un code d'erreur, sa valeur absolue
!>                est retournée.
!>
