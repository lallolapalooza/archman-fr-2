subroutine zcsdts (integer m, integer p, integer q, complex*16, dimension( ldx, * ) x, complex*16, dimension( ldx, * ) xf, integer ldx, complex*16, dimension( ldu1, * ) u1, integer ldu1, complex*16, dimension( ldu2, * ) u2, integer ldu2, complex*16, dimension( ldv1t, * ) v1t, integer ldv1t, complex*16, dimension( ldv2t, * ) v2t, integer ldv2t, double precision, dimension( * ) theta, integer, dimension( * ) iwork, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( 15 ) result)

ZCSDTS

Objectif:

!>
!> ZCSDTS teste ZUNCSD, qui, étant donné une matrice unitaire partitionnée M-par-M
!> X,
!>              Q  M-Q
!>       X = [ X11 X12 ] P   ,
!>           [ X21 X22 ] M-P
!>
!> calcule le CSD (Canonial Symmetric Decomposition)
!>
!>       [ U1    ]**T * [ X11 X12 ] * [ V1    ]
!>       [    U2 ]      [ X21 X22 ]   [    V2 ]
!>
!>                             [  I  0  0 |  0  0  0 ]
!>                             [  0  C  0 |  0 -S  0 ]
!>                             [  0  0  0 |  0  0 -I ]
!>                           = [---------------------] = [ D11 D12 ] .
!>                             [  0  0  0 |  I  0  0 ]   [ D21 D22 ]
!>                             [  0  S  0 |  0  C  0 ]
!>                             [  0  0  I |  0  0  0 ]
!>
!> et aussi SORCSD2BY1, qui, étant donné
!>          Q
!>       [ X11 ] P   ,
!>       [ X21 ] M-P
!>
!> calcule le CSD 2-par-1
!>
!>                                     [  I  0  0 ]
!>                                     [  0  C  0 ]
!>                                     [  0  0  0 ]
!>       [ U1    ]**T * [ X11 ] * V1 = [----------] = [ D11 ] ,
!>       [    U2 ]      [ X21 ]        [  0  0  0 ]   [ D21 ]
!>                                     [  0  S  0 ]
!>                                     [  0  0  I ]
!>

Paramètres

M 
!>          M est un entier (INTEGER)
!>          Le nombre de lignes de la matrice X. M >= 0.
!>

P

!>          P est un entier (INTEGER)
!>          Le nombre de lignes de la matrice X11. P >= 0.
!>

Q

!>          Q est un entier (INTEGER)
!>          Le nombre de colonnes de la matrice X11. Q >= 0.
!>

X

!>          X est un tableau de type COMPLEX*16, de dimension (LDX,M)
!>          La matrice M-par-M X.
!>

XF

!>          XF est un tableau de type COMPLEX*16, de dimension (LDX,M)
!>          Détails du CSD de X, tel que retourné par ZUNCSD;
!>          voir ZUNCSD pour plus de détails.
!>

LDX

!>          LDX est un entier (INTEGER)
!>          La dimension principale des tableaux X et XF.
!>          LDX >= max( 1,M ).
!>

U1

!>          U1 est un tableau de type COMPLEX*16, de dimension (LDU1,P)
!>          La matrice unitaire P-par-P U1.
!>

LDU1

!>          LDU1 est un entier (INTEGER)
!>          La dimension principale du tableau U1. LDU >= max(1,P).
!>

U2

!>          U2 est un tableau de type COMPLEX*16, de dimension (LDU2,M-P)
!>          La matrice unitaire (M-P)-par-(M-P) U2.
!>

LDU2

!>          LDU2 est un entier (INTEGER)
!>          La dimension principale du tableau U2. LDU >= max(1,M-P).
!>

V1T

!>          V1T est un tableau de type COMPLEX*16, de dimension (LDV1T,Q)
!>          La matrice unitaire Q-par-Q V1T.
!>

LDV1T

!>          LDV1T est un entier (INTEGER)
!>          La dimension principale du tableau V1T. LDV1T >=
!>          max(1,Q).
!>

V2T

!>          V2T est un tableau de type COMPLEX*16, de dimension (LDV2T,M-Q)
!>          La matrice unitaire (M-Q)-par-(M-Q) V2T.
!>

LDV2T

!>          LDV2T est un entier (INTEGER)
!>          La dimension principale du tableau V2T. LDV2T >=
!>          max(1,M-Q).
!>

THETA

!>          THETA est un tableau de type DOUBLE PRECISION, de dimension MIN(P,M-P,Q,M-Q)
!>          Les valeurs CS de X; les matrices diagonales essentielles C et
!>          S sont construites à partir de THETA; voir la sous-routine ZUNCSD pour
!>          les détails.
!>

IWORK

!>          IWORK est un tableau de type INTEGER, de dimension (M)
!>

WORK

!>          WORK est un tableau de type COMPLEX*16, de dimension (LWORK)
!>

LWORK

!>          LWORK est un entier (INTEGER)
!>          La dimension du tableau WORK
!>

RWORK

!>          RWORK est un tableau de type DOUBLE PRECISION
!>

RESULT

!>          RESULT est un tableau de type DOUBLE PRECISION, de dimension (15)
!>          Les ratios de test :
!>          Premièrement, le CSD 2-par-2 :
!>          RESULT(1) = norm( U1'*X11*V1 - D11 ) / ( MAX(1,P,Q)*EPS2 )
!>          RESULT(2) = norm( U1'*X12*V2 - D12 ) / ( MAX(1,P,M-Q)*EPS2 )
!>          RESULT(3) = norm( U2'*X21*V1 - D21 ) / ( MAX(1,M-P,Q)*EPS2 )
!>          RESULT(4) = norm( U2'*X22*V2 - D22 ) / ( MAX(1,M-P,M-Q)*EPS2 )
!>          RESULT(5) = norm( I - U1'*U1 ) / ( MAX(1,P)*ULP )
!>          RESULT(6) = norm( I - U2'*U2 ) / ( MAX(1,M-P)*ULP )
!>          RESULT(7) = norm( I - V1T'*V1T ) / ( MAX(1,Q)*ULP )
!>          RESULT(8) = norm( I - V2T'*V2T ) / ( MAX(1,M-Q)*ULP )
!>          RESULT(9) = 0        si THETA est dans l'ordre croissant et
!>                               tous les angles sont dans [0,pi/2]
!>                    = ULPINV   sinon.
!>          Ensuite, le CSD 2-par-1 :
!>          RESULT(10) = norm( U1'*X11*V1 - D11 ) / ( MAX(1,P,Q)*EPS2 )
!>          RESULT(11) = norm( U2'*X21*V1 - D21 ) / ( MAX(1,M-P,Q)*EPS2 )
!>          RESULT(12) = norm( I - U1'*U1 ) / ( MAX(1,P)*ULP )
!>          RESULT(13) = norm( I - U2'*U2 ) / ( MAX(1,M-P)*ULP )
!>          RESULT(14) = norm( I - V1T'*V1T ) / ( MAX(1,Q)*ULP )
!>          RESULT(15) = 0        si THETA est dans l'ordre croissant et
!>                                tous les angles sont dans [0,pi/2]
!>                     = ULPINV   sinon.
!>          ( EPS2 = MAX( norm( I - X'*X ) / M, ULP ). )
!>
