subroutine zchkunhr_col (double precision thresh, logical tsterr, integer nm, integer, dimension( * ) mval, integer nn, integer, dimension( * ) nval, integer nnb, integer, dimension( * ) nbval, integer nout)

ZCHKUNHR_COL

Objectif :

!>
!> ZCHKUNHR_COL teste :
!>   1) ZUNGTSQR et ZUNHR_COL en utilisant ZLATSQR, ZGEMQRT,
!>   2) ZUNGTSQR_ROW et ZUNHR_COL à l'intérieur de ZGETSQRHRT
!>      (qui appelle ZLATSQR, ZUNGTSQR_ROW et ZUNHR_COL) en utilisant ZGEMQRT.
!> Par conséquent, ZLATSQR (partie de ZGEQR), ZGEMQRT (partie de ZGEMQR)
!> doivent être testés avant ce test.
!>
!>

Paramètres

THRESH 
!>          THRESH est DOUBLE PRECISION
!>          La valeur seuil pour les rapports de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour avoir
!>          chaque rapport de test imprimé, utilisez THRESH = 0.
!>

TSTERR

!>          TSTERR est LOGICAL
!>          Indicateur qui spécifie si les sorties d'erreur doivent être testées.
!>

NM

!>          NM est INTEGER
!>          Le nombre de valeurs de M contenues dans le vecteur MVAL.
!>

MVAL

!>          MVAL est un tableau INTEGER, dimension (NM)
!>          Les valeurs de la dimension des lignes de la matrice M.
!>

NN

!>          NN est INTEGER
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau INTEGER, dimension (NN)
!>          Les valeurs de la dimension des colonnes de la matrice N.
!>

NNB

!>          NNB est INTEGER
!>          Le nombre de valeurs de NB contenues dans le vecteur NBVAL.
!>

NBVAL

!>          NBVAL est un tableau INTEGER, dimension (NNB)
!>          Les valeurs de la taille de bloc NB.
!>

NOUT

!>          NOUT est INTEGER
!>          Le numéro de l'unité de sortie.
!>
