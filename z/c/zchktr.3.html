sous-routine zchktr (logique, dimension( * ) dotype, entier nn, entier, dimension( * ) nval, entier nnb, entier, dimension( * ) nbval, entier nns, entier, dimension( * ) nsval, double précision thresh, logique tsterr, entier nmax, complexe*16, dimension( * ) a, complexe*16, dimension( * ) ainv, complexe*16, dimension( * ) b, complexe*16, dimension( * ) x, complexe*16, dimension( * ) xact, complexe*16, dimension( * ) work, double précision, dimension( * ) rwork, entier nout)

ZCHKTR

Objectif :

!>
!> ZCHKTR teste ZTRTRI, -TRS, -RFS, et -CON, ainsi que ZLATRS(3)
!>

Paramètres

DOTYPE 
!>          DOTYPE est un tableau logique, dimension (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .VRAI.; si DOTYPE(j) = .FAUX., alors le type j n'est pas utilisé.
!>

NN

!>          NN est un entier
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau d'entiers, dimension (NN)
!>          Les valeurs de la dimension de colonne de la matrice N.
!>

NNB

!>          NNB est un entier
!>          Le nombre de valeurs de NB contenues dans le vecteur NBVAL.
!>

NBVAL

!>          NBVAL est un tableau d'entiers, dimension (NNB)
!>          Les valeurs de la taille de bloc NB.
!>

NNS

!>          NNS est un entier
!>          Le nombre de valeurs de NRHS contenues dans le vecteur NSVAL.
!>

NSVAL

!>          NSVAL est un tableau d'entiers, dimension (NNS)
!>          Les valeurs du nombre de membres de droite NRHS.
!>

THRESH

!>          THRESH est une double précision
!>          La valeur seuil pour les ratios de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>          chaque ratio de test soit imprimé, utilisez THRESH = 0.
!>

TSTERR

!>          TSTERR est un logique
!>          Indicateur qui détermine si les sorties d'erreur doivent être testées.
!>

NMAX

!>          NMAX est un entier
!>          La dimension principale des tableaux de travail.
!>          NMAX >= la valeur maximale de N dans NVAL.
!>

A

!>          A est un tableau COMPLEX*16, dimension (NMAX*NMAX)
!>

AINV

!>          AINV est un tableau COMPLEX*16, dimension (NMAX*NMAX)
!>

B

!>          B est un tableau COMPLEX*16, dimension (NMAX*NSMAX)
!>          où NSMAX est la plus grande entrée dans NSVAL.
!>

X

!>          X est un tableau COMPLEX*16, dimension (NMAX*NSMAX)
!>

XACT

!>          XACT est un tableau COMPLEX*16, dimension (NMAX*NSMAX)
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, dimension
!>                      (max(NMAX,2*NSMAX))
!>

NOUT

!>          NOUT est un entier
!>          Le numéro de l'unité de sortie.
!>
