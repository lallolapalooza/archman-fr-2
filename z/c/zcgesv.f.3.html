subroutine zcgesv (integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( n, * ) work, complex, dimension( * ) swork, double precision, dimension( * ) rwork, integer iter, integer info)

ZCGESV calcule la solution du système d'équations linéaires A * X = B pour les matrices GE (précision mixte avec raffinement itératif)

Objectif :

!>
!> ZCGESV calcule la solution d'un système d'équations linéaires complexes
!>    A * X = B,
!> où A est une matrice N-par-N et X et B sont des matrices N-par-NRHS.
!>
!> ZCGESV tente d'abord de factoriser la matrice en COMPLEX et utilise cette
!> factorisation dans le cadre d'une procédure de raffinement itératif pour
!> produire une solution avec une qualité d'erreur résiduelle normée de
!> type COMPLEX*16 (voir ci-dessous).
!> Si l'approche échoue, la méthode passe à une factorisation et une résolution
!> en COMPLEX*16.
!>
!> Le raffinement itératif ne sera pas une stratégie gagnante si le ratio
!> performance COMPLEX sur performance COMPLEX*16 est trop faible. Une
!> stratégie raisonnable devrait prendre en compte le nombre de membres
!> de droite et la taille de la matrice. Cela pourrait être fait avec un appel
!> à ILAENV à l'avenir. Jusqu'à présent, nous essayons toujours le raffinement
!> itératif.
!>
!> Le processus de raffinement itératif est arrêté si
!>     ITER > ITERMAX
!> ou pour tous les RHS, nous avons :
!>     RNRM < SQRT(N)*XNRM*ANRM*EPS*BWDMAX
!> où
!>     o ITER est le numéro de l'itération courante dans le processus de
!>       raffinement itératif
!>     o RNRM est la norme infinie du résidu
!>     o XNRM est la norme infinie de la solution
!>     o ANRM est la norme de l'opérateur infini de la matrice A
!>     o EPS est l'epsilon machine retourné par DLAMCH('Epsilon')
!> Les valeurs ITERMAX et BWDMAX sont fixées à 30 et 1.0D+00 respectivement.
!>

Paramètres

N 
!>          N est INTEGER
!>          Le nombre d'équations linéaires, c'est-à-dire l'ordre de la
!>          matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!>

A

!>          A est un tableau COMPLEX*16,
!>          dimension (LDA,N)
!>          À l'entrée, la matrice de coefficients N-par-N A.
!>          À la sortie, si le raffinement itératif a été utilisé avec succès
!>          (INFO = 0 et ITER >= 0, voir la description ci-dessous), alors A
!>          est inchangé ; si la factorisation en double précision a été
!>          utilisée (INFO = 0 et ITER < 0, voir la description ci-dessous),
!>          alors le tableau A contient les facteurs L et U de la factorisation
!>          A = P*L*U ; les éléments diagonaux unitaires de L ne sont pas stockés.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (N)
!>          Les indices de pivot qui définissent la matrice de permutation P ;
!>          la ligne i de la matrice a été échangée avec la ligne IPIV(i).
!>          Correspond soit à la factorisation en simple précision
!>          (si INFO = 0 et ITER >= 0) soit à la factorisation en double précision
!>          (si INFO = 0 et ITER < 0).
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,NRHS)
!>          La matrice de membre de droite N-par-NRHS B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est un tableau COMPLEX*16, dimension (LDX,NRHS)
!>          Si INFO = 0, la matrice solution N-par-NRHS X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension (N,NRHS)
!>          Ce tableau est utilisé pour stocker les vecteurs résiduels.
!>

SWORK

!>          SWORK est un tableau COMPLEX, dimension (N*(N+NRHS))
!>          Ce tableau est utilisé pour utiliser la matrice en simple précision et les
!>          membres de droite ou les solutions en simple précision.
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, dimension (N)
!>

ITER

!>          ITER est INTEGER
!>          < 0 : le raffinement itératif a échoué, une factorisation
!>               en COMPLEX*16 a été effectuée
!>               -1 : la routine est revenue à la précision complète pour
!>                    des raisons spécifiques à l'implémentation ou à la machine
!>               -2 : la réduction de la précision a provoqué un débordement,
!>                    la routine est revenue à la précision complète
!>               -3 : échec de CGETRF
!>               -31 : arrêt du raffinement itératif après la 30ème
!>                    itération
!>          > 0 : le raffinement itératif a été utilisé avec succès.
!>               Retourne le nombre d'itérations
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 : si INFO = i, U(i,i) calculé en COMPLEX*16 est exactement
!>                zéro. La factorisation a été complétée, mais le
!>                facteur U est exactement singulier, donc la solution
!>                n'a pas pu être calculée.
!>
