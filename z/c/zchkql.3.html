!>

Paramètres

DOTYPE 
!>          DOTYPE est un tableau LOGICAL, de dimension (NTYPES).
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .TRUE. ; si DOTYPE(j) = .FALSE., alors le type j n'est pas utilisé.
!>

NM

!>          NM est un INTEGER.
!>          Le nombre de valeurs de M contenues dans le vecteur MVAL.
!>

MVAL

!>          MVAL est un tableau INTEGER, de dimension (NM).
!>          Les valeurs de la dimension des lignes de la matrice M.
!>

NN

!>          NN est un INTEGER.
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau INTEGER, de dimension (NN).
!>          Les valeurs de la dimension des colonnes de la matrice N.
!>

NNB

!>          NNB est un INTEGER.
!>          Le nombre de valeurs de NB et NX contenues dans
!>          les vecteurs NBVAL et NXVAL. Les paramètres de blocage sont utilisés
!>          par paires (NB,NX).
!>

NBVAL

!>          NBVAL est un tableau INTEGER, de dimension (NNB).
!>          Les valeurs de la taille de bloc NB.
!>

NXVAL

!>          NXVAL est un tableau INTEGER, de dimension (NNB).
!>          Les valeurs du point de croisement NX.
!>

NRHS

!>          NRHS est un INTEGER.
!>          Le nombre de vecteurs du second membre à générer pour
!>          chaque système linéaire.
!>

THRESH

!>          THRESH est un DOUBLE PRECISION.
!>          La valeur seuil pour les rapports de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>          chaque rapport de test soit imprimé, utilisez THRESH = 0.
!>

TSTERR

!>          TSTERR est un LOGICAL.
!>          Indicateur indiquant si les sorties d'erreur doivent être testées.
!>

NMAX

!>          NMAX est un INTEGER.
!>          La valeur maximale autorisée pour M ou N, utilisée pour dimensionner
!>          les tableaux de travail.
!>

A

!>          A est un tableau COMPLEX*16, de dimension (NMAX*NMAX).
!>

AF

!>          AF est un tableau COMPLEX*16, de dimension (NMAX*NMAX).
!>

AQ

!>          AQ est un tableau COMPLEX*16, de dimension (NMAX*NMAX).
!>

AL

!>          AL est un tableau COMPLEX*16, de dimension (NMAX*NMAX).
!>

AC

!>          AC est un tableau COMPLEX*16, de dimension (NMAX*NMAX).
!>

B

!>          B est un tableau COMPLEX*16, de dimension (NMAX*NRHS).
!>

X

!>          X est un tableau COMPLEX*16, de dimension (NMAX*NRHS).
!>

XACT

!>          XACT est un tableau COMPLEX*16, de dimension (NMAX*NRHS).
!>

TAU

!>          TAU est un tableau COMPLEX*16, de dimension (NMAX).
!>

WORK

!>          WORK est un tableau COMPLEX*16, de dimension (NMAX*NMAX).
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, de dimension (NMAX).
!>

NOUT

!>          NOUT est un INTEGER.
!>          Le numéro d'unité pour la sortie.
!>
