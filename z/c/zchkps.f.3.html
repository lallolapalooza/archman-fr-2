sous-routine zchkps (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nnb, integer, dimension( * ) nbval, integer nrank, integer, dimension( * ) rankval, double precision thresh, logical tsterr, integer nmax, complex*16, dimension( * ) a, complex*16, dimension( * ) afac, complex*16, dimension( * ) perm, integer, dimension( * ) piv, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer nout)

ZCHKPS

Objectif :

!>
!> ZCHKPS teste ZPSTRF.
!>

Paramètres

DOTYPE 
!>          DOTYPE est un tableau LOGICAL, de dimension (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., alors le type j n'est pas utilisé.
!>

NN

!>          NN est un INTEGER
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau INTEGER, de dimension (NN)
!>          Les valeurs de la dimension de la matrice N.
!>

NNB

!>          NNB est un INTEGER
!>          Le nombre de valeurs de NB contenues dans le vecteur NBVAL.
!>

NBVAL

!>          NBVAL est un tableau INTEGER, de dimension (NNB)
!>          Les valeurs de la taille de bloc NB.
!>

NRANK

!>          NRANK est un INTEGER
!>          Le nombre de valeurs de RANK contenues dans le vecteur RANKVAL.
!>

RANKVAL

!>          RANKVAL est un tableau INTEGER, de dimension (NBVAL)
!>          Les valeurs de la taille de bloc NB.
!>

THRESH

!>          THRESH est un DOUBLE PRECISION
!>          La valeur seuil pour les rapports de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour avoir
!>          chaque rapport de test imprimé, utiliser THRESH = 0.
!>

TSTERR

!>          TSTERR est un LOGICAL
!>          Indicateur qui détermine si les sorties d'erreur doivent être testées.
!>

NMAX

!>          NMAX est un INTEGER
!>          La valeur maximale autorisée pour N, utilisée pour dimensionner les
!>          tableaux de travail.
!>

A

!>          A est un tableau COMPLEX*16, de dimension (NMAX*NMAX)
!>

AFAC

!>          AFAC est un tableau COMPLEX*16, de dimension (NMAX*NMAX)
!>

PERM

!>          PERM est un tableau COMPLEX*16, de dimension (NMAX*NMAX)
!>

PIV

!>          PIV est un tableau INTEGER, de dimension (NMAX)
!>

WORK

!>          WORK est un tableau COMPLEX*16, de dimension (NMAX*3)
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, de dimension (NMAX)
!>

NOUT

!>          NOUT est un INTEGER
!>          Le numéro de l'unité pour la sortie.
!>
