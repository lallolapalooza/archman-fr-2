sous-routine zdrvgb (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nrhs, double precision thresh, logical tsterr, complex*16, dimension( * ) a, integer la, complex*16, dimension( * ) afb, integer lafb, complex*16, dimension( * ) asav, complex*16, dimension( * ) b, complex*16, dimension( * ) bsav, complex*16, dimension( * ) x, complex*16, dimension( * ) xact, double precision, dimension( * ) s, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer nout)

ZDRVGB

But

!>
!> ZDRVGB teste les routines pilotes ZGBSV et -SVX.
!>

Paramètres

DOTYPE 
!>          DOTYPE est un tableau LOGICAL, dimension (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., alors le type j n'est pas utilisé.
!>

NN

!>          NN est un entier
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau d'entiers, dimension (NN)
!>          Les valeurs de la dimension de colonne de matrice N.
!>

NRHS

!>          NRHS est un entier
!>          Le nombre de vecteurs du second membre à générer pour
!>          chaque système linéaire.
!>

THRESH

!>          THRESH est DOUBLE PRECISION
!>          La valeur seuil pour les ratios de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour avoir
!>          chaque ratio de test imprimé, utilisez THRESH = 0.
!>

TSTERR

!>          TSTERR est LOGICAL
!>          Indicateur qui détermine si les sorties d'erreur doivent être testées.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LA)
!>

LA

!>          LA est un entier
!>          La longueur du tableau A. LA >= (2*NMAX-1)*NMAX
!>          où NMAX est la plus grande entrée dans NVAL.
!>

AFB

!>          AFB est un tableau COMPLEX*16, dimension (LAFB)
!>

LAFB

!>          LAFB est un entier
!>          La longueur du tableau AFB. LAFB >= (3*NMAX-2)*NMAX
!>          où NMAX est la plus grande entrée dans NVAL.
!>

ASAV

!>          ASAV est un tableau COMPLEX*16, dimension (LA)
!>

B

!>          B est un tableau COMPLEX*16, dimension (NMAX*NRHS)
!>

BSAV

!>          BSAV est un tableau COMPLEX*16, dimension (NMAX*NRHS)
!>

X

!>          X est un tableau COMPLEX*16, dimension (NMAX*NRHS)
!>

XACT

!>          XACT est un tableau COMPLEX*16, dimension (NMAX*NRHS)
!>

S

!>          S est un tableau DOUBLE PRECISION, dimension (2*NMAX)
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension
!>                      (NMAX*max(3,NRHS,NMAX))
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, dimension
!>                      (NMAX+2*NRHS)
!>

IWORK

!>          IWORK est un tableau d'entiers, dimension (NMAX)
!>

NOUT

!>          NOUT est un entier
!>          Le numéro d'unité pour la sortie.
!>
