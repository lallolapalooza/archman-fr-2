subroutine zdrvrf3 (integer nout, integer nn, integer, dimension( nn ) nval, double precision thresh, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) arf, complex*16, dimension( lda, * ) b1, complex*16, dimension( lda, * ) b2, double precision, dimension( * ) d_work_zlange, complex*16, dimension( * ) z_work_zgeqrf, complex*16, dimension( * ) tau)

ZDRVRF3

Objectif :

!>
!> ZDRVRF3 teste les routines RFP LAPACK :
!>     ZTFSM
!>

Paramètres

NOUT 
!>          NOUT est de type INTEGER
!>                Le numéro de l'unité de sortie.
!>

NN

!>          NN est de type INTEGER
!>                Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau de type INTEGER, dimension (NN)
!>                Les valeurs de la dimension N de la matrice.
!>

THRESH

!>          THRESH est de type DOUBLE PRECISION
!>                La valeur seuil pour les rapports de test. Un résultat est
!>                inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>                chaque rapport de test soit imprimé, utilisez THRESH = 0.
!>

A

!>          A est un tableau de type COMPLEX*16, dimension (LDA,NMAX)
!>

LDA

!>          LDA est de type INTEGER
!>                La dimension principale du tableau A. LDA >= max(1,NMAX).
!>

ARF

!>          ARF est un tableau de type COMPLEX*16, dimension ((NMAX*(NMAX+1))/2).
!>

B1

!>          B1 est un tableau de type COMPLEX*16, dimension (LDA,NMAX)
!>

B2

!>          B2 est un tableau de type COMPLEX*16, dimension (LDA,NMAX)
!>

D_WORK_ZLANGE

!>          D_WORK_ZLANGE est un tableau de type DOUBLE PRECISION, dimension (NMAX)
!>

Z_WORK_ZGEQRF

!>          Z_WORK_ZGEQRF est un tableau de type COMPLEX*16, dimension (NMAX)
!>

TAU

!>          TAU est un tableau de type COMPLEX*16, dimension (NMAX)
!>
