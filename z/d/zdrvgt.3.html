sous-routine zdrvgt (logique, dimension( * ) dotype, entier nn, entier, dimension( * ) nval, entier nrhs, double précision thresh, logique tsterr, complexe*16, dimension( * ) a, complexe*16, dimension( * ) af, complexe*16, dimension( * ) b, complexe*16, dimension( * ) x, complexe*16, dimension( * ) xact, complexe*16, dimension( * ) work, double précision, dimension( * ) rwork, entier, dimension( * ) iwork, entier nout)

ZDRVGT

Objectif :

!>
!> ZDRVGT teste ZGTSV et -SVX.
!>

Paramètres

DOTYPE 
!>          DOTYPE est un tableau LOGIQUE, de dimension (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .TRUE. ; si DOTYPE(j) = .FALSE., alors le type j n'est pas utilisé.
!>

NN

!>          NN est un ENTIER
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau ENTIER, de dimension (NN)
!>          Les valeurs de la dimension de matrice N.
!>

NRHS

!>          NRHS est un ENTIER
!>          Le nombre de membres de droite, NRHS >= 0.
!>

THRESH

!>          THRESH est un DOUBLE PRÉCISION
!>          La valeur seuil pour les ratios de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>          chaque ratio de test soit imprimé, utiliser THRESH = 0.
!>

TSTERR

!>          TSTERR est un LOGIQUE
!>          Indicateur indiquant si les sorties d'erreur doivent être testées.
!>

A

!>          A est un tableau COMPLEX*16, de dimension (NMAX*4)
!>

AF

!>          AF est un tableau COMPLEX*16, de dimension (NMAX*4)
!>

B

!>          B est un tableau COMPLEX*16, de dimension (NMAX*NRHS)
!>

X

!>          X est un tableau COMPLEX*16, de dimension (NMAX*NRHS)
!>

XACT

!>          XACT est un tableau COMPLEX*16, de dimension (NMAX*NRHS)
!>

WORK

!>          WORK est un tableau COMPLEX*16, de dimension
!>                      (NMAX*max(3,NRHS))
!>

RWORK

!>          RWORK est un tableau DOUBLE PRÉCISION, de dimension (NMAX+2*NRHS)
!>

IWORK

!>          IWORK est un tableau ENTIER, de dimension (2*NMAX)
!>

NOUT

!>          NOUT est un ENTIER
!>          Le numéro de l'unité de sortie.
!>
