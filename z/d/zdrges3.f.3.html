subroutine zdrges3 (integer nsizes, integer, dimension( * ) nn, integer ntypes, logical, dimension( * ) dotype, integer, dimension( 4 ) iseed, double precision thresh, integer nounit, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( lda, * ) b, complex*16, dimension( lda, * ) s, complex*16, dimension( lda, * ) t, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( ldq, * ) z, complex*16, dimension( * ) alpha, complex*16, dimension( * ) beta, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( 13 ) result, logical, dimension( * ) bwork, integer info)

ZDRGES3

Objectif :

!>
!> ZDRGES3 teste le pilote de problème de valeurs propres généralisé non symétrique (forme de Schur)
!> ZGGES3.
!>
!> ZGGES3 factorise A et B en Q*S*Z'  et Q*T*Z' , où ' signifie conjugué
!> transposé, S et T sont triangulaires supérieures (c'est-à-dire sous forme de Schur généralisée),
!> et Q et Z sont unitaires. Il calcule également les valeurs propres généralisées
!> (alpha(j),beta(j)), j=1,...,n. Ainsi,
!> w(j) = alpha(j)/beta(j) est une racine de l'équation caractéristique
!>
!>                 det( A - w(j) B ) = 0
!>
!> Facultativement, il réordonne également les valeurs propres pour qu'un
!> cluster de valeurs propres sélectionnées apparaisse dans le bloc diagonal principal des
!> formes de Schur.
!>
!> Lorsque ZDRGES3 est appelé, un nombre de matrices () et un
!> nombre de matrices sont spécifiés. Pour chaque taille ()
!> et chaque TYPE de matrice, une paire de matrices (A, B) sera générée
!> et utilisée pour les tests. Pour chaque paire de matrices, les 13 tests suivants
!> seront effectués et comparés au seuil THRESH, à l'exception
!> des tests (5), (11) et (13).
!>
!>
!> (1)   | A - Q S Z' | / ( |A| n ulp ) (pas de tri des valeurs propres)
!>
!>
!> (2)   | B - Q T Z' | / ( |B| n ulp ) (pas de tri des valeurs propres)
!>
!>
!> (3)   | I - QQ' | / ( n ulp ) (pas de tri des valeurs propres)
!>
!>
!> (4)   | I - ZZ' | / ( n ulp ) (pas de tri des valeurs propres)
!>
!> (5)   si A est sous forme de Schur (c'est-à-dire forme triangulaire) (pas de tri des
!>       valeurs propres)
!>
!> (6)   si les valeurs propres = éléments diagonaux de la forme de Schur (S, T),
!>       c'est-à-dire tester le maximum sur j de D(j) où :
!>
!>                     |alpha(j) - S(j,j)|        |beta(j) - T(j,j)|
!>           D(j) = ------------------------ + -----------------------
!>                  max(|alpha(j)|,|S(j,j)|)   max(|beta(j)|,|T(j,j)|)
!>
!>       (pas de tri des valeurs propres)
!>
!> (7)   | (A,B) - Q (S,T) Z' | / ( |(A,B)| n ulp )
!>       (avec tri des valeurs propres).
!>
!> (8)   | I - QQ' | / ( n ulp ) (avec tri des valeurs propres).
!>
!> (9)   | I - ZZ' | / ( n ulp ) (avec tri des valeurs propres).
!>
!> (10)  si A est sous forme de Schur (c'est-à-dire forme quasi-triangulaire)
!>       (avec tri des valeurs propres).
!>
!> (11)  si les valeurs propres = éléments diagonaux de la forme de Schur (S, T),
!>       c'est-à-dire tester le maximum sur j de D(j) où :
!>
!>                     |alpha(j) - S(j,j)|        |beta(j) - T(j,j)|
!>           D(j) = ------------------------ + -----------------------
!>                  max(|alpha(j)|,|S(j,j)|)   max(|beta(j)|,|T(j,j)|)
!>
!>       (avec tri des valeurs propres).
!>
!> (12)  si le tri a fonctionné et SDIM est le nombre de valeurs propres
!>       qui ont été SÉLECTIONNÉES.
!>
!> Matrices de Test
!> ==============
!>
!> Les tailles des matrices de test sont spécifiées par un tableau
!> NN(1:NSIZES) ; la valeur de chaque élément NN(j) spécifie une taille.
!> Les sont spécifiés par un tableau logique DOTYPE( 1:NTYPES ) ; si
!> DOTYPE(j) est .VRAI., alors le type de matrice sera généré.
!> Actuellement, la liste des types possibles est :
!>
!> (1)  ( 0, 0 )         (une paire de matrices nulles)
!>
!> (2)  ( I, 0 )         (une identité et une matrice nulle)
!>
!> (3)  ( 0, I )         (une identité et une matrice nulle)
!>
!> (4)  ( I, I )         (une paire de matrices identité)
!>
!>         t   t
!> (5)  ( J , J  )       (une paire de blocs de Jordan transposés)
!>
!>                                     t                ( I   0  )
!> (6)  ( X, Y )         où  X = ( J   0  )  et Y = (      t )
!>                                  ( 0   I  )          ( 0   J  )
!>                       et I est une identité k x k et J un bloc de Jordan (k+1)x(k+1)
!>                       ; k=(N-1)/2
!>
!> (7)  ( D, I )         où D est diag( 0, 1,..., N-1 ) (une matrice diagonale
!>                       avec ces entrées diagonales.)
!> (8)  ( I, D )
!>
!> (9)  ( big*D, small*I ) où  est proche du débordement et small=1/big
!>
!> (10) ( small*D, big*I )
!>
!> (11) ( big*I, small*D )
!>
!> (12) ( small*I, big*D )
!>
!> (13) ( big*D, big*I )
!>
!> (14) ( small*D, small*I )
!>
!> (15) ( D1, D2 )        où D1 est diag( 0, 0, 1, ..., N-3, 0 ) et
!>                        D2 est diag( 0, N-3, N-4,..., 1, 0, 0 )
!>           t   t
!> (16) Q ( J , J ) Z     où Q et Z sont des matrices orthogonales aléatoires.
!>
!> (17) Q ( T1, T2 ) Z    où T1 et T2 sont des matrices triangulaires supérieures
!>                        avec des entrées O(1) aléatoires au-dessus de la diagonale
!>                        et des entrées diagonales diag(T1) =
!>                        ( 0, 0, 1, ..., N-3, 0 ) et diag(T2) =
!>                        ( 0, N-3, N-4,..., 1, 0, 0 )
!>
!> (18) Q ( T1, T2 ) Z    diag(T1) = ( 0, 0, 1, 1, s, ..., s, 0 )
!>                        diag(T2) = ( 0, 1, 0, 1,..., 1, 0 )
!>                        s = précision machine.
!>
!> (19) Q ( T1, T2 ) Z    diag(T1)=( 0,0,1,1, 1-d, ..., 1-(N-5)*d=s, 0 )
!>                        diag(T2) = ( 0, 1, 0, 1, ..., 1, 0 )
!>
!>                                                        N-5
!> (20) Q ( T1, T2 ) Z    diag(T1)=( 0, 0, 1, 1, a, ..., a   =s, 0 )
!>                        diag(T2) = ( 0, 1, 0, 1, ..., 1, 0, 0 )
!>
!> (21) Q ( T1, T2 ) Z    diag(T1)=( 0, 0, 1, r1, r2, ..., r(N-4), 0 )
!>                        diag(T2) = ( 0, 1, 0, 1, ..., 1, 0, 0 )
!>                        où r1,..., r(N-4) sont aléatoires.
!>
!> (22) Q ( big*T1, small*T2 ) Z    diag(T1) = ( 0, 0, 1, ..., N-3, 0 )
!>                                  diag(T2) = ( 0, 1, ..., 1, 0, 0 )
!>
!> (23) Q ( small*T1, big*T2 ) Z    diag(T1) = ( 0, 0, 1, ..., N-3, 0 )
!>                                  diag(T2) = ( 0, 1, ..., 1, 0, 0 )
!>
!> (24) Q ( small*T1, small*T2 ) Z  diag(T1) = ( 0, 0, 1, ..., N-3, 0 )
!>                                  diag(T2) = ( 0, 1, ..., 1, 0, 0 )
!>
!> (25) Q ( big*T1, big*T2 ) Z      diag(T1) = ( 0, 0, 1, ..., N-3, 0 )
!>                                  diag(T2) = ( 0, 1, ..., 1, 0, 0 )
!>
!> (26) Q ( T1, T2 ) Z     où T1 et T2 sont des matrices triangulaires supérieures aléatoires.
!>
!>

Paramètres

NSIZES 
!>          NSIZES est un entier (INTEGER)
!>          Le nombre de tailles de matrices à utiliser. S'il est nul,
!>          DDRGES3 ne fait rien. NSIZES >= 0.
!>

NN

!>          NN est un tableau d'entiers (INTEGER), de dimension (NSIZES)
!>          Un tableau contenant les tailles à utiliser pour les matrices.
!>          Les valeurs nulles seront ignorées. NN >= 0.
!>

NTYPES

!>          NTYPES est un entier
!>          Le nombre d'éléments dans DOTYPE. Si c'est zéro, DDRGES3
!>          ne fait rien. Il doit être au moins zéro. Si c'est MAXTYP+1
!>          et NSIZES est 1, alors un type supplémentaire, MAXTYP+1, est
!>          défini, qui consiste à utiliser la matrice présente dans A en entrée.
!>          Ceci n'est utile que si DOTYPE(1:MAXTYP) est .FAUX. et
!>          DOTYPE(MAXTYP+1) est .VRAI. .
!>

DOTYPE

!>          DOTYPE est un tableau LOGIQUE, de dimension (NTYPES)
!>          Si DOTYPE(j) est .VRAI., alors pour chaque taille dans NN une
!>          matrice de cette taille et de type j sera générée.
!>          Si NTYPES est plus petit que le nombre maximum de types
!>          définis (PARAMÈTRE MAXTYP), alors les types NTYPES+1 à
!>          MAXTYP ne seront pas générés. Si NTYPES est plus grand
!>          que MAXTYP, DOTYPE(MAXTYP+1) à DOTYPE(NTYPES)
!>          seront ignorés.
!>

ISEED

!>          ISEED est un tableau d'ENTIERS, de dimension (4)
!>          À l'entrée, ISEED spécifie la graine du générateur de nombres
!>          aléatoires. Les éléments du tableau doivent être compris entre 0 et 4095 ;
!>          sinon, ils seront réduits modulo 4096. De plus, ISEED(4) doit
!>          être impair. Le générateur de nombres aléatoires utilise une séquence
!>          linéaire congruente limitée à de petits entiers, et donc
!>          devrait produire des nombres aléatoires indépendants de la machine. Les
!>          valeurs de ISEED sont modifiées à la sortie et peuvent être utilisées dans
!>          le prochain appel à DDRGES3 pour continuer la même séquence de nombres
!>          aléatoires.
!>

THRESH

!>          THRESH est un DOUBLE PRÉCISION
!>          Un test sera compté comme si le calcul, tel que décrit ci-dessus,
!>          dépassait THRESH. Notez que l'erreur est mise à l'échelle pour être O(1),
!>          donc THRESH devrait être un multiple raisonnablement petit de 1, par
!>          exemple 10 ou 100. En particulier, il ne devrait pas dépendre de la
!>          précision (simple ou double) ou de la taille de la matrice. THRESH >= 0.
!>

NOUNIT

!>          NOUNIT est un ENTIER
!>          Le numéro d'unité FORTRAN pour l'impression des messages d'erreur
!>          (par exemple, si une routine retourne IINFO différent de 0).
!>

A

!>          A est un tableau COMPLEXE*16, de dimension(LDA, max(NN))
!>          Utilisé pour contenir la matrice A originale. Utilisé en entrée uniquement
!>          si NTYPES=MAXTYP+1, DOTYPE(1:MAXTYP)=.FAUX., et
!>          DOTYPE(MAXTYP+1)=.VRAI.
!>

LDA

!>          LDA est un ENTIER
!>          La dimension principale de A, B, S et T.
!>          Elle doit être au moins 1 et au moins max( NN ).
!>

B

!>          B est un tableau COMPLEXE*16, de dimension(LDA, max(NN))
!>          Utilisé pour contenir la matrice B originale. Utilisé en entrée uniquement
!>          si NTYPES=MAXTYP+1, DOTYPE(1:MAXTYP)=.FAUX., et
!>          DOTYPE(MAXTYP+1)=.VRAI.
!>

S

!>          S est un tableau COMPLEXE*16, de dimension (LDA, max(NN))
!>          La matrice forme de Schur calculée à partir de A par ZGGES3. À la sortie, S
!>          contient la matrice forme de Schur correspondant à la matrice
!>          dans A.
!>

T

!>          T est un tableau COMPLEXE*16, de dimension (LDA, max(NN))
!>          La matrice triangulaire supérieure calculée à partir de B par ZGGES3.
!>

Q

!>          Q est un tableau COMPLEXE*16, de dimension (LDQ, max(NN))
!>          La matrice orthogonale (gauche) calculée par ZGGES3.
!>

LDQ

!>          LDQ est un ENTIER
!>          La dimension principale de Q et Z. Elle doit
!>          être au moins 1 et au moins max( NN ).
!>

Z

!>          Z est un tableau COMPLEXE*16, de dimension( LDQ, max(NN) )
!>          La matrice orthogonale (droite) calculée par ZGGES3.
!>

ALPHA

!>          ALPHA est un tableau COMPLEXE*16, de dimension (max(NN))
!>

BETA

!>          BETA est un tableau COMPLEXE*16, de dimension (max(NN))
!>
!>          Les valeurs propres généralisées de (A,B) calculées par ZGGES3.
!>          ALPHA(k) / BETA(k) est la k-ième valeur propre généralisée de A
!>          et B.
!>

WORK

!>          WORK est un tableau COMPLEXE*16, de dimension (LWORK)
!>

LWORK

!>          LWORK est un ENTIER
!>          La dimension du tableau WORK. LWORK >= 3*N*N.
!>

RWORK

!>          RWORK est un tableau DOUBLE PRÉCISION, de dimension ( 8*N )
!>          Espace de travail réel.
!>

RESULT

!>          RESULT est un tableau DOUBLE PRÉCISION, de dimension (15)
!>          Les valeurs calculées par les tests décrits ci-dessus.
!>          Les valeurs sont actuellement limitées à 1/ulp, pour éviter le débordement.
!>

BWORK

!>          BWORK est un tableau LOGIQUE, de dimension (N)
!>

INFO

!>          INFO est un ENTIER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale.
!>          > 0 : une routine a retourné un code d'erreur. INFO est la valeur
!>                absolue de la valeur INFO retournée.
!>
