sous-routine zdrvab (logique, dimensionnelle( * ) dotype, entier nm, entier, dimensionnelle( * ) mval, entier nns, entier, dimensionnelle( * ) nsval, précision double thresh, entier nmax, complexe*16, dimensionnelle( * ) a, complexe*16, dimensionnelle( * ) afac, complexe*16, dimensionnelle( * ) b, complexe*16, dimensionnelle( * ) x, complexe*16, dimensionnelle( * ) work, précision double, dimensionnelle( * ) rwork, complexe, dimensionnelle( * ) swork, entier, dimensionnelle( * ) iwork, entier nout)

!>
!> ZDRVAB teste ZCGESV
!>

Paramètres

DOTYPE 
!>          DOTYPE est un tableau logique, dimensionné (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .VRAI. ; si DOTYPE(j) = .FAUX., alors le type j n'est pas utilisé.
!>

NM

!>          NM est un entier
!>          Le nombre de valeurs de M contenues dans le vecteur MVAL.
!>

MVAL

!>          MVAL est un tableau d'entiers, dimensionné (NM)
!>          Les valeurs de la dimension de ligne de la matrice M.
!>

NNS

!>          NNS est un entier
!>          Le nombre de valeurs de NRHS contenues dans le vecteur NSVAL.
!>

NSVAL

!>          NSVAL est un tableau d'entiers, dimensionné (NNS)
!>          Les valeurs du nombre de membres du second membre NRHS.
!>

THRESH

!>          THRESH est de type DOUBLE PRECISION
!>          La valeur seuil pour les ratios de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>          chaque ratio de test soit imprimé, utilisez THRESH = 0.
!>

NMAX

!>          NMAX est un entier
!>          La valeur maximale permise pour M ou N, utilisée pour dimensionner
!>          les tableaux de travail.
!>

A

!>          A est un tableau de type COMPLEX*16, dimensionné (NMAX*NMAX)
!>

AFAC

!>          AFAC est un tableau de type COMPLEX*16, dimensionné (NMAX*NMAX)
!>

B

!>          B est un tableau de type COMPLEX*16, dimensionné (NMAX*NSMAX)
!>          où NSMAX est la plus grande entrée de NSVAL.
!>

X

!>          X est un tableau de type COMPLEX*16, dimensionné (NMAX*NSMAX)
!>

WORK

!>          WORK est un tableau de type COMPLEX*16, dimensionné
!>                      (NMAX*max(3,NSMAX*2))
!>

RWORK

!>          RWORK est un tableau de type DOUBLE PRECISION, dimensionné
!>                      NMAX
!>

SWORK

!>          SWORK est un tableau de type COMPLEX, dimensionné
!>                      (NMAX*(NSMAX+NMAX))
!>

IWORK

!>          IWORK est un tableau d'entiers, dimensionné
!>                      NMAX
!>

NOUT

!>          NOUT est un entier
!>          Le numéro d'unité pour la sortie.
!>
