sous-routine zdrvls (logique, dimension( * ) dotype, entier nm, entier, dimension( * ) mval, entier nn, entier, dimension( * ) nval, entier nns, entier, dimension( * ) nsval, entier nnb, entier, dimension( * ) nbval, entier, dimension( * ) nxval, double précision thresh, logique tsterr, complexe*16, dimension( * ) a, complexe*16, dimension( * ) copya, complexe*16, dimension( * ) b, complexe*16, dimension( * ) copyb, complexe*16, dimension( * ) c, double précision, dimension( * ) s, double précision, dimension( * ) copys, entier nout)

ZDRVLS

Objectif

!>
!> ZDRVLS teste les routines de pilote de moindres carrés ZGELS, ZGELST,
!> ZGETSLS, ZGELSS, ZGELSY et ZGELSD.
!>

Paramètres

DOTYPE 
!>          DOTYPE est un tableau LOGIQUE, dimension (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .VRAI.; si DOTYPE(j) = .FAUX., alors le type j n'est pas utilisé.
!>          La matrice de type j est générée comme suit :
!>          j=1 : A = U*D*V où U et V sont des matrices unitaires aléatoires
!>               et D a des entrées aléatoires (> 0.1) prises dans une distribution
!>               uniforme (0,1). A est de rang plein.
!>          j=2 : Identique à 1, mais A est mise à l'échelle vers le haut.
!>          j=3 : Identique à 1, mais A est mise à l'échelle vers le bas.
!>          j=4 : A = U*D*V où U et V sont des matrices unitaires aléatoires
!>               et D a 3*min(M,N)/4 entrées aléatoires (> 0.1) prises
!>               dans une distribution uniforme (0,1) et les entrées restantes
!>               sont mises à 0. A est de rang déficient.
!>          j=5 : Identique à 4, mais A est mise à l'échelle vers le haut.
!>          j=6 : Identique à 5, mais A est mise à l'échelle vers le bas.
!>

NM

!>          NM est un ENTIER
!>          Le nombre de valeurs de M contenues dans le vecteur MVAL.
!>

MVAL

!>          MVAL est un tableau ENTIER, dimension (NM)
!>          Les valeurs de la dimension des lignes M de la matrice.
!>

NN

!>          NN est un ENTIER
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau ENTIER, dimension (NN)
!>          Les valeurs de la dimension des colonnes N de la matrice.
!>

NNB

!>          NNB est un ENTIER
!>          Le nombre de valeurs de NB et NX contenues dans les
!>          vecteurs NBVAL et NXVAL. Les paramètres de blocage sont utilisés
!>          par paires (NB,NX).
!>

NBVAL

!>          NBVAL est un tableau ENTIER, dimension (NNB)
!>          Les valeurs de la taille de bloc NB.
!>

NXVAL

!>          NXVAL est un tableau ENTIER, dimension (NNB)
!>          Les valeurs du point de croisement NX.
!>

NNS

!>          NNS est un ENTIER
!>          Le nombre de valeurs de NRHS contenues dans le vecteur NSVAL.
!>

NSVAL

!>          NSVAL est un tableau ENTIER, dimension (NNS)
!>          Les valeurs du nombre de seconds membres NRHS.
!>

THRESH

!>          THRESH est un DOUBLE PRÉCISION
!>          La valeur seuil pour les rapports de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>          chaque rapport de test soit imprimé, utilisez THRESH = 0.
!>

TSTERR

!>          TSTERR est un LOGIQUE
!>          Indicateur indiquant si les sorties d'erreur doivent être testées.
!>

A

!>          A est un tableau COMPLEX*16, dimension (MMAX*NMAX)
!>          où MMAX est la valeur maximale de M dans MVAL et NMAX est la
!>          valeur maximale de N dans NVAL.
!>

COPYA

!>          COPYA est un tableau COMPLEX*16, dimension (MMAX*NMAX)
!>

B

!>          B est un tableau COMPLEX*16, dimension (MMAX*NSMAX)
!>          où MMAX est la valeur maximale de M dans MVAL et NSMAX est la
!>          valeur maximale de NRHS dans NSVAL.
!>

COPYB

!>          COPYB est un tableau COMPLEX*16, dimension (MMAX*NSMAX)
!>

C

!>          C est un tableau COMPLEX*16, dimension (MMAX*NSMAX)
!>

S

!>          S est un tableau DOUBLE PRÉCISION, dimension
!>                      (min(MMAX,NMAX))
!>

COPYS

!>          COPYS est un tableau DOUBLE PRÉCISION, dimension
!>                      (min(MMAX,NMAX))
!>

NOUT

!>          NOUT est un ENTIER
!>          Le numéro de l'unité de sortie.
!>
