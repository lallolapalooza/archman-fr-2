sous-routine zhbtrd (character vect, character uplo, integer n, integer kd, complex*16, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) d, double precision, dimension( * ) e, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( * ) work, integer info)

ZHBTRD

But :

!>
!> ZHBTRD réduit une matrice bande hermitienne complexe A en une forme
!> tridiagonale symétrique réelle T par une transformation unitaire
!> similaire :
!> Q**H * A * Q = T.
!>

Paramètres

VECT 
!>          VECT est CHARACTER*1
!>          = 'N':  ne pas former Q ;
!>          = 'V':  former Q ;
!>          = 'U':  mettre à jour une matrice X, en formant X*Q.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U':  le triangle supérieur de A est stocké ;
!>          = 'L':  le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

KD

!>          KD est INTEGER
!>          Le nombre de superdiagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB est un tableau COMPLEX*16, dimension (LDAB,N)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice
!>          bande hermitienne A, stocké dans les KD+1 premières lignes du
!>          tableau. La j-ième colonne de A est stockée dans la j-ième
!>          colonne du tableau AB comme suit :
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) pour max(1,j-kd)<=i<=j ;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+kd).
!>          À la sortie, les éléments diagonaux de AB sont écrasés par les
!>          éléments diagonaux de la matrice tridiagonale T ; si KD > 0, les
!>          éléments sur la première superdiagonale (si UPLO = 'U') ou la
!>          première sous-diagonale (si UPLO = 'L') sont écrasés par les
!>          éléments hors diagonale de T ; le reste de AB est écrasé par
!>          les valeurs générées pendant la réduction.
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KD+1.
!>

D

!>          D est un tableau DOUBLE PRECISION, dimension (N)
!>          Les éléments diagonaux de la matrice tridiagonale T.
!>

E

!>          E est un tableau DOUBLE PRECISION, dimension (N-1)
!>          Les éléments hors diagonale de la matrice tridiagonale T :
!>          E(i) = T(i,i+1) si UPLO = 'U' ; E(i) = T(i+1,i) si UPLO = 'L'.
!>

Q

!>          Q est un tableau COMPLEX*16, dimension (LDQ,N)
!>          À l'entrée, si VECT = 'U', alors Q doit contenir une matrice
!>          X de taille N par N ; si VECT = 'N' ou 'V', alors Q n'a pas besoin
!>          d'être défini.
!>
!>          À la sortie :
!>          si VECT = 'V', Q contient la matrice unitaire Q de taille N par N ;
!>          si VECT = 'U', Q contient le produit X*Q ;
!>          si VECT = 'N', le tableau Q n'est pas référencé.
!>

LDQ

!>          LDQ est INTEGER
!>          La dimension principale du tableau Q.
!>          LDQ >= 1, et LDQ >= N si VECT = 'V' ou 'U'.
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!>
