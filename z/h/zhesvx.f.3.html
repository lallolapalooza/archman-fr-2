subroutine zhesvx (character fact, character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision rcond, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer info)

ZHESVX calcule la solution du système d'équations linéaires A * X = B pour les matrices HERMITIENNES

Objectif :

!>
!> ZHESVX utilise la factorisation par pivotement diagonal pour calculer la
!> solution d'un système complexe d'équations linéaires A * X = B,
!> où A est une matrice hermitienne N x N et X et B sont des matrices N x NRHS.
!>
!> Des bornes d'erreur sur la solution et une estimation de la condition
!> sont également fournies.
!>

Description :

!>
!> Les étapes suivantes sont effectuées :
!>
!> 1. Si FACT = 'N', la méthode de pivotement diagonal est utilisée pour factoriser A.
!>    La forme de la factorisation est
!>       A = U * D * U**H,  si UPLO = 'U', ou
!>       A = L * D * L**H,  si UPLO = 'L',
!>    où U (ou L) est un produit de matrices triangulaires unitaires par
!>    permutation et supérieures (inférieures), et D est hermitienne et
!>    bloc-diagonale avec des blocs diagonaux de 1x1 et 2x2.
!>
!> 2. Si certains D(i,i)=0, rendant D exactement singulière, alors la routine
!>    retourne avec INFO = i. Sinon, la forme factorisée de A est utilisée
!>    pour estimer le nombre de condition de la matrice A. Si l'inverse
!>    du nombre de condition est inférieur à la précision machine,
!>    INFO = N+1 est retourné comme avertissement, mais la routine continue
!>    à résoudre pour X et calculer les bornes d'erreur comme décrit ci-dessous.
!>
!> 3. Le système d'équations est résolu pour X en utilisant la forme factorisée
!>    de A.
!>
!> 4. L'affinage itératif est appliqué pour améliorer la solution calculée
!>    et calculer des bornes d'erreur et des estimations d'erreur arrière
!>    pour celle-ci.
!>

Paramètres :

FACT 
!>          FACT est CHARACTER*1
!>          Spécifie si la forme factorisée de A a été fournie ou non à l'entrée.
!>          = 'F':  À l'entrée, AF et IPIV contiennent la forme factorisée
!>                  de A. A, AF et IPIV ne seront pas modifiés.
!>          = 'N':  La matrice A sera copiée dans AF et factorisée.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U':  Le triangle supérieur de A est stocké ;
!>          = 'L':  Le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          Le nombre d'équations linéaires, c'est-à-dire l'ordre de
!>          la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de côtés droits, c'est-à-dire le nombre de colonnes
!>          des matrices B et X.  NRHS >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>          La matrice hermitienne A.  Si UPLO = 'U', la partie supérieure
!>          triangulaire N x N de A contient la partie supérieure
!>          triangulaire de la matrice A, et la partie strictement inférieure
!>          triangulaire de A n'est pas référencée.  Si UPLO = 'L', la partie
!>          inférieure triangulaire N x N de A contient la partie inférieure
!>          triangulaire de la matrice A, et la partie strictement supérieure
!>          triangulaire de A n'est pas référencée.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

AF

!>          AF est un tableau COMPLEX*16, dimension (LDAF,N)
!>          Si FACT = 'F', alors AF est un argument d'entrée et contient à
!>          l'entrée la matrice bloc-diagonale D et les multiplicateurs utilisés
!>          pour obtenir le facteur U ou L de la factorisation
!>          A = U*D*U**H ou A = L*D*L**H calculée par ZHETRF.
!>
!>          Si FACT = 'N', alors AF est un argument de sortie et retourne à la
!>          sortie la matrice bloc-diagonale D et les multiplicateurs utilisés
!>          pour obtenir le facteur U ou L de la factorisation
!>          A = U*D*U**H ou A = L*D*L**H.
!>

LDAF

!>          LDAF est INTEGER
!>          La dimension principale du tableau AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (N)
!>          Si FACT = 'F', alors IPIV est un argument d'entrée et contient à
!>          l'entrée les détails des échanges et la structure bloc
!>          de D, tels que déterminés par ZHETRF.
!>          Si IPIV(k) > 0, alors les lignes et colonnes k et IPIV(k) ont été
!>          échangées et D(k,k) est un bloc diagonal 1x1.
!>          Si UPLO = 'U' et IPIV(k) = IPIV(k-1) < 0, alors les lignes et
!>          colonnes k-1 et -IPIV(k) ont été échangées et D(k-1:k,k-1:k)
!>          est un bloc diagonal 2x2.  Si UPLO = 'L' et IPIV(k) =
!>          IPIV(k+1) < 0, alors les lignes et colonnes k+1 et -IPIV(k) ont été
!>          échangées et D(k:k+1,k:k+1) est un bloc diagonal 2x2.
!>
!>          Si FACT = 'N', alors IPIV est un argument de sortie et contient à
!>          la sortie les détails des échanges et la structure bloc
!>          de D, tels que déterminés par ZHETRF.
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,NRHS)
!>          La matrice côté droit B, de taille N x NRHS.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est un tableau COMPLEX*16, dimension (LDX,NRHS)
!>          Si INFO = 0 ou INFO = N+1, la matrice solution X de taille N x NRHS.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

RCOND

!>          RCOND est DOUBLE PRECISION
!>          L'estimation de l'inverse du nombre de condition de la matrice
!>          A.  Si RCOND est inférieur à la précision machine (en
!>          particulier, si RCOND = 0), la matrice est singulière à la précision
!>          de travail. Cette condition est indiquée par un code de retour
!>          INFO > 0.
!>

FERR

!>          FERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur solution
!>          X(j) (la j-ème colonne de la matrice solution X).
!>          Si XTRUE est la vraie solution correspondant à X(j), FERR(j)
!>          est une borne supérieure estimée pour la magnitude du plus grand
!>          élément de (X(j) - XTRUE) divisée par la magnitude du plus grand
!>          élément de X(j). L'estimation est aussi fiable que
!>          l'estimation pour RCOND, et est presque toujours une légère
!>          supérieure de l'erreur réelle.
!>

BERR

!>          BERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>          L'erreur arrière relative composante par composante de chaque
!>          vecteur solution X(j) (c'est-à-dire la plus petite modification
!>          relative dans un élément de A ou B qui rend X(j) une solution exacte).
!>

WORK 

!>          WORK est un tableau COMPLEX*16, dimension (LWORK)
!>          Espace de travail de taille aléatoire.
!>

LWORK

!>          LWORK est INTEGER
!>          La taille de l'espace de travail WORK.  LWORK >= max(1,N).
!>          Pour le calcul optimal, LWORK >= N*log(N) + 2*N.
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, dimension (N)
!>          Tableau de travail de taille aléatoire.
!>

INFO

!>          INFO est INTEGER
!>          = 0:  Exécution réussie.
!>          < 0:  Si INFO = -i, alors le i-ème argument avait un
!>                valeur illégale.
!>          = i > 0: Si INFO = i et 1 <= i <= N, alors D(i,i) est nul.
!>                   La factorisation a été interrompue car la matrice est singulière.
!>                   Le système peut ne pas avoir de solution unique.
!>          = N+1: Le système est exactement singulier. La factorisation a été
!>                 terminée, mais le nombre de condition est très grand.
!>

WORK

!>          WORK est un tableau de type COMPLEX*16, de dimension (MAX(1,LWORK))
!>          En sortie, si INFO = 0, WORK(1) contient la taille optimale de LWORK.
!>

LWORK

!>          LWORK est un INTEGER
!>          La longueur de WORK.  LWORK >= MAX(1,2*N), et pour de meilleures
!>          performances, lorsque FACT = 'N', LWORK >= MAX(1,2*N,N*NB), où
!>          NB est la taille de bloc optimale pour ZHETRF.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message
!>          d'erreur lié à LWORK n'est émis par XERBLA.
!>

RWORK

!>          RWORK est un tableau de type DOUBLE PRECISION, de dimension (N)
!>

INFO

!>          INFO est un INTEGER
!>          = 0: succès
!>          < 0: si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0: si INFO = i, et i est
!>                <= N:  D(i,i) est exactement zéro. La factorisation
!>                       a été complétée mais le facteur D est exactement
!>                       singulier, donc la solution et les bornes d'erreur n'ont pas
!>                       pu être calculées. RCOND = 0 est renvoyé.
!>                = N+1: D est non singulier, mais RCOND est inférieur à la précision machine,
!>                       ce qui signifie que la matrice est singulière
!>                       à la précision de travail. Néanmoins, la
!>                       solution et les bornes d'erreur sont calculées car
!>                       il existe un certain nombre de situations où la
!>                       solution calculée peut être plus précise que ce que
!>                       la valeur de RCOND suggérerait.
!>
