subroutine zhpevd (character jobz, character uplo, integer n, complex*16, dimension( * ) ap, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer lrwork, integer, dimension( * ) iwork, integer liwork, integer info)

ZHPEVD calcule les valeurs propres et, éventuellement, les vecteurs propres gauches et/ou droits pour d'autres matrices

Objectif :

!>
!> ZHPEVD calcule toutes les valeurs propres et, éventuellement, les vecteurs propres
!> d'une matrice hermitienne complexe A en stockage compacté. Si des vecteurs propres sont
!> demandés, il utilise un algorithme de type "diviser pour régner".
!>
!>

Paramètres

JOBZ 
!>          JOBZ est CHARACTER*1
!>          = 'N' : Calculer uniquement les valeurs propres ;
!>          = 'V' : Calculer les valeurs propres et les vecteurs propres.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U' : Le triangle supérieur de A est stocké ;
!>          = 'L' : Le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A. N >= 0.
!>

AP

!>          AP est un tableau de COMPLEX*16, dimension (N*(N+1)/2)
!>          En entrée, le triangle supérieur ou inférieur de la matrice hermitienne
!>          A, stocké en colonne dans un tableau linéaire. La j-ième colonne de A
!>          est stockée dans le tableau AP comme suit :
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L', AP(i + (j-1)*(2*n-j)/2) = A(i,j) pour j<=i<=n.
!>
!>          En sortie, AP est écrasé par les valeurs générées lors de la
!>          réduction à une forme tridiagonale. Si UPLO = 'U', la diagonale
!>          et la première superdiagonale de la matrice tridiagonale T écrasent
!>          les éléments correspondants de A, et si UPLO = 'L', la
!>          diagonale et la première sous-diagonale de T écrasent les
!>          éléments correspondants de A.
!>

W

!>          W est un tableau de DOUBLE PRECISION, dimension (N)
!>          Si INFO = 0, les valeurs propres par ordre croissant.
!>

Z

!>          Z est un tableau de COMPLEX*16, dimension (LDZ, N)
!>          Si JOBZ = 'V', alors si INFO = 0, Z contient les vecteurs propres orthonormés
!>          de la matrice A, la i-ème colonne de Z contenant le vecteur propre associé à W(i).
!>          Si JOBZ = 'N', alors Z n'est pas référencé.
!>

LDZ

!>          LDZ est INTEGER
!>          La dimension principale du tableau Z. LDZ >= 1, et si
!>          JOBZ = 'V', LDZ >= max(1,N).
!>

WORK

!>          WORK est un tableau de COMPLEX*16, dimension (MAX(1,LWORK))
!>          En sortie, si INFO = 0, WORK(1) contient la valeur requise pour LWORK.
!>

LWORK

!>          LWORK est INTEGER
!>          La dimension du tableau WORK.
!>          Si N <= 1,               LWORK doit être au moins 1.
!>          Si JOBZ = 'N' et N > 1, LWORK doit être au moins N.
!>          Si JOBZ = 'V' et N > 1, LWORK doit être au moins 2*N.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la routine
!>          calcule uniquement les tailles requises pour les tableaux WORK, RWORK et
!>          IWORK, renvoie ces valeurs comme premières entrées
!>          des tableaux WORK, RWORK et IWORK, et aucun message d'erreur
!>          lié à LWORK ou LRWORK ou LIWORK n'est émis par XERBLA.
!>

RWORK

!>          RWORK est un tableau de DOUBLE PRECISION, dimension (MAX(1,LRWORK))
!>          En sortie, si INFO = 0, RWORK(1) contient la valeur requise pour LRWORK.
!>

LRWORK

!>          LRWORK est INTEGER
!>          La dimension du tableau RWORK.
!>          Si N <= 1,               LRWORK doit être au moins 1.
!>          Si JOBZ = 'N' et N > 1, LRWORK doit être au moins N.
!>          Si JOBZ = 'V' et N > 1, LRWORK doit être au moins
!>                    1 + 5*N + 2*N**2.
!>
!>          Si LRWORK = -1, alors une requête d'espace de travail est supposée ; la
!>          routine calcule uniquement les tailles requises pour les tableaux WORK, RWORK
!>          et IWORK, renvoie ces valeurs comme premières entrées
!>          des tableaux WORK, RWORK et IWORK, et aucun message d'erreur
!>          lié à LWORK ou LRWORK ou LIWORK n'est émis par XERBLA.
!>

IWORK

!>          IWORK est un tableau de INTEGER, dimension (MAX(1,LIWORK))
!>          En sortie, si INFO = 0, IWORK(1) contient la valeur requise pour LIWORK.
!>

LIWORK

!>          LIWORK est INTEGER
!>          La dimension du tableau IWORK.
!>          Si JOBZ  = 'N' ou N <= 1, LIWORK doit être au moins 1.
!>          Si JOBZ  = 'V' et N > 1, LIWORK doit être au moins 3 + 5*N.
!>
!>          Si LIWORK = -1, alors une requête d'espace de travail est supposée ; la
!>          routine calcule uniquement les tailles requises pour les tableaux WORK, RWORK
!>          et IWORK, renvoie ces valeurs comme premières entrées
!>          des tableaux WORK, RWORK et IWORK, et aucun message d'erreur
!>          lié à LWORK ou LRWORK ou LIWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale.
!>          > 0 : si INFO = i, l'algorithme n'a pas convergé ; i
!>                éléments hors diagonale d'une forme tridiagonale intermédiaire
!>                n'ont pas convergé vers zéro.
!>
