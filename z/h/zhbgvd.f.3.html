sous-programme zhbgvd (caractère jobz, caractère uplo, entier n, entier ka, entier kb, complexe*16, dimension( ldab, * ) ab, entier ldab, complexe*16, dimension( ldbb, * ) bb, entier ldbb, double précision, dimension( * ) w, complexe*16, dimension( ldz, * ) z, entier ldz, complexe*16, dimension( * ) work, entier lwork, double précision, dimension( * ) rwork, entier lrwork, entier, dimension( * ) iwork, entier liwork, entier info)

ZHBGVD

But :

!>
!> ZHBGVD calcule toutes les valeurs propres, et optionnellement, les vecteurs propres
!> d'un problème propre généralisé Hermitien défini et bande complexe, de
!> la forme A*x=(lambda)*B*x. Ici, A et B sont supposés Hermitiens
!> et bande, et B est également définie positive. Si des vecteurs propres sont
!> désirés, il utilise un algorithme de type diviser pour régner.
!>
!>

Paramètres

JOBZ 
!>          JOBZ est CHARACTER*1
!>          = 'N' : Calculer uniquement les valeurs propres ;
!>          = 'V' : Calculer les valeurs propres et les vecteurs propres.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U' : Les triangles supérieurs de A et B sont stockés ;
!>          = 'L' : Les triangles inférieurs de A et B sont stockés.
!>

N

!>          N est INTEGER
!>          L'ordre des matrices A et B. N >= 0.
!>

KA

!>          KA est INTEGER
!>          Le nombre de super-diagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'. KA >= 0.
!>

KB

!>          KB est INTEGER
!>          Le nombre de super-diagonales de la matrice B si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'. KB >= 0.
!>

AB

!>          AB est un tableau COMPLEX*16, dimension (LDAB, N)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice bande
!>          Hermitienne A, stocké dans les ka+1 premières lignes du tableau. La
!>          colonne j de A est stockée dans la colonne j du tableau AB
!>          comme suit :
!>          si UPLO = 'U', AB(ka+1+i-j,j) = A(i,j) pour max(1,j-ka)<=i<=j ;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+ka).
!>
!>          À la sortie, le contenu de AB est détruit.
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB. LDAB >= KA+1.
!>

BB

!>          BB est un tableau COMPLEX*16, dimension (LDBB, N)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice bande
!>          Hermitienne B, stocké dans les kb+1 premières lignes du tableau. La
!>          colonne j de B est stockée dans la colonne j du tableau BB
!>          comme suit :
!>          si UPLO = 'U', BB(kb+1+i-j,j) = B(i,j) pour max(1,j-kb)<=i<=j ;
!>          si UPLO = 'L', BB(1+i-j,j)    = B(i,j) pour j<=i<=min(n,j+kb).
!>
!>          À la sortie, le facteur S de la factorisation de Cholesky divisée
!>          B = S**H*S, tel que retourné par ZPBSTF.
!>

LDBB

!>          LDBB est INTEGER
!>          La dimension principale du tableau BB. LDBB >= KB+1.
!>

W

!>          W est un tableau DOUBLE PRECISION, dimension (N)
!>          Si INFO = 0, les valeurs propres par ordre croissant.
!>

Z

!>          Z est un tableau COMPLEX*16, dimension (LDZ, N)
!>          Si JOBZ = 'V', alors si INFO = 0, Z contient la matrice Z des
!>          vecteurs propres, la colonne i de Z contenant le
!>          vecteur propre associé à W(i). Les vecteurs propres sont
!>          normalisés de sorte que Z**H*B*Z = I.
!>          Si JOBZ = 'N', alors Z n'est pas référencé.
!>

LDZ

!>          LDZ est INTEGER
!>          La dimension principale du tableau Z. LDZ >= 1, et si
!>          JOBZ = 'V', LDZ >= N.
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO=0, WORK(1) contient le LWORK optimal.
!>

LWORK

!>          LWORK est INTEGER
!>          La dimension du tableau WORK.
!>          Si N <= 1,               LWORK >= 1.
!>          Si JOBZ = 'N' et N > 1, LWORK >= N.
!>          Si JOBZ = 'V' et N > 1, LWORK >= 2*N**2.
!>
!>          Si LWORK = -1, alors une requête de mémoire tampon est supposée ; la routine
!>          calcule uniquement les tailles optimales des tableaux WORK, RWORK et
!>          IWORK, retourne ces valeurs comme premières entrées
!>          des tableaux WORK, RWORK et IWORK, et aucun message d'erreur
!>          lié à LWORK ou LRWORK ou LIWORK n'est émis par XERBLA.
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, dimension (MAX(1,LRWORK))
!>          À la sortie, si INFO=0, RWORK(1) contient le LRWORK optimal.
!>

LRWORK

!>          LRWORK est INTEGER
!>          La dimension du tableau RWORK.
!>          Si N <= 1,               LRWORK >= 1.
!>          Si JOBZ = 'N' et N > 1, LRWORK >= N.
!>          Si JOBZ = 'V' et N > 1, LRWORK >= 1 + 5*N + 2*N**2.
!>
!>          Si LRWORK = -1, alors une requête de mémoire tampon est supposée ; la
!>          routine calcule uniquement les tailles optimales des tableaux WORK, RWORK
!>          et IWORK, retourne ces valeurs comme premières entrées
!>          des tableaux WORK, RWORK et IWORK, et aucun message d'erreur
!>          lié à LWORK ou LRWORK ou LIWORK n'est émis par XERBLA.
!>

IWORK

!>          IWORK est un tableau INTEGER, dimension (MAX(1,LIWORK))
!>          À la sortie, si INFO=0, IWORK(1) contient le LIWORK optimal.
!>

LIWORK

!>          LIWORK est INTEGER
!>          La dimension du tableau IWORK.
!>          Si JOBZ = 'N' ou N <= 1, LIWORK >= 1.
!>          Si JOBZ = 'V' et N > 1, LIWORK >= 3 + 5*N.
!>
!>          Si LIWORK = -1, alors une requête de mémoire tampon est supposée ; la
!>          routine calcule uniquement les tailles optimales des tableaux WORK, RWORK
!>          et IWORK, retourne ces valeurs comme premières entrées
!>          des tableaux WORK, RWORK et IWORK, et aucun message d'erreur
!>          lié à LWORK ou LRWORK ou LIWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 : si INFO = i, et i est :
!>             <= N : l'algorithme n'a pas convergé :
!>                    i éléments hors diagonale d'une forme tridiagonale intermédiaire
!>                    n'ont pas convergé vers zéro ;
!>             > N :   si INFO = N + i, pour 1 <= i <= N, alors ZPBSTF
!>                    a retourné INFO = i : B n'est pas définie positive.
!>                    La factorisation de B n'a pas pu être complétée et
!>                    aucune valeur propre ou vecteur propre n'a été calculé.
!>
