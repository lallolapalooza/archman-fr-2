subroutine zher2k (uplo, trans, n, k, alpha, a, lda, b, ldb, beta, c, ldc)
ZHER2K

Documentation de la Fonction/Sous-programme

sous-programme zher2k (caractère uplo, caractère trans, entier n, entier k, complexe*16 alpha, complexe*16, dimension(lda,*) a, entier lda, complexe*16, dimension(ldb,*) b, entier ldb, double précision beta, complexe*16, dimension(ldc,*) c, entier ldc)

ZHER2K

But :

!>
!> ZHER2K  effectue l'une des opérations hermitiennes de rang 2k
!>
!>    C := alpha*A*B**H + conjg( alpha )*B*A**H + beta*C,
!>
!> ou
!>
!>    C := alpha*A**H*B + conjg( alpha )*B**H*A + beta*C,
!>
!> où  alpha et beta  sont des scalaires avec  beta  réel,  C est une matrice
!> hermitienne de taille n par n et  A et B sont des matrices de taille n par k
!> dans le premier cas et de taille k par n dans le second cas.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>           En entrée,   UPLO  spécifie si la partie  triangulaire  supérieure
!>           ou inférieure du tableau  C  doit être référencée comme suit :
!>
!>              UPLO = 'U' ou 'u'   Seule la partie triangulaire supérieure de C
!>                                  doit être référencée.
!>
!>              UPLO = 'L' ou 'l'   Seule la partie triangulaire inférieure de C
!>                                  doit être référencée.
!>

TRANS

!>          TRANS est CHARACTER*1
!>           En entrée,  TRANS  spécifie l'opération à effectuer comme suit :
!>
!>              TRANS = 'N' ou 'n'    C := alpha*A*B**H          +
!>                                         conjg( alpha )*B*A**H +
!>                                         beta*C.
!>
!>              TRANS = 'C' ou 'c'    C := alpha*A**H*B          +
!>                                         conjg( alpha )*B**H*A +
!>                                         beta*C.
!>

N

!>          N est INTEGER
!>           En entrée,  N spécifie l'ordre de la matrice C.  N doit être
!>           au moins zéro.
!>

K

!>          K est INTEGER
!>           En entrée, avec  TRANS = 'N' ou 'n',  K  spécifie le nombre
!>           de  colonnes  des matrices  A et B,  et en entrée avec
!>           TRANS = 'C' ou 'c',  K  spécifie le nombre de lignes des
!>           matrices  A et B.  K doit être au moins zéro.
!>

ALPHA

!>          ALPHA est COMPLEX*16.
!>           En entrée, ALPHA spécifie le scalaire alpha.
!>

A

!>          A est un tableau COMPLEX*16, de dimension ( LDA, ka ), où ka est
!>           k  lorsque  TRANS = 'N' ou 'n',  et est  n  sinon.
!>           Avant l'entrée avec  TRANS = 'N' ou 'n',  la partie principale n par k
!>           du tableau  A  doit contenir la matrice  A,  sinon
!>           la partie principale k par n du tableau  A  doit contenir la
!>           matrice A.
!>

LDA

!>          LDA est INTEGER
!>           En entrée, LDA spécifie la première dimension de A telle que déclarée
!>           dans le programme appelant.   Lorsque  TRANS = 'N' ou 'n'
!>           alors  LDA doit être au moins  max( 1, n ), sinon  LDA doit
!>           être au moins  max( 1, k ).
!>

B

!>          B est un tableau COMPLEX*16, de dimension ( LDB, kb ), où kb est
!>           k  lorsque  TRANS = 'N' ou 'n',  et est  n  sinon.
!>           Avant l'entrée avec  TRANS = 'N' ou 'n',  la partie principale n par k
!>           du tableau  B  doit contenir la matrice  B,  sinon
!>           la partie principale k par n du tableau  B  doit contenir la
!>           matrice B.
!>

LDB

!>          LDB est INTEGER
!>           En entrée, LDB spécifie la première dimension de B telle que déclarée
!>           dans le programme appelant.   Lorsque  TRANS = 'N' ou 'n'
!>           alors  LDB doit être au moins  max( 1, n ), sinon  LDB doit
!>           être au moins  max( 1, k ).
!>           Inchangé en sortie.
!>

BETA

!>          BETA est DOUBLE PRÉCISION.
!>           En entrée, BETA spécifie le scalaire beta.
!>

C

!>          C est un tableau COMPLEX*16, de dimension ( LDC, N )
!>           Avant l'entrée avec  UPLO = 'U' ou 'u',  la partie supérieure n par n
!>           du tableau C doit contenir la partie triangulaire supérieure
!>           de la matrice hermitienne  et la partie strictement inférieure de C
!>           n'est pas référencée.  En sortie, la partie supérieure
!>           du tableau  C est écrasée par la partie supérieure de la matrice mise à jour.
!>           Avant l'entrée avec  UPLO = 'L' ou 'l',  la partie inférieure n par n
!>           du tableau C doit contenir la partie triangulaire inférieure
!>           de la matrice hermitienne  et la partie strictement supérieure de C
!>           n'est pas référencée.  En sortie, la partie inférieure
!>           du tableau  C est écrasée par la partie inférieure de la matrice mise à jour.
!>           Notez que les parties imaginaires des éléments diagonaux n'ont pas
!>           besoin d'être définies,  elles sont supposées être nulles,  et en sortie
!>           elles sont mises à zéro.
!>

LDC

!>          LDC est INTEGER
!>           En entrée, LDC spécifie la première dimension de C telle que déclarée
!>           dans le programme appelant.   LDC  doit être au moins
!>           max( 1, n ).
!>
