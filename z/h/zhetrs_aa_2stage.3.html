sous-programme zhetrs_aa_2stage (caractere uplo, entier n, entier nrhs, complexe*16, dimension( lda, * ) a, entier lda, complexe*16, dimension( * ) tb, entier ltb, entier, dimension( * ) ipiv, entier, dimension( * ) ipiv2, complexe*16, dimension( ldb, * ) b, entier ldb, entier info)

ZHETRS_AA_2STAGE

Objectif :

!>
!> ZHETRS_AA_2STAGE résout un système d'équations linéaires A*X = B avec une
!> matrice hermitienne A en utilisant la factorisation A = U**H*T*U ou
!> A = L*T*L**H calculée par ZHETRF_AA_2STAGE.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si les détails de la factorisation sont stockés
!>          sous forme de matrice triangulaire supérieure ou inférieure.
!>          = 'U' :  Triangulaire supérieure, forme est A = U**H*T*U ;
!>          = 'L' :  Triangulaire inférieure, forme est A = L*T*L**H.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de côtés droits, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>          Détails des facteurs calculés par ZHETRF_AA_2STAGE.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

TB

!>          TB est un tableau COMPLEX*16, dimension (LTB)
!>          Détails des facteurs calculés par ZHETRF_AA_2STAGE.
!>

LTB

!>          LTB est INTEGER
!>          La taille du tableau TB. LTB >= 4*N.
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (N)
!>          Détails des échanges calculés par
!>          ZHETRF_AA_2STAGE.
!>

IPIV2

!>          IPIV2 est un tableau INTEGER, dimension (N)
!>          Détails des échanges calculés par
!>          ZHETRF_AA_2STAGE.
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,NRHS)
!>          À l'entrée, la matrice du côté droit B.
!>          À la sortie, la matrice solution X.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

INFO

!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!>
