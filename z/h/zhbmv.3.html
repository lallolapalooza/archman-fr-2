subroutine zhbmv (uplo, n, k, alpha, a, lda, x, incx, beta, y, incy)
ZHBMV

Documentation fonction/sous-programme

sous-programme zhbmv (caractère uplo, entier n, entier k, complexe*16 alpha, complexe*16, dimension(lda,*) a, entier lda, complexe*16, dimension(*) x, entier incx, complexe*16 beta, complexe*16, dimension(*) y, entier incy)

ZHBMV

But :

!>
!> ZHBMV  effectue l'opération matrice-vecteur
!>
!>    y := alpha*A*x + beta*y,
!>
!> où alpha et beta sont des scalaires, x et y sont des vecteurs de n éléments et
!> A est une matrice bande hermitienne n par n, avec k super-diagonales.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>           À l'entrée, UPLO spécifie si la partie triangulaire supérieure ou inférieure
!>           de la matrice bande A est fournie comme suit :
!>
!>              UPLO = 'U' ou 'u'   La partie triangulaire supérieure de A est
!>                                  fournie.
!>
!>              UPLO = 'L' ou 'l'   La partie triangulaire inférieure de A est
!>                                  fournie.
!>

N

!>          N est INTEGER
!>           À l'entrée, N spécifie l'ordre de la matrice A.
!>           N doit être supérieur ou égal à zéro.
!>

K

!>          K est INTEGER
!>           À l'entrée, K spécifie le nombre de super-diagonales de la
!>           matrice A. K doit satisfaire 0 .le. K.
!>

ALPHA

!>          ALPHA est COMPLEX*16
!>           À l'entrée, ALPHA spécifie le scalaire alpha.
!>

A

!>          A est un tableau COMPLEX*16, de dimension ( LDA, N )
!>           Avant l'entrée avec UPLO = 'U' ou 'u', la partie principale ( k + 1 )
!>           par n du tableau A doit contenir la partie triangulaire supérieure
!>           bande de la matrice hermitienne, fournie colonne par
!>           colonne, avec la diagonale principale de la matrice dans la ligne
!>           ( k + 1 ) du tableau, la première super-diagonale commençant à la
!>           position 2 dans la ligne k, et ainsi de suite. Le triangle supérieur gauche k par k
!>           du tableau A n'est pas référencé.
!>           Le segment de programme suivant transférera la partie triangulaire supérieure
!>           d'une matrice bande hermitienne du stockage de matrice complète conventionnelle
!>           vers le stockage bande :
!>
!>                 DO 20, J = 1, N
!>                    M = K + 1 - J
!>                    DO 10, I = MAX( 1, J - K ), J
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           Avant l'entrée avec UPLO = 'L' ou 'l', la partie principale ( k + 1 )
!>           par n du tableau A doit contenir la partie triangulaire inférieure
!>           bande de la matrice hermitienne, fournie colonne par
!>           colonne, avec la diagonale principale de la matrice dans la ligne 1 du
!>           tableau, la première sous-diagonale commençant à la position 1 dans
!>           la ligne 2, et ainsi de suite. Le triangle inférieur droit k par k du
!>           tableau A n'est pas référencé.
!>           Le segment de programme suivant transférera la partie triangulaire inférieure
!>           d'une matrice bande hermitienne du stockage de matrice complète conventionnelle
!>           vers le stockage bande :
!>
!>                 DO 20, J = 1, N
!>                    M = 1 - J
!>                    DO 10, I = J, MIN( N, J + K )
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           Notez que les parties imaginaires des éléments diagonaux n'ont pas
!>           besoin d'être définies et sont supposées être nulles.
!>

LDA

!>          LDA est INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle que déclarée
!>           dans le programme appelant (sous-programme). LDA doit être au moins
!>           ( k + 1 ).
!>

X

!>          X est un tableau COMPLEX*16, de dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           Avant l'entrée, le tableau incrémenté X doit contenir
!>           le vecteur x.
!>

INCX

!>          INCX est INTEGER
!>           À l'entrée, INCX spécifie l'incrément pour les éléments de
!>           X. INCX ne doit pas être zéro.
!>

BETA

!>          BETA est COMPLEX*16
!>           À l'entrée, BETA spécifie le scalaire beta.
!>

Y

!>          Y est un tableau COMPLEX*16, de dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           Avant l'entrée, le tableau incrémenté Y doit contenir
!>           le vecteur y. À la sortie, Y est écrasé par le vecteur y mis à jour.
!>

INCY

!>          INCY est INTEGER
!>           À l'entrée, INCY spécifie l'incrément pour les éléments de
!>           Y. INCY ne doit pas être zéro.
!>
