subroutine zher2k (uplo, trans, n, k, alpha, a, lda, b, ldb, beta, c, ldc)
ZHER2K

Documentation de la fonction/sous-programme

sous-programme zher2k (character uplo, character trans, integer n, integer k, complex*16 alpha, complex*16, dimension(lda,*) a, integer lda, complex*16, dimension(ldb,*) b, integer ldb, double precision beta, complex*16, dimension(ldc,*) c, integer ldc)

ZHER2K

Objectif :

!>
!> ZHER2K  effectue l'une des opérations hermitiennes de rang 2k
!>
!>    C := alpha*A*B**H + conjg( alpha )*B*A**H + beta*C,
!>
!> ou
!>
!>    C := alpha*A**H*B + conjg( alpha )*B**H*A + beta*C,
!>
!> où  alpha et beta  sont des scalaires avec  beta  réel,  C est une matrice
!> hermitienne de taille n x n et  A et B  sont des matrices de taille n x k
!> dans le premier cas et de taille k x n dans le second cas.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>           À l'entrée,  UPLO  spécifie si la partie  supérieure  ou inférieure
!>           du tableau  C  doit être référencée comme suit :
!>
!>              UPLO = 'U' ou 'u'   Seule la partie triangulaire supérieure de  C
!>                                  doit être référencée.
!>
!>              UPLO = 'L' ou 'l'   Seule la partie triangulaire inférieure de  C
!>                                  doit être référencée.
!>

TRANS

!>          TRANS est CHARACTER*1
!>           À l'entrée,  TRANS  spécifie l'opération à effectuer comme suit :
!>
!>              TRANS = 'N' ou 'n'    C := alpha*A*B**H          +
!>                                         conjg( alpha )*B*A**H +
!>                                         beta*C.
!>
!>              TRANS = 'C' ou 'c'    C := alpha*A**H*B          +
!>                                         conjg( alpha )*B**H*A +
!>                                         beta*C.
!>

N

!>          N est INTEGER
!>           À l'entrée,  N  spécifie l'ordre de la matrice C.  N doit être
!>           au moins zéro.
!>

K

!>          K est INTEGER
!>           À l'entrée avec  TRANS = 'N' ou 'n',  K  spécifie le nombre
!>           de colonnes des matrices  A et B,  et à l'entrée avec
!>           TRANS = 'C' ou 'c',  K  spécifie le nombre de lignes des
!>           matrices  A et B.  K doit être au moins zéro.
!>

ALPHA

!>          ALPHA est COMPLEX*16 .
!>           À l'entrée, ALPHA spécifie le scalaire alpha.
!>

A

!>          A est un tableau COMPLEX*16, de dimensions ( LDA, ka ), où ka est
!>           k  lorsque  TRANS = 'N' ou 'n',  et est  n  sinon.
!>           Avant l'entrée avec  TRANS = 'N' ou 'n',  la partie principale n x k
!>           du tableau  A  doit contenir la matrice  A,  sinon la partie
!>           principale k x n du tableau  A  doit contenir la matrice A.
!>

LDA

!>          LDA est INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle que déclarée
!>           dans le programme (sous-)appelant.  Lorsque  TRANS = 'N' ou 'n'
!>           alors  LDA doit être au moins  max( 1, n ), sinon  LDA doit
!>           être au moins  max( 1, k ).
!>

B

!>          B est un tableau COMPLEX*16, de dimensions ( LDB, kb ), où kb est
!>           k  lorsque  TRANS = 'N' ou 'n',  et est  n  sinon.
!>           Avant l'entrée avec  TRANS = 'N' ou 'n',  la partie principale n x k
!>           du tableau  B  doit contenir la matrice  B,  sinon la partie
!>           principale k x n du tableau  B  doit contenir la matrice B.
!>

LDB

!>          LDB est INTEGER
!>           À l'entrée, LDB spécifie la première dimension de B telle que déclarée
!>           dans le programme (sous-)appelant.  Lorsque  TRANS = 'N' ou 'n'
!>           alors  LDB doit être au moins  max( 1, n ), sinon  LDB doit
!>           être au moins  max( 1, k ).
!>           Inchangé à la sortie.
!>

BETA

!>          BETA est DOUBLE PRECISION .
!>           À l'entrée, BETA spécifie le scalaire beta.
!>

C

!>          C est un tableau COMPLEX*16, de dimensions ( LDC, N )
!>           Avant l'entrée avec  UPLO = 'U' ou 'u',  la partie supérieure n x n
!>           du tableau C doit contenir la partie triangulaire supérieure
!>           de la matrice  hermitienne et la partie strictement inférieure
!>           de C n'est pas référencée. À la sortie, la partie triangulaire
!>           supérieure du tableau  C  est écrasée par la partie triangulaire
!>           supérieure de la matrice mise à jour.
!>           Avant l'entrée avec  UPLO = 'L' ou 'l',  la partie inférieure n x n
!>           du tableau C doit contenir la partie triangulaire inférieure
!>           de la matrice  hermitienne et la partie strictement supérieure
!>           de C n'est pas référencée. À la sortie, la partie triangulaire
!>           inférieure du tableau  C  est écrasée par la partie triangulaire
!>           inférieure de la matrice mise à jour.
!>           Notez que les parties imaginaires des éléments diagonaux n'ont
!>           pas besoin d'être définies,  elles sont supposées être nulles,
!>           et à la sortie elles sont mises à zéro.
!>

LDC

!>          LDC est INTEGER
!>           À l'entrée, LDC spécifie la première dimension de C telle que déclarée
!>           dans le programme (sous-)appelant.  LDC  doit être  au moins
!>           max( 1, n ).
!>
