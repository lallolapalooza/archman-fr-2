subroutine zhegvx (integer itype, character jobz, character range, character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, double precision vl, double precision vu, integer il, integer iu, double precision abstol, integer m, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

ZHEGVX

Objectif :

!>
!> ZHEGVX calcule les valeurs propres sélectionnées, et éventuellement les vecteurs propres,
!> d'un problème propre généralisé hermitien-défini complexe, de la forme
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  ou  B*A*x=(lambda)*x. Ici, A et
!> B sont supposés hermitiens et B est aussi défini positif.
!> Les valeurs propres et les vecteurs propres peuvent être sélectionnés en spécifiant
!> soit une plage de valeurs, soit une plage d'indices pour les valeurs propres désirées.
!>

Paramètres

ITYPE 
!>          ITYPE est de type INTEGER
!>          Spécifie le type de problème à résoudre :
!>          = 1 :  A*x = (lambda)*B*x
!>          = 2 :  A*B*x = (lambda)*x
!>          = 3 :  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ est de type CHARACTER*1
!>          = 'N' : Calculer uniquement les valeurs propres ;
!>          = 'V' : Calculer les valeurs propres et les vecteurs propres.
!>

RANGE

!>          RANGE est de type CHARACTER*1
!>          = 'A' : toutes les valeurs propres seront trouvées.
!>          = 'V' : toutes les valeurs propres dans l'intervalle ouvert à droite (VL,VU]
!>                 seront trouvées.
!>          = 'I' : les valeurs propres de l'indice IL à l'indice IU seront trouvées.
!>

UPLO

!>          UPLO est de type CHARACTER*1
!>          = 'U' : les triangles supérieurs de A et B sont stockés ;
!>          = 'L' : les triangles inférieurs de A et B sont stockés.
!>

N

!>          N est de type INTEGER
!>          L'ordre des matrices A et B. N >= 0.
!>

A

!>          A est un tableau de type COMPLEX*16, de dimension (LDA, N)
!>          En entrée, la matrice hermitienne A. Si UPLO = 'U', la partie
!>          supérieure triangulaire N x N de A contient la partie supérieure
!>          triangulaire de la matrice A. Si UPLO = 'L', la partie
!>          inférieure triangulaire N x N de A contient la partie inférieure
!>          triangulaire de la matrice A.
!>
!>          En sortie, le triangle inférieur (si UPLO='L') ou le triangle
!>          supérieur (si UPLO='U') de A, y compris la diagonale, est détruit.
!>

LDA

!>          LDA est de type INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,N).
!>

B

!>          B est un tableau de type COMPLEX*16, de dimension (LDB, N)
!>          En entrée, la matrice hermitienne B. Si UPLO = 'U', la partie
!>          supérieure triangulaire N x N de B contient la partie supérieure
!>          triangulaire de la matrice B. Si UPLO = 'L', la partie
!>          inférieure triangulaire N x N de B contient la partie inférieure
!>          triangulaire de la matrice B.
!>
!>          En sortie, si INFO <= N, la partie de B contenant la matrice est
!>          écrasée par le facteur triangulaire U ou L de la factorisation de Cholesky
!>          B = U**H*U ou B = L*L**H.
!>

LDB

!>          LDB est de type INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,N).
!>

VL

!>          VL est de type DOUBLE PRECISION
!>
!>          Si RANGE='V', la borne inférieure de l'intervalle
!>          à rechercher pour les valeurs propres. VL < VU.
!>          Non référencé si RANGE = 'A' ou 'I'.
!>

VU

!>          VU est de type DOUBLE PRECISION
!>
!>          Si RANGE='V', la borne supérieure de l'intervalle
!>          à rechercher pour les valeurs propres. VL < VU.
!>          Non référencé si RANGE = 'A' ou 'I'.
!>

IL

!>          IL est de type INTEGER
!>
!>          Si RANGE='I', l'indice de la
!>          plus petite valeur propre à retourner.
!>          1 <= IL <= IU <= N, si N > 0 ; IL = 1 et IU = 0 si N = 0.
!>          Non référencé si RANGE = 'A' ou 'V'.
!>

IU

!>          IU est de type INTEGER
!>
!>          Si RANGE='I', l'indice de la
!>          plus grande valeur propre à retourner.
!>          1 <= IL <= IU <= N, si N > 0 ; IL = 1 et IU = 0 si N = 0.
!>          Non référencé si RANGE = 'A' ou 'V'.
!>

ABSTOL

!>          ABSTOL est de type DOUBLE PRECISION
!>          La tolérance d'erreur absolue pour les valeurs propres.
!>          Une valeur propre approximative est acceptée comme convergée
!>          lorsqu'elle est déterminée à se trouver dans un intervalle [a,b]
!>          de largeur inférieure ou égale à
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          où EPS est la précision machine. Si ABSTOL est inférieur ou
!>          égal à zéro, alors EPS*|T| sera utilisé à sa place,
!>          où |T| est la norme 1 de la matrice tridiagonale obtenue
!>          en réduisant C à une forme tridiagonale, où C est la matrice symétrique
!>          du problème symétrique standard auquel le problème généralisé
!>          est transformé.
!>
!>          Les valeurs propres seront calculées avec la plus grande précision
!>          lorsque ABSTOL est défini à deux fois le seuil de sous-flux 2*DLAMCH('S'),
!>          et non zéro.
!>          Si cette routine retourne avec INFO>0, indiquant que certains
!>          vecteurs propres n'ont pas convergé, essayez de définir ABSTOL à
!>          2*DLAMCH('S').
!>

M

!>          M est de type INTEGER
!>          Le nombre total de valeurs propres trouvées. 0 <= M <= N.
!>          Si RANGE = 'A', M = N, et si RANGE = 'I', M = IU-IL+1.
!>

W

!>          W est un tableau de type DOUBLE PRECISION, de dimension (N)
!>          Les M premiers éléments contiennent les valeurs propres sélectionnées
!>          par ordre croissant.
!>

Z

!>          Z est un tableau de type COMPLEX*16, de dimension (LDZ, max(1,M))
!>          Si JOBZ = 'N', alors Z n'est pas référencé.
!>          Si JOBZ = 'V', alors si INFO = 0, les M premières colonnes de Z
!>          contiennent les vecteurs propres orthonormés de la matrice A
!>          correspondant aux valeurs propres sélectionnées, avec la i-ème
!>          colonne de Z contenant le vecteur propre associé à W(i).
!>          Les vecteurs propres sont normalisés comme suit :
!>          si ITYPE = 1 ou 2, Z**T*B*Z = I ;
!>          si ITYPE = 3, Z**T*inv(B)*Z = I.
!>
!>          Si un vecteur propre ne parvient pas à converger, alors cette colonne de Z
!>          contient la dernière approximation du vecteur propre, et l'indice
!>          du vecteur propre est retourné dans IFAIL.
!>          Note : l'utilisateur doit s'assurer qu'au moins max(1,M) colonnes sont
!>          fournies dans le tableau Z ; si RANGE = 'V', la valeur exacte de M
!>          n'est pas connue à l'avance et une borne supérieure doit être utilisée.
!>

LDZ

!>          LDZ est de type INTEGER
!>          La dimension principale du tableau Z. LDZ >= 1, et si
!>          JOBZ = 'V', LDZ >= max(1,N).
!>

WORK

!>          WORK est un tableau de type COMPLEX*16, de dimension



NAME


zhegvx - calcul de certains des valeurs propres et vecteurs propres d'une paire générale de matrices hermitiennes complexe, et résout l'un des systèmes linéaires associés

Function/Subroutine Documentation


JOBZ : CHARACTER*1


!>          = 'N':  Compute only the eigenvalues and eigenvectors of the
!>                  generalized eigenvalue problem A*x = lambda*B*x.
!>          = 'V':  Compute the eigenvalues and eigenvectors of the
!>                  generalized eigenvalue problem A*x = lambda*B*x.
!>


RANGE : CHARACTER*1


!>          = 'A': all eigenvalues will be found.
!>          = 'V': all eigenvalues in the interval (VL,VU] will be found.
!>          = 'I': if IL .le. IU are given, all eigenvalues in the
!>                 left most 11111 eigenvalues will be found.
!>


UPLO : CHARACTER*1


!>          = 'U':  Upper triangle of A and B are used for storage.
!>          = 'L':  Lower triangle of A and B are used for storage.
!>


N : INTEGER


!>          The order of the matrices A and B. N >= 0.
!>


A : COMPLEX*16 array, dimension (LDA,N)


!>          On entry with JOBZ = 'S', the upper or lower triangle of
!>          symmetric matrix A.
!>          On exit, A is overwritten by details of its factorization.
!>


LDA : INTEGER


!>          The leading dimension of A. LDA >= max(1,N).
!>


B : COMPLEX*16 array, dimension (LDB,N)


!>          On entry with JOBZ = 'S', the upper or lower triangle of
!>          symmetric matrix B.
!>          On exit, B is overwritten by details of its factorization.
!>


LDB : INTEGER


!>          The leading dimension of B. LDB >= max(1,N).
!>


VL : DOUBLE PRECISION


!>


VU : DOUBLE PRECISION


!>


IL : INTEGER


!>


IU : INTEGER


!>


OFFST : DOUBLE PRECISION


!>


V : COMPLEX*16 array, dimension (LDV,*)


!>          On exit, if JOBZ = 'V', then
!>             the approximate eigenvectors, orthonormalized
!>             with respect to B, are stored in the columns of V.
!>             All the vectors are computed if RANGE = 'A' or 'I'
!>             or RANGE = 'V' and VU - VL is sufficiently large
!>             (V contains at least unspecified number of vectors).
!>             The number of vectors computed is M.
!>          If JOBZ = 'N', then the array V is not referenced.
!>


LDV : INTEGER


!>          The number of rows of V. LDV >= 1.
!>          If JOBZ = 'V', LDV >= N.
!>


WORK : COMPLEX*16 array, dimension (LWORK)


!>          On exit, if INFO = 0, WORK(1) returns the optimal LWORK.
!> 


LWORK


!>          LWORK is INTEGER
!>          The length of the array WORK.  LWORK >= max(1,2*N).
!>          For optimal efficiency, LWORK >= (NB+1)*N,
!>          where NB is the blocksize for ZHETRD returned by ILAENV.
!>
!>          If LWORK = -1, then a workspace query is assumed; the routine
!>          only calculates the optimal size of the WORK array, returns
!>          this value as the first entry of the WORK array, and no error
!>          message related to LWORK is issued by XERBLA.
!> 


RWORK


!>          RWORK is DOUBLE PRECISION array, dimension (7*N)
!> 


IWORK


!>          IWORK is INTEGER array, dimension (5*N)
!> 


IFAIL


!>          IFAIL is INTEGER array, dimension (N)
!>          If JOBZ = 'V', then if INFO = 0, the first M elements of
!>          IFAIL are zero.  If INFO > 0, then IFAIL contains the
!>          indices of the eigenvectors that failed to converge.
!>          If JOBZ = 'N', then IFAIL is not referenced.
!> 


INFO


!>          INFO is INTEGER
!>          = 0:  successful exit
!>          < 0:  if INFO = -i, the i-th argument had an illegal value
!>          > 0:  ZPOTRF or ZHEEVX returned an error code:
!>             <= N:  if INFO = i, ZHEEVX failed to converge;
!>                    i eigenvectors failed to converge.  Their indices
!>                    are stored in array IFAIL.
!>             > N:   if INFO = N + i, for 1 <= i <= N, then the leading
!>                    principal minor of order i of B is not positive.
!>                    The factorization of B could not be completed and
!>                    no eigenvalues or eigenvectors were computed.
!>
