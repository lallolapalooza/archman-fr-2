sous-routine zhet01_3 (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldafac, * ) afac, integer ldafac, complex*16, dimension( * ) e, integer, dimension( * ) ipiv, complex*16, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) rwork, double precision resid)

ZHET01_3

But :

!>
!> ZHET01_3 reconstruit une matrice hermitienne indéfinie A à partir de sa
!> factorisation par blocs L*D*L' ou U*D*U' calculée par ZHETRF_RK
!> (ou ZHETRF_BK) et calcule le résidu
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> où C est la matrice reconstruite et EPS est le epsilon machine.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si la partie triangulaire supérieure ou inférieure de la
!>          matrice hermitienne A est stockée :
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

N

!>          N est INTEGER
!>          Le nombre de lignes et de colonnes de la matrice A.  N >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>          La matrice hermitienne originale A.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC est un tableau COMPLEX*16, dimension (LDAFAC,N)
!>          Diagonale de la matrice diagonale par blocs D et facteurs U ou L
!>          calculés par ZHETRF_RK et ZHETRF_BK :
!>            a) UNIQUEMENT les éléments diagonaux de la matrice hermitienne
!>               diagonale par blocs D sur la diagonale de A, c'est-à-dire
!>               D(k,k) = A(k,k) ;
!>               (les éléments de la supradiagonale (ou sous-diagonale) de D
!>                doivent être fournis en entrée dans le tableau E), et
!>            b) Si UPLO = 'U': facteur U dans la partie supradiagonale de A.
!>               Si UPLO = 'L': facteur L dans la partie sous-diagonale de A.
!>

LDAFAC

!>          LDAFAC est INTEGER
!>          La dimension principale du tableau AFAC.
!>          LDAFAC >= max(1,N).
!>

E

!>          E est un tableau COMPLEX*16, dimension (N)
!>          En entrée, contient les éléments de la supradiagonale (ou sous-diagonale)
!>          de la matrice hermitienne diagonale par blocs D
!>          avec des blocs diagonaux de taille 1x1 ou 2x2, où
!>          Si UPLO = 'U': E(i) = D(i-1,i),i=2:N, E(1) n'est pas référencé ;
!>          Si UPLO = 'L': E(i) = D(i+1,i),i=1:N-1, E(N) n'est pas référencé.
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (N)
!>          Les indices de pivot de ZHETRF_RK (ou ZHETRF_BK).
!>

C

!>          C est un tableau COMPLEX*16, dimension (LDC,N)
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale du tableau C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, dimension (N)
!>

RESID

!>          RESID est DOUBLE PRECISION
!>          Si UPLO = 'L', norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          Si UPLO = 'U', norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
