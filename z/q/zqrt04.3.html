subroutine zqrt04 (integer m, integer n, integer nb, double precision, dimension(6) result)

ZQRT04

But:

!>
!> ZQRT04 teste ZGEQRT et ZGEMQRT.
!>

Paramètres

M 
!>          M est un ENTIER
!>          Nombre de lignes dans la matrice de test.
!>

N

!>          N est un ENTIER
!>          Nombre de colonnes dans la matrice de test.
!>

NB

!>          NB est un ENTIER
!>          Taille du bloc de la matrice de test. NB <= Min(M,N).
!>

RESULT

!>          RESULT est un tableau DOUBLE PRÉCISION, dimension (6)
!>          Résultats de chacun des six tests ci-dessous.
!>
!>          RESULT(1) = | A - Q R |
!>          RESULT(2) = | I - Q^H Q |
!>          RESULT(3) = | Q C - Q C |
!>          RESULT(4) = | Q^H C - Q^H C |
!>          RESULT(5) = | C Q - C Q |
!>          RESULT(6) = | C Q^H - C Q^H |
!>
