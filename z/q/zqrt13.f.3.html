subroutine zqrt13 (integer scale, integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision norma, integer, dimension( 4 ) iseed)

ZQRT13

But :

!>
!> ZQRT13 génère une matrice de rang plein qui peut être mise à l'échelle pour
!> avoir une norme grande ou petite.
!>

Paramètres

SCALE 
!>          SCALE est un entier (INTEGER)
!>          SCALE = 1 : matrice normalement mise à l'échelle
!>          SCALE = 2 : matrice mise à l'échelle vers le haut
!>          SCALE = 3 : matrice mise à l'échelle vers le bas
!>

M

!>          M est un entier (INTEGER)
!>          Le nombre de lignes de la matrice A.
!>

N

!>          N est un entier (INTEGER)
!>          Le nombre de colonnes de A.
!>

A

!>          A est un tableau COMPLEX*16, de dimension (LDA,N)
!>          La matrice M par N, A.
!>

LDA

!>          LDA est un entier (INTEGER)
!>          La dimension principale du tableau A.
!>

NORMA

!>          NORMA est un double précision (DOUBLE PRECISION)
!>          La norme 1 de A.
!>

ISEED

!>          ISEED est un tableau d'entiers, de dimension (4)
!>          Graine pour le générateur de nombres aléatoires
!>
