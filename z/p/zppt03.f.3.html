subroutine zppt03 (character uplo, integer n, complex*16, dimension( * ) a, complex*16, dimension( * ) ainv, complex*16, dimension( ldwork, * ) work, integer ldwork, double precision, dimension( * ) rwork, double precision rcond, double precision resid)

ZPPT03

Objectif :

!>
!> ZPPT03 calcule le résidu d'une matrice hermitienne empaquetée multipliée par son
!> inverse :
!>    norm( I - A*AINV ) / ( N * norm(A) * norm(AINV) * EPS ),
!> où EPS est l'epsilon machine.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si la partie triangulaire supérieure ou inférieure de la
!>          matrice hermitienne A est stockée :
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

N

!>          N est INTEGER
!>          Le nombre de lignes et de colonnes de la matrice A.  N >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (N*(N+1)/2)
!>          La matrice hermitienne originale A, stockée comme une matrice
!>          triangulaire empaquetée.
!>

AINV

!>          AINV est un tableau COMPLEX*16, dimension (N*(N+1)/2)
!>          L'inverse (Hermitien) de la matrice A, stocké comme une matrice
!>          triangulaire empaquetée.
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension (LDWORK,N)
!>

LDWORK

!>          LDWORK est INTEGER
!>          La dimension principale du tableau WORK.  LDWORK >= max(1,N).
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, dimension (N)
!>

RCOND

!>          RCOND est DOUBLE PRECISION
!>          L'inverse du nombre de conditionnement de A, calculé comme
!>          ( 1/norm(A) ) / norm(AINV).
!>

RESID

!>          RESID est DOUBLE PRECISION
!>          norm(I - A*AINV) / ( N * norm(A) * norm(AINV) * EPS )
!>
