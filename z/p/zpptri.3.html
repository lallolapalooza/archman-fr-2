subroutine zpptri (character uplo, integer n, complex*16, dimension( * ) ap, integer info)

ZPPTRI

Objectif:

!>
!> ZPPTRI calcule l'inverse d'une matrice complexe hermitienne définie positive
!> A en utilisant la factorisation de Cholesky A = U**H*U ou A = L*L**H
!> calculée par ZPPTRF.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U':  Le facteur triangulaire supérieur est stocké dans AP;
!>          = 'L':  Le facteur triangulaire inférieur est stocké dans AP.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

AP

!>          AP est un tableau COMPLEX*16, de dimension (N*(N+1)/2)
!>          En entrée, le facteur triangulaire U ou L de la factorisation de Cholesky
!>          A = U**H*U ou A = L*L**H, empaqueté par colonnes comme
!>          un tableau linéaire. La j-ème colonne de U ou L est stockée dans
!>          le tableau AP comme suit :
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = U(i,j) pour 1<=i<=j;
!>          si UPLO = 'L', AP(i + (j-1)*(2n-j)/2) = L(i,j) pour j<=i<=n.
!>
!>          En sortie, le triangle supérieur ou inférieur de l'inverse
!>          (Hermitienne) de A, écrasant le facteur d'entrée U ou L.
!>

INFO

!>          INFO est INTEGER
!>          = 0:  exécution réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0:  si INFO = i, l'élément (i,i) du facteur U ou L est
!>                zéro, et l'inverse n'a pas pu être calculé.
!>
