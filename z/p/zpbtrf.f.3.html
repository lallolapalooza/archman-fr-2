subroutine zpbtrf (character uplo, integer n, integer kd, complex*16, dimension( ldab, * ) ab, integer ldab, integer info)

ZPBTRF

Objectif :

!>
!> ZPBTRF calcule la factorisation de Cholesky d'une matrice bande complexe hermitienne
!> positive définie A.
!>
!> La factorisation a la forme
!>    A = U**H * U,  si UPLO = 'U', ou
!>    A = L  * L**H,  si UPLO = 'L',
!> où U est une matrice triangulaire supérieure et L est une matrice triangulaire inférieure.
!>

Paramètres :

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U':  La partie triangulaire supérieure de A est stockée ;
!>          = 'L':  La partie triangulaire inférieure de A est stockée.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

KD

!>          KD est INTEGER
!>          Le nombre de super-diagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB est COMPLEX*16 array, dimension (LDAB,N)
!>          À l'entrée, la partie triangulaire supérieure ou inférieure de la matrice bande hermitienne
!>          A, stockée dans les KD+1 premières lignes du tableau. La
!>          j-ème colonne de A est stockée dans la j-ème colonne du tableau AB
!>          comme suit :
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) pour max(1,j-kd)<=i<=j ;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+kd).
!>
!>          À la sortie, si INFO = 0, le facteur triangulaire U ou L de la
!>          factorisation de Cholesky A = U**H*U ou A = L*L**H de la matrice bande
!>          A, dans le même format de stockage que A.
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KD+1.
!>

INFO

!>          INFO est INTEGER
!>          = 0:  terminaison réussie
!>          < 0:  si INFO = -i, le i-ème argument était illégal
!>          > 0:  si INFO = i, le mineur principal principal d'ordre i
!>                n'est pas positif, et la factorisation n'a pas pu être
!>                achevée.
!>

Détails supplémentaires :

!>
!>  Le schéma de stockage bande est illustré par l'exemple suivant, quand
!>  N = 6, KD = 2, et UPLO = 'U' :
!>
!>  À l'entrée :                       À la sortie :
!>
!>      *    *   a13  a24  a35  a46      *    *   u13  u24  u35  u46
!>      *   a12  a23  a34  a45  a56      *   u12  u23  u34  u45  u56
!>     a11  a22  a33  a44  a55  a66     u11  u22  u33  u44  u55  u66
!>
!>  De même, si UPLO = 'L' le format de A est le suivant :
!>
!>  À l'entrée :                       À la sortie :
!>
!>     a11  a22  a33  a44  a55  a66     l11  l22  l33  l44  l55  l66
!>     a21  a32  a43  a54  a65   *      l21  l32  l43  l54  l65   *
!>     a31  a42  a53  a64   *    *      l31  l42  l53  l64   *    *
!>
!>  Les éléments du tableau marqués * ne sont pas utilisés par la routine.
!>
