sous-routine zpot02 (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) rwork, double precision resid)

ZPOT02

But :

!>
!> ZPOT02 calcule le résidu pour la résolution d'un système hermitien
!> d'équations linéaires  A*x = b :
!>
!>    RESID = norm(B - A*X) / ( norm(A) * norm(X) * EPS ),
!>
!> où EPS est l'epsilon machine.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si la partie triangulaire supérieure ou inférieure de la
!>          matrice hermitienne A est stockée :
!>          = 'U' : Triangulaire supérieure
!>          = 'L' : Triangulaire inférieure
!>

N

!>          N est INTEGER
!>          Le nombre de lignes et de colonnes de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de colonnes de B, la matrice des seconds membres.
!>          NRHS >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>          La matrice hermitienne originale A.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N)
!>

X

!>          X est un tableau COMPLEX*16, dimension (LDX,NRHS)
!>          Les vecteurs solution calculés pour le système d'équations
!>          linéaires.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.   LDX >= max(1,N).
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,NRHS)
!>          En entrée, les vecteurs seconds membres pour le système d'équations
!>          linéaires.
!>          En sortie, B est écrasé par la différence B - A*X.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, dimension (N)
!>

RESID

!>          RESID est DOUBLE PRECISION
!>          Le maximum sur le nombre de seconds membres de
!>          norm(B - A*X) / ( norm(A) * norm(X) * EPS ).
!>
