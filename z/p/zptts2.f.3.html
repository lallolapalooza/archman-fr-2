subroutine zptts2 (integer iuplo, integer n, integer nrhs, double precision, dimension( * ) d, complex*16, dimension( * ) e, complex*16, dimension( ldb, * ) b, integer ldb)

ZPTTS2 résout un système tridiagonal de la forme AX=B en utilisant la factorisation L D LH calculée par spttrf.

But :

!>
!> ZPTTS2 résout un système tridiagonal de la forme
!>    A * X = B
!> en utilisant la factorisation A = U**H *D*U ou A = L*D*L**H calculée par ZPTTRF.
!> D est une matrice diagonale spécifiée dans le vecteur D, U (ou L) est une matrice
!> bidiagonale unitaire dont la superdiagonale (sous-diagonale) est spécifiée dans
!> le vecteur E, et X et B sont des matrices N par NRHS.
!>

Paramètres

IUPLO 
!>          IUPLO est INTEGER
!>          Spécifie la forme de la factorisation et si le vecteur E est la
!>          superdiagonale du facteur bidiagonal supérieur U ou la sous-diagonale
!>          du facteur bidiagonal inférieur L.
!>          = 1 :  A = U**H *D*U, E est la superdiagonale de U
!>          = 0 :  A = L*D*L**H, E est la sous-diagonale de L
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice tridiagonale A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de côtés droits, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!>

D

!>          D est un tableau de type DOUBLE PRECISION, de dimension (N)
!>          Les n éléments diagonaux de la matrice diagonale D issue de la
!>          factorisation A = U**H *D*U ou A = L*D*L**H.
!>

E

!>          E est un tableau de type COMPLEX*16, de dimension (N-1)
!>          Si IUPLO = 1, les (n-1) éléments superdiagonaux du facteur bidiagonal unitaire
!>          U issu de la factorisation A = U**H*D*U.
!>          Si IUPLO = 0, les (n-1) éléments sous-diagonaux du facteur bidiagonal unitaire
!>          L issu de la factorisation A = L*D*L**H.
!>

B

!>          B est un tableau de type COMPLEX*16, de dimension (LDB,NRHS)
!>          À l'entrée, les vecteurs de côté droit B pour le système d'équations
!>          linéaires.
!>          À la sortie, les vecteurs solution, X.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>
