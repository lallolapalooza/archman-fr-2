subroutine zpotri (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer info)

ZPOTRI

But:

!>
!> ZPOTRI calcule l'inverse d'une matrice hermitienne définie positive complexe A
!> en utilisant la factorisation de Cholesky A = U**H*U ou A = L*L**H
!> calculée par ZPOTRF.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U':  La partie triangulaire supérieure de A est stockée ;
!>          = 'L':  La partie triangulaire inférieure de A est stockée.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

A

!>          A est un tableau COMPLEX*16, de dimension (LDA,N)
!>          En entrée, le facteur triangulaire U ou L de la factorisation de
!>          Cholesky A = U**H*U ou A = L*L**H, tel que calculé par
!>          ZPOTRF.
!>          En sortie, la partie triangulaire supérieure ou inférieure de
!>          l'inverse (Hermitienne) de A, écrasant le facteur d'entrée U ou L.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

INFO

!>          INFO est INTEGER
!>          = 0:  Succès de l'opération
!>          < 0:  Si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0:  Si INFO = i, l'élément (i,i) du facteur U ou L est
!>                zéro, et l'inverse n'a pas pu être calculé.
!>
