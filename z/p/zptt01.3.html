subroutine zptt01 (integer n, double precision, dimension( * ) d, complex*16, dimension( * ) e, double precision, dimension( * ) df, complex*16, dimension( * ) ef, complex*16, dimension( * ) work, double precision resid)

ZPTT01

Objectif :

!>
!> ZPTT01 reconstruit une matrice tridiagonale A à partir de sa factorisation L*D*L'
!> et calcule le résidu
!>    norme(L*D*L' - A) / ( n * norme(A) * EPS ),
!> où EPS est l'epsilon machine.
!>

Paramètres

N 
!>          N est un entier
!>          L'ordre de la matrice A.
!>

D

!>          D est un tableau de type DOUBLE PRECISION, de dimension (N)
!>          Les n éléments diagonaux de la matrice tridiagonale A.
!>

E

!>          E est un tableau de type COMPLEX*16, de dimension (N-1)
!>          Les (n-1) éléments sous-diagonaux de la matrice tridiagonale A.
!>

DF

!>          DF est un tableau de type DOUBLE PRECISION, de dimension (N)
!>          Les n éléments diagonaux du facteur L de la factorisation L*D*L'
!>          de A.
!>

EF

!>          EF est un tableau de type COMPLEX*16, de dimension (N-1)
!>          Les (n-1) éléments sous-diagonaux du facteur L de la
!>          factorisation L*D*L' de A.
!>

WORK

!>          WORK est un tableau de type COMPLEX*16, de dimension (2*N)
!>

RESID

!>          RESID est de type DOUBLE PRECISION
!>          norme(L*D*L' - A) / (n * norme(A) * EPS)
!>
