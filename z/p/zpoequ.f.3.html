subroutine zpoequ (integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) s, double precision scond, double precision amax, integer info)

ZPOEQU

Objectif:

!>
!> ZPOEQU calcule les échelles de lignes et de colonnes destinées à équilibrer
!> une matrice hermitienne définie positive A et à réduire son nombre de conditionnement
!> (par rapport à la norme deux). S contient les facteurs d'échelle,
!> S(i) = 1/sqrt(A(i,i)), choisis de sorte que la matrice mise à l'échelle B avec
!> les éléments B(i,j) = S(i)*A(i,j)*S(j) ait des uns sur la diagonale. Ce
!> choix de S place le nombre de conditionnement de B dans un facteur N du
!> plus petit nombre de conditionnement possible sur toutes les mises à l'échelle
!> diagonales possibles.
!>

Paramètres

N 
!>          N est un entier (INTEGER)
!>          L'ordre de la matrice A. N >= 0.
!>

A

!>          A est un tableau COMPLEX*16, de dimension (LDA,N)
!>          La matrice hermitienne définie positive N x N dont les facteurs d'échelle
!>          doivent être calculés. Seuls les éléments diagonaux de A
!>          sont référencés.
!>

LDA

!>          LDA est un entier (INTEGER)
!>          La dimension principale du tableau A. LDA >= max(1,N).
!>

S

!>          S est un tableau DOUBLE PRECISION, de dimension (N)
!>          Si INFO = 0, S contient les facteurs d'échelle pour A.
!>

SCOND

!>          SCOND est un DOUBLE PRECISION
!>          Si INFO = 0, S contient le rapport du plus petit S(i) au
!>          plus grand S(i). Si SCOND >= 0.1 et AMAX n'est ni trop
!>          grand ni trop petit, il n'est pas utile de mettre à l'échelle avec S.
!>

AMAX

!>          AMAX est un DOUBLE PRECISION
!>          Valeur absolue de l'élément de matrice le plus grand. Si AMAX est très
!>          proche du débordement ou très proche du sous-débordement, la matrice
!>          doit être mise à l'échelle.
!>

INFO

!>          INFO est un entier (INTEGER)
!>          = 0: sortie réussie
!>          < 0: si INFO = -i, le i-ème argument était illégal
!>          > 0: si INFO = i, le i-ème élément diagonal est non positif.
!>
