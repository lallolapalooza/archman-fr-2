subroutine zpprfs (character uplo, integer n, integer nrhs, complex*16, dimension( * ) ap, complex*16, dimension( * ) afp, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZPPRFS

Objectif :

!>
!> ZPPRFS améliore la solution calculée d'un système d'équations linéaires
!> lorsque la matrice des coefficients est hermitienne définie positive
!> et condensée, et fournit des bornes d'erreur et des estimations d'erreur
!> à rebours pour la solution.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U': Le triangle supérieur de A est stocké ;
!>          = 'L': Le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de colonnes
!>          des matrices B et X.  NRHS >= 0.
!>

AP

!>          AP est un tableau COMPLEX*16, de dimension (N*(N+1)/2)
!>          Le triangle supérieur ou inférieur de la matrice hermitienne A,
!>          condensé par colonne dans un tableau linéaire. La j-ième colonne de A
!>          est stockée dans le tableau AP comme suit :
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L', AP(i + (j-1)*(2n-j)/2) = A(i,j) pour j<=i<=n.
!>

AFP

!>          AFP est un tableau COMPLEX*16, de dimension (N*(N+1)/2)
!>          Le facteur triangulaire U ou L de la factorisation de Cholesky
!>          A = U**H*U ou A = L*L**H, tel que calculé par DPPTRF/ZPPTRF,
!>          condensé par colonne dans un tableau linéaire, dans le même format que A
!>          (voir AP).
!>

B

!>          B est un tableau COMPLEX*16, de dimension (LDB,NRHS)
!>          La matrice des seconds membres B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est un tableau COMPLEX*16, de dimension (LDX,NRHS)
!>          En entrée, la matrice solution X, telle que calculée par ZPPTRS.
!>          En sortie, la matrice solution améliorée X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

FERR

!>          FERR est un tableau DOUBLE PRECISION, de dimension (NRHS)
!>          La borne d'erreur directe estimée pour chaque vecteur solution
!>          X(j) (la j-ième colonne de la matrice solution X).
!>          Si XTRUE est la vraie solution correspondant à X(j), FERR(j)
!>          est une borne supérieure estimée de la magnitude du plus grand
!>          élément de (X(j) - XTRUE) divisée par la magnitude du plus grand
!>          élément de X(j). L'estimation est aussi fiable que
!>          l'estimation de RCOND, et est presque toujours une légère
!>          surestimation de l'erreur réelle.
!>

BERR

!>          BERR est un tableau DOUBLE PRECISION, de dimension (NRHS)
!>          L'erreur résiduelle relative composante par composante de chaque
!>          vecteur solution X(j) (c'est-à-dire le plus petit changement relatif
!>          dans un élément de A ou B qui rend X(j) une solution exacte).
!>

WORK

!>          WORK est un tableau COMPLEX*16, de dimension (2*N)
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, de dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0:  Succès de l'exécution
!>          < 0:  Si INFO = -i, le i-ème argument avait une valeur illégale
!>

Paramètres internes :

!>  ITMAX est le nombre maximum d'étapes de raffinement itératif.
!>
