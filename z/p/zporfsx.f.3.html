subroutine zporfsx (character uplo, character equed, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldaf, * ) af, integer ldaf, double precision, dimension( * ) s, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision rcond, double precision, dimension( * ) berr, integer n_err_bnds, double precision, dimension( nrhs, * ) err_bnds_norm, double precision, dimension( nrhs, * ) err_bnds_comp, integer nparams, double precision, dimension(*) params, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZPORFSX

But :

!>
!>    ZPORFSX améliore la solution calculée d'un système d'équations linéaires
!>    lorsque la matrice des coefficients est hermitienne définie positive, et fournit des bornes d'erreur et des estimations d'erreur à posteriori pour la solution. En plus de la borne d'erreur globale, le code fournit une borne d'erreur composante par composante maximale si possible. Voir
!>    les commentaires pour ERR_BNDS_NORM et ERR_BNDS_COMP pour les détails des
!>    bornes d'erreur.
!>
!>    Le système d'équations linéaires d'origine peut avoir été équilibré
!>    avant d'appeler cette routine, comme décrit par les arguments EQUED et S
!>    ci-dessous. Dans ce cas, la solution et les bornes d'erreur retournées sont
!>    pour le système non équilibré d'origine.
!> 
!>     Certains paramètres optionnels sont regroupés dans le tableau PARAMS. Ces
!>     paramètres déterminent comment le raffinement est effectué, mais souvent les
!>     valeurs par défaut sont acceptables. Si les valeurs par défaut sont acceptables, les utilisateurs
!>     peuvent passer NPARAMS = 0, ce qui empêche le code source d'accéder
!>     à l'argument PARAMS.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>       = 'U' :  Le triangle supérieur de A est stocké ;
!>       = 'L' :  Le triangle inférieur de A est stocké.
!>

EQUED

!>          EQUED est CHARACTER*1
!>     Spécifie la forme d'équilibration qui a été effectuée sur A
!>     avant d'appeler cette routine. Ceci est nécessaire pour calculer
!>     la solution et les bornes d'erreur correctement.
!>       = 'N' :  Aucune équilibration
!>       = 'Y' :  Équilibration des lignes et des colonnes, c'est-à-dire que A a été
!>               remplacé par diag(S) * A * diag(S).
!>               Le second membre B a été modifié en conséquence.
!>

N

!>          N est INTEGER
!>     L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>     Le nombre de seconds membres, c'est-à-dire le nombre de colonnes
!>     des matrices B et X.  NRHS >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>     La matrice hermitienne A.  Si UPLO = 'U', la partie triangulaire
!>     supérieure N par N de A contient la partie triangulaire supérieure
!>     de la matrice A, et la partie strictement triangulaire inférieure de A
!>     n'est pas référencée.  Si UPLO = 'L', la partie triangulaire
!>     inférieure N par N de A contient la partie triangulaire inférieure
!>     de la matrice A, et la partie strictement triangulaire supérieure de A n'est
!>     pas référencée.
!>

LDA

!>          LDA est INTEGER
!>     La dimension principale du tableau A.  LDA >= max(1,N).
!>

AF

!>          AF est un tableau COMPLEX*16, dimension (LDAF,N)
!>     Le facteur triangulaire U ou L de la factorisation de Cholesky
!>     A = U**H*U ou A = L*L**H, tel que calculé par ZPOTRF.
!>

LDAF

!>          LDAF est INTEGER
!>     La dimension principale du tableau AF.  LDAF >= max(1,N).
!>

S

!>          S est un tableau DOUBLE PRECISION, dimension (N)
!>     Les facteurs d'échelle pour A.  Si EQUED = 'Y', A est multiplié à
!>     gauche et à droite par diag(S).  S est un argument d'entrée si FACT =
!>     'F' ; sinon, S est un argument de sortie.  Si FACT = 'F' et EQUED
!>     = 'Y', chaque élément de S doit être positif.  Si S est en sortie, chaque
!>     élément de S est une puissance de la base. Si S est en entrée, chaque
!>     élément de S doit être une puissance de la base pour assurer une solution fiable
!>     et des estimations d'erreur. La mise à l'échelle par des puissances de la base ne provoque
!>     pas d'erreurs d'arrondi sauf si le résultat déborde (sous- ou sur-).
!>     Les erreurs d'arrondi pendant la mise à l'échelle conduisent à un raffinement avec une matrice qui
!>     n'est pas équivalente à la matrice d'entrée, produisant des estimations d'erreur
!>     qui peuvent ne pas être fiables.
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,NRHS)
!>     La matrice B du second membre.
!>

LDB

!>          LDB est INTEGER
!>     La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est un tableau COMPLEX*16, dimension (LDX,NRHS)
!>     En entrée, la matrice solution X, telle que calculée par ZGETRS.
!>     En sortie, la matrice solution améliorée X.
!>

LDX

!>          LDX est INTEGER
!>     La dimension principale du tableau X.  LDX >= max(1,N).
!>

RCOND

!>          RCOND est DOUBLE PRECISION
!>     Conditionnement inverse échelonné. Ceci est une estimation du
!>     conditionnement inverse de Skeel de la matrice A après
!>     équilibration (si effectuée).  Si cela est inférieur à la précision machine
!>     (en particulier, si c'est zéro), la matrice est singulière
!>     à la précision de travail.  Notez que l'erreur peut toujours être faible même
!>     si ce nombre est très petit et que la matrice semble mal
!>     conditionnée.
!>

BERR

!>          BERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>     Erreur relative composante par composante à posteriori. Ceci est l'
!>     erreur relative composante par composante à posteriori de chaque vecteur solution X(j)
!>     (c'est-à-dire, le plus petit changement relatif dans un élément de A ou B qui
!>     rend X(j) une solution exacte).
!>

N_ERR_BNDS

!>          N_ERR_BNDS est INTEGER
!>     Nombre d'estimations d'erreur à retourner pour chaque second membre
!>     et chaque type (global ou composante par composante). Voir ERR_BNDS_NORM et
!>     ERR_BNDS_COMP ci-dessous.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM est un tableau DOUBLE PRECISION, dimension (NRHS, N_ERR_BNDS)
!>     Pour chaque second membre, ce tableau contient des informations sur
!>     diverses bornes d'erreur et conditionnements correspondant à l'
!>     erreur relative globale, qui est définie comme suit :
!>
!>     Erreur relative globale dans le i-ème vecteur solution :
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     Le tableau est indexé par le type d'information d'erreur comme décrit
!>     ci-dessous. Il y a actuellement jusqu'à trois éléments d'information
!>     retournés.
!>
!>     Le premier indice dans ERR_BNDS_NORM(i,:) correspond au i-ème
!>     second membre.
!>
!>     Le second indice dans ERR_BNDS_NORM(:,err) contient les trois champs
!>     suivants :
!>     err = 1  booléen. Faites confiance à la réponse

NOM

ZPORFSX — Résout un système d'équations linéaires A*X = B pour un matrice A hermitienne définie positive, et calcule des bornes pour l'erreur relative.

SYNTAXE

 SUBROUTINE ZPORFSX(
 UPLO, N, NRHS, A, LDA, AF, LDAF, IPIV, B, LDB, X, LDX, FERR, BERR,
 CND, INFO, PARAMS, WORK, RWORK
 )

EXPLICATION DES ARGUMENTS

UPLO
CHARACTER*1. Spécifie si la matrice A est triangulaire supérieure ou inférieure.
'U': A est triangulaire supérieure.
'L': A est triangulaire inférieure.
N
INTEGER. L'ordre de la matrice A. 0 <= N.
NRHS
INTEGER. Le nombre de colonnes de la matrice B. Le nombre de systèmes à résoudre.
0 <= NRHS.
A
COMPLEX*16 array, dimension (LDA, N). La matrice hermitienne définie positive.
LDA
INTEGER. Le premier indice de A.
AF
COMPLEX*16 array, dimension (LDAF, N). La factorisation LU de A, telle que retournée par ZPORFA.
LDAF
INTEGER. Le premier indice de AF.
IPIV
INTEGER array, dimension (N). Les informations sur les permutations telles que retournées par ZPORFA. Il est supposé que IPIV(k) est le rang de la ligne/colonne échangée avec la k-ième ligne/colonne.
B
COMPLEX*16 array, dimension (LDB, NRHS). Sur l'entrée, la matrice du second membre.
LDB
INTEGER. Le premier indice de B.
X
COMPLEX*16 array, dimension (LDX, NRHS). Sur la sortie, la solution des systèmes linéaires Ax = b.
LDX
INTEGER. Le premier indice de X.
FERR
DOUBLE PRECISION array, dimension (NRHS). Sur la sortie, FERR(j) est une borne supérieure estimée pour l'erreur relative sur la j-ième colonne de la solution X. La borne relative est définie comme max(|XTRUE - X| / |X|).
BERR
DOUBLE PRECISION array, dimension (NRHS). Sur la sortie, BERR(j) est une borne supérieure estimée pour l'erreur relative par composante sur la j-ième colonne de la solution X. La borne relative est définie comme max(|XTRUE - X| / |X|).
CND
DOUBLE PRECISION array, dimension (NRHS). Sur la sortie, CND(j) est une estimation du nombre de conditionnement de la j-ième colonne de la matrice.
INFO
INTEGER. Sur la sortie, INFO est un entier et vaut 0 en cas de succès, et un entier négatif en cas d'échec.
PARAMS
DOUBLE PRECISION array, dimension (3). Paramètres optionnels.
WORK
COMPLEX*16 array, dimension (2*N).
RWORK
DOUBLE PRECISION array, dimension (2*N).

DETAILS

Les arguments optionnels PARAMS sont utilisés pour spécifier des paramètres de l'algorithme.

PARAMS(1) : Indique si le raffinement itératif doit être effectué.

   = 1.0D+0 : Raffinement itératif effectué (défaut).
   = 0.0D+0 : Aucun raffinement itératif effectué. Aucune borne d'erreur
              calculée.

PARAMS(2) : Nombre maximal de calculs de résidus autorisés pour le raffinement itératif.

   = 10 : Nombre maximal de calculs de résidus (défaut).
   = 100 : Permet la convergence même avec des factorisations approximatives.

PARAMS(3) : Indique si une solution avec une petite erreur relative par composante doit être recherchée.

   = 1.0 : Tenter de trouver une solution avec une petite erreur relative par
          composante (défaut).
   = 0.0 : Ne pas rechercher une solution avec une petite erreur relative par
          composante.

Les matrices de sortie FERR, BERR et CND contiennent des informations sur l'erreur.

FERR
Contient des estimations des erreurs relatives par colonne.
BERR
Contient des estimations des erreurs relatives par composante.
CND
Contient des estimations des nombres de conditionnement.

Les matrices ERR_BNDS_NORM et ERR_BNDS_COMP (non directement accessibles via l'interface mais utilisées en interne) contiennent des informations plus détaillées sur les erreurs.

ERR_BNDS_NORM

!>          ERR_BNDS_NORM est un tableau DOUBLE PRECISION, dimension (NRHS, 3)
!>     Pour chaque côté droit, ce tableau contient des informations sur
!>     diverses bornes d'erreur et nombres de conditionnement correspondant
!>     à l'erreur relative normwise, qui est définie comme suit :
!>
!>     Erreur relative normwise dans la i-ème colonne de la solution :
!>                    norm(XTRUE - X, inf)
!>             ---------------------------
!>                        norm(X, inf)
!>
!>     Le tableau est indexé par le côté droit i (sur lequel l'erreur
!>     relative normwise dépend), et le type d'information d'erreur comme
!>     décrit ci-dessous. Il y a actuellement jusqu'à trois
!>     informations retournées pour chaque côté droit.
!>
!>     Le premier indice dans ERR_BNDS_NORM(i,:) correspond au i-ème
!>     côté droit.
!>
!>     Le deuxième indice dans ERR_BNDS_NORM(:,err) contient les trois
!>     champs suivants :
!>     err = 1  booléen. Faites confiance à la réponse si le
!>              nombre de conditionnement réciproque est inférieur au seuil
!>              sqrt(n) * dlamch('Epsilon').
!>
!>     err = 2  borne d'erreur : La borne d'erreur avant estimée,
!>              presque certainement dans un facteur de 10 de l'erreur
!>              vraie tant que la prochaine entrée est supérieure au seuil
!>              sqrt(n) * dlamch('Epsilon'). Cette borne d'erreur ne devrait
!>              être fiable que si le booléen précédent est vrai.
!>
!>     err = 3  Nombre de conditionnement réciproque : Estimation du nombre
!>              de conditionnement réciproque normwise. Comparé au seuil
!>              sqrt(n) * dlamch('Epsilon') pour déterminer si l'estimation
!>              d'erreur est . Ces nombres de conditionnement
!>              réciproques sont 1 / (norm(Z^{-1},inf) * norm(Z,inf)) pour une
!>              matrice Z correctement mise à l'échelle.
!>              Soit Z = S*A, où S met à l'échelle chaque ligne par une puissance
!>              de la base de façon à ce que toutes les sommes de lignes absolues
!>              de Z soient approximativement 1.
!>
!>     Voir la note de travail Lapack 165 pour plus de détails et des
!>     précautions supplémentaires.
!>

ERR_BNDS_COMP

!>          ERR_BNDS_COMP est un tableau DOUBLE PRECISION, dimension (NRHS, 3)
!>     Pour chaque côté droit, ce tableau contient des informations sur
!>     diverses bornes d'erreur et nombres de conditionnement correspondant
!>     à l'erreur relative par composante, qui est définie comme suit :
!>
!>     Erreur relative par composante dans la i-ème colonne de la solution :
!>                    abs(XTRUE(j,i) - X(j,i))
!>             max_j ----------------------
!>                         abs(X(j,i))
!>
!>     Le tableau est indexé par le côté droit i (sur lequel l'erreur
!>     relative par composante dépend), et le type d'information d'erreur comme
!>     décrit ci-dessous. Il y a actuellement jusqu'à trois
!>     informations retournées pour chaque côté droit. Si
!>     la précision par composante n'est pas demandée (PARAMS(3) = 0.0), alors
!>     ERR_BNDS_COMP n'est pas accédé. Si N_ERR_BNDS < 3, alors au plus
!>     les premières (:,N_ERR_BNDS) entrées sont retournées.
!>
!>     Le premier indice dans ERR_BNDS_COMP(i,:) correspond au i-ème
!>     côté droit.
!>
!>     Le deuxième indice dans ERR_BNDS_COMP(:,err) contient les trois
!>     champs suivants :
!>     err = 1  booléen. Faites confiance à la réponse si le
!>              nombre de conditionnement réciproque est inférieur au seuil
!>              sqrt(n) * dlamch('Epsilon').
!>
!>     err = 2  borne d'erreur : La borne d'erreur avant estimée,
!>              presque certainement dans un facteur de 10 de l'erreur
!>              vraie tant que la prochaine entrée est supérieure au seuil
!>              sqrt(n) * dlamch('Epsilon'). Cette borne d'erreur ne devrait
!>              être fiable que si le booléen précédent est vrai.
!>
!>     err = 3  Nombre de conditionnement réciproque : Estimation du nombre
!>              de conditionnement réciproque par composante. Comparé au seuil
!>              sqrt(n) * dlamch('Epsilon') pour déterminer si l'estimation
!>              d'erreur est . Ces nombres de conditionnement
!>              réciproques sont 1 / (norm(Z^{-1},inf) * norm(Z,inf)) pour une
!>              matrice Z correctement mise à l'échelle.
!>              Soit Z = S*(A*diag(x)), où x est la solution pour
!>              le côté droit courant et S met à l'échelle chaque ligne de
!>              A*diag(x) par une puissance de la base de façon à ce que toutes
!>              les sommes de lignes absolues de Z soient approximativement 1.
!>
!>     Voir la note de travail Lapack 165 pour plus de détails et des
!>     précautions supplémentaires.
!>

NPARAMS

!>          NPARAMS est un entier
!>     Spécifie le nombre de paramètres définis dans PARAMS. Si <= 0,
!>     le tableau PARAMS n'est jamais référencé et les valeurs par défaut sont
!>     utilisées.
!>

PARAMS

!>          PARAMS est un tableau DOUBLE PRECISION, dimension NPARAMS
!>     Spécifie les paramètres de l'algorithme. Si une entrée est < 0.0, alors
!>     cette entrée sera remplie avec la valeur par défaut utilisée pour ce
!>     paramètre. Seules les positions jusqu'à NPARAMS sont accédées ; les
!>     valeurs par défaut sont utilisées pour les paramètres de numéro supérieur.
!>
!>       PARAMS(LA_LINRX_ITREF_I = 1) : Indique si le raffinement itératif
!>            doit être effectué ou non.
!>         Défaut: 1.0D+0
!>            = 0.0:  Aucun raffinement n'est effectué, et aucune borne d'erreur
!>                    n'est calculée.
!>            = 1.0:  Utilise l'algorithme de raffinement en double précision,
!>                    éventuellement avec des calculs en double précision doublés
!>                    si l'environnement de compilation ne supporte pas la
!>                    double précision.
!>              (autres valeurs sont réservées pour une utilisation future)
!>
!>       PARAMS(LA_LINRX_ITHRESH_I = 2) : Nombre maximal de calculs de résidus
!>            autorisés pour le raffinement.
!>         Défaut: 10
!>         Agressif: Défini à 100 pour permettre la convergence en utilisant
!>                     des factorisations approximatives ou des factorisations
!>                     autres que LU. Si la factorisation utilise une technique
!>                     autre que l'élimination gaussienne, les garanties dans
!>                     err_bnds_norm et err_bnds_comp peuvent ne plus être
!>                     fiables.
!>
!>       PARAMS(LA_LINRX_CWISE_I = 3) : Indicateur déterminant si le code
!>            tente de trouver une solution avec une petite erreur relative par
!>            composante dans l'algorithme en double précision.
!>            Positif est vrai, 0.0 est faux.
!>         Défaut: 1.0 (tenter la convergence par composante)
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension (2*N)
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, dimension (2*N)
!>

INFO

!>          INFO est un entier
!>       = 0:  Sortie réussie. La solution pour chaque côté droit est
!>         garantie.
!>       < 0:  Si INFO = -i, le i-ème argument avait une valeur illégale.
!>       > 0 et <= N:  U(INFO,INFO) est exactement zéro. La factorisation
!>         a été complétée, mais le facteur U est exactement singulier, donc
!>         la solution et les bornes d'erreur n'ont pas pu être calculées. RCOND = 0
!>         est retourné.
!>       = N+J: La solution correspondant au J-ième côté droit n'est
!>         pas garantie. Les solutions correspondant aux autres côtés droits
!>         K avec K > J peuvent également ne pas être garanties, mais
!>         seul le premier côté droit de ce type est rapporté. Si une petite
!>         erreur par composante n'est pas demandée (PARAMS(3) = 0.0) alors
!>         le J-ième côté droit est le premier avec une borne d'erreur normwise
!>         qui n'est pas garantie (le plus petit J tel que
!>         ERR_BNDS_NORM(J,1) = 0.0). Par défaut (PARAMS(3) = 1.0)
!>         le J-ième côté droit est le premier avec une borne d'erreur normwise ou
!>         par composante qui n'est pas garantie (le plus petit
!>         J tel que ERR_BNDS_NORM(J,1) = 0.0 ou
!>         ERR_BNDS_COMP(J,1) = 0.0). Voir la définition de
!>         ERR_BNDS_NORM(:,1) et ERR_BNDS_COMP(:,1). Pour obtenir des informations
!>         sur tous les côtés droits, vérifiez ERR_BNDS_NORM ou
!>         ERR_BNDS_COMP.
!>
