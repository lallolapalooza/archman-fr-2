subroutine zpbrfs (character uplo, integer n, integer kd, integer nrhs, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( ldafb, * ) afb, integer ldafb, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZPBRFS

Objectif :

!>
!> ZPBRFS améliore la solution calculée d'un système d'équations linéaires
!> lorsque la matrice des coefficients est hermitienne définie positive
!> et bande, et fournit des bornes d'erreur et des estimations d'erreur arrière
!> pour la solution.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U' : Le triangle supérieur de A est stocké ;
!>          = 'L' : Le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

KD

!>          KD est INTEGER
!>          Le nombre de super-diagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de côtés droits, c'est-à-dire le nombre de colonnes
!>          des matrices B et X.  NRHS >= 0.
!>

AB

!>          AB est COMPLEX*16 array, dimension (LDAB,N)
!>          Le triangle supérieur ou inférieur de la matrice bande hermitienne A,
!>          stocké dans les KD+1 premières lignes du tableau. La j-ième colonne
!>          de A est stockée dans la j-ième colonne du tableau AB comme suit :
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) pour max(1,j-kd)<=i<=j ;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KD+1.
!>

AFB

!>          AFB est COMPLEX*16 array, dimension (LDAFB,N)
!>          Le facteur triangulaire U ou L de la factorisation de Cholesky
!>          A = U**H*U ou A = L*L**H de la matrice bande A calculée par
!>          ZPBTRF, dans le même format de stockage que A (voir AB).
!>

LDAFB

!>          LDAFB est INTEGER
!>          La dimension principale du tableau AFB.  LDAFB >= KD+1.
!>

B

!>          B est COMPLEX*16 array, dimension (LDB,NRHS)
!>          La matrice du côté droit B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est COMPLEX*16 array, dimension (LDX,NRHS)
!>          En entrée, la matrice solution X, telle que calculée par ZPBTRS.
!>          En sortie, la matrice solution améliorée X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

FERR

!>          FERR est DOUBLE PRECISION array, dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur solution
!>          X(j) (la j-ième colonne de la matrice solution X).
!>          Si XTRUE est la vraie solution correspondant à X(j), FERR(j)
!>          est une borne supérieure estimée de la magnitude du plus grand
!>          élément de (X(j) - XTRUE) divisée par la magnitude du plus grand
!>          élément de X(j). L'estimation est aussi fiable que
!>          l'estimation de RCOND, et est presque toujours une légère
!>          surestimation de l'erreur réelle.
!>

BERR

!>          BERR est DOUBLE PRECISION array, dimension (NRHS)
!>          L'erreur arrière relative composante par composante de chaque vecteur solution
!>          X(j) (c'est-à-dire le plus petit changement relatif dans
!>          tout élément de A ou B qui rend X(j) une solution exacte).
!>

WORK

!>          WORK est COMPLEX*16 array, dimension (2*N)
!>

RWORK

!>          RWORK est DOUBLE PRECISION array, dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 : succès
!>          < 0 : si INFO = -i, le i-ième argument avait une valeur illégale
!>

Paramètres internes :

!>  ITMAX est le nombre maximum d'étapes de raffinement itératif.
!>
