subroutine zpftrs (transr, uplo, n, nrhs, a, b, ldb, info)

Documentation de la fonction/sous-programme

sous-routine zpftrs (caractere transr, caractere uplo, entier n, entier nrhs, complexe*16, dimension( 0: * ) a, complexe*16, dimension( ldb, * ) b, entier ldb, entier info)

ZPFTRS

Objectif :

!>
!> ZPFTRS résout un système d'équations linéaires A*X = B avec une matrice
!> hermitienne positive définie A en utilisant la factorisation de Cholesky
!> A = U**H*U ou A = L*L**H calculée par ZPFTRF.
!>

Paramètres

TRANSR 
!>          TRANSR est CHARACTER*1
!>          = 'N' :  La représentation normale TRANSR de RFP A est stockée ;
!>          = 'C' :  La représentation conjuguée-transposée TRANSR de RFP A est stockée.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U' : Le triangle supérieur de RFP A est stocké ;
!>          = 'L' : Le triangle inférieur de RFP A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de côtés droits, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension ( N*(N+1)/2 ) ;
!>          Le facteur triangulaire U ou L de la factorisation de Cholesky
!>          de RFP A = U**H*U ou RFP A = L*L**H, tel que calculé par ZPFTRF.
!>          Voir la note ci-dessous pour plus de détails sur RFP A.
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,NRHS)
!>          À l'entrée, la matrice des côtés droits B.
!>          À la sortie, la matrice solution X.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

INFO

!>          INFO est INTEGER
!>          = 0 : succès de l'exécution
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>

Détails supplémentaires :

!>
!>  Nous considérons d'abord le format standard compact lorsque N est pair.
!>  Nous donnons un exemple où N = 6.
!>
!>      AP est Supérieur           AP est Inférieur
!>
!>   00 01 02 03 04 05       00
!>      11 12 13 14 15       10 11
!>         22 23 24 25       20 21 22
!>            33 34 35       30 31 32 33
!>               44 45       40 41 42 43 44
!>                  55       50 51 52 53 54 55
!>
!>
!>  Soit TRANSR = 'N'. RFP contient AP comme suit :
!>  Pour UPLO = 'U', le trapèze supérieur A(0:5,0:2) consiste en les trois
!>  dernières colonnes de AP supérieur. Le triangle inférieur A(4:6,0:2) consiste en
!>  la conjuguée-transposée des trois premières colonnes de AP supérieur.
!>  Pour UPLO = 'L', le trapèze inférieur A(1:6,0:2) consiste en les trois
!>  premières colonnes de AP inférieur. Le triangle supérieur A(0:2,0:2) consiste en
!>  la conjuguée-transposée des trois dernières colonnes de AP inférieur.
!>  Pour dénoter le conjugué, nous plaçons -- au-dessus de l'élément. Ceci couvre le
!>  cas N pair et TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>                                -- -- --
!>        03 04 05                33 43 53
!>                                   -- --
!>        13 14 15                00 44 54
!>                                      --
!>        23 24 25                10 11 55
!>
!>        33 34 35                20 21 22
!>        --
!>        00 44 45                30 31 32
!>        -- --
!>        01 11 55                40 41 42
!>        -- -- --
!>        02 12 22                50 51 52
!>
!>  Maintenant, soit TRANSR = 'C'. RFP A dans les deux cas UPLO est juste la conjuguée-
!>  transposée de RFP A ci-dessus. On obtient donc :
!>
!>
!>           RFP A                   RFP A
!>
!>     -- -- -- --                -- -- -- -- -- --
!>     03 13 23 33 00 01 02    33 00 10 20 30 40 50
!>     -- -- -- -- --                -- -- -- -- --
!>     04 14 24 34 44 11 12    43 44 11 21 31 41 51
!>     -- -- -- -- -- --                -- -- -- --
!>     05 15 25 35 45 55 22    53 54 55 22 32 42 52
!>
!>
!>  Nous considérons ensuite le format standard compact lorsque N est impair.
!>  Nous donnons un exemple où N = 5.
!>
!>     AP est Supérieur            AP est Inférieur
!>
!>   00 01 02 03 04              00
!>      11 12 13 14              10 11
!>         22 23 24              20 21 22
!>            33 34              30 31 32 33
!>               44              40 41 42 43 44
!>
!>
!>  Soit TRANSR = 'N'. RFP contient AP comme suit :
!>  Pour UPLO = 'U', le trapèze supérieur A(0:4,0:2) consiste en les trois
!>  dernières colonnes de AP supérieur. Le triangle inférieur A(3:4,0:1) consiste en
!>  la conjuguée-transposée des deux premières colonnes de AP supérieur.
!>  Pour UPLO = 'L', le trapèze inférieur A(0:4,0:2) consiste en les trois
!>  premières colonnes de AP inférieur. Le triangle supérieur A(0:1,1:2) consiste en
!>  la conjuguée-transposée des deux dernières colonnes de AP inférieur.
!>  Pour dénoter le conjugué, nous plaçons -- au-dessus de l'élément. Ceci couvre le
!>  cas N impair et TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>                                   -- --
!>        02 03 04                00 33 43
!>                                      --
!>        12 13 14                10 11 44
!>
!>        22 23 24                20 21 22
!>        --
!>        00 33 34                30 31 32
!>        -- --
!>        01 11 44                40 41 42
!>
!>  Maintenant, soit TRANSR = 'C'. RFP A dans les deux cas UPLO est juste la conjuguée-
!>  transposée de RFP A ci-dessus. On obtient donc :
!>
!>
!>           RFP A                   RFP A
!>
!>     -- -- --                   -- -- -- -- -- --
!>     02 12 22 00 01             00 10 20 30 40 50
!>     -- -- -- --                   -- -- -- -- --
!>     03 13 23 33 11             33 11 21 31 41 51
!>     -- -- -- -- --                   -- -- -- --
!>     04 14 24 34 44             43 44 22 32 42 52
!>
