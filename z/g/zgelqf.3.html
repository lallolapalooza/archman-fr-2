subroutine zgelqf (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( * ) work, integer lwork, integer info)

ZGELQF

Objectif:

!>
!> ZGELQF calcule une factorisation LQ d'une matrice complexe M-par-N A:
!>
!>    A = ( L 0 ) *  Q
!>
!> où:
!>
!>    Q est une matrice orthogonale N-par-N;
!>    L est une matrice triangulaire inférieure M-par-M;
!>    0 est une matrice nulle M-par-(N-M), si M < N.
!>
!>

Paramètres

M 
!>          M est un ENTIER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est un ENTIER
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

A

!>          A est un tableau de COMPLEX*16, de dimension (LDA,N)
!>          À l'entrée, la matrice M-par-N A.
!>          À la sortie, les éléments sur et sous la diagonale du tableau
!>          contiennent la matrice trapézoïdale inférieure L de taille m-par-min(m,n) (L est
!>          triangulaire inférieure si m <= n); les éléments au-dessus de la diagonale,
!>          avec le tableau TAU, représentent la matrice unitaire Q comme un
!>          produit de réflecteurs élémentaires (voir Détails supplémentaires).
!>

LDA

!>          LDA est un ENTIER
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!>

TAU

!>          TAU est un tableau de COMPLEX*16, de dimension (min(M,N))
!>          Les facteurs scalaires des réflecteurs élémentaires (voir Détails
!>          supplémentaires).
!>

WORK

!>          WORK est un tableau de COMPLEX*16, de dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) contient la valeur optimale de LWORK.
!>

LWORK

!>          LWORK est un ENTIER
!>          La dimension du tableau WORK.
!>          LWORK >= 1, si MIN(M,N) = 0, et LWORK >= M, sinon.
!>          Pour des performances optimales, LWORK >= M*NB, où NB est
!>          la taille de bloc optimale.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée; la routine
!>          calcule seulement la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message d'erreur
!>          concernant LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est un ENTIER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>

Détails supplémentaires:

!>
!>  La matrice Q est représentée comme un produit de réflecteurs élémentaires
!>
!>     Q = H(k)**H . . . H(2)**H H(1)**H, où k = min(m,n).
!>
!>  Chaque H(i) a la forme
!>
!>     H(i) = I - tau * v * v**H
!>
!>  où tau est un scalaire complexe, et v est un vecteur complexe avec
!>  v(1:i-1) = 0 et v(i) = 1; conjg(v(i+1:n)) est stocké à la sortie dans
!>  A(i,i+1:n), et tau dans TAU(i).
!>
