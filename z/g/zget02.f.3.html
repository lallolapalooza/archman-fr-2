subroutine zget02 (character trans, integer m, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) rwork, double precision resid)

ZGET02

But :

!>
!> ZGET02 calcule le résidu pour une solution d'un système d'équations linéaires
!> op(A)*X = B:
!>    RESID = norm(B - op(A)*X) / ( norm(op(A)) * norm(X) * EPS ),
!> où op(A) = A, A**T, ou A**H, selon TRANS, et EPS est l'epsilon machine.
!>

Paramètres

TRANS 
!>          TRANS est CHARACTER*1
!>          Spécifie la forme du système d'équations :
!>          = 'N' :  A    * X = B  (Pas de transposée)
!>          = 'T' :  A**T * X = B  (Transposée)
!>          = 'C' :  A**H * X = B  (Conjuguée transposée)
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de colonnes de B, la matrice des seconds membres.
!>          NRHS >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>          La matrice originale A de dimensions M x N.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!>

X

!>          X est un tableau COMPLEX*16, dimension (LDX,NRHS)
!>          Les vecteurs solution calculés pour le système d'équations linéaires.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  Si TRANS = 'N',
!>          LDX >= max(1,N) ; si TRANS = 'T' ou 'C', LDX >= max(1,M).
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,NRHS)
!>          En entrée, les vecteurs du second membre pour le système d'équations linéaires.
!>          En sortie, B est écrasé par la différence B - op(A)*X.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  SI TRANS = 'N',
!>          LDB >= max(1,M) ; si TRANS = 'T' ou 'C', LDB >= max(1,N).
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, dimension (M)
!>

RESID

!>          RESID est DOUBLE PRECISION
!>          Le maximum sur le nombre de seconds membres de
!>          norm(B - op(A)*X) / ( norm(op(A)) * norm(X) * EPS ).
!>
