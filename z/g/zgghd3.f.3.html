subroutine zgghd3 (compq, compz, n, ilo, ihi, a, lda, b, ldb, q, ldq, z, ldz, work, lwork, info)

Documentation de la fonction/sous-programme

sous-programme zgghd3 (character compq, character compz, integer n, integer ilo, integer ihi, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, integer lwork, integer info)

ZGGHD3

Objectif:

!>
!> ZGGHD3 réduit une paire de matrices complexes (A,B) à la forme de Hessenberg généralisée
!> généralisée supérieure en utilisant des transformations unitaires, où A est une
!> matrice générale et B est triangulaire supérieure. La forme du problème de valeurs propres généralisé est
!>    A*x = lambda*B*x,
!> et B est typiquement rendue triangulaire supérieure en calculant sa factorisation QR
!> et en déplaçant la matrice unitaire Q du côté gauche
!> de l'équation.
!>
!> Ce sous-programme réduit simultanément A à une matrice de Hessenberg H :
!>    Q**H*A*Z = H
!> et transforme B en une autre matrice triangulaire supérieure T :
!>    Q**H*B*Z = T
!> afin de réduire le problème à sa forme standard
!>    H*y = lambda*T*y
!> où y = Z**H*x.
!>
!> Les matrices unitaires Q et Z sont déterminées comme des produits de rotations de Givens.
!> Elles peuvent soit être formées explicitement, soit être
!> post-multipliées dans les matrices d'entrée Q1 et Z1, de sorte que
!>      Q1 * A * Z1**H = (Q1*Q) * H * (Z1*Z)**H
!>      Q1 * B * Z1**H = (Q1*Q) * T * (Z1*Z)**H
!> Si Q1 est la matrice unitaire de la factorisation QR de B dans le
!> problème original A*x = lambda*B*x, alors ZGGHD3 réduit le problème original
!> à la forme de Hessenberg généralisée.
!>
!> Il s'agit d'une variante bloquée de CGGHRD, utilisant des multiplications matrice-matrice
!> pour certaines parties du calcul afin d'améliorer les performances.
!>

Paramètres

COMPQ 
!>          COMPQ est CHARACTER*1
!>          = 'N': ne pas calculer Q ;
!>          = 'I': Q est initialisée à la matrice unité, et la
!>                 matrice unitaire Q est retournée ;
!>          = 'V': Q doit contenir une matrice unitaire Q1 à l'entrée,
!>                 et le produit Q1*Q est retourné.
!>

COMPZ

!>          COMPZ est CHARACTER*1
!>          = 'N': ne pas calculer Z ;
!>          = 'I': Z est initialisée à la matrice unité, et la
!>                 matrice unitaire Z est retournée ;
!>          = 'V': Z doit contenir une matrice unitaire Z1 à l'entrée,
!>                 et le produit Z1*Z est retourné.
!>

N

!>          N est INTEGER
!>          L'ordre des matrices A et B.  N >= 0.
!>

ILO

!>          ILO est INTEGER
!>

IHI

!>          IHI est INTEGER
!>
!>          ILO et IHI marquent les lignes et les colonnes de A à réduire.
!>          On suppose que A est déjà triangulaire supérieure
!>          dans les lignes et colonnes 1:ILO-1 et IHI+1:N.  ILO et IHI sont
!>          normalement définis par un appel précédent à ZGGBAL ; sinon, ils
!>          devraient être définis à 1 et N respectivement.
!>          1 <= ILO <= IHI <= N, si N > 0 ; ILO=1 et IHI=0, si N=0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA, N)
!>          À l'entrée, la matrice générale N-par-N à réduire.
!>          À la sortie, le triangle supérieur et la première sous-diagonale de A
!>          sont écrasés par la matrice de Hessenberg supérieure H, et le
!>          reste est mis à zéro.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB, N)
!>          À l'entrée, la matrice triangulaire supérieure N-par-N B.
!>          À la sortie, la matrice triangulaire supérieure T = Q**H B Z. Les
!>          éléments sous la diagonale sont mis à zéro.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

Q

!>          Q est un tableau COMPLEX*16, dimension (LDQ, N)
!>          À l'entrée, si COMPQ = 'V', la matrice unitaire Q1, typiquement
!>          de la factorisation QR de B.
!>          À la sortie, si COMPQ='I', la matrice unitaire Q, et si
!>          COMPQ = 'V', le produit Q1*Q.
!>          Non référencé si COMPQ='N'.
!>

LDQ

!>          LDQ est INTEGER
!>          La dimension principale du tableau Q.
!>          LDQ >= N si COMPQ='V' ou 'I' ; LDQ >= 1 sinon.
!>

Z

!>          Z est un tableau COMPLEX*16, dimension (LDZ, N)
!>          À l'entrée, si COMPZ = 'V', la matrice unitaire Z1.
!>          À la sortie, si COMPZ='I', la matrice unitaire Z, et si
!>          COMPZ = 'V', le produit Z1*Z.
!>          Non référencé si COMPZ='N'.
!>

LDZ

!>          LDZ est INTEGER
!>          La dimension principale du tableau Z.
!>          LDZ >= N si COMPZ='V' ou 'I' ; LDZ >= 1 sinon.
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) contient la taille optimale LWORK.
!>

LWORK

!>          LWORK est INTEGER
!>          La longueur du tableau WORK. LWORK >= 1.
!>          Pour des performances optimales, LWORK >= 6*N*NB, où NB est la
!>          taille de bloc optimale.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, retourne
!>          cette valeur comme première entrée du tableau WORK, et aucun message d'erreur
!>          relatif à LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0 :  fin de l'exécution réussie.
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale.
!>
