subroutine zgerfs (character trans, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZGERFS

But:

!>
!> ZGERFS améliore la solution calculée d'un système d'équations linéaires
!> et fournit des bornes d'erreur et des estimations d'erreur arrière pour
!> la solution.
!>

Paramètres

TRANS 
!>          TRANS est CHARACTER*1
!>          Spécifie la forme du système d'équations :
!>          = 'N':  A * X = B     (Pas de transposée)
!>          = 'T':  A**T * X = B  (Transposée)
!>          = 'C':  A**H * X = B  (Conjuguée transposée)
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de côtés droits, c'est-à-dire le nombre de colonnes
!>          des matrices B et X.  NRHS >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>          La matrice originale A de taille N x N.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

AF

!>          AF est un tableau COMPLEX*16, dimension (LDAF,N)
!>          Les facteurs L et U de la factorisation A = P*L*U
!>          tels que calculés par ZGETRF.
!>

LDAF

!>          LDAF est INTEGER
!>          La dimension principale du tableau AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (N)
!>          Les indices de pivot de ZGETRF ; pour 1<=i<=N, la ligne i de la
!>          matrice a été échangée avec la ligne IPIV(i).
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,NRHS)
!>          La matrice du côté droit B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est un tableau COMPLEX*16, dimension (LDX,NRHS)
!>          À l'entrée, la matrice solution X, telle que calculée par ZGETRS.
!>          En sortie, la matrice solution améliorée X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

FERR

!>          FERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur solution
!>          X(j) (la j-ième colonne de la matrice solution X).
!>          Si XTRUE est la vraie solution correspondant à X(j), FERR(j)
!>          est une borne supérieure estimée pour la magnitude du plus grand
!>          élément de (X(j) - XTRUE) divisé par la magnitude du plus grand
!>          élément de X(j). L'estimation est aussi fiable que
!>          l'estimation de RCOND, et est presque toujours une légère
!>          surestimation de l'erreur vraie.
!>

BERR

!>          BERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>          L'erreur arrière relative composante par composante de chaque vecteur solution
!>          X(j) (c'est-à-dire le plus petit changement relatif dans
!>          tout élément de A ou B qui rend X(j) une solution exacte).
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension (2*N)
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>

Paramètres internes:

!>  ITMAX est le nombre maximum d'étapes de raffinement itératif.
!>
