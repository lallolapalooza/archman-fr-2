sous-routine zgrqts (integer m, integer p, integer n, complex*16, dimension( lda, * ) a, complex*16, dimension( lda, * ) af, complex*16, dimension( lda, * ) q, complex*16, dimension( lda, * ) r, integer lda, complex*16, dimension( * ) taua, complex*16, dimension( ldb, * ) b, complex*16, dimension( ldb, * ) bf, complex*16, dimension( ldb, * ) z, complex*16, dimension( ldb, * ) t, complex*16, dimension( ldb, * )bwk, integer ldb, complex*16, dimension( * ) taub, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( 4 ) result)

ZGRQTS

But :

!>
!> ZGRQTS teste ZGGRQF, qui calcule la factorisation GRQ d'une
!> matrice M-par-N A et d'une matrice P-par-N B : A = R*Q et B = Z*T*Q.
!>

Paramètres

M 
!>          M est un entier (INTEGER)
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

P

!>          P est un entier (INTEGER)
!>          Le nombre de lignes de la matrice B.  P >= 0.
!>

N

!>          N est un entier (INTEGER)
!>          Le nombre de colonnes des matrices A et B.  N >= 0.
!>

A

!>          A est un tableau de complexes*16 (COMPLEX*16), dimension (LDA,N)
!>          La matrice M-par-N A.
!>

AF

!>          AF est un tableau de complexes*16 (COMPLEX*16), dimension (LDA,N)
!>          Détails de la factorisation GRQ de A et B, tels que renvoyés
!>          par ZGGRQF, voir CGGRQF pour plus de détails.
!>

Q

!>          Q est un tableau de complexes*16 (COMPLEX*16), dimension (LDA,N)
!>          La matrice unitaire N-par-N Q.
!>

R

!>          R est un tableau de complexes*16 (COMPLEX*16), dimension (LDA,MAX(M,N))
!>

LDA

!>          LDA est un entier (INTEGER)
!>          La dimension principale des tableaux A, AF, R et Q.
!>          LDA >= max(M,N).
!>

TAUA

!>          TAUA est un tableau de complexes*16 (COMPLEX*16), dimension (min(M,N))
!>          Les facteurs scalaires des réflecteurs élémentaires, tels que renvoyés
!>          par DGGQRC.
!>

B

!>          B est un tableau de complexes*16 (COMPLEX*16), dimension (LDB,N)
!>          À l'entrée, la matrice P-par-N A.
!>

BF

!>          BF est un tableau de complexes*16 (COMPLEX*16), dimension (LDB,N)
!>          Détails de la factorisation GQR de A et B, tels que renvoyés
!>          par ZGGRQF, voir CGGRQF pour plus de détails.
!>

Z

!>          Z est un tableau de double précision (DOUBLE PRECISION), dimension (LDB,P)
!>          La matrice unitaire P-par-P Z.
!>

T

!>          T est un tableau de complexes*16 (COMPLEX*16), dimension (LDB,max(P,N))
!>

BWK

!>          BWK est un tableau de complexes*16 (COMPLEX*16), dimension (LDB,N)
!>

LDB

!>          LDB est un entier (INTEGER)
!>          La dimension principale des tableaux B, BF, Z et T.
!>          LDB >= max(P,N).
!>

TAUB

!>          TAUB est un tableau de complexes*16 (COMPLEX*16), dimension (min(P,N))
!>          Les facteurs scalaires des réflecteurs élémentaires, tels que renvoyés
!>          par DGGRQF.
!>

WORK

!>          WORK est un tableau de complexes*16 (COMPLEX*16), dimension (LWORK)
!>

LWORK

!>          LWORK est un entier (INTEGER)
!>          La dimension du tableau WORK, LWORK >= max(M,P,N)**2.
!>

RWORK

!>          RWORK est un tableau de double précision (DOUBLE PRECISION), dimension (M)
!>

RESULT

!>          RESULT est un tableau de double précision (DOUBLE PRECISION), dimension (4)
!>          Les rapports de test :
!>            RESULT(1) = norm( R - A*Q' ) / ( MAX(M,N)*norm(A)*ULP)
!>            RESULT(2) = norm( T*Q - Z'*B ) / (MAX(P,N)*norm(B)*ULP)
!>            RESULT(3) = norm( I - Q'*Q ) / ( N*ULP )
!>            RESULT(4) = norm( I - Z'*Z ) / ( P*ULP )
!>
