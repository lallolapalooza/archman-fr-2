subroutine zgttrf (integer n, complex*16, dimension( * ) dl, complex*16, dimension( * ) d, complex*16, dimension( * ) du, complex*16, dimension( * ) du2, integer, dimension( * ) ipiv, integer info)

ZGTTRF

But :

!>
!> ZGTTRF calcule une factorisation LU d'une matrice tridiagonale complexe A
!> en utilisant l'élimination avec pivot partiel et échanges de lignes.
!>
!> La factorisation est de la forme
!>    A = L * U
!> où L est un produit de matrices de permutation et de matrices bidiagonales
!> unitaires inférieures, et U est triangulaire supérieure avec des éléments
!> non nuls uniquement sur la diagonale principale et les deux superdiagonales
!> supérieures.
!>

Paramètres

N 
!>          N est un entier (INTEGER).
!>          L'ordre de la matrice A.
!>

DL

!>          DL est un tableau de complexes*16 (COMPLEX*16 array), de dimension (N-1).
!>          À l'entrée, DL doit contenir les (n-1) éléments sous-diagonaux de
!>          A.
!>
!>          À la sortie, DL est écrasé par les (n-1) multiplicateurs qui
!>          définissent la matrice L issue de la factorisation LU de A.
!>

D

!>          D est un tableau de complexes*16 (COMPLEX*16 array), de dimension (N).
!>          À l'entrée, D doit contenir les éléments diagonaux de A.
!>
!>          À la sortie, D est écrasé par les n éléments diagonaux de la
!>          matrice triangulaire supérieure U issue de la factorisation LU de A.
!>

DU

!>          DU est un tableau de complexes*16 (COMPLEX*16 array), de dimension (N-1).
!>          À l'entrée, DU doit contenir les (n-1) éléments super-diagonaux
!>          de A.
!>
!>          À la sortie, DU est écrasé par les (n-1) éléments de la première
!>          superdiagonale de U.
!>

DU2

!>          DU2 est un tableau de complexes*16 (COMPLEX*16 array), de dimension (N-2).
!>          À la sortie, DU2 est écrasé par les (n-2) éléments de la
!>          deuxième superdiagonale de U.
!>

IPIV

!>          IPIV est un tableau entier (INTEGER array), de dimension (N).
!>          Les indices de pivot ; pour 1 <= i <= n, la ligne i de la matrice a
!>          été échangée avec la ligne IPIV(i). IPIV(i) sera toujours égal à
!>          i ou i+1 ; IPIV(i) = i indique qu'un échange de ligne n'était pas
!>          nécessaire.
!>

INFO

!>          INFO est un entier (INTEGER).
!>          = 0 : sortie réussie.
!>          < 0 : si INFO = -k, le k-ième argument avait une valeur illégale.
!>          > 0 : si INFO = k, U(k,k) est exactement zéro. La factorisation
!>                a été complétée, mais le facteur U est exactement
!>                singulier, et une division par zéro se produira si elle est utilisée
!>                pour résoudre un système d'équations.
!>
