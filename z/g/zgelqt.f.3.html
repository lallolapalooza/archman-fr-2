subroutine zgelqt (integer m, integer n, integer mb, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( * ) work, integer info)

ZGELQT

Objectif:

!>
!> ZGELQT calcule une factorisation LQ bloquée d'une matrice complexe M par N, A,
!> en utilisant la représentation WY compacte de Q.
!>

Paramètres

M 
!>          M est un entier (INTEGER)
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est un entier (INTEGER)
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

MB

!>          MB est un entier (INTEGER)
!>          La taille de bloc à utiliser dans la factorisation QR bloquée.  MIN(M,N) >= MB >= 1.
!>

A

!>          A est un tableau de type COMPLEX*16, de dimension (LDA,N)
!>          À l'entrée, la matrice A de M par N.
!>          À la sortie, les éléments sur et sous la diagonale du tableau
!>          contiennent la matrice trapézoïdale inférieure L de M par MIN(M,N) (L est
!>          triangulaire inférieure si M <= N) ; les éléments au-dessus de la diagonale
!>          sont les lignes de V.
!>

LDA

!>          LDA est un entier (INTEGER)
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!>

T

!>          T est un tableau de type COMPLEX*16, de dimension (LDT,MIN(M,N))
!>          Les réflecteurs de blocs triangulaires supérieurs stockés sous forme compacte
!>          comme une séquence de blocs triangulaires supérieurs.  Voir ci-dessous
!>          pour plus de détails.
!>

LDT

!>          LDT est un entier (INTEGER)
!>          La dimension principale du tableau T.  LDT >= MB.
!>

WORK

!>          WORK est un tableau de type COMPLEX*16, de dimension (MB*N)
!>

INFO

!>          INFO est un entier (INTEGER)
!>          = 0:  Sortie réussie
!>          < 0:  Si INFO = -i, le i-ème argument avait une valeur illégale
!>

Détails supplémentaires:

!>
!>  La matrice V stocke les réflecteurs élémentaires H(i) dans la i-ème ligne
!>  au-dessus de la diagonale. Par exemple, si M=5 et N=3, la matrice V est
!>
!>               V = (  1  v1 v1 v1 v1 )
!>                   (     1  v2 v2 v2 )
!>                   (         1 v3 v3 )
!>
!>
!>  où les vi représentent les vecteurs qui définissent H(i), qui sont retournés
!>  dans la matrice A.  Les 1 le long de la diagonale de V ne sont pas stockés dans A.
!>  Soit K=MIN(M,N).  Le nombre de blocs est B = ceiling(K/MB), où chaque
!>  bloc est d'ordre MB sauf le dernier bloc, qui est d'ordre
!>  IB = K - (B-1)*MB.  Pour chacun des B blocs, un facteur de réflecteur
!>  de bloc triangulaire supérieur est calculé : T1, T2, ..., TB.  Les T de taille
!>  MBxMB (et IBxIB pour le dernier bloc) sont stockés dans la matrice T de taille
!>  MBxK comme suit :
!>
!>               T = (T1 T2 ... TB).
!>
