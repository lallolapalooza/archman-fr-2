subroutine zgemv (trans, m, n, alpha, a, lda, x, incx, beta, y, incy)
ZGEMV

Documentation de la fonction/sous-routine

sous-routine zgemv (caractère trans, entier m, entier n, complexe*16 alpha, complexe*16, dimension(lda,*) a, entier lda, complexe*16, dimension(*) x, entier incx, complexe*16 beta, complexe*16, dimension(*) y, entier incy)

ZGEMV

Objectif:

!>
!> ZGEMV  effectue l'une des opérations matricielles-vectorielles suivantes :
!>
!>    y := alpha*A*x + beta*y,   ou   y := alpha*A**T*x + beta*y,   ou
!>
!>    y := alpha*A**H*x + beta*y,
!>
!> où alpha et beta sont des scalaires, x et y sont des vecteurs et A est une
!> matrice de taille m par n.
!>

Paramètres

TRANS 
!>          TRANS est CHARACTER*1
!>           À l'entrée, TRANS spécifie l'opération à effectuer comme suit :
!>
!>              TRANS = 'N' ou 'n'   y := alpha*A*x + beta*y.
!>
!>              TRANS = 'T' ou 't'   y := alpha*A**T*x + beta*y.
!>
!>              TRANS = 'C' ou 'c'   y := alpha*A**H*x + beta*y.
!>

M

!>          M est INTEGER
!>           À l'entrée, M spécifie le nombre de lignes de la matrice A.
!>           M doit être supérieur ou égal à zéro.
!>

N

!>          N est INTEGER
!>           À l'entrée, N spécifie le nombre de colonnes de la matrice A.
!>           N doit être supérieur ou égal à zéro.
!>

ALPHA

!>          ALPHA est COMPLEX*16
!>           À l'entrée, ALPHA spécifie le scalaire alpha.
!>

A

!>          A est un tableau de COMPLEX*16, de dimension ( LDA, N )
!>           Avant l'entrée, la partie principale de taille m par n du tableau A doit
!>           contenir la matrice des coefficients.
!>

LDA

!>          LDA est INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle que déclarée
!>           dans le programme appelant (sous-programme). LDA doit être supérieur ou égal à
!>           max( 1, m ).
!>

X

!>          X est un tableau de COMPLEX*16, de dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCX ) ) lorsque TRANS = 'N' ou 'n'
!>           et au moins
!>           ( 1 + ( m - 1 )*abs( INCX ) ) autrement.
!>           Avant l'entrée, le tableau X incrémenté doit contenir le
!>           vecteur x.
!>

INCX

!>          INCX est INTEGER
!>           À l'entrée, INCX spécifie l'incrément pour les éléments de
!>           X. INCX ne doit pas être zéro.
!>

BETA

!>          BETA est COMPLEX*16
!>           À l'entrée, BETA spécifie le scalaire beta. Lorsque BETA est
!>           fourni comme zéro, Y n'a pas besoin d'être défini à l'entrée.
!>

Y

!>          Y est un tableau de COMPLEX*16, de dimension au moins
!>           ( 1 + ( m - 1 )*abs( INCY ) ) lorsque TRANS = 'N' ou 'n'
!>           et au moins
!>           ( 1 + ( n - 1 )*abs( INCY ) ) autrement.
!>           Avant l'entrée, avec BETA non nul, le tableau Y incrémenté
!>           doit contenir le vecteur y. À la sortie, Y est écrasé par le
!>           vecteur y mis à jour.
!>           Si m ou n est zéro, Y n'est pas référencé et la fonction
!>           revient rapidement.
!>

INCY

!>          INCY est INTEGER
!>           À l'entrée, INCY spécifie l'incrément pour les éléments de
!>           Y. INCY ne doit pas être zéro.
!>
