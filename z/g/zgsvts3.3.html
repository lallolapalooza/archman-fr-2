sous-programme zgsvts3 (integer m, integer p, integer n, complex*16, dimension( lda, * ) a, complex*16, dimension( lda, * ) af, integer lda, complex*16, dimension( ldb, * ) b, complex*16, dimension( ldb, * ) bf, integer ldb, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( ldv, * ) v, integer ldv, complex*16, dimension( ldq, * ) q, integer ldq, double precision, dimension( * ) alpha, double precision, dimension( * ) beta, complex*16, dimension( ldr, * ) r, integer ldr, integer, dimension( * ) iwork, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( 6 ) result)

ZGSVTS3

Objectif:

!>
!> ZGSVTS3 teste ZGGSVD3, qui calcule la GSVD d'une matrice M-par-N A
!> et d'une matrice P-par-N B:
!>              U'*A*Q = D1*R et V'*B*Q = D2*R.
!>

Paramètres

M 
!>          M est un ENTIER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

P

!>          P est un ENTIER
!>          Le nombre de lignes de la matrice B.  P >= 0.
!>

N

!>          N est un ENTIER
!>          Le nombre de colonnes des matrices A et B.  N >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,M)
!>          La matrice A de taille M x N.
!>

AF

!>          AF est un tableau COMPLEX*16, dimension (LDA,N)
!>          Détails de la GSVD de A et B, tels que retournés par ZGGSVD3,
!>          voir ZGGSVD3 pour plus de détails.
!>

LDA

!>          LDA est un ENTIER
!>          La dimension principale des tableaux A et AF.
!>          LDA >= max( 1,M ).
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,P)
!>          À l'entrée, la matrice B de taille P x N.
!>

BF

!>          BF est un tableau COMPLEX*16, dimension (LDB,N)
!>          Détails de la GSVD de A et B, tels que retournés par ZGGSVD3,
!>          voir ZGGSVD3 pour plus de détails.
!>

LDB

!>          LDB est un ENTIER
!>          La dimension principale des tableaux B et BF.
!>          LDB >= max(1,P).
!>

U

!>          U est un tableau COMPLEX*16, dimension(LDU,M)
!>          La matrice unitaire U de taille M x M.
!>

LDU

!>          LDU est un ENTIER
!>          La dimension principale du tableau U. LDU >= max(1,M).
!>

V

!>          V est un tableau COMPLEX*16, dimension(LDV,M)
!>          La matrice unitaire V de taille P x P.
!>

LDV

!>          LDV est un ENTIER
!>          La dimension principale du tableau V. LDV >= max(1,P).
!>

Q

!>          Q est un tableau COMPLEX*16, dimension(LDQ,N)
!>          La matrice unitaire Q de taille N x N.
!>

LDQ

!>          LDQ est un ENTIER
!>          La dimension principale du tableau Q. LDQ >= max(1,N).
!>

ALPHA

!>          ALPHA est un tableau de DOUBLE PRECISION, dimension (N)
!>

BETA

!>          BETA est un tableau de DOUBLE PRECISION, dimension (N)
!>
!>          Les paires de valeurs singulières généralisées de A et B, les
!>          matrices "diagonales" D1 et D2 sont construites à partir de
!>          ALPHA et BETA, voir la sous-routine ZGGSVD3 pour les détails.
!>

R

!>          R est un tableau COMPLEX*16, dimension(LDQ,N)
!>          La matrice triangulaire supérieure R.
!>

LDR

!>          LDR est un ENTIER
!>          La dimension principale du tableau R. LDR >= max(1,N).
!>

IWORK

!>          IWORK est un tableau ENTIER, dimension (N)
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension (LWORK)
!>

LWORK

!>          LWORK est un ENTIER
!>          La dimension du tableau WORK,
!>          LWORK >= max(M,P,N)*max(M,P,N).
!>

RWORK

!>          RWORK est un tableau de DOUBLE PRECISION, dimension (max(M,P,N))
!>

RESULT

!>          RESULT est un tableau de DOUBLE PRECISION, dimension (6)
!>          Les rapports de test:
!>          RESULT(1) = norm( U'*A*Q - D1*R ) / ( MAX(M,N)*norm(A)*ULP)
!>          RESULT(2) = norm( V'*B*Q - D2*R ) / ( MAX(P,N)*norm(B)*ULP)
!>          RESULT(3) = norm( I - U'*U ) / ( M*ULP )
!>          RESULT(4) = norm( I - V'*V ) / ( P*ULP )
!>          RESULT(5) = norm( I - Q'*Q ) / ( N*ULP )
!>          RESULT(6) = 0        si ALPHA est en ordre décroissant;
!>                    = ULPINV   sinon.
!>
