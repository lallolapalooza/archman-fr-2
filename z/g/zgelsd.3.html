subroutine zgelsd (integer m, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) s, double precision rcond, integer rank, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer info)

ZGELSD calcule la solution de norme minimale d'un problème de moindres carrés linéaires pour les matrices GE

Objectif :

!>
!> ZGELSD calcule la solution de norme minimale d'un problème linéaire de
!> moindres carrés :
!>     minimiser la norme 2 de | b - A*x |
!> en utilisant la décomposition en valeurs singulières (SVD) de A. A est une
!> matrice M par N qui peut être de rang déficient.
!>
!> Plusieurs vecteurs second membre b et vecteurs solution x peuvent être
!> gérés en un seul appel ; ils sont stockés sous forme de colonnes de la
!> matrice second membre B de taille M x NRHS et de la matrice solution X de
!> taille N x NRHS.
!>
!> Le problème est résolu en trois étapes :
!> (1) Réduire la matrice de coefficients A en forme bidiagonale avec des
!>     transformations de Householder, réduisant le problème original en un
!>     problème BLS (Bidiagonal Least Squares).
!> (2) Résoudre le BLS en utilisant une approche diviser pour régner.
!> (3) Appliquer en retour toutes les transformations de Householder pour
!>     résoudre le problème de moindres carrés original.
!>
!> Le rang effectif de A est déterminé en considérant comme nulles les
!> valeurs singulières inférieures à RCOND fois la plus grande valeur
!> singulière.
!>
!>

Paramètres

M 
!>          M est INTEGER
!>          Le nombre de lignes de la matrice A. M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice A. N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de colonnes
!>          des matrices B et X. NRHS >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>          En entrée, la matrice A de taille M x N.
!>          En sortie, A a été détruit.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,M).
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,NRHS)
!>          En entrée, la matrice second membre B de taille M x NRHS.
!>          En sortie, B est écrasé par la matrice solution X de taille N x NRHS.
!>          Si m >= n et RANK = n, la somme des carrés des résidus
!>          pour la solution dans la i-ème colonne est donnée par la somme
!>          des carrés des modules des éléments n+1:m de cette colonne.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,M,N).
!>

S

!>          S est un tableau DOUBLE PRECISION, dimension (min(M,N))
!>          Les valeurs singulières de A en ordre décroissant.
!>          Le conditionnement de A dans la norme 2 = S(1)/S(min(m,n)).
!>

RCOND

!>          RCOND est DOUBLE PRECISION
!>          RCOND est utilisé pour déterminer le rang effectif de A.
!>          Les valeurs singulières S(i) <= RCOND*S(1) sont traitées comme nulles.
!>          Si RCOND < 0, la précision machine est utilisée à la place.
!>

RANK

!>          RANK est INTEGER
!>          Le rang effectif de A, c'est-à-dire le nombre de valeurs singulières
!>          supérieures à RCOND*S(1).
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension (MAX(1,LWORK))
!>          En sortie, si INFO = 0, WORK(1) contient la valeur optimale de LWORK.
!>

LWORK

!>          LWORK est INTEGER
!>          La dimension du tableau WORK. LWORK doit être au moins 1.
!>          La quantité exacte minimale d'espace de travail nécessaire dépend de M,
!>          N et NRHS. Tant que LWORK est au moins
!>              2*N + N*NRHS
!>          si M est supérieur ou égal à N, ou
!>              2*M + M*NRHS
!>          si M est inférieur à N, le code s'exécutera correctement.
!>          Pour de bonnes performances, LWORK devrait généralement être plus grand.
!>
!>          Si LWORK = -1, une requête d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK et les tailles
!>          minimales des tableaux RWORK et IWORK, et renvoie ces valeurs comme
!>          premières entrées des tableaux WORK, RWORK et IWORK, sans message
!>          d'erreur relatif à LWORK par XERBLA.
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, dimension (MAX(1,LRWORK))
!>          LRWORK >=
!>             10*N + 2*N*SMLSIZ + 8*N*NLVL + 3*SMLSIZ*NRHS +
!>             MAX( (SMLSIZ+1)**2, N*(1+NRHS) + 2*NRHS )
!>          si M est supérieur ou égal à N, ou
!>             10*M + 2*M*SMLSIZ + 8*M*NLVL + 3*SMLSIZ*NRHS +
!>             MAX( (SMLSIZ+1)**2, N*(1+NRHS) + 2*NRHS )
!>          si M est inférieur à N, le code s'exécutera correctement.
!>          SMLSIZ est retourné par ILAENV et correspond à la taille maximale des
!>          sous-problèmes au bas de l'arbre de calcul (généralement environ 25), et
!>             NLVL = MAX( 0, INT( LOG_2( MIN( M,N )/(SMLSIZ+1) ) ) + 1 )
!>          En sortie, si INFO = 0, RWORK(1) contient la valeur minimale de LRWORK.
!>

IWORK

!>          IWORK est un tableau INTEGER, dimension (MAX(1,LIWORK))
!>          LIWORK >= max(1, 3*MINMN*NLVL + 11*MINMN),
!>          où MINMN = MIN( M,N ).
!>          En sortie, si INFO = 0, IWORK(1) contient la valeur minimale de LIWORK.
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale.
!>          > 0 : l'algorithme de calcul de la SVD n'a pas convergé ;
!>                si INFO = i, i éléments hors diagonale d'une forme bidiagonale
!>                intermédiaire n'ont pas convergé vers zéro.
!>
