subroutine zgbbrd (character vect, integer m, integer n, integer ncc, integer kl, integer ku, complex*16, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) d, double precision, dimension( * ) e, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( ldpt, * ) pt, integer ldpt, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZGBBRD

Objectif :

!>
!> ZGBBRD réduit une matrice bande générale complexe m par n, A, en une forme
!> bidiagonale réelle supérieure B par une transformation unitaire : Q**H * A * P = B.
!>
!> La routine calcule B, et éventuellement forme Q ou P**H, ou calcule
!> Q**H*C pour une matrice C donnée.
!>

Paramètres

VECT 
!>          VECT est CHARACTER*1
!>          Spécifie si les matrices Q et P**H doivent être formées ou non.
!>          = 'N' : ne forme ni Q ni P**H ;
!>          = 'Q' : forme Q uniquement ;
!>          = 'P' : forme P**H uniquement ;
!>          = 'B' : forme les deux.
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice A. M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice A. N >= 0.
!>

NCC

!>          NCC est INTEGER
!>          Le nombre de colonnes de la matrice C. NCC >= 0.
!>

KL

!>          KL est INTEGER
!>          Le nombre de sous-diagonales de la matrice A. KL >= 0.
!>

KU

!>          KU est INTEGER
!>          Le nombre de super-diagonales de la matrice A. KU >= 0.
!>

AB

!>          AB est COMPLEX*16 array, dimension (LDAB,N)
!>          En entrée, la matrice bande A de m lignes et n colonnes, stockée
!>          dans les lignes 1 à KL+KU+1. La j-ième colonne de A est stockée
!>          dans la j-ième colonne du tableau AB comme suit :
!>          AB(ku+1+i-j,j) = A(i,j) pour max(1,j-ku) <= i <= min(m,j+kl).
!>          En sortie, A est écrasée par des valeurs générées pendant la
!>          réduction.
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau A. LDAB >= KL+KU+1.
!>

D

!>          D est DOUBLE PRECISION array, dimension (min(M,N))
!>          Les éléments diagonaux de la matrice bidiagonale B.
!>

E

!>          E est DOUBLE PRECISION array, dimension (min(M,N)-1)
!>          Les éléments super-diagonaux de la matrice bidiagonale B.
!>

Q

!>          Q est COMPLEX*16 array, dimension (LDQ,M)
!>          Si VECT = 'Q' ou 'B', la matrice unitaire Q de m x m.
!>          Si VECT = 'N' ou 'P', le tableau Q n'est pas référencé.
!>

LDQ

!>          LDQ est INTEGER
!>          La dimension principale du tableau Q.
!>          LDQ >= max(1,M) si VECT = 'Q' ou 'B' ; LDQ >= 1 sinon.
!>

PT

!>          PT est COMPLEX*16 array, dimension (LDPT,N)
!>          Si VECT = 'P' ou 'B', la matrice unitaire P' (transposée et conjuguée de P).
!>          Si VECT = 'N' ou 'Q', le tableau PT n'est pas référencé.
!>

LDPT

!>          LDPT est INTEGER
!>          La dimension principale du tableau PT.
!>          LDPT >= max(1,N) si VECT = 'P' ou 'B' ; LDPT >= 1 sinon.
!>

C

!>          C est COMPLEX*16 array, dimension (LDC,NCC)
!>          En entrée, une matrice C de m lignes et ncc colonnes.
!>          En sortie, C est écrasée par Q**H*C.
!>          C n'est pas référencée si NCC = 0.
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale du tableau C.
!>          LDC >= max(1,M) si NCC > 0 ; LDC >= 1 si NCC = 0.
!>

WORK

!>          WORK est COMPLEX*16 array, dimension (max(M,N))
!>

RWORK

!>          RWORK est DOUBLE PRECISION array, dimension (max(M,N))
!>

INFO

!>          INFO est INTEGER
!>          = 0 : exécution réussie.
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale.
!>
