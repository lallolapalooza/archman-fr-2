subroutine zgbmv (trans, m, n, kl, ku, alpha, a, lda, x, incx, beta, y, incy)
ZGBMV

Documentation Fonction/Sous-programme

sous-programme zgbmv (caractère trans, entier m, entier n, entier kl, entier ku, complexe*16 alpha, complexe*16, dimension(lda,*) a, entier lda, complexe*16, dimension(*) x, entier incx, complexe*16 beta, complexe*16, dimension(*) y, entier incy)

ZGBMV

Objectif :

!>
!> ZGBMV  effectue l'une des opérations matriciel-vecteur suivantes :
!>
!>    y := alpha*A*x + beta*y,   ou   y := alpha*A**T*x + beta*y,   ou
!>
!>    y := alpha*A**H*x + beta*y,
!>
!> où alpha et beta sont des scalaires, x et y sont des vecteurs et A est une
!> matrice bande de taille m par n, avec kl sous-diagonales et ku super-diagonales.
!>

Paramètres

TRANS 
!>          TRANS est CHARACTER*1
!>           À l'entrée, TRANS spécifie l'opération à effectuer comme suit :
!>
!>              TRANS = 'N' ou 'n'   y := alpha*A*x + beta*y.
!>
!>              TRANS = 'T' ou 't'   y := alpha*A**T*x + beta*y.
!>
!>              TRANS = 'C' ou 'c'   y := alpha*A**H*x + beta*y.
!>

M

!>          M est INTEGER
!>           À l'entrée, M spécifie le nombre de lignes de la matrice A.
!>           M doit être supérieur ou égal à zéro.
!>

N

!>          N est INTEGER
!>           À l'entrée, N spécifie le nombre de colonnes de la matrice A.
!>           N doit être supérieur ou égal à zéro.
!>

KL

!>          KL est INTEGER
!>           À l'entrée, KL spécifie le nombre de sous-diagonales de la
!>           matrice A. KL doit satisfaire  0 .le. KL.
!>

KU

!>          KU est INTEGER
!>           À l'entrée, KU spécifie le nombre de super-diagonales de la
!>           matrice A. KU doit satisfaire  0 .le. KU.
!>

ALPHA

!>          ALPHA est COMPLEX*16
!>           À l'entrée, ALPHA spécifie le scalaire alpha.
!>

A

!>          A est un tableau de COMPLEX*16, dimension ( LDA, N )
!>           Avant l'entrée, la partie principale ( kl + ku + 1 ) par N du
!>           tableau A doit contenir la matrice des coefficients, fournie
!>           colonne par colonne, avec la diagonale principale de la matrice
!>           dans la ligne ( ku + 1 ) du tableau, la première super-diagonale
!>           commençant à la position 2 dans la ligne ku, la première sous-diagonale
!>           commençant à la position 1 dans la ligne ( ku + 2 ), et ainsi de suite.
!>           Les éléments du tableau A qui ne correspondent pas aux éléments
!>           de la matrice bande (comme le triangle supérieur gauche ku par ku)
!>           ne sont pas référencés.
!>           Le segment de programme suivant transférera une matrice bande
!>           depuis un stockage de matrice complète conventionnel vers un stockage bande :
!>
!>                 DO 20, J = 1, N
!>                    K = KU + 1 - J
!>                    DO 10, I = MAX( 1, J - KU ), MIN( M, J + KL )
!>                       A( K + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>

LDA

!>          LDA est INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle que déclarée
!>           dans le programme appelant (sous-programme). LDA doit être au moins
!>           égal à ( kl + ku + 1 ).
!>

X

!>          X est un tableau de COMPLEX*16, dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCX ) ) lorsque TRANS = 'N' ou 'n'
!>           et au moins
!>           ( 1 + ( m - 1 )*abs( INCX ) ) sinon.
!>           Avant l'entrée, le tableau X incrémenté doit contenir le
!>           vecteur x.
!>

INCX

!>          INCX est INTEGER
!>           À l'entrée, INCX spécifie l'incrément pour les éléments de
!>           X. INCX ne doit pas être zéro.
!>

BETA

!>          BETA est COMPLEX*16
!>           À l'entrée, BETA spécifie le scalaire beta. Lorsque BETA est
!>           fourni comme zéro, Y n'a pas besoin d'être défini à l'entrée.
!>

Y

!>          Y est un tableau de COMPLEX*16, dimension au moins
!>           ( 1 + ( m - 1 )*abs( INCY ) ) lorsque TRANS = 'N' ou 'n'
!>           et au moins
!>           ( 1 + ( n - 1 )*abs( INCY ) ) sinon.
!>           Avant l'entrée, le tableau Y incrémenté doit contenir le
!>           vecteur y. À la sortie, Y est écrasé par le vecteur y mis à jour.
!>           Si m ou n est zéro, alors Y n'est pas référencé et la fonction
!>           effectue un retour rapide.
!>

INCY

!>          INCY est INTEGER
!>           À l'entrée, INCY spécifie l'incrément pour les éléments de
!>           Y. INCY ne doit pas être zéro.
!>
