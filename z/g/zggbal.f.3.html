Documentation de la fonction/sous-programme

subroutine zggbal (character job, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, integer ilo, integer ihi, double precision, dimension( * ) lscale, double precision, dimension( * ) rscale, double precision, dimension( * ) work, integer info)

ZGGBAL

But :

!>
!> ZGGBAL équilibre une paire de matrices complexes générales (A,B). Cela
!> implique, d'abord, de permuter A et B par des transformations de similarité pour
!> isoler les valeurs propres dans les éléments de 1 à ILO-1 et de IHI+1 à N
!> sur la diagonale ; et ensuite, d'appliquer une transformation de similarité diagonale
!> aux lignes et colonnes ILO à IHI pour rendre les lignes et colonnes aussi proches que possible en norme. Les deux étapes sont facultatives.
!>
!> L'équilibrage peut réduire la norme 1 des matrices et améliorer la
!> précision des valeurs propres et/ou des vecteurs propres calculés dans le
!> problème de valeurs propres généralisé A*x = lambda*B*x.
!>

Paramètres

JOB 
!>          JOB est CHARACTER*1
!>          Spécifie les opérations à effectuer sur A et B :
!>          = 'N' :  rien :  simplement régler ILO = 1, IHI = N, LSCALE(I) = 1.0
!>                  et RSCALE(I) = 1.0 pour i=1,...,N ;
!>          = 'P' :  seulement permutation ;
!>          = 'S' :  seulement mise à l'échelle ;
!>          = 'B' :  permutation et mise à l'échelle des deux.
!>

N

!>          N est INTEGER
!>          L'ordre des matrices A et B.  N >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>          En entrée, la matrice A d'entrée.
!>          En sortie, A est écrasée par la matrice équilibrée.
!>          Si JOB = 'N', A n'est pas référencé.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,N).
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,N)
!>          En entrée, la matrice B d'entrée.
!>          En sortie, B est écrasée par la matrice équilibrée.
!>          Si JOB = 'N', B n'est pas référencé.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,N).
!>

ILO

!>          ILO est INTEGER
!>

IHI

!>          IHI est INTEGER
!>          ILO et IHI sont définis comme des entiers tels qu'en sortie
!>          A(i,j) = 0 et B(i,j) = 0 si i > j et
!>          j = 1,...,ILO-1 ou i = IHI+1,...,N.
!>          Si JOB = 'N' ou 'S', ILO = 1 et IHI = N.
!>

LSCALE

!>          LSCALE est un tableau DOUBLE PRECISION, dimension (N)
!>          Détails des permutations et des facteurs de mise à l'échelle appliqués
!>          au côté gauche de A et B. Si P(j) est l'indice de la
!>          ligne échangée avec la ligne j, et D(j) est le facteur de mise à l'échelle
!>          appliqué à la ligne j, alors
!>            LSCALE(j) = P(j)    pour J = 1,...,ILO-1
!>                      = D(j)    pour J = ILO,...,IHI
!>                      = P(j)    pour J = IHI+1,...,N.
!>          L'ordre dans lequel les échanges sont effectués est N vers IHI+1,
!>          puis 1 vers ILO-1.
!>

RSCALE

!>          RSCALE est un tableau DOUBLE PRECISION, dimension (N)
!>          Détails des permutations et des facteurs de mise à l'échelle appliqués
!>          au côté droit de A et B. Si P(j) est l'indice de la
!>          colonne échangée avec la colonne j, et D(j) est le facteur de mise à l'échelle
!>          appliqué à la colonne j, alors
!>            RSCALE(j) = P(j)    pour J = 1,...,ILO-1
!>                      = D(j)    pour J = ILO,...,IHI
!>                      = P(j)    pour J = IHI+1,...,N.
!>          L'ordre dans lequel les échanges sont effectués est N vers IHI+1,
!>          puis 1 vers ILO-1.
!>

WORK

!>          WORK est un tableau DOUBLE PRECISION, dimension (lwork)
!>          lwork doit être au moins max(1,6*N) lorsque JOB = 'S' ou 'B', et
!>          au moins 1 lorsque JOB = 'N' ou 'P'.
!>

INFO

!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale.
!>
