sous-routine zgqrts (integer n, integer m, integer p, complex*16, dimension( lda, * ) a, complex*16, dimension( lda, * ) af, complex*16, dimension( lda, * ) q, complex*16, dimension( lda, * ) r, integer lda, complex*16, dimension( * ) taua, complex*16, dimension( ldb, * ) b, complex*16, dimension( ldb, * ) bf, complex*16, dimension( ldb, * ) z, complex*16, dimension( ldb, * ) t, complex*16, dimension( ldb, * )bwk, integer ldb, complex*16, dimension( * ) taub, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( 4 ) result)

ZGQRTS

But:

!>
!> ZGQRTS teste ZGGQRF, qui calcule la factorisation GQR d'une matrice
!> N-par-M A et d'une matrice N-par-P B: A = Q*R et B = Q*T*Z.
!>

Paramètres

N 
!>          N est un entier
!>          Le nombre de lignes des matrices A et B.  N >= 0.
!>

M

!>          M est un entier
!>          Le nombre de colonnes de la matrice A.  M >= 0.
!>

P

!>          P est un entier
!>          Le nombre de colonnes de la matrice B.  P >= 0.
!>

A

!>          A est un tableau de COMPLEX*16, dimension (LDA,M)
!>          La matrice A de N lignes et M colonnes.
!>

AF

!>          AF est un tableau de COMPLEX*16, dimension (LDA,N)
!>          Détails de la factorisation GQR de A et B, tels que retournés
!>          par ZGGQRF, voir CGGQRF pour plus de détails.
!>

Q

!>          Q est un tableau de COMPLEX*16, dimension (LDA,N)
!>          La matrice unitaire Q de M lignes et M colonnes.
!>

R

!>          R est un tableau de COMPLEX*16, dimension (LDA,MAX(M,N))
!>

LDA

!>          LDA est un entier
!>          La dimension principale des tableaux A, AF, R et Q.
!>          LDA >= max(M,N).
!>

TAUA

!>          TAUA est un tableau de COMPLEX*16, dimension (min(M,N))
!>          Les facteurs scalaires des réflecteurs élémentaires, tels que retournés
!>          par ZGGQRF.
!>

B

!>          B est un tableau de COMPLEX*16, dimension (LDB,P)
!>          À l'entrée, la matrice A de N lignes et P colonnes.
!>

BF

!>          BF est un tableau de COMPLEX*16, dimension (LDB,N)
!>          Détails de la factorisation GQR de A et B, tels que retournés
!>          par ZGGQRF, voir CGGQRF pour plus de détails.
!>

Z

!>          Z est un tableau de COMPLEX*16, dimension (LDB,P)
!>          La matrice unitaire Z de P lignes et P colonnes.
!>

T

!>          T est un tableau de COMPLEX*16, dimension (LDB,max(P,N))
!>

BWK

!>          BWK est un tableau de COMPLEX*16, dimension (LDB,N)
!>

LDB

!>          LDB est un entier
!>          La dimension principale des tableaux B, BF, Z et T.
!>          LDB >= max(P,N).
!>

TAUB

!>          TAUB est un tableau de COMPLEX*16, dimension (min(P,N))
!>          Les facteurs scalaires des réflecteurs élémentaires, tels que retournés
!>          par DGGRQF.
!>

WORK

!>          WORK est un tableau de COMPLEX*16, dimension (LWORK)
!>

LWORK

!>          LWORK est un entier
!>          La dimension du tableau WORK, LWORK >= max(N,M,P)**2.
!>

RWORK

!>          RWORK est un tableau de DOUBLE PRECISION, dimension (max(N,M,P))
!>

RESULT

!>          RESULT est un tableau de DOUBLE PRECISION, dimension (4)
!>          Les ratios de test:
!>            RESULT(1) = norm( R - Q'*A ) / ( MAX(M,N)*norm(A)*ULP)
!>            RESULT(2) = norm( T*Z - Q'*B ) / (MAX(P,N)*norm(B)*ULP)
!>            RESULT(3) = norm( I - Q'*Q ) / ( M*ULP )
!>            RESULT(4) = norm( I - Z'*Z ) / ( P*ULP )
!>
