sous-routine zget10 (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, double precision result)

ZGET10

But :

!>
!> ZGET10 compare deux matrices A et B et calcule le rapport
!> RESULT = norm( A - B ) / ( norm(A) * M * EPS )
!>

Paramètres

M 
!>          M est de type INTEGER
!>          Le nombre de lignes des matrices A et B.
!>

N

!>          N est de type INTEGER
!>          Le nombre de colonnes des matrices A et B.
!>

A

!>          A est un tableau de type COMPLEX*16, dimension (LDA,N)
!>          La matrice A de taille m x n.
!>

LDA

!>          LDA est de type INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!>

B

!>          B est un tableau de type COMPLEX*16, dimension (LDB,N)
!>          La matrice B de taille m x n.
!>

LDB

!>          LDB est de type INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,M).
!>

WORK

!>          WORK est un tableau de type COMPLEX*16, dimension (M)
!>

RWORK

!>          RWORK est un tableau de type COMPLEX*16, dimension (M)
!>

RESULT

!>          RESULT est de type DOUBLE PRECISION
!>          RESULT = norm( A - B ) / ( norm(A) * M * EPS )
!>
