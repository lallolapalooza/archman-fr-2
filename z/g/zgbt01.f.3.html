Documentation de la fonction/sous-programme

subroutine zgbt01 (integer m, integer n, integer kl, integer ku, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, complex*16, dimension( * ) work, double precision resid)

ZGBT01

But:

!>
!> ZGBT01 reconstruit une matrice bande A à partir de sa factorisation L*U et
!> calcule le résidu :
!>    norme(L*U - A) / ( N * norme(A) * EPS ),
!> où EPS est le epsilon machine.
!>
!> L'expression L*U - A est calculée colonne par colonne, donc A et
!> AFAC ne sont pas modifiés.
!>

Paramètres

M 
!>          M est un ENTIER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est un ENTIER
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

KL

!>          KL est un ENTIER
!>          Le nombre de sous-diagonales dans la bande de A.  KL >= 0.
!>

KU

!>          KU est un ENTIER
!>          Le nombre de super-diagonales dans la bande de A.  KU >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>          La matrice originale A en stockage bande, stockée dans les lignes 1 à
!>          KL+KU+1.
!>

LDA

!>          LDA est un ENTIER.
!>          La dimension principale du tableau A.  LDA >= max(1,KL+KU+1).
!>

AFAC

!>          AFAC est un tableau COMPLEX*16, dimension (LDAFAC,N)
!>          La forme factorisée de la matrice A.  AFAC contient les facteurs de
!>          bande L et U de la factorisation L*U, comme calculé par
!>          ZGBTRF.  U est stocké comme une matrice triangulaire supérieure bande avec
!>          KL+KU super-diagonales dans les lignes 1 à KL+KU+1, et les
!>          multiplicateurs utilisés pendant la factorisation sont stockés dans les lignes
!>          KL+KU+2 à 2*KL+KU+1.  Voir ZGBTRF pour plus de détails.
!>

LDAFAC

!>          LDAFAC est un ENTIER
!>          La dimension principale du tableau AFAC.
!>          LDAFAC >= max(1,2*KL*KU+1).
!>

IPIV

!>          IPIV est un tableau ENTIER, dimension (min(M,N))
!>          Les indices de pivot de ZGBTRF.
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension (2*KL+KU+1)
!>

RESID

!>          RESID est un DOUBLE PRÉCISION
!>          norme(L*U - A) / ( N * norme(A) * EPS )
!>
