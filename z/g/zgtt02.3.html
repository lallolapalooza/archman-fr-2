subroutine zgtt02 (character trans, integer n, integer nrhs, complex*16, dimension( * ) dl, complex*16, dimension( * ) d, complex*16, dimension( * ) du, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldb, * ) b, integer ldb, double precision resid)

ZGTT02

Objectif :

!>
!> ZGTT02 calcule le résidu de la solution d'un système d'équations tridiagonales :
!>    RESID = norm(B - op(A)*X) / (norm(op(A)) * norm(X) * EPS),
!> où EPS est l'epsilon machine.
!>

Paramètres

TRANS 
!>          TRANS est CHARACTER
!>          Spécifie la forme du résidu.
!>          = 'N':  B - A    * X  (Pas de transposée)
!>          = 'T':  B - A**T * X  (Transposée)
!>          = 'C':  B - A**H * X  (Transposée conjuguée)
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          des matrices B et X.  NRHS >= 0.
!>

DL

!>          DL est un tableau COMPLEX*16, dimension (N-1)
!>          Les (n-1) éléments sous-diagonaux de A.
!>

D

!>          D est un tableau COMPLEX*16, dimension (N)
!>          Les éléments diagonaux de A.
!>

DU

!>          DU est un tableau COMPLEX*16, dimension (N-1)
!>          Les (n-1) éléments super-diagonaux de A.
!>

X

!>          X est un tableau COMPLEX*16, dimension (LDX,NRHS)
!>          La solution calculée, vecteurs X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,NRHS)
!>          À l'entrée, les vecteurs membres de droite pour le système d'équations
!>          linéaires.
!>          À la sortie, B est écrasé par la différence B - op(A)*X.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

RESID

!>          RESID est DOUBLE PRECISION
!>          norm(B - op(A)*X) / (norm(op(A)) * norm(X) * EPS)
!>
