subroutine zgtrfs (character trans, integer n, integer nrhs, complex*16, dimension( * ) dl, complex*16, dimension( * ) d, complex*16, dimension( * ) du, complex*16, dimension( * ) dlf, complex*16, dimension( * ) df, complex*16, dimension( * ) duf, complex*16, dimension( * ) du2, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZGTRFS

Purpose:

!>
!> ZGTRFS améliore la solution calculée d'un système d'équations linéaires
!> lorsque la matrice des coefficients est tridiagonale, et fournit des
!> bornes d'erreur et des estimations d'erreur arrière pour la solution.
!>

Paramètres

TRANS 
!>          TRANS est CHARACTER*1
!>          Spécifie la forme du système d'équations :
!>          = 'N':  A * X = B     (Pas de transposée)
!>          = 'T':  A**T * X = B  (Transposée)
!>          = 'C':  A**H * X = B  (Conjuguée transposée)
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!>

DL

!>          DL est un tableau COMPLEX*16, dimension (N-1)
!>          Les éléments sous-diagonaux (N-1) de A.
!>

D

!>          D est un tableau COMPLEX*16, dimension (N)
!>          Les éléments diagonaux de A.
!>

DU

!>          DU est un tableau COMPLEX*16, dimension (N-1)
!>          Les éléments superdiagonaux (N-1) de A.
!>

DLF

!>          DLF est un tableau COMPLEX*16, dimension (N-1)
!>          Les multiplicateurs (N-1) qui définissent la matrice L de la
!>          factorisation LU de A calculée par ZGTTRF.
!>

DF

!>          DF est un tableau COMPLEX*16, dimension (N)
!>          Les N éléments diagonaux de la matrice triangulaire supérieure U de
!>          la factorisation LU de A.
!>

DUF

!>          DUF est un tableau COMPLEX*16, dimension (N-1)
!>          Les éléments (N-1) de la première superdiagonale de U.
!>

DU2

!>          DU2 est un tableau COMPLEX*16, dimension (N-2)
!>          Les éléments (N-2) de la deuxième superdiagonale de U.
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (N)
!>          Les indices de pivot ; pour 1 <= i <= n, la ligne i de la matrice a
!>          été échangée avec la ligne IPIV(i).  IPIV(i) est toujours égal à
!>          i ou i+1 ; IPIV(i) = i indique qu'un échange de ligne n'a pas
!>          été nécessaire.
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,NRHS)
!>          La matrice membre de droite B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est un tableau COMPLEX*16, dimension (LDX,NRHS)
!>          En entrée, la matrice solution X, telle que calculée par ZGTTRS.
!>          En sortie, la matrice solution améliorée X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

FERR

!>          FERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur solution
!>          X(j) (la j-ème colonne de la matrice solution X).
!>          Si XTRUE est la vraie solution correspondant à X(j), FERR(j)
!>          est une borne supérieure estimée pour la magnitude du plus grand
!>          élément dans (X(j) - XTRUE) divisé par la magnitude du plus grand
!>          élément dans X(j).  L'estimation est aussi fiable que
!>          l'estimation de RCOND, et est presque toujours légèrement
!>          supérieure à l'erreur vraie.
!>

BERR

!>          BERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>          L'erreur arrière relative composante par composante de chaque vecteur solution
!>          X(j) (c'est-à-dire le plus petit changement relatif dans
!>          n'importe quel élément de A ou B qui fait de X(j) une solution exacte).
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension (2*N)
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0:  sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>

Paramètres internes :

!>  ITMAX est le nombre maximum d'étapes de raffinement itératif.
!>
