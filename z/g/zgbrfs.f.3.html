subroutine zgbrfs (character trans, integer n, integer kl, integer ku, integer nrhs, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( ldafb, * ) afb, integer ldafb, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZGBRFS

Purpose:

!>
!> ZGBRFS améliore la solution calculée d'un système d'équations linéaires
!> lorsque la matrice des coefficients est bande, et fournit des bornes d'erreur
!> et des estimations d'erreur arrière pour la solution.
!>

Paramètres

TRANS 
!>          TRANS est CHARACTER*1
!>          Spécifie la forme du système d'équations :
!>          = 'N':  A * X = B     (Pas de transposée)
!>          = 'T':  A**T * X = B  (Transposée)
!>          = 'C':  A**H * X = B  (Conjuguée transposée)
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

KL

!>          KL est INTEGER
!>          Le nombre de sous-diagonales dans la bande de A.  KL >= 0.
!>

KU

!>          KU est INTEGER
!>          Le nombre de super-diagonales dans la bande de A.  KU >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          des matrices B et X.  NRHS >= 0.
!>

AB

!>          AB est un tableau COMPLEX*16, de dimension (LDAB,N)
!>          La matrice bande originale A, stockée dans les lignes 1 à KL+KU+1.
!>          La j-ème colonne de A est stockée dans la j-ème colonne du
!>          tableau AB comme suit :
!>          AB(ku+1+i-j,j) = A(i,j) pour max(1,j-ku) <= i <= min(n,j+kl).
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KL+KU+1.
!>

AFB

!>          AFB est un tableau COMPLEX*16, de dimension (LDAFB,N)
!>          Détails de la factorisation LU de la matrice bande A, comme
!>          calculé par ZGBTRF.  U est stockée comme une matrice bande triangulaire supérieure
!>          avec KL+KU super-diagonales dans les lignes 1 à KL+KU+1, et
!>          les multiplicateurs utilisés lors de la factorisation sont stockés dans
!>          les lignes KL+KU+2 à 2*KL+KU+1.
!>

LDAFB

!>          LDAFB est INTEGER
!>          La dimension principale du tableau AFB.  LDAFB >= 2*KL*KU+1.
!>

IPIV

!>          IPIV est un tableau INTEGER, de dimension (N)
!>          Les indices de pivot de ZGBTRF ; pour 1<=i<=N, la ligne i de la
!>          matrice a été échangée avec la ligne IPIV(i).
!>

B

!>          B est un tableau COMPLEX*16, de dimension (LDB,NRHS)
!>          La matrice des membres de droite B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est un tableau COMPLEX*16, de dimension (LDX,NRHS)
!>          En entrée, la matrice solution X, telle que calculée par ZGBTRS.
!>          En sortie, la matrice solution améliorée X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

FERR

!>          FERR est un tableau DOUBLE PRECISION, de dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur solution
!>          X(j) (la j-ème colonne de la matrice solution X).
!>          Si XTRUE est la vraie solution correspondant à X(j), FERR(j)
!>          est une borne supérieure estimée pour la magnitude du plus grand
!>          élément de (X(j) - XTRUE) divisé par la magnitude du plus grand
!>          élément de X(j). L'estimation est aussi fiable que
!>          l'estimation pour RCOND, et est presque toujours une légère
!>          surostimation de l'erreur réelle.
!>

BERR

!>          BERR est un tableau DOUBLE PRECISION, de dimension (NRHS)
!>          L'erreur arrière relative composante par composante de chaque vecteur solution
!>          X(j) (c'est-à-dire la plus petite modification relative de
!>          n'importe quel élément de A ou B qui fait de X(j) une solution exacte).
!>

WORK

!>          WORK est un tableau COMPLEX*16, de dimension (2*N)
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, de dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0:  sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>

Paramètres internes:

!>  ITMAX est le nombre maximum d'étapes de raffinement itératif.
!>

Gestion des paquets

Arch Linux utilise le gestionnaire de paquets pacman. Vous pouvez l'utiliser pour installer, supprimer et mettre à jour des logiciels.

Commandes courantes :

sudo pacman -Syu    # Synchroniser les bases de données des paquets et mettre à jour le système
sudo pacman -S nom_du_paquet # Installer un paquet
sudo pacman -R nom_du_paquet # Supprimer un paquet
sudo pacman -Ss terme_de_recherche # Rechercher un paquet

Pour en savoir plus, consultez la page de manuel de pacman : man pacman.

Configuration du système
