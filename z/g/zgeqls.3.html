subroutine zgeqls (m, n, nrhs, a, lda, tau, b, ldb, work, lwork, info)
ZGEQLS

Documentation de la fonction/sous-routine

sous-routine zgeqls (integer m, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( lwork ) work, integer lwork, integer info)

ZGEQLS

But :

!>
!> Résoudre le problème des moindres carrés
!>     min || A*X - B ||
!> en utilisant la factorisation QL
!>     A = Q*L
!> calculée par ZGEQLF.
!>

Paramètres

M 
!>          M est un entier
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est un entier
!>          Le nombre de colonnes de la matrice A.  M >= N >= 0.
!>

NRHS

!>          NRHS est un entier
!>          Le nombre de colonnes de B.  NRHS >= 0.
!>

A

!>          A est un tableau COMPLEX*16, de dimension (LDA,N)
!>          Détails de la factorisation QL de la matrice originale A retournée
!>          par ZGEQLF.
!>

LDA

!>          LDA est un entier
!>          La dimension principale du tableau A.  LDA >= M.
!>

TAU

!>          TAU est un tableau COMPLEX*16, de dimension (N)
!>          Détails de la matrice orthogonale Q.
!>

B

!>          B est un tableau COMPLEX*16, de dimension (LDB,NRHS)
!>          En entrée, la matrice second membre B de dimensions m x nrhs.
!>          En sortie, la matrice solution X de dimensions n x nrhs, stockée
!>          dans les lignes m-n+1:m.
!>

LDB

!>          LDB est un entier
!>          La dimension principale du tableau B. LDB >= M.
!>

WORK

!>          WORK est un tableau COMPLEX*16, de dimension (LWORK)
!>

LWORK

!>          LWORK est un entier
!>          La longueur du tableau WORK.  LWORK doit être au moins NRHS,
!>          et devrait être au moins NRHS*NB, où NB est la taille du bloc
!>          pour cet environnement.
!>

INFO

!>          INFO est un entier
!>          = 0 : succès
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>
