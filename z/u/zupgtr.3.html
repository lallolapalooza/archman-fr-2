subroutine zupgtr (character uplo, integer n, complex*16, dimension( * ) ap, complex*16, dimension( * ) tau, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( * ) work, integer info)

ZUPGTR

Objectif :

!>
!> ZUPGTR génère une matrice unitaire complexe Q qui est définie comme le
!> produit de n-1 réflecteurs élémentaires H(i) d'ordre n, tel que retourné par
!> ZHPTRD en utilisant un stockage compact :
!>
!> si UPLO = 'U', Q = H(n-1) . . . H(2) H(1),
!>
!> si UPLO = 'L', Q = H(1) H(2) . . . H(n-1).
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U': Stockage triangulaire supérieur compact utilisé dans l'appel
!>                 précédent à ZHPTRD ;
!>          = 'L': Stockage triangulaire inférieur compact utilisé dans l'appel
!>                 précédent à ZHPTRD.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice Q. N >= 0.
!>

AP

!>          AP est COMPLEX*16 array, dimension (N*(N+1)/2)
!>          Les vecteurs qui définissent les réflecteurs élémentaires, tels que
!>          retournés par ZHPTRD.
!>

TAU

!>          TAU est COMPLEX*16 array, dimension (N-1)
!>          TAU(i) doit contenir le facteur scalaire du réflecteur
!>          élémentaire H(i), tel que retourné par ZHPTRD.
!>

Q

!>          Q est COMPLEX*16 array, dimension (LDQ,N)
!>          La matrice unitaire Q de taille N*N.
!>

LDQ

!>          LDQ est INTEGER
!>          La dimension principale du tableau Q. LDQ >= max(1,N).
!>

WORK

!>          WORK est COMPLEX*16 array, dimension (N-1)
!>

INFO

!>          INFO est INTEGER
!>          = 0:  sortie réussie
!>          < 0:  si INFO = -i, alors le i-ème argument avait une valeur illégale
!>
