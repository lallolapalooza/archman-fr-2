sous-routine zunbdb6 (integer m1, integer m2, integer n, complex*16, dimension(*) x1, integer incx1, complex*16, dimension(*) x2, integer incx2, complex*16, dimension(ldq1,*) q1, integer ldq1, complex*16, dimension(ldq2,*) q2, integer ldq2, complex*16, dimension(*) work, integer lwork, integer info)

ZUNBDB6

Objectif :

!>
!> ZUNBDB6 orthogonalise le vecteur colonne
!>      X = [ X1 ]
!>          [ X2 ]
!> par rapport aux colonnes de
!>      Q = [ Q1 ] .
!>          [ Q2 ]
!> Les colonnes de Q doivent être orthonormées. Le vecteur orthogonalisé sera
!> nul si et seulement s'il se trouve entièrement dans le sous-espace engendré par Q.
!>
!> La projection est calculée avec au plus deux itérations de l'algorithme
!> classique de Gram-Schmidt, voir
!> * L. Giraud, J. Langou, M. Rozložník.
!>   2002. CERFACS Technical Report No. TR/PA/02/33. URL :
!>   https://www.cerfacs.fr/algor/reports/2002/TR_PA_02_33.pdf
!>
!>

Paramètres

M1 
!>          M1 est INTEGER
!>           La dimension de X1 et le nombre de lignes de Q1. 0 <= M1.
!>

M2

!>          M2 est INTEGER
!>           La dimension de X2 et le nombre de lignes de Q2. 0 <= M2.
!>

N

!>          N est INTEGER
!>           Le nombre de colonnes de Q1 et Q2. 0 <= N.
!>

X1

!>          X1 est un tableau de type COMPLEX*16, de dimension (M1)
!>           À l'entrée, la partie supérieure du vecteur à orthogonaliser.
!>           À la sortie, la partie supérieure du vecteur projeté.
!>

INCX1

!>          INCX1 est INTEGER
!>           Incrément pour les entrées de X1.
!>

X2

!>          X2 est un tableau de type COMPLEX*16, de dimension (M2)
!>           À l'entrée, la partie inférieure du vecteur à orthogonaliser. À la
!>           sortie, la partie inférieure du vecteur projeté.
!>

INCX2

!>          INCX2 est INTEGER
!>           Incrément pour les entrées de X2.
!>

Q1

!>          Q1 est un tableau de type COMPLEX*16, de dimension (LDQ1, N)
!>           La partie supérieure de la matrice de base orthonormée.
!>

LDQ1

!>          LDQ1 est INTEGER
!>           La dimension principale de Q1. LDQ1 >= M1.
!>

Q2

!>          Q2 est un tableau de type COMPLEX*16, de dimension (LDQ2, N)
!>           La partie inférieure de la matrice de base orthonormée.
!>

LDQ2

!>          LDQ2 est INTEGER
!>           La dimension principale de Q2. LDQ2 >= M2.
!>

WORK

!>          WORK est un tableau de type COMPLEX*16, de dimension (LWORK)
!>

LWORK

!>          LWORK est INTEGER
!>           La dimension du tableau WORK. LWORK >= N.
!>

INFO

!>          INFO est INTEGER
!>           = 0 : succès.
!>           < 0 : si INFO = -i, alors le i-ème argument avait une valeur illégale.
!>
