subroutine zbdt01 (m, n, kd, a, lda, q, ldq, d, e, pt, ldpt, work, rwork, resid)
ZBDT01

Documentation de la fonction/sous-programme

sous-programme zbdt01 (entier m, entier n, entier kd, complexe*16, dimension( lda, * ) a, entier lda, complexe*16, dimension( ldq, * ) q, entier ldq, double précision, dimension( * ) d, double précision, dimension( * ) e, complexe*16, dimension( ldpt, * ) pt, entier ldpt, complexe*16, dimension( * ) work, double précision, dimension( * ) rwork, double précision resid)

ZBDT01

Objectif:

!>
!> ZBDT01 reconstruit une matrice générale A à partir de sa forme
!> bidiagonale:
!>    A = Q * B * P**H
!> où Q (m par min(m,n)) et P**H (min(m,n) par n) sont des matrices unitaires
!> et B est bidiagonale.
!>
!> Le ratio de test pour vérifier la réduction est
!>    RESID = norm(A - Q * B * P**H) / ( n * norm(A) * EPS )
!> où EPS est la précision machine.
!>

Paramètres

M 
!>          M est ENTIER
!>          Le nombre de lignes des matrices A et Q.
!>

N

!>          N est ENTIER
!>          Le nombre de colonnes des matrices A et P**H.
!>

KD

!>          KD est ENTIER
!>          Si KD = 0, B est diagonale et le tableau E n'est pas référencé.
!>          Si KD = 1, la réduction a été effectuée par xGEBRD; B est
!>          bidiagonale supérieure si M >= N, et bidiagonale inférieure si M < N.
!>          Si KD = -1, la réduction a été effectuée par xGBBRD; B est
!>          toujours bidiagonale supérieure.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>          La matrice A de taille m par n.
!>

LDA

!>          LDA est ENTIER
!>          La dimension principale du tableau A. LDA >= max(1,M).
!>

Q

!>          Q est un tableau COMPLEX*16, dimension (LDQ,N)
!>          La matrice unitaire Q de taille m par min(m,n) dans la réduction
!>          A = Q * B * P**H.
!>

LDQ

!>          LDQ est ENTIER
!>          La dimension principale du tableau Q. LDQ >= max(1,M).
!>

D

!>          D est un tableau DOUBLE PRECISION, dimension (min(M,N))
!>          Les éléments diagonaux de la matrice bidiagonale B.
!>

E

!>          E est un tableau DOUBLE PRECISION, dimension (min(M,N)-1)
!>          Les éléments superdiagonaux de la matrice bidiagonale B si
!>          m >= n, ou les éléments subdiagonaux de B si m < n.
!>

PT

!>          PT est un tableau COMPLEX*16, dimension (LDPT,N)
!>          La matrice unitaire P**H de taille min(m,n) par n dans la réduction
!>          A = Q * B * P**H.
!>

LDPT

!>          LDPT est ENTIER
!>          La dimension principale du tableau PT.
!>          LDPT >= max(1,min(M,N)).
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension (M+N)
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, dimension (M)
!>

RESID

!>          RESID est DOUBLE PRECISION
!>          Le ratio de test:
!>          norm(A - Q * B * P**H) / ( n * norm(A) * EPS )
!>
