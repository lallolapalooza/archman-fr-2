subroutine zstegr (character jobz, character range, integer n, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision vl, double precision vu, integer il, integer iu, double precision abstol, integer m, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, integer, dimension( * ) isuppz, double precision, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

ZSTEGR

But :

!>
!> ZSTEGR calcule les valeurs propres sélectionnées et, éventuellement, les vecteurs propres
!> d'une matrice tridiagonale symétrique réelle T. Toute matrice non réduite de ce type possède
!> un ensemble bien défini de valeurs propres réelles deux à deux distinctes, les vecteurs propres
!> réels correspondants sont deux à deux orthogonaux.
!>
!> Le spectre peut être calculé complètement ou partiellement en spécifiant
!> soit un intervalle (VL,VU] soit une plage d'indices IL:IU pour les valeurs propres souhaitées.
!>
!> ZSTEGR est une routine de compatibilité qui s'appuie sur la routine améliorée ZSTEMR.
!> Voir ZSTEMR pour plus de détails.
!>
!> Un changement important est que le paramètre ABSTOL ne présente plus d'avantage et n'est donc plus utilisé.
!>
!> Note : ZSTEGR et ZSTEMR ne fonctionnent que sur des machines qui respectent
!> la norme IEEE-754 pour le traitement des infinis et des NaNs. L'exécution normale peut créer ces valeurs exceptionnelles et donc
!> s'interrompre en raison d'une exception de virgule flottante dans les environnements qui
!> ne sont pas conformes à la norme IEEE-754.
!>

Paramètres

JOBZ 
!>          JOBZ est CHARACTER*1
!>          = 'N':  Calcul des valeurs propres uniquement ;
!>          = 'V':  Calcul des valeurs propres et des vecteurs propres.
!>

RANGE

!>          RANGE est CHARACTER*1
!>          = 'A': toutes les valeurs propres seront trouvées.
!>          = 'V': toutes les valeurs propres dans l'intervalle semi-ouvert (VL,VU]
!>                 seront trouvées.
!>          = 'I': les valeurs propres de l'indice IL à IU seront trouvées.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice. N >= 0.
!>

D

!>          D est un tableau DOUBLE PRECISION, dimension (N)
!>          En entrée, les N éléments diagonaux de la matrice tridiagonale
!>          T. En sortie, D est écrasé.
!>

E

!>          E est un tableau DOUBLE PRECISION, dimension (N)
!>          En entrée, les N-1 éléments sous-diagonaux de la matrice tridiagonale
!>          T dans les éléments 1 à N-1 de E. E(N) n'a pas besoin d'être défini à l'entrée,
!>          mais est utilisé en interne comme espace de travail.
!>          En sortie, E est écrasé.
!>

VL

!>          VL est DOUBLE PRECISION
!>
!>          Si RANGE='V', la borne inférieure de l'intervalle à
!>          rechercher pour les valeurs propres. VL < VU.
!>          Non référencé si RANGE = 'A' ou 'I'.
!>

VU

!>          VU est DOUBLE PRECISION
!>
!>          Si RANGE='V', la borne supérieure de l'intervalle à
!>          rechercher pour les valeurs propres. VL < VU.
!>          Non référencé si RANGE = 'A' ou 'I'.
!>

IL

!>          IL est INTEGER
!>
!>          Si RANGE='I', l'indice de la
!>          plus petite valeur propre à retourner.
!>          1 <= IL <= IU <= N, si N > 0.
!>          Non référencé si RANGE = 'A' ou 'V'.
!>

IU

!>          IU est INTEGER
!>
!>          Si RANGE='I', l'indice de la
!>          plus grande valeur propre à retourner.
!>          1 <= IL <= IU <= N, si N > 0.
!>          Non référencé si RANGE = 'A' ou 'V'.
!>

ABSTOL

!>          ABSTOL est DOUBLE PRECISION
!>          Non utilisé. Était la tolérance d'erreur absolue pour les
!>          valeurs propres/vecteurs propres dans les versions précédentes.
!>

M

!>          M est INTEGER
!>          Le nombre total de valeurs propres trouvées. 0 <= M <= N.
!>          Si RANGE = 'A', M = N, et si RANGE = 'I', M = IU-IL+1.
!>

W

!>          W est un tableau DOUBLE PRECISION, dimension (N)
!>          Les M premiers éléments contiennent les valeurs propres sélectionnées en
!>          ordre croissant.
!>

Z

!>          Z est un tableau COMPLEX*16, dimension (LDZ, max(1,M) )
!>          Si JOBZ = 'V', et si INFO = 0, alors les M premières colonnes de Z
!>          contiennent les vecteurs propres orthonormés de la matrice T
!>          correspondant aux valeurs propres sélectionnées, avec la i-ème
!>          colonne de Z contenant le vecteur propre associé à W(i).
!>          Si JOBZ = 'N', alors Z n'est pas référencé.
!>          Note : l'utilisateur doit s'assurer qu'au moins max(1,M) colonnes sont
!>          fournies dans le tableau Z ; si RANGE = 'V', la valeur exacte de M
!>          n'est pas connue à l'avance et une limite supérieure doit être utilisée.
!>          Fournir N colonnes est toujours sûr.
!>

LDZ

!>          LDZ est INTEGER
!>          La dimension principale du tableau Z. LDZ >= 1, et si
!>          JOBZ = 'V', alors LDZ >= max(1,N).
!>

ISUPPZ

!>          ISUPPZ est un tableau INTEGER, dimension ( 2*max(1,M) )
!>          Le support des vecteurs propres dans Z, c'est-à-dire les indices
!>          indiquant les éléments non nuls dans Z. Le i-ème vecteur propre calculé
!>          est non nul uniquement dans les éléments ISUPPZ( 2*i-1 ) à
!>          ISUPPZ( 2*i ). Ceci est pertinent dans le cas où la matrice
!>          est divisée. ISUPPZ n'est accédé que lorsque JOBZ est 'V' et N > 0.
!>

WORK

!>          WORK est un tableau DOUBLE PRECISION, dimension (LWORK)
!>          En sortie, si INFO = 0, WORK(1) contient le LWORK optimal
!>          (et minimal).
!>

LWORK

!>          LWORK est INTEGER
!>          La dimension du tableau WORK. LWORK >= max(1,18*N)
!>          si JOBZ = 'V', et LWORK >= max(1,12*N) si JOBZ = 'N'.
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!>

IWORK

!>          IWORK est un tableau INTEGER, dimension (LIWORK)
!>          En sortie, si INFO = 0, IWORK(1) contient le LIWORK optimal.
!>

LIWORK

!>          LIWORK est INTEGER
!>          La dimension du tableau IWORK. LIWORK >= max(1,10*N)
!>          si les vecteurs propres sont désirés, et LIWORK >= max(1,8*N)
!>          si seules les valeurs propres doivent être calculées.
!>          Si LIWORK = -1, alors une requête d'espace de travail est supposée ; la
!>          routine calcule uniquement la taille optimale du tableau IWORK,
!>          renvoie cette valeur comme première entrée du tableau IWORK, et
!>          aucun message d'erreur lié à LIWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          En sortie, INFO
!>          = 0:  fin de l'exécution réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0:  si INFO = 1X, erreur interne dans DLARRE,
!>                si INFO = 2X, erreur interne dans ZLARRV.
!>                Ici, le chiffre X = ABS( IINFO ) < 10, où IINFO est
!>                le code d'erreur non nul renvoyé par DLARRE ou
!>
