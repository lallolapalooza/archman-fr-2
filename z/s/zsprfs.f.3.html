subroutine zsprfs (character uplo, integer n, integer nrhs, complex*16, dimension( * ) ap, complex*16, dimension( * ) afp, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZSPRFS

But:

!>
!> ZSPRFS améliore la solution calculée d'un système d'équations linéaires
!> lorsque la matrice des coefficients est symétrique indéfinie
!> et compactée, et fournit des bornes d'erreur et des estimations d'erreur retour
!> pour la solution.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U':  Le triangle supérieur de A est stocké;
!>          = 'L':  Le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          des matrices B et X.  NRHS >= 0.
!>

AP

!>          AP est tableau COMPLEX*16, dimension (N*(N+1)/2)
!>          Le triangle supérieur ou inférieur de la matrice symétrique A, compacté
!>          par colonne dans un tableau linéaire. La j-ème colonne de A est stockée
!>          dans le tableau AP comme suit:
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) pour 1<=i<=j;
!>          si UPLO = 'L', AP(i + (j-1)*(2*n-j)/2) = A(i,j) pour j<=i<=n.
!>

AFP

!>          AFP est tableau COMPLEX*16, dimension (N*(N+1)/2)
!>          La forme factorisée de la matrice A. AFP contient la matrice
!>          triangulaire par blocs D et les multiplicateurs utilisés pour obtenir
!>          le facteur U ou L à partir de la factorisation A = U*D*U**T ou
!>          A = L*D*L**T comme calculé par ZSPTRF, stocké comme une matrice
!>          triangulaire compactée.
!>

IPIV

!>          IPIV est tableau INTEGER, dimension (N)
!>          Détails des échanges et de la structure par blocs de D
!>          déterminés par ZSPTRF.
!>

B

!>          B est tableau COMPLEX*16, dimension (LDB,NRHS)
!>          La matrice du membre de droite B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est tableau COMPLEX*16, dimension (LDX,NRHS)
!>          À l'entrée, la matrice solution X, comme calculée par ZSPTRS.
!>          À la sortie, la matrice solution améliorée X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

FERR

!>          FERR est tableau DOUBLE PRECISION, dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur solution
!>          X(j) (la j-ème colonne de la matrice solution X).
!>          Si XTRUE est la vraie solution correspondant à X(j), FERR(j)
!>          est une borne supérieure estimée pour la magnitude du plus grand
!>          élément dans (X(j) - XTRUE) divisée par la magnitude du plus grand
!>          élément dans X(j).  L'estimation est aussi fiable que
!>          l'estimation pour RCOND, et est presque toujours une légère
!>          surestimation de l'erreur réelle.
!>

BERR

!>          BERR est tableau DOUBLE PRECISION, dimension (NRHS)
!>          L'erreur retour relative composante par composante de chaque vecteur
!>          solution X(j) (c'est-à-dire le plus petit changement relatif dans
!>          tout élément de A ou B qui fait de X(j) une solution exacte).
!>

WORK

!>          WORK est tableau COMPLEX*16, dimension (2*N)
!>

RWORK

!>          RWORK est tableau DOUBLE PRECISION, dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0:  succès de l'exécution
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>

Paramètres internes:

!>  ITMAX est le nombre maximal d'étapes de raffinement itératif.
!>
