subroutine zsycon_rook (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision anorm, double precision rcond, complex*16, dimension( * ) work, integer info)

ZSYCON_ROOK

But :

!>
!> ZSYCON_ROOK estime la réciproque du nombre de conditionnement (en
!> norme 1) d'une matrice symétrique complexe A, en utilisant la factorisation
!> A = U*D*U**T ou A = L*D*L**T calculée par ZSYTRF_ROOK.
!>
!> Une estimation est obtenue pour norm(inv(A)), et la réciproque du
!> nombre de conditionnement est calculée comme RCOND = 1 / (ANORM * norm(inv(A))).
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si les détails de la factorisation sont stockés
!>          sous forme de matrice triangulaire supérieure ou inférieure.
!>          = 'U':  Triangulaire supérieure, forme est A = U*D*U**T ;
!>          = 'L':  Triangulaire inférieure, forme est A = L*D*L**T.
!>

N

!>          N est INTEGER
!>          Ordre de la matrice A.  N >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>          La matrice diagonale par blocs D et les multiplicateurs utilisés pour
!>          obtenir le facteur U ou L tels que calculés par ZSYTRF_ROOK.
!>

LDA

!>          LDA est INTEGER
!>          Dimension principale du tableau A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (N)
!>          Détails des échanges et structure en blocs de D
!>          tels que déterminés par ZSYTRF_ROOK.
!>

ANORM

!>          ANORM est DOUBLE PRECISION
!>          La norme 1 de la matrice originale A.
!>

RCOND

!>          RCOND est DOUBLE PRECISION
!>          La réciproque du nombre de conditionnement de la matrice A,
!>          calculée comme RCOND = 1/(ANORM * AINVNM), où AINVNM est une
!>          estimation de la norme 1 de inv(A) calculée dans cette routine.
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension (2*N)
!>

INFO

!>          INFO est INTEGER
!>          = 0:  Sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>
