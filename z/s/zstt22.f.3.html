subroutine zstt22 (n, m, kband, ad, ae, sd, se, u, ldu, work, ldwork, rwork, result)
ZSTT22

Documentation de la fonction/sous-routine

sous-routine zstt22 (integer n, integer m, integer kband, double precision, dimension( * ) ad, double precision, dimension( * ) ae, double precision, dimension( * ) sd, double precision, dimension( * ) se, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( ldwork, * ) work, integer ldwork, double precision, dimension( * ) rwork, double precision, dimension( 2 ) result)

ZSTT22

Objectif :

!>
!> ZSTT22  vérifie un ensemble de M valeurs propres et vecteurs propres,
!>
!>     A U = U S
!>
!> où A est hermitienne tridiagonale, les colonnes de U sont unitaires,
!> et S est diagonale (si KBAND=0) ou hermitienne tridiagonale (si KBAND=1).
!> Deux tests sont effectués :
!>
!>    RESULT(1) = | U* A U - S | / ( |A| m ulp )
!>
!>    RESULT(2) = | I - U*U | / ( m ulp )
!>

Paramètres

N 
!>          N est INTEGER
!>          La taille de la matrice. Si elle est nulle, ZSTT22 ne fait rien.
!>          Elle doit être supérieure ou égale à zéro.
!>

M

!>          M est INTEGER
!>          Le nombre de paires propres à vérifier. Si il est nul, ZSTT22
!>          ne fait rien. Il doit être supérieur ou égal à zéro.
!>

KBAND

!>          KBAND est INTEGER
!>          La largeur de bande de la matrice S. Elle ne peut être que zéro ou un.
!>          Si elle est zéro, alors S est diagonale, et SE n'est pas référencé. Si
!>          elle est un, alors S est hermitienne tri-diagonale.
!>

AD

!>          AD est un tableau de type DOUBLE PRECISION, dimension (N)
!>          La diagonale de la matrice A originale (non factorisée). A
!>          est supposée être hermitienne tridiagonale.
!>

AE

!>          AE est un tableau de type DOUBLE PRECISION, dimension (N)
!>          La partie non diagonale de la matrice A originale. A
!>          est supposée être hermitienne tridiagonale. AE(1) est ignoré,
!>          AE(2) est l'élément (1,2) et (2,1), etc.
!>

SD

!>          SD est un tableau de type DOUBLE PRECISION, dimension (N)
!>          La diagonale de la matrice S (tri-)diagonale (hermitienne).
!>

SE

!>          SE est un tableau de type DOUBLE PRECISION, dimension (N)
!>          La partie non diagonale de la matrice S (tri-)diagonale (hermitienne).
!>          Non référencé si KBSND=0. Si KBAND=1, alors AE(1) est
!>          ignoré, SE(2) est l'élément (1,2) et (2,1), etc.
!>

U

!>          U est un tableau de type DOUBLE PRECISION, dimension (LDU, N)
!>          La matrice unitaire dans la décomposition.
!>

LDU

!>          LDU est un INTEGER
!>          La dimension principale de U. LDU doit être au moins N.
!>

WORK

!>          WORK est un tableau de type COMPLEX*16, dimension (LDWORK, M+1)
!>

LDWORK

!>          LDWORK est un INTEGER
!>          La dimension principale de WORK. LDWORK doit être au moins
!>          max(1,M).
!>

RWORK

!>          RWORK est un tableau de type DOUBLE PRECISION, dimension (N)
!>

RESULT

!>          RESULT est un tableau de type DOUBLE PRECISION, dimension (2)
!>          Les valeurs calculées par les deux tests décrits ci-dessus. Les
!>          valeurs sont actuellement limitées à 1/ulp, pour éviter le débordement.
!>
