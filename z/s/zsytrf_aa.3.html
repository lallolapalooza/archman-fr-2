subroutine zsytrf_aa (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( * ) work, integer lwork, integer info)

ZSYTRF_AA

Objectif:

!>
!> ZSYTRF_AA calcule la factorisation d'une matrice complexe symétrique A
!> en utilisant l'algorithme d'Aasen. La forme de la factorisation est
!>
!>    A = U**T*T*U  ou  A = L*T*L**T
!>
!> où U (ou L) est un produit de matrices triangulaires unitaires supérieures (inférieures)
!> et de matrices de permutation, et T est une matrice symétrique tridiagonale complexe.
!>
!> Ceci est la version bloquée de l'algorithme, qui appelle les BLAS de niveau 3.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U':  Le triangle supérieur de A est stocké ;
!>          = 'L':  Le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

A

!>          A est un tableau COMPLEX*16, de dimension (LDA,N)
!>          À l'entrée, la matrice symétrique A. Si UPLO = 'U', la partie
!>          triangulaire supérieure N-par-N de A contient la partie
!>          triangulaire supérieure de la matrice A, et la partie strictement
!>          triangulaire inférieure de A n'est pas référencée. Si UPLO = 'L', la partie
!>          triangulaire inférieure N-par-N de A contient la partie
!>          triangulaire inférieure de la matrice A, et la partie strictement
!>          triangulaire supérieure de A n'est pas référencée.
!>
!>          À la sortie, la matrice tridiagonale est stockée dans les diagonales
!>          et les sous-diagonales de A juste en dessous (ou au-dessus) des diagonales,
!>          et L est stockée en dessous (ou au-dessus) des sous-diagonales, lorsque UPLO
!>          est 'L' (ou 'U').
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV est un tableau INTEGER, de dimension (N)
!>          À la sortie, il contient les détails des échanges, c'est-à-dire
!>          que la ligne et la colonne k de A ont été échangées avec la
!>          ligne et la colonne IPIV(k).
!>

WORK

!>          WORK est un tableau COMPLEX*16, de dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie la valeur optimale de LWORK.
!>

LWORK

!>          LWORK est INTEGER
!>          La longueur de WORK. LWORK >=MAX(1,2*N). Pour des performances optimales
!>          LWORK >= N*(1+NB), où NB est la taille de bloc optimale.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la routine
!>          calcule seulement la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0:  Sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument était invalide.
!>
