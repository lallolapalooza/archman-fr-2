subroutine zsysv_aa_2stage (caractère uplo, entier n, entier nrhs, complexe*16, dimension( lda, * ) a, entier lda, complexe*16, dimension( * ) tb, entier ltb, entier, dimension( * ) ipiv, entier, dimension( * ) ipiv2, complexe*16, dimension( ldb, * ) b, entier ldb, complexe*16, dimension( * ) work, entier lwork, entier info)

ZSYSV_AA_2STAGE calcule la solution du système d'équations linéaires A * X = B pour les matrices SY

Objectif :

!>
!> ZSYSV_AA_2STAGE calcule la solution d'un système complexe d'équations
!> linéaires
!>    A * X = B,
!> où A est une matrice symétrique de taille N x N et X et B sont des
!> matrices N x NRHS.
!>
!> L'algorithme à 2 étapes d'Aasen est utilisé pour factoriser A comme
!>    A = U**T * T * U,  si UPLO = 'U', ou
!>    A = L * T * L**T,  si UPLO = 'L',
!> où U (ou L) est un produit de matrices de permutation et triangulaires
!> unitaires supérieures (inférieures), et T est symétrique et bande.
!> La matrice T est ensuite factorisée LU avec pivotage partiel. La forme
!> factorisée de A est ensuite utilisée pour résoudre le système
!> d'équations A * X = B.
!>
!> C'est la version bloquée de l'algorithme, appelant le BLAS de niveau 3.
!>

Paramètres

UPLO 
!>          UPLO est CARACTERE*1
!>          = 'U':  Le triangle supérieur de A est stocké ;
!>          = 'L':  Le triangle inférieur de A est stocké.
!>

N

!>          N est ENTIER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est ENTIER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>          À l'entrée, la matrice symétrique A.  Si UPLO = 'U', la partie
!>          supérieure triangulaire de taille N x N de A contient la partie
!>          supérieure triangulaire de la matrice A, et la partie strictement
!>          inférieure triangulaire de A n'est pas référencée. Si UPLO = 'L',
!>          la partie inférieure triangulaire de taille N x N de A contient la
!>          partie inférieure triangulaire de la matrice A, et la partie
!>          strictement supérieure triangulaire de A n'est pas référencée.
!>
!>          À la sortie, L est stocké sous (ou au-dessus) des blocs sous-diagonaux,
!>          quand UPLO est 'L' (ou 'U').
!>

LDA

!>          LDA est ENTIER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

TB

!>          TB est un tableau COMPLEX*16, dimension (LTB)
!>          À la sortie, détails de la factorisation LU de la matrice bande.
!>

LTB

!>          LTB est ENTIER
!>          La taille du tableau TB. LTB >= 4*N, utilisé en interne
!>          pour sélectionner NB tel que LTB >= (3*NB+1)*N.
!>
!>          Si LTB = -1, alors une requête d'espace de travail est supposée ; la
!>          routine calcule seulement la taille optimale de LTB,
!>          retourne cette valeur comme première entrée de TB, et
!>          aucun message d'erreur relatif à LTB n'est émis par XERBLA.
!>

IPIV

!>          IPIV est un tableau ENTIER, dimension (N)
!>          À la sortie, il contient les détails des échanges, c'est-à-dire
!>          la ligne et la colonne k de A ont été échangées avec la
!>          ligne et la colonne IPIV(k).
!>

IPIV2

!>          IPIV2 est un tableau ENTIER, dimension (N)
!>          À la sortie, il contient les détails des échanges, c'est-à-dire
!>          la ligne et la colonne k de T ont été échangées avec la
!>          ligne et la colonne IPIV(k).
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,NRHS)
!>          À l'entrée, le membre de droite B.
!>          À la sortie, la matrice solution X.
!>

LDB

!>          LDB est ENTIER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

WORK

!>          WORK est un espace de travail COMPLEX*16 de taille LWORK
!>

LWORK

!>          LWORK est ENTIER
!>          La taille de WORK. LWORK >= N, utilisé en interne pour
!>          sélectionner NB tel que LWORK >= N*NB.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la
!>          routine calcule seulement la taille optimale du tableau WORK,
!>          retourne cette valeur comme première entrée du tableau WORK, et
!>          aucun message d'erreur relatif à LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est ENTIER
!>          = 0:  fin réussie
!>          < 0:  si INFO = -i, le i-ème argument était invalide.
!>          > 0:  si INFO = i, la factorisation LU bande a échoué sur la i-ème colonne
!>
