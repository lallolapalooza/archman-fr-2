subroutine zsytrf_rook (uplo, n, a, lda, ipiv, work, lwork, info)
ZSYTRF_ROOK

Documentation de fonction/sous-routine

sous-routine zsytrf_rook (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( * ) work, integer lwork, integer info)

ZSYTRF_ROOK

But :

!>
!> ZSYTRF_ROOK calcule la factorisation d'une matrice complexe symétrique A
!> en utilisant la méthode de pivotement diagonal bornée de Bunch-Kaufman ().
!> La forme de la factorisation est
!>
!>    A = U*D*U**T  ou  A = L*D*L**T
!>
!> où U (ou L) est un produit de matrices de permutation et de matrices triangulaires
!> supérieures (inférieures) unitaires, et D est symétrique et bloc-diagonale avec
!> des blocs diagonaux de 1x1 et 2x2.
!>
!> C'est la version bloquée de l'algorithme, appelant le BLAS de niveau 3.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U' : Le triangle supérieur de A est stocké ;
!>          = 'L' : Le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A. N >= 0.
!>

A

!>          A est un tableau COMPLEX*16, de dimension (LDA,N)
!>          À l'entrée, la matrice symétrique A. Si UPLO = 'U', la partie
!>          supérieure triangulaire N x N de A contient la partie triangulaire
!>          supérieure de la matrice A, et la partie strictement inférieure
!>          triangulaire de A n'est pas référencée. Si UPLO = 'L', la partie
!>          inférieure triangulaire N x N de A contient la partie triangulaire
!>          inférieure de la matrice A, et la partie strictement supérieure
!>          triangulaire de A n'est pas référencée.
!>
!>          À la sortie, la matrice bloc-diagonale D et les multiplicateurs utilisés
!>          pour obtenir le facteur U ou L (voir ci-dessous pour plus de détails).
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,N).
!>

IPIV

!>          IPIV est un tableau INTEGER, de dimension (N)
!>          Détails des échanges et de la structure de bloc de D.
!>
!>          Si UPLO = 'U' :
!>               Si IPIV(k) > 0, alors les lignes et colonnes k et IPIV(k)
!>               ont été échangées et D(k,k) est un bloc diagonal 1x1.
!>
!>               Si IPIV(k) < 0 et IPIV(k-1) < 0, alors les lignes et
!>               colonnes k et -IPIV(k) ont été échangées et les lignes et
!>               colonnes k-1 et -IPIV(k-1) ont été échangées,
!>               D(k-1:k,k-1:k) est un bloc diagonal 2x2.
!>
!>          Si UPLO = 'L' :
!>               Si IPIV(k) > 0, alors les lignes et colonnes k et IPIV(k)
!>               ont été échangées et D(k,k) est un bloc diagonal 1x1.
!>
!>               Si IPIV(k) < 0 et IPIV(k+1) < 0, alors les lignes et
!>               colonnes k et -IPIV(k) ont été échangées et les lignes et
!>               colonnes k+1 et -IPIV(k+1) ont été échangées,
!>               D(k:k+1,k:k+1) est un bloc diagonal 2x2.
!>

WORK

!>          WORK est un tableau COMPLEX*16, de dimension (MAX(1,LWORK)).
!>          À la sortie, si INFO = 0, WORK(1) renvoie la valeur optimale de LWORK.
!>

LWORK

!>          LWORK est INTEGER
!>          La longueur de WORK. LWORK >=1. Pour de meilleures performances,
!>          LWORK >= N*NB, où NB est la taille de bloc renvoyée par ILAENV.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, renvoie
!>          cette valeur dans la première entrée du tableau WORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 : si INFO = i, D(i,i) est exactement zéro. La factorisation
!>                a été terminée, mais la matrice bloc-diagonale D est
!>                exactement singulière, et une division par zéro se produira si elle
!>                est utilisée pour résoudre un système d'équations.
!>

Détails supplémentaires :

!>
!>  Si UPLO = 'U', alors A = U*D*U**T, où
!>     U = P(n)*U(n)* ... *P(k)U(k)* ...,
!>  c'est-à-dire que U est un produit de termes P(k)*U(k), où k décroît de n à
!>  1 par pas de 1 ou 2, et D est une matrice bloc-diagonale avec des
!>  blocs diagonaux D(k) de 1x1 et 2x2. P(k) est une matrice de permutation telle que
!>  définie par IPIV(k), et U(k) est une matrice triangulaire supérieure unitaire, telle que
!>  si le bloc diagonal D(k) est de taille s (s = 1 ou 2), alors
!>
!>             (   I    v    0   )   k-s
!>     U(k) =  (   0    I    0   )   s
!>             (   0    0    I   )   n-k
!>                k-s   s   n-k
!>
!>  Si s = 1, D(k) écrase A(k,k), et v écrase A(1:k-1,k).
!>  Si s = 2, le triangle supérieur de D(k) écrase A(k-1,k-1), A(k-1,k),
!>  et A(k,k), et v écrase A(1:k-2,k-1:k).
!>
!>  Si UPLO = 'L', alors A = L*D*L**T, où
!>     L = P(1)*L(1)* ... *P(k)*L(k)* ...,
!>  c'est-à-dire que L est un produit de termes P(k)*L(k), où k augmente de 1 à
!>  n par pas de 1 ou 2, et D est une matrice bloc-diagonale avec des
!>  blocs diagonaux D(k) de 1x1 et 2x2. P(k) est une matrice de permutation telle que
!>  définie par IPIV(k), et L(k) est une matrice triangulaire inférieure unitaire, telle que
!>  si le bloc diagonal D(k) est de taille s (s = 1 ou 2), alors
!>
!>             (   I    0     0   )  k-1
!>     L(k) =  (   0    I     0   )  s
!>             (   0    v     I   )  n-k-s+1
!>                k-1   s  n-k-s+1
!>
!>  Si s = 1, D(k) écrase A(k,k), et v écrase A(k+1:n,k).
!>  Si s = 2, le triangle inférieur de D(k) écrase A(k,k), A(k+1,k),
!>  et A(k+1,k+1), et v écrase A(k+2:n,k:k+1).
!>
