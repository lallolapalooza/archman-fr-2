subroutine zsyequb (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) s, double precision scond, double precision amax, complex*16, dimension( * ) work, integer info)

ZSYEQUB

Purpose:

!>
!> ZSYEQUB calcule les mises à l'échelle des lignes et des colonnes destinées à équilibrer
!> une matrice symétrique A (par rapport à la norme euclidienne) et à réduire
!> son nombre de conditionnement. Les facteurs d'échelle S sont calculés par l'algorithme
!> BIN (voir les références) de sorte que la matrice mise à l'échelle B avec les éléments
!> B(i,j) = S(i)*A(i,j)*S(j) ait un nombre de conditionnement dans un facteur N du
!> plus petit nombre de conditionnement possible sur toutes les mises à l'échelle diagonales possibles.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U': le triangle supérieur de A est stocké ;
!>          = 'L': le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A. N >= 0.
!>

A

!>          A est un tableau COMPLEX*16, de dimension (LDA,N)
!>          La matrice symétrique de taille N x N dont les facteurs de mise à l'échelle
!>          doivent être calculés.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,N).
!>

S

!>          S est un tableau DOUBLE PRECISION, de dimension (N)
!>          Si INFO = 0, S contient les facteurs d'échelle pour A.
!>

SCOND

!>          SCOND est DOUBLE PRECISION
!>          Si INFO = 0, S contient le ratio du plus petit S(i) au
!>          plus grand S(i). Si SCOND >= 0.1 et AMAX n'est ni trop
!>          grand ni trop petit, il n'est pas utile de mettre à l'échelle par S.
!>

AMAX

!>          AMAX est DOUBLE PRECISION
!>          La plus grande valeur absolue de tout élément de la matrice. Si AMAX est
!>          très proche du dépassement de capacité ou très proche du sous-dépassement de capacité,
!>          la matrice doit être mise à l'échelle.
!>

WORK

!>          WORK est un tableau COMPLEX*16, de dimension (2*N)
!>

INFO

!>          INFO est INTEGER
!>          = 0: sortie réussie
!>          < 0: si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0: si INFO = i, le i-ème élément diagonal est non positif.
!>
