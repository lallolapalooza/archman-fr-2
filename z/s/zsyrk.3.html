subroutine zsyrk (uplo, trans, n, k, alpha, a, lda, beta, c, ldc)
ZSYRK

Documentation de la fonction/sous-programme

sous-programme zsyrk (caractere uplo, caractere trans, entier n, entier k, complexe*16 alpha, complexe*16, dimension(lda,*) a, entier lda, complexe*16 beta, complexe*16, dimension(ldc,*) c, entier ldc)

ZSYRK

Objectif :

!>
!> ZSYRK  effectue une des opérations symétriques de rang k
!>
!>    C := alpha*A*A**T + beta*C,
!>
!> ou
!>
!>    C := alpha*A**T*A + beta*C,
!>
!> où  alpha et beta  sont des scalaires,  C est une matrice symétrique
!> de taille n par n et  A est une matrice de taille n par k
!> dans le premier cas et une matrice de taille k par n
!> dans le second cas.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>           À l'entrée,  UPLO  spécifie si la partie
!>           triangulaire  supérieure ou inférieure  du tableau
!>           C  doit être référencée comme suit :
!>
!>              UPLO = 'U' ou 'u'   Seule la partie triangulaire supérieure de C
!>                                  doit être référencée.
!>
!>              UPLO = 'L' ou 'l'   Seule la partie triangulaire inférieure de C
!>                                  doit être référencée.
!>

TRANS

!>          TRANS est CHARACTER*1
!>           À l'entrée,  TRANS  spécifie l'opération à effectuer comme suit :
!>
!>              TRANS = 'N' ou 'n'   C := alpha*A*A**T + beta*C.
!>
!>              TRANS = 'T' ou 't'   C := alpha*A**T*A + beta*C.
!>

N

!>          N est INTEGER
!>           À l'entrée,  N spécifie l'ordre de la matrice C.  N doit être
!>           supérieur ou égal à zéro.
!>

K

!>          K est INTEGER
!>           À l'entrée avec  TRANS = 'N' ou 'n',  K  spécifie le nombre
!>           de colonnes de la matrice A, et à l'entrée avec
!>           TRANS = 'T' ou 't',  K  spécifie le nombre de lignes de la
!>           matrice A.  K doit être supérieur ou égal à zéro.
!>

ALPHA

!>          ALPHA est COMPLEX*16
!>           À l'entrée, ALPHA spécifie le scalaire alpha.
!>

A

!>          A est un tableau COMPLEX*16, de dimension ( LDA, ka ), où ka est
!>           k  quand  TRANS = 'N' ou 'n',  et est  n  sinon.
!>           Avant l'entrée avec  TRANS = 'N' ou 'n',  la partie
!>           principale n par k du tableau  A  doit contenir la matrice  A,
!>           sinon la partie principale k par n du tableau  A  doit contenir
!>           la matrice A.
!>

LDA

!>          LDA est INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle que déclarée
!>           dans le (sous-)programme appelant.  Quand TRANS = 'N' ou 'n'
!>           alors  LDA doit être supérieur ou égal à  max( 1, n ), sinon LDA doit
!>           être supérieur ou égal à  max( 1, k ).
!>

BETA

!>          BETA est COMPLEX*16
!>           À l'entrée, BETA spécifie le scalaire beta.
!>

C

!>          C est un tableau COMPLEX*16, de dimension ( LDC, N )
!>           Avant l'entrée avec  UPLO = 'U' ou 'u',  la partie
!>           principale n par n  du tableau C doit contenir la partie
!>           triangulaire supérieure  de la matrice symétrique  et la partie
!>           strictement triangulaire inférieure de C n'est pas référencée.  À la sortie,
!>           la partie triangulaire supérieure du tableau  C est écrasée par la
!>           partie triangulaire supérieure de la matrice mise à jour.
!>           Avant l'entrée avec  UPLO = 'L' ou 'l',  la partie
!>           principale n par n  du tableau C doit contenir la partie
!>           triangulaire inférieure  de la matrice symétrique  et la partie
!>           strictement triangulaire supérieure de C n'est pas référencée.  À la sortie,
!>           la partie triangulaire inférieure du tableau  C est écrasée par la
!>           partie triangulaire inférieure de la matrice mise à jour.
!>

LDC

!>          LDC est INTEGER
!>           À l'entrée, LDC spécifie la première dimension de C telle que déclarée
!>           dans le (sous-)programme appelant.  LDC doit être supérieur ou égal
!>           à  max( 1, n ).
!>
