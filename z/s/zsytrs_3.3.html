subroutine zsytrs_3 (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) e, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZSYTRS_3

Objectif:

!> ZSYTRS_3 résout un système d'équations linéaires A * X = B avec une matrice
!> complexe symétrique A en utilisant la factorisation calculée
!> par ZSYTRF_RK ou ZSYTRF_BK:
!>
!>    A = P*U*D*(U**T)*(P**T) ou A = P*L*D*(L**T)*(P**T),
!>
!> où U (ou L) est une matrice triangulaire unitaire supérieure (ou inférieure),
!> U**T (ou L**T) est la transposée de U (ou L), P est une matrice de permutation,
!> P**T est la transposée de P, et D est symétrique et bloc-diagonale avec des
!> blocs diagonaux de taille 1x1 et 2x2.
!>
!> Cet algorithme utilise le BLAS de niveau 3.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si les détails de la factorisation sont
!>          stockés sous forme de matrice triangulaire supérieure ou inférieure:
!>          = 'U':  Triangulaire supérieure, forme est A = P*U*D*(U**T)*(P**T);
!>          = 'L':  Triangulaire inférieure, forme est A = P*L*D*(L**T)*(P**T).
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>          La diagonale de la matrice bloc-diagonale D et les facteurs U ou L
!>          tels que calculés par ZSYTRF_RK et ZSYTRF_BK:
!>            a) SEULEMENT les éléments diagonaux de la matrice bloc-diagonale
!>               symétrique D sur la diagonale de A, c'est-à-dire D(k,k) = A(k,k);
!>               (les éléments superdiagonaux (ou sous-diagonaux) de D
!>                doivent être fournis en entrée dans le tableau E), et
!>            b) Si UPLO = 'U': le facteur U dans la partie superdiagonale de A.
!>               Si UPLO = 'L': le facteur L dans la partie sous-diagonale de A.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

E

!>          E est un tableau COMPLEX*16, dimension (N)
!>          En entrée, contient les éléments superdiagonaux (ou sous-diagonaux)
!>          de la matrice bloc-diagonale symétrique D
!>          avec des blocs diagonaux de taille 1x1 ou 2x2, où
!>          Si UPLO = 'U': E(i) = D(i-1,i),i=2:N, E(1) non référencé;
!>          Si UPLO = 'L': E(i) = D(i+1,i),i=1:N-1, E(N) non référencé.
!>
!>          NOTE: Pour un bloc diagonal de taille 1x1 D(k), où
!>          1 <= k <= N, l'élément E(k) n'est pas référencé dans les deux
!>          cas UPLO = 'U' ou UPLO = 'L'.
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (N)
!>          Détails des échanges et de la structure de blocs de D
!>          tels que déterminés par ZSYTRF_RK ou ZSYTRF_BK.
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,NRHS)
!>          En entrée, la matrice membre de droite B.
!>          En sortie, la matrice solution X.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

INFO

!>          INFO est INTEGER
!>          = 0:  succès de l'exécution
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>
