subroutine zlavsp (character uplo, character trans, character diag, integer n, integer nrhs, complex*16, dimension( * ) a, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZLAVSP

Purpose:

!>
!>    ZLAVSP  effectue l'une des opérations matricielles-vecteur
!>       x := A*x  ou  x := A^T*x,
!>    où x est un vecteur de N éléments et  A est l'un des facteurs
!>    de la factorisation symétrique calculée par ZSPTRF.
!>    ZSPTRF produit une factorisation de la forme
!>         U * D * U^T     ou     L * D * L^T,
!>    où U (ou L) est un produit de matrices de permutation et triangulaires
!>    unitaires supérieures (inférieures), U^T (ou L^T) est la transposée de
!>    U (ou L), et D est symétrique et bloc diagonale avec des blocs
!>    diagonaux de 1x1 et 2x2. Les multiplicateurs pour les transformations
!>    et les parties triangulaires supérieures ou inférieures des blocs
!>    diagonaux sont stockés colonne par colonne au format compacté
!>    dans le tableau linéaire A.
!>
!>    Si TRANS = 'N' ou 'n', ZLAVSP multiplie soit par U, soit par U * D
!>    (ou L, ou L * D).
!>    Si TRANS = 'C' ou 'c', ZLAVSP multiplie soit par U^T, soit par D * U^T
!>    (ou L^T, soit par D * L^T ).
!> 
!>  UPLO   - CHARACTER*1
!>           En entrée, UPLO spécifie si la matrice triangulaire
!>           stockée dans A est triangulaire supérieure ou inférieure.
!>              UPLO = 'U' ou 'u'   La matrice est triangulaire supérieure.
!>              UPLO = 'L' ou 'l'   La matrice est triangulaire inférieure.
!>           Non modifié en sortie.
!>
!>  TRANS  - CHARACTER*1
!>           En entrée, TRANS spécifie l'opération à effectuer comme
!>           suit :
!>              TRANS = 'N' ou 'n'   x := A*x.
!>              TRANS = 'T' ou 't'   x := A^T*x.
!>           Non modifié en sortie.
!>
!>  DIAG   - CHARACTER*1
!>           En entrée, DIAG spécifie si les blocs diagonaux sont
!>           supposés être des matrices unitaires, comme suit :
!>              DIAG = 'U' ou 'u'   Les blocs diagonaux sont des matrices unitaires.
!>              DIAG = 'N' ou 'n'   Les blocs diagonaux sont non unitaires.
!>           Non modifié en sortie.
!>
!>  N      - INTEGER
!>           En entrée, N spécifie l'ordre de la matrice A.
!>           N doit être supérieur ou égal à zéro.
!>           Non modifié en sortie.
!>
!>  NRHS   - INTEGER
!>           En entrée, NRHS spécifie le nombre de membres de droite,
!>           c'est-à-dire le nombre de vecteurs x à multiplier par A.
!>           NRHS doit être supérieur ou égal à zéro.
!>           Non modifié en sortie.
!>
!>  A      - Tableau COMPLEX*16, dimension( N*(N+1)/2 )
!>           En entrée, A contient une matrice bloc diagonale et les
!>           multiplicateurs des transformations utilisées pour l'obtenir,
!>           stockés comme une matrice triangulaire compactée.
!>           Non modifié en sortie.
!>
!>  IPIV   - Tableau INTEGER, dimension( N )
!>           En entrée, IPIV contient le vecteur des indices de pivot comme
!>           déterminé par ZSPTRF.
!>           Si IPIV( K ) = K, aucun échange n'a été effectué.
!>           Si IPIV( K ) <> K mais IPIV( K ) > 0, alors la ligne K a été inter-
!>           changée avec la ligne IPIV( K ) et un bloc pivot de 1x1 a été utilisé.
!>           Si IPIV( K ) < 0 et UPLO = 'U', alors la ligne K-1 a été échangée
!>           avec la ligne | IPIV( K ) | et un bloc pivot de 2x2 a été utilisé.
!>           Si IPIV( K ) < 0 et UPLO = 'L', alors la ligne K+1 a été échangée
!>           avec la ligne | IPIV( K ) | et un bloc pivot de 2x2 a été utilisé.
!>
!>  B      - Tableau COMPLEX*16, dimension( LDB, NRHS )
!>           En entrée, B contient NRHS vecteurs de longueur N.
!>           En sortie, B est écrasé par le produit A * B.
!>
!>  LDB    - INTEGER
!>           En entrée, LDB contient la dimension principale de B telle que
!>           déclarée dans le programme appelant. LDB doit être supérieur ou
!>           égal à max( 1, N ).
!>           Non modifié en sortie.
!>
!>  INFO   - INTEGER
!>           INFO est le drapeau d'erreur.
!>           En sortie, une valeur de 0 indique une sortie réussie.
!>           Une valeur négative, par exemple -K, indique que le K-ème argument
!>           a une valeur illégale.
!>
