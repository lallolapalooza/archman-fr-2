Documentation de la fonction/sous-routine

subroutine zlqt01 (integer m, integer n, complex*16, dimension( lda, * ) a, complex*16, dimension( lda, * ) af, complex*16, dimension( lda, * ) q, complex*16, dimension( lda, * ) l, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( * ) result)

ZLQT01

But :

!>
!> ZLQT01 teste ZGELQF, qui calcule la factorisation LQ d'une matrice m-par-n A, et teste partiellement ZUNGLQ qui forme la matrice orthogonale n-par-n Q.
!>
!> ZLQT01 compare L avec A*Q', et vérifie que Q est orthogonale.
!>

Paramètres

M 
!>          M est un entier (INTEGER)
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est un entier (INTEGER)
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>          La matrice m-par-n A.
!>

AF

!>          AF est un tableau COMPLEX*16, dimension (LDA,N)
!>          Détails de la factorisation LQ de A, tel que retourné par ZGELQF.
!>          Voir ZGELQF pour plus de détails.
!>

Q

!>          Q est un tableau COMPLEX*16, dimension (LDA,N)
!>          La matrice orthogonale n-par-n Q.
!>

L

!>          L est un tableau COMPLEX*16, dimension (LDA,max(M,N))
!>

LDA

!>          LDA est un entier (INTEGER)
!>          La dimension principale des tableaux A, AF, Q et L.
!>          LDA >= max(M,N).
!>

TAU

!>          TAU est un tableau COMPLEX*16, dimension (min(M,N))
!>          Les facteurs scalaires des réflecteurs élémentaires, tels que retournés
!>          par ZGELQF.
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension (LWORK)
!>

LWORK

!>          LWORK est un entier (INTEGER)
!>          La dimension du tableau WORK.
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, dimension (max(M,N))
!>

RESULT

!>          RESULT est un tableau DOUBLE PRECISION, dimension (2)
!>          Les rapports de test :
!>          RESULT(1) = norm( L - A*Q' ) / ( N * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q*Q' ) / ( N * EPS )
!>
