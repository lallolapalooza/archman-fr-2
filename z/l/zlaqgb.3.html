subroutine zlaqgb (integer m, integer n, integer kl, integer ku, complex*16, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) r, double precision, dimension( * ) c, double precision rowcnd, double precision colcnd, double precision amax, character equed)

ZLAQGB met à l'échelle une matrice bande générale, en utilisant les facteurs de mise à l'échelle des lignes et des colonnes calculés par sgbequ.

Objectif :

!>
!> ZLAQGB équilibre une matrice bande générale M par N de A avec KL
!> sous-diagonales et KU super-diagonales en utilisant les facteurs de mise à l'échelle
!> des lignes et des colonnes dans les vecteurs R et C.
!>

Paramètres

M 
!>          M est un entier (INTEGER)
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est un entier (INTEGER)
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

KL

!>          KL est un entier (INTEGER)
!>          Le nombre de sous-diagonales dans la bande de A.  KL >= 0.
!>

KU

!>          KU est un entier (INTEGER)
!>          Le nombre de super-diagonales dans la bande de A.  KU >= 0.
!>

AB

!>          AB est un tableau COMPLEX*16, de dimension (LDAB,N)
!>          En entrée, la matrice A en stockage bande, dans les lignes 1 à KL+KU+1.
!>          La j-ième colonne de A est stockée dans la j-ième colonne du
!>          tableau AB comme suit :
!>          AB(ku+1+i-j,j) = A(i,j) pour max(1,j-ku)<=i<=min(m,j+kl)
!>
!>          En sortie, la matrice équilibrée, dans le même format de stockage
!>          que A.  Voir EQUED pour la forme de la matrice équilibrée.
!>

LDAB

!>          LDAB est un entier (INTEGER)
!>          La dimension principale du tableau AB.  LDA >= KL+KU+1.
!>

R

!>          R est un tableau DOUBLE PRECISION, de dimension (M)
!>          Les facteurs d'échelle des lignes de A.
!>

C

!>          C est un tableau DOUBLE PRECISION, de dimension (N)
!>          Les facteurs d'échelle des colonnes de A.
!>

ROWCND

!>          ROWCND est DOUBLE PRECISION
!>          Rapport du plus petit R(i) au plus grand R(i).
!>

COLCND

!>          COLCND est DOUBLE PRECISION
!>          Rapport du plus petit C(i) au plus grand C(i).
!>

AMAX

!>          AMAX est DOUBLE PRECISION
!>          Valeur absolue de la plus grande entrée de la matrice.
!>

EQUED

!>          EQUED est CHARACTER*1
!>          Spécifie la forme d'équilibrage qui a été effectuée.
!>          = 'N' : Pas d'équilibrage
!>          = 'R' : Équilibrage des lignes, c'est-à-dire que A a été pré-multiplié par
!>                  diag(R).
!>          = 'C' : Équilibrage des colonnes, c'est-à-dire que A a été post-multiplié
!>                  par diag(C).
!>          = 'B' : Équilibrage des lignes et des colonnes, c'est-à-dire que A a été
!>                  remplacé par diag(R) * A * diag(C).
!>

Paramètres internes :

!>  THRESH est une valeur seuil utilisée pour décider si la mise à l'échelle
!>  des lignes ou des colonnes doit être effectuée en fonction du rapport des
!>  facteurs d'échelle des lignes ou des colonnes. Si ROWCND < THRESH, la mise à l'échelle
!>  des lignes est effectuée, et si COLCND < THRESH, la mise à l'échelle
!>  des colonnes est effectuée.
!>
!>  LARGE et SMALL sont des valeurs seuils utilisées pour décider si la mise à l'échelle
!>  des lignes doit être effectuée en fonction de la taille absolue du plus grand élément
!>  de la matrice. Si AMAX > LARGE ou AMAX < SMALL, la mise à l'échelle
!>  des lignes est effectuée.
!>
