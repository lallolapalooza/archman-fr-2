subroutine zlacp2 (character uplo, integer m, integer n, double precision, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb)

ZLACP2 copie tout ou partie d'une matrice bidimensionnelle réelle A dans une matrice complexe B.

Objectif :

!>
!> ZLACP2 copie tout ou partie d'une matrice bidimensionnelle réelle A dans une
!> matrice complexe B.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie la partie de la matrice A à copier dans B.
!>          = 'U' : partie triangulaire supérieure
!>          = 'L' : partie triangulaire inférieure
!>          Sinon : toute la matrice A
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

A

!>          A est un tableau de type DOUBLE PRECISION, de dimension (LDA,N)
!>          La matrice A de taille m x n. Si UPLO = 'U', seul le trapèze supérieur
!>          est accédé ; si UPLO = 'L', seul le trapèze inférieur est
!>          accédé.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!>

B

!>          B est un tableau de type COMPLEX*16, de dimension (LDB,N)
!>          En sortie, B = A dans les emplacements spécifiés par UPLO.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,M).
!>
