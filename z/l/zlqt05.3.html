subroutine zlqt05 (integer m, integer n, integer l, integer nb, double precision, dimension(6) result)

ZLQT05

But:

!>
!> ZQRT05 teste ZTPLQT et ZTPMLQT.
!>

Paramètres

M 
!>          M est un ENTIER
!>          Nombre de lignes dans la partie inférieure de la matrice de test.
!>

N

!>          N est un ENTIER
!>          Nombre de colonnes dans la matrice de test.
!>

L

!>          L est un ENTIER
!>          Le nombre de lignes de la partie trapézoïdale supérieure de la
!>          matrice de test inférieure.  0 <= L <= M.
!>

NB

!>          NB est un ENTIER
!>          Taille de bloc de la matrice de test.  NB <= N.
!>

RESULT

!>          RESULT est un tableau DOUBLE PRECISION, de dimension (6)
!>          Résultats de chacun des six tests ci-dessous.
!>
!>          RESULT(1) = | A - Q R |
!>          RESULT(2) = | I - Q^H Q |
!>          RESULT(3) = | Q C - Q C |
!>          RESULT(4) = | Q^H C - Q^H C |
!>          RESULT(5) = | C Q - C Q |
!>          RESULT(6) = | C Q^H - C Q^H |
!>
