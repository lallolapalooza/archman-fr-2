Documentation de fonction/sous-routine

subroutine zlatzm (character side, integer m, integer n, complex*16, dimension( * ) v, integer incv, complex*16 tau, complex*16, dimension( ldc, * ) c1, complex*16, dimension( ldc, * ) c2, integer ldc, complex*16, dimension( * ) work)

ZLATZM

Objectif:

!>
!> Cette sous-routine est obsolète et a été remplacée par la sous-routine ZUNMRZ.
!>
!> ZLATZM applique une matrice de Householder générée par ZTZRQF à une matrice.
!>
!> Soit P = I - tau*u*u**H,   u = ( 1 ),
!>                               ( v )
!> où v est un vecteur de taille (m-1) si SIDE = 'L', ou un vecteur de taille (n-1) si
!> SIDE = 'R'.
!>
!> Si SIDE est égal à 'L', soit
!>        C = [ C1 ] 1
!>            [ C2 ] m-1
!>              n
!> Alors C est écrasé par P*C.
!>
!> Si SIDE est égal à 'R', soit
!>        C = [ C1, C2 ] m
!>               1  n-1
!> Alors C est écrasé par C*P.
!>

Paramètres

SIDE 
!>          SIDE est CHARACTER*1
!>          = 'L': forme P * C
!>          = 'R': forme C * P
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice C.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice C.
!>

V

!>          V est un tableau COMPLEX*16, dimension
!>                  (1 + (M-1)*abs(INCV)) si SIDE = 'L'
!>                  (1 + (N-1)*abs(INCV)) si SIDE = 'R'
!>          Le vecteur v dans la représentation de P. V n'est pas utilisé
!>          si TAU = 0.
!>

INCV

!>          INCV est INTEGER
!>          L'incrément entre les éléments de v. INCV <> 0
!>

TAU

!>          TAU est COMPLEX*16
!>          La valeur tau dans la représentation de P.
!>

C1

!>          C1 est un tableau COMPLEX*16, dimension
!>                         (LDC,N) si SIDE = 'L'
!>                         (M,1)   si SIDE = 'R'
!>          À l'entrée, le vecteur colonne C1 de taille n si SIDE = 'L', ou le
!>          vecteur colonne C1 de taille m si SIDE = 'R'.
!>
!>          À la sortie, la première ligne de P*C si SIDE = 'L', ou la première
!>          colonne de C*P si SIDE = 'R'.
!>

C2

!>          C2 est un tableau COMPLEX*16, dimension
!>                         (LDC, N)   si SIDE = 'L'
!>                         (LDC, N-1) if SIDE = 'R'
!>          À l'entrée, la matrice C2 de taille (m - 1) x n si SIDE = 'L', ou la
!>          matrice C2 de taille m x (n - 1) si SIDE = 'R'.
!>
!>          À la sortie, les lignes 2:m de P*C si SIDE = 'L', ou les colonnes 2:m de C*P
!>          si SIDE = 'R'.
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale des tableaux C1 et C2.
!>          LDC >= max(1,M).
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension
!>                      (N) si SIDE = 'L'
!>                      (M) si SIDE = 'R'
!>
