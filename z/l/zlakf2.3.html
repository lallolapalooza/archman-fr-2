subroutine zlakf2 (m, n, a, lda, b, d, e, z, ldz)
ZLAKF2

Documentation de la fonction/sous-programme

sous-programme zlakf2 (entier m, entier n, complexe*16, dimension( lda, * ) a, entier lda, complexe*16, dimension( lda, * ) b, complexe*16, dimension( lda, * ) d, complexe*16, dimension( lda, * ) e, complexe*16, dimension( ldz, * ) z, entier ldz)

ZLAKF2

Objectif:

!>
!> Forme la matrice 2*M*N par 2*M*N
!>
!>        Z = [ kron(In, A)  -kron(B', Im) ]
!>            [ kron(In, D)  -kron(E', Im) ],
!>
!> où In est la matrice identité de taille n et X' est la transposée
!> de X. kron(X, Y) est le produit de Kronecker entre les matrices X
!> et Y.
!>

Paramètres

M 
!>          M est ENTIER
!>          Taille de la matrice, doit être >= 1.
!>

N

!>          N est ENTIER
!>          Taille de la matrice, doit être >= 1.
!>

A

!>          A est COMPLEX*16, dimension ( LDA, M )
!>          La matrice A dans la matrice de sortie Z.
!>

LDA

!>          LDA est ENTIER
!>          La dimension principale de A, B, D, et E. ( LDA >= M+N )
!>

B

!>          B est COMPLEX*16, dimension ( LDA, N )
!>

D

!>          D est COMPLEX*16, dimension ( LDA, M )
!>

E

!>          E est COMPLEX*16, dimension ( LDA, N )
!>
!>          Les matrices utilisées pour former la matrice de sortie Z.
!>

Z

!>          Z est COMPLEX*16, dimension ( LDZ, 2*M*N )
!>          La matrice résultante de Kronecker M*N*2 par M*N*2 (voir ci-dessus.)
!>

LDZ

!>          LDZ est ENTIER
!>          La dimension principale de Z. ( LDZ >= 2*M*N )
!>
