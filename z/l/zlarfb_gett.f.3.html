subroutine zlarfb_gett (character ident, integer m, integer n, integer k, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldwork, * ) work, integer ldwork)

ZLARFB_GETT

Purpose:

!>
!> ZLARFB_GETT applique un bloc reflector Householder complexe H de la
!> gauche a une matrice K+M par N complexe composee de deux matrices blocs:
!> une matrice K par N a trapeze superieure A stockee dans le tableau A, et
!> une matrice rectangulaire M par (N-K) B, stockee dans le tableau B. Le
!> bloc reflector H est stocke dans une representation WY compacte, ou les
!> reflectors elementaires sont dans les tableaux A, B et T. Voir la section
!> Details supplementaires.
!>

Parameters

IDENT 
!>          IDENT est CHARACTER*1
!>          Si IDENT = pas 'I', ou pas 'i', alors V1 est une matrice unite
!>             triangulaire inferieure et stockee dans le bloc gauche K par K
!>             de la matrice d'entree A,
!>          Si IDENT = 'I' ou 'i', alors V1 est une matrice identite et
!>             n'est pas stockee.
!>          Voir la section Details supplementaires.
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice B.
!>          M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes des matrices A et B.
!>          N >= 0.
!>

K

!>          K est INTEGER
!>          Le nombre de lignes de la matrice A.
!>          K est aussi l'ordre de la matrice T, c'est-a-dire le nombre
!>          de reflectors elementaires dont le produit definit le bloc
!>          reflector. 0 <= K <= N.
!>

T

!>          T est un tableau COMPLEX*16, dimension (LDT,K)
!>          La matrice triangulaire superieure T de taille K par K dans la
!>          representation du bloc reflector.
!>

LDT

!>          LDT est INTEGER
!>          La dimension principale du tableau T. LDT >= K.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>
!>          A l'entree :
!>           a) Dans la partie trapezoide superieure K par N de A : matrice d'entree A.
!>           b) Dans les colonnes sous la diagonale : colonnes de V1
!>              (les uns ne sont pas stockes sur la diagonale).
!>
!>          A la sortie :
!>            A est ecrasé par le produit rectangulaire K par N H*A.
!>
!>          Voir la section Details supplementaires.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,K).
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,N)
!>
!>          A l'entree :
!>            a) Dans le bloc droit M par (N-K) : matrice d'entree B.
!>            b) Dans le bloc gauche M par N : colonnes de V2.
!>
!>          A la sortie :
!>            B est ecrasé par le produit rectangulaire M par N H*B.
!>
!>          Voir la section Details supplementaires.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,M).
!>

WORK

!>          WORK est un tableau COMPLEX*16,
!>          dimension (LDWORK,max(K,N-K))
!>

LDWORK

!>          LDWORK est INTEGER
!>          La dimension principale du tableau WORK. LDWORK>=max(1,K).
!>

Further Details:

!>
!>    (1) Description de l'Operation Algebrique.
!>
!>    La matrice A est une matrice K par N composee de deux matrices colonnes
!>    en blocs, A1, qui est K par K, et A2, qui est K par (N-K):
!>    A = ( A1, A2 ).
!>    La matrice B est une matrice M par N composee de deux matrices colonnes
!>    en blocs, B1, qui est M par K, et B2, qui est M par (N-K):
!>    B = ( B1, B2 ).
!>
!>    Effectuer l'operation:
!>
!>       ( A_out ) := H * ( A_in ) = ( I - V * T * V**H ) * ( A_in ) =
!>       ( B_out )        ( B_in )                          ( B_in )
!>                  = ( I - ( V1 ) * T * ( V1**H, V2**H ) ) * ( A_in )
!>                          ( V2 )                            ( B_in )
!>     A l'entree :
!>
!>    a) ( A_in )  est compose de deux blocs de colonnes:
!>       ( B_in )
!>
!>       ( A_in ) = (( A1_in ) ( A2_in )) = (( A1_in ) ( A2_in ))
!>       ( B_in )   (( B1_in ) ( B2_in ))   ((     0 ) ( B2_in )),
!>
!>       ou les blocs de colonnes sont :
!>
!>       (  A1_in )  est une matrice triangulaire superieure K par K stockee dans
!>                   la partie triangulaire superieure du tableau A(1:K,1:K).
!>       (  B1_in )  est une matrice ZERO rectangulaire M par K et n'est pas stockee.
!>
!>       ( A2_in )  est une matrice rectangulaire K par (N-K) stockee
!>                  dans le tableau A(1:K,K+1:N).
!>       ( B2_in )  est une matrice rectangulaire M par (N-K) stockee
!>                  dans le tableau B(1:M,K+1:N).
!>
!>    b) V = ( V1 )
!>           ( V2 )
!>
!>       ou :
!>       1) si IDENT == 'I', V1 est une matrice identite K par K, non stockee;
!>       2) si IDENT != 'I', V1 est une matrice triangulaire inferieure unitaire K par K,
!>          stockee dans la partie triangulaire inferieure du tableau
!>          A(1:K,1:K) (les uns ne sont pas stockes),
!>       et V2 est une matrice rectangulaire M par K stockee dans le tableau
!>                 B(1:M,1:K),
!>                 (car a l'entree B1_in est une matrice zero rectangulaire qui n'est pas
!>                  stockee et l'espace est utilise pour stocker V2).
!>
!>    c) T est une matrice triangulaire superieure K par K stockee
!>       dans le tableau T(1:K,1:K).
!>
!>    A la sortie :
!>
!>    a) ( A_out ) est compose de deux blocs de colonnes:
!>       ( B_out )
!>
!>       ( A_out ) = (( A1_out ) ( A2_out ))
!>       ( B_out )   (( B1_out ) ( B2_out )),
!>
!>       ou les blocs de colonnes sont :
!>
!>       ( A1_out )  est une matrice carree K par K, ou une matrice
!>                   triangulaire superieure K par K, si V1 est une
!>                   matrice identite. AiOut est stockee dans le tableau
!>                   A(1:K,1:K).
!>       ( B1_out )  est une matrice rectangulaire M par K stockee
!>                   dans le tableau B(1:M,K:N).
!>
!>       ( A2_out )  est une matrice rectangulaire K par (N-K) stockee
!>                   dans le tableau A(1:K,K+1:N).
!>       ( B2_out )  est une matrice rectangulaire M par (N-K) stockee
!>                   dans le tableau B(1:M,K+1:N).
!>
!>
!>    L'operation ci-dessus peut etre representee comme la meme operation
!>    sur chaque bloc de colonnes:
!>
!>       ( A1_out ) := H * ( A1_in ) = ( I - V * T * V**H ) * ( A1_in )
!>       ( B1_out )        (     0 )                          (     0 )
!>
!>       ( A2_out ) := H * ( A2_in ) = ( I - V * T * V**H ) * ( A2_in )
!>       ( B2_out )        ( B2_in )                          ( B2_in )
!>
!>    Si IDENT != 'I':
!>
!>       Le calcul pour le bloc de colonne 1:
!>
!>       A1_out: = A1_in - V1*T*(V1**H)*A1_in
!>
!>       B1_out: = - V2*T*(V1**H)*A1_in
!>
!>       Le calcul pour le bloc de colonne 2, qui existe si N > K:
!>
!>       A2_out: = A2_in - V1*T*( (V1**H)*A2_in + (V2**H)*B2_in )
!>
!>       B2_out: = B2_in - V2*T*( (V1**H)*A2_in + (V2**H)*B2_in )
!>
!>    Si IDENT == 'I':
!>
!>       L'operation pour le bloc de colonne 1:
!>
!>       A1_out: = A1_in - V1*T*A1_in
!>
!>       B1_out: = - V2*T*A1_in
!>
!>       Le calcul pour le bloc de colonne 2, qui existe si N > K:
!>
!>       A2_out: = A2_in - V1*T*( (V1**H)*A2_in + (V2**H)*B2_in )
!>
!>       B2_out: = B2_in - V2*T*( (V1**H)*A2_in + (V2**H)*B2_in )
!>
!>
!> (2) Representation WY de la Multiplication de Matrice.
!>
!>    Pour une matrice A de taille M x N, on souhaite calculer
!>    A*H, ou H est un bloc reflector Householder.
!>
!>    H = I - V * T * V**H
!>
!>    ou V = ( V1 ) est une matrice K x K (ou K x (K+1) si on compte
!>           ( V2 )   les uns implicites sur la diagonale de V1)
!>    et T est une matrice triangulaire superieure K x K.
!>
!>    L'operation A*H peut s'ecrire A*(I - V*T*V**H) = A - (A*V)*T*(V**H).
!>    Le calcul est optimise par blocs pour eviter de former explicitement V.
!>    L'algorithme utilise est :
!>
!>    Calculer W = A*V. Ceci est fait par blocs:
!>    W1 = A1*V1 + A2*V2. Ici, A1 est KxK, A2 est Kx(N-K), V1 est KxK, V2 est
!>    MxK. Donc, W1 est KxK.
!>    W2 = B1*V1 + B2*V2. B1 est MxK (zero), B2 est Mx(N-K). Donc, W2 est Mx(N-K).
!>    Le calcul A*V est fait via A_ = A*V, la partie gauche est A(1:K,1:N), la partie droite B(1:M,K+1:N).
!>    W = ( W1 ) est une matrice (K+M) x K.
!>        ( W2 )
!>    Calculer W = W*T. Ceci est fait par blocs :
!>    W1 = W1*T. W1 est KxK, T est KxK. Donc W1 est KxK.
!>    W2 = W2*T. W2 est Mx(N-K), T est KxK. Ceci n'est pas possible directement.
!>    L'operation correcte est W := W*T (on suppose que T est KxK).
!>    Cela signifie que W est KxK.
!>    Puis calculer W**H.
!>    Ensuite, la partie droite du produit A*V, A_part2 = A*V.
!>    A_part2 = ( A2 ) est une matrice (K+M) x (N-K).
!>              ( B2 )
!>    Calculer X = W**H * A_part2.
!>    X1 = T**H * V1**H * A2. (KxK) * (KxK) * (Kx(N-K)) = Kx(N-K).
!>    X2 = T**H * V2**H * B2. (KxK) * (KxM) * (Mx(N-K)) = Kx(N-K).
!>    L'operation finale est A_out := A_in - X.
!>
!>    Finalement, A := A - W * T * V**H.
!>
!>    L'implementation utilise une representation en bloc.
!>
!>    L'operation est Aout = A - V * T * V^H * A.
!>
!>    Si IDENT != 'I', V1 est triangulaire inferieure et V2 est M x K.
!>    Si IDENT == 'I', V1 est une matrice identite.
!>
!>    Les etapes sont:
!>    1. Calculer Y = T * V^H * A.
!>       - Calculer X = V^H * A.
!>         - X1 = V1^H * A1 + V2^H * B1 (B1 est zero ici). Donc X1 = V1^H * A1.
!>         - X2 = V1^H * A2 + V2^H * B2.
!>       - Calculer Y = T * X.
!>         - Y1 = T * X1.
!>         - Y2 = T * X2.
!>    2. Calculer Aout = A - V * Y.
!>       - A1_out = A1 - V1 * Y1.
!>       - B1_out = B1 - V2 * Y1. (B1 est zero).
!>       - A2_out = A2 - V1 * Y2.
!>       - B2_out = B2 - V2 * Y2.
!>
!>    Les tableaux V1 et V2 sont stockes dans A et B.
!>    Si IDENT != 'I', V1 est stockee dans la partie triangulaire inferieure
!>    de A(1:K,1:K) (sans la diagonale).
!>    V2 est stockee dans B(1:M,1:K).
!>
!>    Le calcul est realise en utilisant un tableau de travail 'WORK'.
!>
!>    - Si IDENT != 'I':
!>      - Calculer W = A2. Puis W = V1^H * W. W est K x (N-K).
!>      - Calculer X = B2. Puis X = V2^H * X. X est K x (N-K).
!>      - Calculer Z = W + X. Z est K x (N-K).
!>      - Calculer Z = T * Z. Z est K x (N-K).
!>      - Calculer A2_out = A2 - Z.
!>      - Calculer B2_out = B2 - Z.
!>      - Calculer W = A1. Puis W = V1^H * W. W est K x K.
!>      - Calculer B1_out = -V2 * W. B1_out est M x K.
!>      - Calculer A1_out = A1 - W.
!>
!>    - Si IDENT == 'I':
!>      - Calculer W = A2. Puis W = V1^H * W (V1 est I). W = A2.
!>      - Calculer X = B2. Puis X = V2^H * X. X est K x (N-K).
!>      - Calculer Z = W + X. Z est K x (N-K).
!>      - Calculer Z = T * Z. Z est K x (N-K).
!>      - Calculer A2_out = A2 - Z.
!>      - Calculer B2_out = B2 - Z.
!>      - Calculer W = A1. Puis W = V1^H * W (V1 est I). W = A1.
!>      - Calculer B1_out = -V2 * W. B1_out est M x K.
!>      - Calculer A1_out = A1 - W.
!>
!>    L'ordre des operations est important pour eviter d'ecraser les donnees
!>    necessaires.
!>
!>    Le tableau WORK est utilise pour stocker les resultats intermediaires.
!>
!>    Le tableau WORK a une dimension (LDWORK, max(K, N-K)).
!>    Si N > K:
!>      WORK(1:K, 1:N-K) stocke les resultats des multiplications avec V^H.
!>      La premiere partie est V1^H * A2, la deuxieme est V2^H * B2.
!>      Puis T est applique a ces resultats.
!>      Finalement, les resultats sont soustraits de A2 et B2.
!>
!>    La partie gauche (A1, B1) est traitee separement.
!>    Si IDENT != 'I', V1^H * A1 est calcule et stocke dans WORK(1:K, 1:K).
!>    Puis B1_out = -V2 * WORK(1:K, 1:K) est calcule.
!>    Finalement, A1_out = A1 - WORK(1:K, 1:K) est calcule.
!>
!>    Si IDENT == 'I', V1 est une matrice identite, donc V1^H * A1 = A1.
!>    Les etapes sont similaires mais sans l'application de V1^H.
!>
!>
!>    Reference:
!>    P.82, BLAS-like routines for decompositon,
!>    High-Performance Scientific Computing, Vol. 2, Chap. 14, pp. 77-101.
!>    Ed. J. Dongarra, P. Constantine.
!>
!>

NAME

ZLARFB - Routine de mise à jour de blocs de matrices pour le calcul de la factorisation d'une matrice bande.

SYNOPSIS

SUBROUTINE ZLARFB( SIDE, TRANS, DIRECT, STORE, M, N, K, V, LDV, T, LDT,
     C, LDC, WORK, LDWORK )

Parameters

CHARACTER SIDE

Indique si le bloc de matrice V est appliqué à gauche ou à droite.

SIDE = 'L' : la transformation de Householder est appliquée à gauche, de la forme Q*C.

SIDE = 'R' : la transformation de Householder est appliquée à droite, de la forme C*Q.

CHARACTER TRANS

Indique si la transformation T est appliquée sous forme normale ou transposée/conjuguée.

TRANS = 'N' : la transformation T est appliquée sous forme normale.

TRANS = 'C' : la transformation T est appliquée sous forme transposée/conjuguée.

CHARACTER DIRECT

Indique si les transformations de Householder sont calculées par blocs.

DIRECT = 'F' : avant (forward).

DIRECT = 'B' : arrière (backward).

CHARACTER STORE

Indique comment les matrices V et T sont stockées.

STORE = 'C' : les matrices V et T sont stockées par colonnes.

STORE = 'R' : les matrices V et T sont stockées par lignes.

INTEGER M

Nombre de lignes de la matrice C.

INTEGER N

Nombre de colonnes de la matrice C.

INTEGER K

Nombre de colonnes de la matrice V et nombre de lignes de la matrice T.

COMPLEX*16, DIMENSION ( LDV, * ) V

Matrice de transformation.

Si DIRECT = 'F', V contient les vecteurs de Householder.

Si DIRECT = 'B', V contient les vecteurs de Householder.

INTEGER LDV

Taille de la première dimension de V.

COMPLEX*16, DIMENSION ( LDT, * ) T

Matrice triangulaire.

Si STORE = 'C', T est une matrice triangulaire supérieure.

Si STORE = 'R', T est une matrice triangulaire inférieure.

INTEGER LDT

Taille de la première dimension de T.

COMPLEX*16, DIMENSION ( LDC, * ) C

Matrice de mise à jour.

INTEGER LDC

Taille de la première dimension de C.

COMPLEX*16, DIMENSION ( LDWORK, * ) WORK

Tableau de travail.

INTEGER LDWORK

Taille de la première dimension de WORK.

Function/Subroutine Documentation

Le présent document décrit l'implémentation de la routine ZLARFB, qui est utilisée pour appliquer une transformation de Householder à un bloc de matrices. La routine est conçue pour être utilisée dans des algorithmes tels que la factorisation LU par blocs.

Le calcul de ZLARFB peut être exprimé comme suit:

Si SIDE = 'L', la transformation est Q*C, où

    Q = [ I  0 ] * [ I - V*T*V' ]
        [ 0  I ]   [ 0  I      ]

et V est une matrice de taille M-by-K, et T est une matrice triangulaire de taille K-by-K.

Si SIDE = 'R', la transformation est C*Q, où

    Q = [ I  0 ] * [ I  0      ]
        [ 0  I ]   [ -V*T*V' I ]

et V est une matrice de taille N-by-K, et T est une matrice triangulaire de taille K-by-K.

Les paramètres DIRECT et STORE déterminent la manière dont les transformations sont appliquées et comment les matrices V et T sont stockées.

Voici les formules de mise à jour pour le calcul des blocs de matrices.

       A2_out: = A2_in - T*( A2_in + (V2**H)*B2_in )
!>
!>       B2_out: = B2_in - V2*T*( A2_in + (V2**H)*B2_in )
!>
!>    (2) Description of the Algorithmic Computation.
!>
!>    In the first step, we compute column block 2, i.e. A2 and B2.
!>    Here, we need to use the K-by-(N-K) rectangular workspace
!>    matrix W2 that is of the same size as the matrix A2.
!>    W2 is stored in the array WORK(1:K,1:(N-K)).
!>
!>    In the second step, we compute column block 1, i.e. A1 and B1.
!>    Here, we need to use the K-by-K square workspace matrix W1
!>    that is of the same size as the as the matrix A1.
!>    W1 is stored in the array WORK(1:K,1:K).
!>
!>    NOTE: Hence, in this routine, we need the workspace array WORK
!>    only of size WORK(1:K,1:max(K,N-K)) so it can hold both W2 from
!>    the first step and W1 from the second step.
!>
!>    Case (A), when V1 is unit lower-triangular, i.e. IDENT != 'I',
!>    more computations than in the Case (B).
!>
!>    if( IDENT != 'I' ) then
!>     if ( N > K ) then
!>       (First Step - column block 2)
!>       col2_(1) W2: = A2
!>       col2_(2) W2: = (V1**H) * W2 = (unit_lower_tr_of_(A1)**H) * W2
!>       col2_(3) W2: = W2 + (V2**H) * B2 = W2 + (B1**H) * B2
!>       col2_(4) W2: = T * W2
!>       col2_(5) B2: = B2 - V2 * W2 = B2 - B1 * W2
!>       col2_(6) W2: = V1 * W2 = unit_lower_tr_of_(A1) * W2
!>       col2_(7) A2: = A2 - W2
!>     else
!>       (Second Step - column block 1)
!>       col1_(1) W1: = A1
!>       col1_(2) W1: = (V1**H) * W1 = (unit_lower_tr_of_(A1)**H) * W1
!>       col1_(3) W1: = T * W1
!>       col1_(4) B1: = - V2 * W1 = - B1 * W1
!>       col1_(5) square W1: = V1 * W1 = unit_lower_tr_of_(A1) * W1
!>       col1_(6) square A1: = A1 - W1
!>     end if
!>    end if
!>
!>    Case (B), when V1 is an identity matrix, i.e. IDENT == 'I',
!>    less computations than in the Case (A)
!>
!>    if( IDENT == 'I' ) then
!>     if ( N > K ) then
!>       (First Step - column block 2)
!>       col2_(1) W2: = A2
!>       col2_(3) W2: = W2 + (V2**H) * B2 = W2 + (B1**H) * B2
!>       col2_(4) W2: = T * W2
!>       col2_(5) B2: = B2 - V2 * W2 = B2 - B1 * W2
!>       col2_(7) A2: = A2 - W2
!>     else
!>       (Second Step - column block 1)
!>       col1_(1) W1: = A1
!>       col1_(3) W1: = T * W1
!>       col1_(4) B1: = - V2 * W1 = - B1 * W1
!>       col1_(6) upper-triangular_of_(A1): = A1 - W1
!>     end if
!>    end if
!>
!>    Combine these cases (A) and (B) together, this is the resulting
!>    algorithm:
!>
!>    if ( N > K ) then
!>
!>      (First Step - column block 2)
!>
!>      col2_(1)  W2: = A2
!>      if( IDENT != 'I' ) then
!>        col2_(2)  W2: = (V1**H) * W2
!>                      = (unit_lower_tr_of_(A1)**H) * W2
!>      end if
!>      col2_(3)  W2: = W2 + (V2**H) * B2 = W2 + (B1**H) * B2]
!>      col2_(4)  W2: = T * W2
