subroutine zlacpy (uplo, m, n, a, lda, b, ldb)
ZLACPY copie tout ou partie d'un tableau bidimensionnel sur un autre.

Documentation de la fonction/sous-programme

sous-programme zlacpy (caractère uplo, entier m, entier n, complexe*16, dimension( lda, * ) a, entier lda, complexe*16, dimension( ldb, * ) b, entier ldb)

ZLACPY copie tout ou partie d'un tableau bidimensionnel sur un autre.

Objectif :

!>
!> ZLACPY copie tout ou partie d'une matrice bidimensionnelle A vers une autre
!> matrice B.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie quelle partie de la matrice A doit être copiée vers B.
!>          = 'U' :      Partie triangulaire supérieure
!>          = 'L' :      Partie triangulaire inférieure
!>          Sinon :    La totalité de la matrice A
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>          La matrice A de m lignes et n colonnes.  Si UPLO = 'U', seule la
!>          trapèze supérieure est accédée ; si UPLO = 'L', seule la
!>          trapèze inférieure est accédée.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,N)
!>          En sortie, B = A dans les emplacements spécifiés par UPLO.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,M).
!>
