subroutine zlagge (m, n, kl, ku, d, a, lda, iseed, work, info)
ZLAGGE

Documentation de la fonction/sous-programme

sous-programme zlagge (integer m, integer n, integer kl, integer ku, double precision, dimension( * ) d, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( 4 ) iseed, complex*16, dimension( * ) work, integer info)

ZLAGGE

Objectif :

!>
!> ZLAGGE génère une matrice complexe générale m par n, A, en pré- et post-
!> multipliant une matrice diagonale réelle D par des matrices unitaires aléatoires :
!> A = U*D*V. Les bandes inférieure et supérieure peuvent ensuite être réduites à
!> kl et ku par des transformations unitaires supplémentaires.
!>

Paramètres

M 
!>          M est un entier (INTEGER)
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est un entier (INTEGER)
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

KL

!>          KL est un entier (INTEGER)
!>          Le nombre de sous-diagonales non nulles à l'intérieur de la bande de A.
!>          0 <= KL <= M-1.
!>

KU

!>          KU est un entier (INTEGER)
!>          Le nombre de super-diagonales non nulles à l'intérieur de la bande de A.
!>          0 <= KU <= N-1.
!>

D

!>          D est un tableau de type DOUBLE PRECISION, de dimension (min(M,N))
!>          Les éléments diagonaux de la matrice diagonale D.
!>

A

!>          A est un tableau de type COMPLEX*16, de dimension (LDA,N)
!>          La matrice m par n générée A.
!>

LDA

!>          LDA est un entier (INTEGER)
!>          La dimension principale du tableau A.  LDA >= M.
!>

ISEED

!>          ISEED est un tableau d'entiers (INTEGER), de dimension (4)
!>          En entrée, la graine du générateur de nombres aléatoires ; les éléments
!>          du tableau doivent être compris entre 0 et 4095, et ISEED(4) doit être
!>          impair.
!>          En sortie, la graine est mise à jour.
!>

WORK

!>          WORK est un tableau de type COMPLEX*16, de dimension (M+N)
!>

INFO

!>          INFO est un entier (INTEGER)
!>          = 0 : succès
!>          < 0 : si INFO = -i, le i-ème argument était invalide
!>
