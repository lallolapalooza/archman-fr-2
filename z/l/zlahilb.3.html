subroutine zlahilb (n, nrhs, a, lda, x, ldx, b, ldb, work, info, path)
ZLAHILB

Documentation de la fonction/sous-programme

sous-programme zlahilb (integer n, integer nrhs, complex*16, dimension(lda,n) a, integer lda, complex*16, dimension(ldx, nrhs) x, integer ldx, complex*16, dimension(ldb, nrhs) b, integer ldb, double precision, dimension(n) work, integer info, character*3 path)

ZLAHILB

Objectif:

!>
!> ZLAHILB génère une matrice de Hilbert mise à l'échelle N par N dans A ainsi que
!> NRHS côtés droits dans B et des solutions dans X tels que A*X=B.
!>
!> La matrice de Hilbert est mise à l'échelle par M = PPCM(1, 2, ..., 2*N-1) de sorte que toutes
!> les entrées soient des entiers. Les côtés droits sont les NRHS premières
!> colonnes de la matrice identité multipliée par M, et les solutions sont les
!> premières NRHS colonnes de la matrice de Hilbert inverse.
!>
!> Le nombre de condition de la matrice de Hilbert croît exponentiellement avec
!> sa taille, environ comme O(e ** (3.5*N)). De plus, les matrices de Hilbert inverses
!> au-delà d'une dimension relativement petite ne peuvent pas être
!> générées exactement sans précision supplémentaire. La précision est épuisée
!> lorsque la plus grande entrée dans la matrice de Hilbert inverse est supérieure à
!> 2 à la puissance du nombre de bits dans la fraction du type de données
!> utilisé plus un, soit 24 pour la précision simple.
!>
!> En précision simple, la solution générée est exacte pour N <= 6 et a
!> une erreur composante par composante faible pour 7 <= N <= 11.
!>

Paramètres

N 
!>          N est un ENTIER
!>          La dimension de la matrice A.
!>

NRHS

!>          NRHS est un ENTIER
!>          Le nombre de côtés droits demandés.
!>

A

!>          A est un tableau COMPLEXE, dimension (LDA, N)
!>          La matrice de Hilbert mise à l'échelle générée.
!>

LDA

!>          LDA est un ENTIER
!>          La dimension de tête du tableau A. LDA >= N.
!>

X

!>          X est un tableau COMPLEXE, dimension (LDX, NRHS)
!>          Les solutions exactes générées. Actuellement, les NRHS premières
!>          colonnes de la matrice de Hilbert inverse.
!>

LDX

!>          LDX est un ENTIER
!>          La dimension de tête du tableau X. LDX >= N.
!>

B

!>          B est un tableau RÉEL, dimension (LDB, NRHS)
!>          Les côtés droits générés. Actuellement, les NRHS premières
!>          colonnes de PPCM(1, 2, ..., 2*N-1) multiplié par la matrice identité.
!>

LDB

!>          LDB est un ENTIER
!>          La dimension de tête du tableau B. LDB >= N.
!>

WORK

!>          WORK est un tableau RÉEL, dimension (N)
!>

INFO

!>          INFO est un ENTIER
!>          = 0 : sortie réussie
!>          = 1 : N est trop grand ; les données sont toujours générées mais peuvent ne pas
!>               être exactes.
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>

PATH

!>          PATH est une CHAÎNE de caractères de longueur 3
!>          Le nom du chemin LAPACK.
!>
