subroutine ztplqt (integer m, integer n, integer l, integer mb, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( * ) work, integer info)

ZTPLQT

Objectif :

!>
!> ZTPLQT calcule une factorisation LQ bloquée d'une matrice complexe
!>  C, qui est composée d'une matrice triangulaire A et d'une matrice pentagonale B,
!>  en utilisant la représentation WY compacte pour Q.
!>

Paramètres

M 
!>          M est un entier (INTEGER)
!>          Le nombre de lignes de la matrice B, et l'ordre de la
!>          matrice triangulaire A.
!>          M >= 0.
!>

N

!>          N est un entier (INTEGER)
!>          Le nombre de colonnes de la matrice B.
!>          N >= 0.
!>

L

!>          L est un entier (INTEGER)
!>          Le nombre de lignes de la partie trapézoïdale inférieure de B.
!>          MIN(M,N) >= L >= 0.  Voir Détails supplémentaires.
!>

MB

!>          MB est un entier (INTEGER)
!>          La taille de bloc à utiliser dans la factorisation LQ bloquée.  M >= MB >= 1.
!>

A

!>          A est un tableau de type COMPLEX*16, de dimension (LDA,M)
!>          En entrée, la matrice triangulaire inférieure A de taille M x M.
!>          En sortie, les éléments sur et sous la diagonale du tableau
!>          contiennent la matrice triangulaire inférieure L.
!>

LDA

!>          LDA est un entier (INTEGER)
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!>

B

!>          B est un tableau de type COMPLEX*16, de dimension (LDB,N)
!>          En entrée, la matrice pentagonale B de taille M x N. Les L premières
!>          colonnes sont rectangulaires, et les N-L dernières colonnes sont
!>          trapézoïdales inférieures.
!>          En sortie, B contient la matrice pentagonale V.  Voir Détails supplémentaires.
!>

LDB

!>          LDB est un entier (INTEGER)
!>          La dimension principale du tableau B.  LDB >= max(1,M).
!>

T

!>          T est un tableau de type COMPLEX*16, de dimension (LDT,N)
!>          Les réflecteurs de bloc triangulaires inférieurs stockés sous forme compacte
!>          en séquence de blocs triangulaires supérieurs.  Voir Détails supplémentaires.
!>

LDT

!>          LDT est un entier (INTEGER)
!>          La dimension principale du tableau T.  LDT >= MB.
!>

WORK

!>          WORK est un tableau de type COMPLEX*16, de dimension (MB*M)
!>

INFO

!>          INFO est un entier (INTEGER)
!>          = 0 : succès de l'exécution
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>

Détails supplémentaires :

!>
!>  La matrice d'entrée C est une matrice M x (M+N)
!>
!>               C = [ A ] [ B ]
!>
!>
!>  où A est une matrice M x M triangulaire inférieure, et B est une matrice pentagonale
!>  M x N composée d'une matrice rectangulaire M x (N-L), B1, à gauche d'une
!>  matrice trapézoïdale inférieure M x L, B2 :
!>          [ B ] = [ B1 ] [ B2 ]
!>                   [ B1 ]  <- rectangulaire M x (N-L)
!>                   [ B2 ]  <-     trapézoïdale inférieure M x L.
!>
!>  La matrice trapézoïdale inférieure B2 est constituée des L premières colonnes d'une
!>  matrice triangulaire inférieure M x M, où 0 <= L <= MIN(M,N). Si L=0,
!>  B est rectangulaire M x N ; si M=L=N, B est triangulaire inférieure.
!>
!>  La matrice W stocke les réflecteurs élémentaires H(i) dans la i-ème ligne
!>  au-dessus de la diagonale (de A) dans la matrice d'entrée C de taille M x (M+N)
!>            [ C ] = [ A ] [ B ]
!>                   [ A ]  <- M x M triangulaire inférieure
!>                   [ B ]  <- M x N pentagonale
!>
!>  de sorte que W peut être représentée comme
!>            [ W ] = [ I ] [ V ]
!>                   [ I ]  <- identité, M x M
!>                   [ V ]  <- M x N, même forme que B.
!>
!>  Ainsi, toutes les informations nécessaires pour W sont contenues en sortie dans B,
!>  que nous appelons V ci-dessus. Notez que V a la même forme que B ; c'est-à-dire,
!>            [ V ] = [ V1 ] [ V2 ]
!>                   [ V1 ] <- M x (N-L) rectangulaire
!>                   [ V2 ] <-     M x L trapézoïdale inférieure.
!>
!>  Les lignes de V représentent les vecteurs qui définissent les H(i).
!>
!>  Le nombre de blocs est B = plafond(M/MB), où chaque
!>  bloc est d'ordre MB sauf le dernier bloc, qui est d'ordre
!>  IB = M - (M-1)*MB. Pour chacun des B blocs, un facteur de réflecteur
!>  de bloc triangulaire supérieur est calculé : T1, T2, ..., TB. Les T de taille
!>  MB x MB (et IB x IB pour le dernier bloc) sont stockés dans la matrice T de taille MB x N comme
!>
!>               T = [T1 T2 ... TB].
!>
