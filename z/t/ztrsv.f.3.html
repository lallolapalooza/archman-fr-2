subroutine ztrsv (caractère uplo, caractère trans, caractère diag, entier n, complexe*16, dimension(lda,*) a, entier lda, complexe*16, dimension(*) x, entier incx)

ZTRSV

But :

!>
!> ZTRSV  résout l'un des systèmes d'équations
!>
!>    A*x = b,   ou   A**T*x = b,   ou   A**H*x = b,
!>
!> où b et x sont des vecteurs de n éléments et A est une matrice
!> triangulaire unitaire, ou non unitaire, supérieure ou inférieure de taille n x n.
!>
!> Aucun test de singularité ou de quasi-singularité n'est inclus dans cette
!> routine. De tels tests doivent être effectués avant d'appeler cette routine.
!>

Paramètres

UPLO 
!>          UPLO est CARACTERE*1
!>           À l'entrée, UPLO spécifie si la matrice est une matrice supérieure ou
!>           inférieure comme suit :
!>
!>              UPLO = 'U' ou 'u'   A est une matrice triangulaire supérieure.
!>
!>              UPLO = 'L' ou 'l'   A est une matrice triangulaire inférieure.
!>

TRANS

!>          TRANS est CARACTERE*1
!>           À l'entrée, TRANS spécifie les équations à résoudre comme suit :
!>
!>              TRANS = 'N' ou 'n'   A*x = b.
!>
!>              TRANS = 'T' ou 't'   A**T*x = b.
!>
!>              TRANS = 'C' ou 'c'   A**H*x = b.
!>

DIAG

!>          DIAG est CARACTERE*1
!>           À l'entrée, DIAG spécifie si A est unitaire ou non
!>           triangulaire comme suit :
!>
!>              DIAG = 'U' ou 'u'   A est supposée être unitaire triangulaire.
!>
!>              DIAG = 'N' ou 'n'   A n'est pas supposée être unitaire
!>                                  triangulaire.
!>

N

!>          N est ENTIER
!>           À l'entrée, N spécifie l'ordre de la matrice A.
!>           N doit être supérieur ou égal à zéro.
!>

A

!>          A est un tableau COMPLEX*16, dimension ( LDA, N )
!>           Avant l'entrée avec UPLO = 'U' ou 'u', la partie principale n x n
!>           supérieure triangulaire du tableau A doit contenir la matrice
!>           triangulaire supérieure et la partie strictement inférieure
!>           triangulaire de A n'est pas référencée.
!>           Avant l'entrée avec UPLO = 'L' ou 'l', la partie principale n x n
!>           inférieure triangulaire du tableau A doit contenir la matrice
!>           triangulaire inférieure et la partie strictement supérieure
!>           triangulaire de A n'est pas référencée.
!>           Notez que lorsque DIAG = 'U' ou 'u', les éléments diagonaux de
!>           A ne sont pas non plus référencés, mais sont supposés être l'unité.
!>

LDA

!>          LDA est ENTIER
!>           À l'entrée, LDA spécifie la première dimension de A telle que déclarée
!>           dans le programme appelant (sous-programme). LDA doit être
!>           au moins max( 1, n ).
!>

X

!>          X est un tableau COMPLEX*16, dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           Avant l'entrée, le tableau incrémenté X doit contenir le vecteur
!>           de membre droit b de n éléments. À la sortie, X est écrasé
!>           par le vecteur solution x.
!>

INCX

!>          INCX est ENTIER
!>           À l'entrée, INCX spécifie l'incrément pour les éléments de
!>           X. INCX ne doit pas être nul.
!>
