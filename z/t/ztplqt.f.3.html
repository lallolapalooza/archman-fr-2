subroutine ztplqt (m, n, l, mb, a, lda, b, ldb, t, ldt, work, info)
ZTPLQT

Documentation de la fonction/sous-routine

sous-routine ztplqt (entier m, entier n, entier l, entier mb, complexe*16, dimension( lda, * ) a, entier lda, complexe*16, dimension( ldb, * ) b, entier ldb, complexe*16, dimension( ldt, * ) t, entier ldt, complexe*16, dimension( * ) work, entier info)

ZTPLQT

Purpose:

!>
!> ZTPLQT calcule une factorisation LQ bloquée d'une matrice complexe
!>  C, qui est composée d'une matrice triangulaire A et d'une matrice
!>  pentagonale B, en utilisant la représentation WY compacte pour Q.
!>

Paramètres

M 
!>          M est ENTIER
!>          Le nombre de lignes de la matrice B, et l'ordre de la
!>          matrice triangulaire A.
!>          M >= 0.
!>

N

!>          N est ENTIER
!>          Le nombre de colonnes de la matrice B.
!>          N >= 0.
!>

L

!>          L est ENTIER
!>          Le nombre de lignes de la partie trapézoïdale inférieure de B.
!>          MIN(M,N) >= L >= 0. Voir Détails supplémentaires.
!>

MB

!>          MB est ENTIER
!>          La taille du bloc à utiliser dans le QR bloqué.  M >= MB >= 1.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,M)
!>          À l'entrée, la matrice triangulaire inférieure M-par-M A.
!>          À la sortie, les éléments sur et sous la diagonale du tableau
!>          contiennent la matrice triangulaire inférieure L.
!>

LDA

!>          LDA est ENTIER
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,N)
!>          À l'entrée, la matrice pentagonale M-par-N B. Les N-L premières
!>          colonnes sont rectangulaires, et les L dernières colonnes sont
!>          trapézoïdales inférieures.
!>          À la sortie, B contient la matrice pentagonale V. Voir Détails
!>          supplémentaires.
!>

LDB

!>          LDB est ENTIER
!>          La dimension principale du tableau B.  LDB >= max(1,M).
!>

T

!>          T est un tableau COMPLEX*16, dimension (LDT,N)
!>          Les réflecteurs de blocs triangulaires inférieurs stockés sous forme
!>          compacte comme une séquence de blocs triangulaires supérieurs.
!>          Voir Détails supplémentaires.
!>

LDT

!>          LDT est ENTIER
!>          La dimension principale du tableau T.  LDT >= MB.
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension (MB*M)
!>

INFO

!>          INFO est ENTIER
!>          = 0:  sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>

Détails supplémentaires:

!>
!>  La matrice d'entrée C est une matrice M-par-(M+N)
!>
!>               C = [ A ] [ B ]
!>
!>
!>  où A est une matrice M-par-M triangulaire inférieure, et B est une matrice
!>  pentagonale M-par-N composée d'une matrice rectangulaire M-par-(N-L) B1
!>  à gauche d'une matrice trapézoïdale inférieure M-par-L B2 :
!>          [ B ] = [ B1 ] [ B2 ]
!>                   [ B1 ]  <- rectangulaire M-par-(N-L)
!>                   [ B2 ]  <-     trapézoïdale inférieure M-par-L.
!>
!>  La matrice trapézoïdale inférieure B2 consiste des L premières colonnes
!>  d'une matrice triangulaire inférieure M-par-M, où 0 <= L <= MIN(M,N).
!>  Si L=0, B est rectangulaire M-par-N ; si M=L=N, B est triangulaire inférieure.
!>
!>  La matrice W stocke les réflecteurs élémentaires H(i) dans la i-ème ligne
!>  au-dessus de la diagonale (de A) dans la matrice d'entrée C M-par-(M+N)
!>            [ C ] = [ A ] [ B ]
!>                   [ A ]  <- triangulaire inférieure M-par-M
!>                   [ B ]  <- pentagonale M-par-N
!>
!>  de sorte que W peut être représentée comme
!>            [ W ] = [ I ] [ V ]
!>                   [ I ]  <- identité, M-par-M
!>                   [ V ]  <- M-par-N, même forme que B.
!>
!>  Ainsi, toutes les informations nécessaires pour W sont contenues à la sortie
!>  dans B, que nous appelons V ci-dessus. Notons que V a la même forme que B ;
!>  c'est-à-dire,
!>            [ V ] = [ V1 ] [ V2 ]
!>                   [ V1 ] <- rectangulaire M-par-(N-L)
!>                   [ V2 ] <-     trapézoïdale inférieure M-par-L.
!>
!>  Les lignes de V représentent les vecteurs qui définissent les H(i).
!>
!>  Le nombre de blocs est B = ceiling(M/MB), où chaque
!>  bloc est d'ordre MB sauf le dernier bloc, qui est d'ordre
!>  IB = M - (M-1)*MB. Pour chacun des B blocs, un facteur de réflecteur
!>  de bloc triangulaire supérieur est calculé : T1, T2, ..., TB. Les T MB-par-MB
!>  (et IB-par-IB pour le dernier bloc) sont stockés dans la matrice T
!>  MB-par-N comme
!>
!>               T = [T1 T2 ... TB].
!>
