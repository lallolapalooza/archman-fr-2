sous-programme ztbt03 (caractère uplo, caractère trans, caractère diag, entier n, entier kd, entier nrhs, complexe*16, dimension( ldab, * ) ab, entier ldab, double précision scale, double précision, dimension( * ) cnorm, double précision tscal, complexe*16, dimension( ldx, * ) x, entier ldx, complexe*16, dimension( ldb, * ) b, entier ldb, complexe*16, dimension( * ) work, double précision resid)

ZTBT03

Objectif :

!>
!> ZTBT03 calcule le résidu pour la solution d'un système triangulaire mis à l'échelle
!> d'équations  A*x = s*b,  A**T *x = s*b,  ou  A**H *x = s*b
!> lorsque A est une matrice triangulaire bande. Ici A**T  désigne la transposée
!> de A, A**H désigne la transposée conjuguée de A, s est un scalaire, et
!> x et b sont des matrices N par NRHS. Le rapport de test est le maximum sur
!> le nombre de seconds membres de
!>    norm(s*b - op(A)*x) / ( norm(op(A)) * norm(x) * EPS ),
!> où op(A) désigne A, A**T, ou A**H, et EPS est l'epsilon machine.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si la matrice A est triangulaire supérieure ou inférieure.
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

TRANS

!>          TRANS est CHARACTER*1
!>          Spécifie l'opération appliquée à A.
!>          = 'N':  A *x = s*b     (Pas de transposée)
!>          = 'T':  A**T *x = s*b  (Transposée)
!>          = 'C':  A**H *x = s*b  (Transposée conjuguée)
!>

DIAG

!>          DIAG est CHARACTER*1
!>          Spécifie si la matrice A est unitaire ou non.
!>          = 'N':  Non unitaire
!>          = 'U':  Unitaire
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

KD

!>          KD est INTEGER
!>          Le nombre de super-diagonales ou sous-diagonales de la
!>          matrice triangulaire bande A.  KD >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de colonnes
!>          des matrices X et B.  NRHS >= 0.
!>

AB

!>          AB est un tableau COMPLEX*16, dimension (LDAB,N)
!>          La matrice triangulaire bande supérieure ou inférieure A, stockée dans les
!>          kd+1 premières lignes du tableau. La j-ième colonne de A est stockée
!>          dans la j-ième colonne du tableau AB comme suit :
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) pour max(1,j-kd)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KD+1.
!>

SCALE

!>          SCALE est DOUBLE PRECISION
!>          Le facteur de mise à l'échelle s utilisé pour résoudre le système triangulaire.
!>

CNORM

!>          CNORM est un tableau DOUBLE PRECISION, dimension (N)
!>          Les normes 1 des colonnes de A, sans compter la diagonale.
!>

TSCAL

!>          TSCAL est DOUBLE PRECISION
!>          Le facteur de mise à l'échelle utilisé pour calculer les normes 1 dans CNORM.
!>          CNORM contient en fait les normes des colonnes de TSCAL*A.
!>

X

!>          X est un tableau COMPLEX*16, dimension (LDX,NRHS)
!>          Les vecteurs solution calculés pour le système d'équations
!>          linéaires.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,NRHS)
!>          Les vecteurs du second membre pour le système d'équations
!>          linéaires.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension (N)
!>

RESID

!>          RESID est DOUBLE PRECISION
!>          Le maximum sur le nombre de seconds membres de
!>          norm(op(A)*x - s*b) / ( norm(op(A)) * norm(x) * EPS ).
!>
