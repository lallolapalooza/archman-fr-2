subroutine ztprfb (character side, character trans, character direct, character storev, integer m, integer n, integer k, integer l, complex*16, dimension( ldv, * ) v, integer ldv, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldwork, * ) work, integer ldwork)

ZTPRFB applique un réflecteur de bloc triangulaire-pentagonal complexe à une matrice complexe, qui est composée de deux blocs.

But :

!>
!> ZTPRFB applique un réflecteur de bloc complexe H ou sa
!> transposée conjuguée H**H à une matrice complexe C, qui est composée de deux
!> blocs A et B, soit par la gauche soit par la droite.
!>
!>

Paramètres

SIDE 
!>          SIDE est CHARACTER*1
!>          = 'L' : appliquer H ou H**H par la gauche
!>          = 'R' : appliquer H ou H**H par la droite
!>

TRANS

!>          TRANS est CHARACTER*1
!>          = 'N' : appliquer H (pas de transposition)
!>          = 'C' : appliquer H**H (transposée conjuguée)
!>

DIRECT

!>          DIRECT est CHARACTER*1
!>          Indique comment H est formé à partir d'un produit de
!>          réflecteurs élémentaires
!>          = 'F' : H = H(1) H(2) . . . H(k) (en avant)
!>          = 'B' : H = H(k) . . . H(2) H(1) (en arrière)
!>

STOREV

!>          STOREV est CHARACTER*1
!>          Indique comment les vecteurs qui définissent les réflecteurs
!>          élémentaires sont stockés :
!>          = 'C' : Colonnes
!>          = 'R' : Lignes
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice B.
!>          M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice B.
!>          N >= 0.
!>

K

!>          K est INTEGER
!>          L'ordre de la matrice T, c'est-à-dire le nombre de réflecteurs
!>          élémentaires dont le produit définit le réflecteur de bloc.
!>          K >= 0.
!>

L

!>          L est INTEGER
!>          L'ordre de la partie trapézoïdale de V.
!>          K >= L >= 0.  Voir Détails supplémentaires.
!>

V

!>          V est un tableau COMPLEX*16, dimension
!>                                (LDV,K) si STOREV = 'C'
!>                                (LDV,M) si STOREV = 'R' et SIDE = 'L'
!>                                (LDV,N) si STOREV = 'R' et SIDE = 'R'
!>          La matrice pentagonale V, qui contient les réflecteurs élémentaires
!>          H(1), H(2), ..., H(K).  Voir Détails supplémentaires.
!>

LDV

!>          LDV est INTEGER
!>          La dimension principale du tableau V.
!>          Si STOREV = 'C' et SIDE = 'L', LDV >= max(1,M) ;
!>          si STOREV = 'C' et SIDE = 'R', LDV >= max(1,N) ;
!>          si STOREV = 'R', LDV >= K.
!>

T

!>          T est un tableau COMPLEX*16, dimension (LDT,K)
!>          La matrice triangulaire T de taille K par K dans la représentation
!>          du réflecteur de bloc.
!>

LDT

!>          LDT est INTEGER
!>          La dimension principale du tableau T.
!>          LDT >= K.
!>

A

!>          A est un tableau COMPLEX*16, dimension
!>          (LDA,N) si SIDE = 'L' ou (LDA,K) si SIDE = 'R'
!>          À l'entrée, la matrice A de taille K par N ou M par K.
!>          À la sortie, A est écrasée par le bloc correspondant de
!>          H*C ou H**H*C ou C*H ou C*H**H.  Voir Détails supplémentaires.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.
!>          Si SIDE = 'L', LDA >= max(1,K) ;
!>          Si SIDE = 'R', LDA >= max(1,M).
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,N)
!>          À l'entrée, la matrice B de taille M par N.
!>          À la sortie, B est écrasée par le bloc correspondant de
!>          H*C ou H**H*C ou C*H ou C*H**H.  Voir Détails supplémentaires.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.
!>          LDB >= max(1,M).
!>

WORK

!>          WORK est un tableau COMPLEX*16, dimension
!>          (LDWORK,N) si SIDE = 'L',
!>          (LDWORK,K) si SIDE = 'R'.
!>

LDWORK

!>          LDWORK est INTEGER
!>          La dimension principale du tableau WORK.
!>          Si SIDE = 'L', LDWORK >= K ;
!>          si SIDE = 'R', LDWORK >= M.
!>

Détails supplémentaires :

!>
!>  La matrice C est une matrice composite formée à partir des blocs A et B.
!>  Le bloc B est de taille M par N ; si SIDE = 'R', A est de taille M par K,
!>  et si SIDE = 'L', A est de taille K par N.
!>
!>  Si SIDE = 'R' et DIRECT = 'F', C = [A B].
!>
!>  Si SIDE = 'L' et DIRECT = 'F', C = [A]
!>                                      [B].
!>
!>  Si SIDE = 'R' et DIRECT = 'B', C = [B A].
!>
!>  Si SIDE = 'L' et DIRECT = 'B', C = [B]
!>                                      [A].
!>
!>  La matrice pentagonale V est composée d'un bloc rectangulaire V1 et d'un
!>  bloc trapézoïdal V2.  La taille du bloc trapézoïdal est déterminée par
!>  le paramètre L, où 0<=L<=K.  Si L=K, le bloc V2 de V est triangulaire ;
!>  si L=0, il n'y a pas de bloc trapézoïdal, ainsi V = V1 est rectangulaire.
!>
!>  Si DIRECT = 'F' et STOREV = 'C' :  V = [V1]
!>                                         [V2]
!>     - V2 est trapézoïdal supérieur (premières L lignes de la triangulaire supérieure K par K)
!>
!>  Si DIRECT = 'F' et STOREV = 'R' :  V = [V1 V2]
!>
!>     - V2 est trapézoïdal inférieur (premières L colonnes de la triangulaire inférieure K par K)
!>
!>  Si DIRECT = 'B' et STOREV = 'C' :  V = [V2]
!>                                         [V1]
!>     - V2 est trapézoïdal inférieur (dernières L lignes de la triangulaire inférieure K par K)
!>
!>  Si DIRECT = 'B' et STOREV = 'R' :  V = [V2 V1]
!>
!>     - V2 est trapézoïdal supérieur (dernières L colonnes de la triangulaire supérieure K par K)
!>
!>  Si STOREV = 'C' et SIDE = 'L', V est M par K avec V2 L par K.
!>
!>  Si STOREV = 'C' et SIDE = 'R', V est N par K avec V2 L par K.
!>
!>  Si STOREV = 'R' et SIDE = 'L', V est K par M avec V2 K par L.
!>
!>  Si STOREV = 'R' et SIDE = 'R', V est K par N avec V2 K par L.
!>
