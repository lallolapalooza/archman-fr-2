subroutine ztrsyl (trana, tranb, isgn, m, n, a, lda, b, ldb, c, ldc, scale, info)
ZTRSYL

Documentation de la fonction/sous-programme

sous-programme ztrsyl (caractère trana, caractère tranb, entier isgn, entier m, entier n, complexe*16, dimension( lda, * ) a, entier lda, complexe*16, dimension( ldb, * ) b, entier ldb, complexe*16, dimension( ldc, * ) c, entier ldc, double précision scale, entier info)

ZTRSYL

Objectif :

!>
!> ZTRSYL résout l'équation matricielle complexe de Sylvester :
!>
!>    op(A)*X + X*op(B) = scale*C ou
!>    op(A)*X - X*op(B) = scale*C,
!>
!> où op(A) = A ou A**H, et A et B sont toutes deux triangulaires supérieures. A est
!> de taille M par M et B est de taille N par N ; C et la solution X sont
!> de taille M par N ; et scale est un facteur d'échelle en sortie, fixé <= 1 pour éviter
!> le débordement dans X.
!>

Paramètres

TRANA 
!>          TRANA est CHARACTER*1
!>          Spécifie l'option op(A) :
!>          = 'N': op(A) = A    (Pas de transposée)
!>          = 'C': op(A) = A**H (Transposée conjuguée)
!>

TRANB

!>          TRANB est CHARACTER*1
!>          Spécifie l'option op(B) :
!>          = 'N': op(B) = B    (Pas de transposée)
!>          = 'C': op(B) = B**H (Transposée conjuguée)
!>

ISGN

!>          ISGN est INTEGER
!>          Spécifie le signe dans l'équation :
!>          = +1 : résoudre op(A)*X + X*op(B) = scale*C
!>          = -1 : résoudre op(A)*X - X*op(B) = scale*C
!>

M

!>          M est INTEGER
!>          L'ordre de la matrice A, et le nombre de lignes des
!>          matrices X et C. M >= 0.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice B, et le nombre de colonnes des
!>          matrices X et C. N >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,M)
!>          La matrice triangulaire supérieure A.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,M).
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,N)
!>          La matrice triangulaire supérieure B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,N).
!>

C

!>          C est un tableau COMPLEX*16, dimension (LDC,N)
!>          À l'entrée, la matrice C du membre de droite de taille M par N.
!>          À la sortie, C est écrasée par la matrice solution X.
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale du tableau C. LDC >= max(1,M)
!>

SCALE

!>          SCALE est DOUBLE PRECISION
!>          Le facteur d'échelle, scale, fixé <= 1 pour éviter le débordement dans X.
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>          = 1 : A et B ont des valeurs propres communes ou très proches ; des valeurs perturbées
!>               ont été utilisées pour résoudre l'équation (mais les matrices
!>               A et B restent inchangées).
!>
