sous-programme ztrt05 (caractère uplo, caractère trans, caractère diag, entier n, entier nrhs, complexe*16, dimension( lda, * ) a, entier lda, complexe*16, dimension( ldb, * ) b, entier ldb, complexe*16, dimension( ldx, * ) x, entier ldx, complexe*16, dimension( ldxact, * ) xact, entier ldxact, double précision, dimension( * ) ferr, double précision, dimension( * ) berr, double précision, dimension( * ) reslts)

ZTRT05

But :

!>
!> ZTRT05 teste les bornes d'erreur issues du raffinement itératif pour la
!> solution calculée d'un système d'équations A*X = B, où A est une matrice
!> triangulaire de taille n par n.
!>
!> RESLTS(1) = test de la borne d'erreur
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> Une grande valeur est retournée si ce ratio n'est pas inférieur à un.
!>
!> RESLTS(2) = résidu du raffinement itératif
!>           = le maximum de BERR / ( (n+1)*EPS + (*) ), où
!>             (*) = (n+1)*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si la matrice A est triangulaire supérieure ou inférieure.
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

TRANS

!>          TRANS est CHARACTER*1
!>          Spécifie la forme du système d'équations.
!>          = 'N':  A * X = B  (Pas de transposée)
!>          = 'T':  A'* X = B  (Transposée)
!>          = 'C':  A'* X = B  (Conjuguée transposée = Transposée)
!>

DIAG

!>          DIAG est CHARACTER*1
!>          Spécifie si la matrice A est unitaire triangulaire ou non.
!>          = 'N':  Triangulaire non unitaire
!>          = 'U':  Triangulaire unitaire
!>

N

!>          N est INTEGER
!>          Le nombre de lignes des matrices X, B et XACT, et l'ordre de
!>          la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de colonnes des matrices X, B et XACT.
!>          NRHS >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>          La matrice triangulaire A.  Si UPLO = 'U', la partie supérieure
!>          triangulaire gauche de taille n par n du tableau A contient la
!>          matrice triangulaire supérieure, et la partie strictement
!>          triangulaire inférieure de A n'est pas référencée.  Si UPLO = 'L',
!>          la partie inférieure triangulaire gauche de taille n par n du
!>          tableau A contient la matrice triangulaire inférieure, et la
!>          partie strictement triangulaire supérieure de A n'est pas
!>          référencée.  Si DIAG = 'U', les éléments diagonaux de A ne sont
!>          pas non plus référencés et sont supposés être 1.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,NRHS)
!>          Les vecteurs du second membre pour le système d'équations
!>          linéaires.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est un tableau COMPLEX*16, dimension (LDX,NRHS)
!>          Les vecteurs solution calculés. Chaque vecteur est stocké comme
!>          une colonne de la matrice X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

XACT

!>          XACT est un tableau COMPLEX*16, dimension (LDXACT,NRHS)
!>          Les vecteurs solution exacts. Chaque vecteur est stocké comme
!>          une colonne de la matrice XACT.
!>

LDXACT

!>          LDXACT est INTEGER
!>          La dimension principale du tableau XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>          Les bornes d'erreur avant estimées pour chaque vecteur solution
!>          X.  Si XTRUE est la vraie solution, FERR borne la magnitude de
!>          la plus grande entrée dans (X - XTRUE) divisée par la magnitude
!>          de la plus grande entrée dans X.
!>

BERR

!>          BERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>          L'erreur arrière relative composante par composante de chaque
!>          vecteur solution (c'est-à-dire, le plus petit changement relatif
!>          dans n'importe quelle entrée de A ou B qui rend X une solution
!>          exacte).
!>

RESLTS

!>          RESLTS est un tableau DOUBLE PRECISION, dimension (2)
!>          Le maximum sur les vecteurs solutions NRHS des ratios :
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( (n+1)*EPS + (*) )
!>
