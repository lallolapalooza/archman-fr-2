subroutine ztbt06 (double precision rcond, double precision rcondc, character uplo, character diag, integer n, integer kd, complex*16, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) rwork, double precision rat)

ZTBT06

But :

!>
!> ZTBT06 calcule un rapport de test comparant RCOND (le nombre réciproque
!> de conditionnement d'une matrice triangulaire A) et RCONDC, l'estimation
!> calculée par ZTBCON. Les informations sur la matrice triangulaire A sont
!> utilisées si une estimation est nulle et l'autre non nulle pour décider si
!> un sous-dépassement dans l'estimation est justifié.
!>

Paramètres

RCOND 
!>          RCOND est DOUBLE PRECISION
!>          L'estimation du nombre réciproque de conditionnement obtenue en
!>          formant l'inverse explicite de la matrice A et en calculant
!>          RCOND = 1/( norm(A) * norm(inv(A)) ).
!>

RCONDC

!>          RCONDC est DOUBLE PRECISION
!>          L'estimation du nombre réciproque de conditionnement calculée par
!>          ZTBCON.
!>

UPLO

!>          UPLO est CHARACTER
!>          Spécifie si la matrice A est triangulaire supérieure ou inférieure.
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

DIAG

!>          DIAG est CHARACTER
!>          Spécifie si la matrice A est unitaire ou non.
!>          = 'N':  Triangulaire non unitaire
!>          = 'U':  Triangulaire unitaire
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

KD

!>          KD est INTEGER
!>          Le nombre de superdiagonales ou sous-diagonales de la
!>          bande triangulaire A.  KD >= 0.
!>

AB

!>          AB est tableau COMPLEX*16, dimension (LDAB,N)
!>          La matrice de bande triangulaire supérieure ou inférieure A, stockée dans les
!>          kd+1 premières lignes du tableau. La j-ième colonne de A est stockée
!>          dans la j-ième colonne du tableau AB comme suit :
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) pour max(1,j-kd)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KD+1.
!>

RWORK

!>          RWORK est tableau DOUBLE PRECISION, dimension (N)
!>

RAT

!>          RAT est DOUBLE PRECISION
!>          Le rapport de test.  Si RCOND et RCONDC sont tous deux non nuls,
!>             RAT = MAX( RCOND, RCONDC )/MIN( RCOND, RCONDC ) - 1.
!>          Si RAT = 0, les deux estimations sont exactement identiques.
!>
