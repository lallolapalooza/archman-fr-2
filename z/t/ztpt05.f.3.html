sous-routine ztpt05 (character uplo, character trans, character diag, integer n, integer nrhs, complex*16, dimension( * ) ap, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldxact, * ) xact, integer ldxact, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) reslts)

ZTPT05

Objectif :

!>
!> ZTPT05 teste les bornes d'erreur de l'affinage itératif pour la
!> solution calculée d'un système d'équations A*X = B, où A est une
!> matrice triangulaire en format de stockage compacté.
!>
!> RESLTS(1) = test de la borne d'erreur
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> Une grande valeur est retournée si ce ratio n'est pas inférieur à un.
!>
!> RESLTS(2) = résidu de la routine d'affinage itératif
!>           = le maximum de BERR / ( (n+1)*EPS + (*) ), où
!>             (*) = (n+1)*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si la matrice A est triangulaire supérieure ou inférieure.
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

TRANS

!>          TRANS est CHARACTER*1
!>          Spécifie la forme du système d'équations.
!>          = 'N':  A * X = B  (Pas de transposée)
!>          = 'T':  A'* X = B  (Transposée)
!>          = 'C':  A'* X = B  (Conjuguée transposée = Transposée)
!>

DIAG

!>          DIAG est CHARACTER*1
!>          Spécifie si la matrice A est unitaire ou non.
!>          = 'N':  Triangulaire non unitaire
!>          = 'U':  Triangulaire unitaire
!>

N

!>          N est INTEGER
!>          Le nombre de lignes des matrices X, B et XACT, et l'ordre de
!>          la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de colonnes des matrices X, B et XACT.
!>          NRHS >= 0.
!>

AP

!>          AP est un tableau COMPLEX*16, dimension (N*(N+1)/2)
!>          La matrice triangulaire supérieure ou inférieure A, stockée en
!>          colonne compactée dans un tableau linéaire. La j-ième colonne
!>          de A est stockée dans le tableau AP comme suit :
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) pour 1<=i<=j;
!>          si UPLO = 'L', AP(i + (j-1)*(2n-j)/2) = A(i,j) pour j<=i<=n.
!>          Si DIAG = 'U', les éléments diagonaux de A ne sont pas référencés
!>          et sont supposés être 1.
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,NRHS)
!>          Les vecteurs du second membre pour le système d'équations
!>          linéaires.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est un tableau COMPLEX*16, dimension (LDX,NRHS)
!>          Les vecteurs solutions calculés. Chaque vecteur est stocké comme
!>          une colonne de la matrice X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

XACT

!>          XACT est un tableau COMPLEX*16, dimension (LDXACT,NRHS)
!>          Les vecteurs solutions exacts. Chaque vecteur est stocké comme
!>          une colonne de la matrice XACT.
!>

LDXACT

!>          LDXACT est INTEGER
!>          La dimension principale du tableau XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>          Les bornes d'erreur avant estimées pour chaque vecteur solution
!>          X. Si XTRUE est la vraie solution, FERR borne la magnitude de
!>          la plus grande entrée de (X - XTRUE) divisée par la magnitude
!>          de la plus grande entrée de X.
!>

BERR

!>          BERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>          L'erreur relative composante par composante de chaque vecteur
!>          solution (c'est-à-dire le plus petit changement relatif dans
!>          une entrée de A ou B qui rend X une solution exacte).
!>

RESLTS

!>          RESLTS est un tableau DOUBLE PRECISION, dimension (2)
!>          Le maximum sur les NRHS vecteurs solutions des ratios :
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( (n+1)*EPS + (*) )
!>
