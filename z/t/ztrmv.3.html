subroutine ztrmv (caractère uplo, caractère trans, caractère diag, entier n, complexe*16, dimension(lda,*) a, entier lda, complexe*16, dimension(*) x, entier incx)

ZTRMV

But :

!>
!> ZTRMV  effectue une des opérations matricielles-vectorielles suivantes
!>
!>    x := A*x,   ou   x := A**T*x,   ou   x := A**H*x,
!>
!> où x est un vecteur de n éléments et A est une matrice triangulaire unitaire ou non, supérieure ou inférieure, de taille n par n.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>           À l'entrée, UPLO spécifie si la matrice est supérieure ou
!>           inférieure triangulaire comme suit :
!>
!>              UPLO = 'U' ou 'u'   A est une matrice triangulaire supérieure.
!>
!>              UPLO = 'L' ou 'l'   A est une matrice triangulaire inférieure.
!>

TRANS

!>          TRANS est CHARACTER*1
!>           À l'entrée, TRANS spécifie l'opération à effectuer comme suit :
!>
!>              TRANS = 'N' ou 'n'   x := A*x.
!>
!>              TRANS = 'T' ou 't'   x := A**T*x.
!>
!>              TRANS = 'C' ou 'c'   x := A**H*x.
!>

DIAG

!>          DIAG est CHARACTER*1
!>           À l'entrée, DIAG spécifie si A est unitaire triangulaire ou non
!>           comme suit :
!>
!>              DIAG = 'U' ou 'u'   A est supposée être unitaire triangulaire.
!>
!>              DIAG = 'N' ou 'n'   A n'est pas supposée être unitaire
!>                                  triangulaire.
!>

N

!>          N est INTEGER
!>           À l'entrée, N spécifie l'ordre de la matrice A.
!>           N doit être supérieur ou égal à zéro.
!>

A

!>          A est un tableau COMPLEX*16, de dimension ( LDA, N ).
!>           Avant l'entrée avec  UPLO = 'U' ou 'u', la partie supérieure triangulaire principale de taille n par n du tableau A doit contenir la matrice triangulaire supérieure et la partie strictement inférieure triangulaire de A n'est pas référencée.
!>           Avant l'entrée avec UPLO = 'L' ou 'l', la partie inférieure triangulaire principale de taille n par n du tableau A doit contenir la matrice triangulaire inférieure et la partie strictement supérieure triangulaire de A n'est pas référencée.
!>           Notez que lorsque DIAG = 'U' ou 'u', les éléments diagonaux de
!>           A ne sont pas non plus référencés, mais sont supposés être l'unité.
!>

LDA

!>          LDA est INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle que déclarée
!>           dans le programme appelant (sous-programme). LDA doit être au moins
!>           max( 1, n ).
!>

X

!>          X est un tableau COMPLEX*16, de dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           Avant l'entrée, le tableau X incrémenté doit contenir le vecteur
!>           x de n éléments. À la sortie, X est écrasé par le vecteur transformé x.
!>

INCX

!>          INCX est INTEGER
!>           À l'entrée, INCX spécifie l'incrément pour les éléments de
!>           X. INCX ne doit pas être nul.
!>
