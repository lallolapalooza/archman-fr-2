subroutine ztrtrs (uplo, trans, diag, n, nrhs, a, lda, b, ldb, info)
ZTRTRS

Documentation de fonction/sous-routine

sous-routine ztrtrs (caractère uplo, caractère trans, caractère diag, entier n, entier nrhs, complexe*16, dimension( lda, * ) a, entier lda, complexe*16, dimension( ldb, * ) b, entier ldb, entier info)

ZTRTRS

Objectif :

!>
!> ZTRTRS résout un système triangulaire de la forme
!>
!>    A * X = B,  A**T * X = B,  ou  A**H * X = B,
!>
!> où A est une matrice triangulaire d'ordre N, et B est une matrice N-par-NRHS.
!>
!> Cette sous-routine vérifie que A est non singulière, mais les appelants doivent noter que seule la
!> singularité exacte est détectée. Il est concevable que l'un ou plusieurs éléments diagonaux de A soient
!> des nombres anormalement petits sans que cette sous-routine ne signale d'erreur.
!>
!> Si une perte possible de précision numérique due à des matrices quasi-singulières est une préoccupation,
!> l'appelant doit vérifier que A est non singulière dans une certaine tolérance avant d'appeler cette sous-routine.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U':  A est triangulaire supérieure ;
!>          = 'L':  A est triangulaire inférieure.
!>

TRANS

!>          TRANS est CHARACTER*1
!>          Spécifie la forme du système d'équations :
!>          = 'N':  A * X = B     (Pas de transposée)
!>          = 'T':  A**T * X = B  (Transposée)
!>          = 'C':  A**H * X = B  (Transposée conjuguée)
!>

DIAG

!>          DIAG est CHARACTER*1
!>          = 'N':  A est triangulaire non-unitaire ;
!>          = 'U':  A est triangulaire unitaire.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!>

A

!>          A est un tableau COMPLEX*16, dimension (LDA,N)
!>          La matrice triangulaire A.  Si UPLO = 'U', la partie supérieure
!>          triangulaire de taille N-par-N du tableau A contient la matrice
!>          triangulaire supérieure, et la partie strictement triangulaire inférieure
!>          de A n'est pas référencée.  Si UPLO = 'L', la partie inférieure
!>          triangulaire de taille N-par-N du tableau A contient la matrice
!>          triangulaire inférieure, et la partie strictement triangulaire supérieure
!>          de A n'est pas référencée.  Si DIAG = 'U', les éléments diagonaux de A
!>          ne sont pas non plus référencés et sont supposés être 1.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

B

!>          B est un tableau COMPLEX*16, dimension (LDB,NRHS)
!>          À l'entrée, la matrice membre de droite B.
!>          À la sortie, si INFO = 0, la matrice solution X.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

INFO

!>          INFO est INTEGER
!>          = 0:  sortie réussie
!>          < 0: si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0:  si INFO = i, le i-ème élément diagonal de A est exactement zéro,
!>               indiquant que la matrice est singulière et que les solutions
!>               X n'ont pas été calculées.
!>
