MLX4DV_INIT_OBJ(3) Manuel des fonctions de bibliothèque MLX4DV_INIT_OBJ(3)

NOM

mlx4dv_init_obj - Initialiser un objet mlx4 direct verbs à partir de structures ibv_xxx

SYNTAXE

#include <infiniband/mlx4dv.h>

int mlx4dv_init_obj(struct mlx4dv_obj *obj, uint64_t obj_type);

DESCRIPTION

mlx4dv_init_obj() Cette fonction initialisera les structures mlx4dv_xxx en fonction du type fourni. Les informations d'initialisation sont extraites des structures ibv_xx fournies en entrée.

La demande d'informations sur le CQ le marque comme appartenant aux direct verbs pour toutes les actions liées à l'indice du consommateur. Le type d'initialisation peut être une combinaison de plusieurs types.

struct mlx4dv_qp {
uint32_t         *rdb;
uint32_t         *sdb;
struct {
uint32_t        wqe_cnt;
int             wqe_shift;
int             offset;
} sq;
struct {
uint32_t        wqe_cnt;
int             wqe_shift;
int             offset;
} rq;
struct {
void            *buf;
size_t          length;
} buf;
uint64_t        comp_mask; /* Utiliser enum mlx4dv_qp_comp_mask */
off_t           uar_mmap_offset; /* Si MLX4DV_QP_MASK_UAR_MMAP_OFFSET est défini dans comp_mask, ceci contiendra l'offset mmap de *sdb* */
};
struct mlx4dv_cq {
struct {
void            *buf;
size_t          length;
} buf;
uint32_t        cqe_cnt;
uint32_t        cqn;
uint32_t        *set_ci_db;
uint32_t        *arm_db;
int             arm_sn;
int             cqe_size;
uint64_t        comp_mask; /* Utiliser enum mlx4dv_cq_comp_mask */
void            *cq_uar;
};
struct mlx4dv_srq {
struct {
void            *buf;
size_t          length;
} buf;
int             wqe_shift;
int             head;
int             tail;
uint32_t        *db;
uint64_t        comp_mask;
};
struct mlx4dv_rwq {
__be32          *rdb;
struct {
uint32_t        wqe_cnt;
int             wqe_shift;
int             offset;
} rq;
struct {
void            *buf;
size_t          length;
} buf;
uint64_t        comp_mask;
};
struct mlx4dv_obj {
struct {
struct ibv_qp      *in;
struct mlx4dv_qp   *out;
} qp;
struct {
struct ibv_cq      *in;
struct mlx4dv_cq   *out;
} cq;
};
enum mlx4dv_obj_type {
MLX4DV_OBJ_QP   = 1 << 0,
MLX4DV_OBJ_CQ   = 1 << 1,
MLX4DV_OBJ_SRQ  = 1 << 2,
};

VALEUR RETOUR

0 en cas de succès ou la valeur d'errno en cas d'échec (qui indique la raison de l'échec).

NOTES


* Les masques de compatibilité (comp_mask) sont des champs d'entrée/sortie.
