MLX5DV_QUERY_DEVICE(3) Manuel des fonctions de bibliothèque MLX5DV_QUERY_DEVICE(3)

NOM

mlx5dv_query_device - interroger les capacités de l'appareil spécifiques à mlx5

SYNTAXE

#include <infiniband/mlx5dv.h>

int mlx5dv_query_device(struct ibv_context *ctx_in,
                        struct mlx5dv_context *attrs_out);

DESCRIPTION

mlx5dv_query_device() interroge les informations spécifiques au matériel qui sont importantes pour le chemin des données, mais qui ne sont pas fournies par ibv_query_device(3).

Cette fonction renvoie la version, les drapeaux et le masque de compatibilité. La version représente le format des structures matérielles internes que mlx5dv.h représente. L'ajout de nouveaux champs aux structures existantes est géré par le champ comp_mask.

struct mlx5dv_sw_parsing_caps {
uint32_t sw_parsing_offloads; /* Utiliser enum mlx5dv_sw_parsing_offloads */
uint32_t supported_qpts;
};
struct mlx5dv_striding_rq_caps {
uint32_t min_single_stride_log_num_of_bytes; /* taille min. en log de chaque pas */
uint32_t max_single_stride_log_num_of_bytes; /* taille max. en log de chaque pas */
uint32_t min_single_wqe_log_num_of_strides; /* nombre min. de pas par WQE en log */
uint32_t max_single_wqe_log_num_of_strides; /* nombre max. de pas par WQE en log */
uint32_t supported_qpts;
};
struct mlx5dv_dci_streams_caps {
uint8_t max_log_num_concurent; /* nombre max. en log de flux différents concurrents pouvant être gérés par le matériel */
uint8_t max_log_num_errored; /* nombre max. de canaux de flux d'erreur DCI pris en charge par DCI avant qu'un DCI ne passe à un état d'erreur */
};
struct mlx5dv_context {
uint8_t         version;
uint64_t        flags;
uint64_t        comp_mask; /* Utiliser enum mlx5dv_context_comp_mask */
struct mlx5dv_cqe_comp_caps     cqe_comp_caps;
struct mlx5dv_sw_parsing_caps sw_parsing_caps;
uint32_t	tunnel_offloads_caps;
uint32_t        max_dynamic_bfregs /* nombre max. de registres blue-flame pouvant être alloués dynamiquement */
uint64_t        max_clock_info_update_nsec;
uint32_t        flow_action_flags; /* utiliser enum mlx5dv_flow_action_cap_flags */
uint32_t        dc_odp_caps; /* utiliser enum ibv_odp_transport_cap_bits */
void		*hca_core_clock; /* pointe vers un emplacement mémoire mappé à l'horloge centrale du HCA */
struct mlx5dv_sig_caps sig_caps;
size_t max_wr_memcpy_length; /* longueur max. prise en charge par le WR DMA memcpy */
struct mlx5dv_crypto_caps crypto_caps;
uint64_t max_dc_rd_atom; /* Nombre maximum de lectures/atomiques RDMA en attente par QP DC en tant que demandeur */
uint64_t max_dc_init_rd_atom; /* Nombre maximum de lectures/atomiques RDMA en attente par QP DC en tant que répondeur */
struct mlx5dv_reg reg_c0; /* valeur et masque pour faire correspondre le trafic sortant de vport local dans FDB */
struct mlx5dv_ooo_recv_wrs_caps ooo_recv_wrs_caps; /* Nombre maximum de WRs en attente par type de QP hors séquence */
};
enum mlx5dv_context_flags {
/*
 * Ce drapeau indique si la version 0 ou 1 de CQE est nécessaire.
 */
 MLX5DV_CONTEXT_FLAGS_CQE_V1 = (1 << 0),
 MLX5DV_CONTEXT_FLAGS_OBSOLETE    =  (1 << 1), /* Obsolète, ne pas utiliser */
 MLX5DV_CONTEXT_FLAGS_MPW_ALLOWED  = (1 << 2), /* Le WQE multi-paquets est autorisé */
 MLX5DV_CONTEXT_FLAGS_ENHANCED_MPW = (1 << 3), /* Le WQE multi-paquets amélioré est pris en charge ou non */
 MLX5DV_CONTEXT_FLAGS_CQE_128B_COMP = (1 << 4), /* Prend en charge la compression CQE 128B */
 MLX5DV_CONTEXT_FLAGS_CQE_128B_PAD = (1 << 5), /* Prend en charge le remplissage CQE 128B */
 MLX5DV_CONTEXT_FLAGS_PACKET_BASED_CREDIT_MODE = (1 << 6), /* Prend en charge le mode de crédit basé sur les paquets en QP RC */
/*
 * Si CQ a été créé avec IBV_WC_EX_WITH_COMPLETION_TIMESTAMP_WALLCLOCK, les horodatages CQE seront au format temps réel.
 */
 MLX5DV_CONTEXT_FLAGS_REAL_TIME_TS = (1 << 7),
 MLX5DV_CONTEXT_FLAGS_BLUEFLAME = (1 << 8), /* Indique si BlueFlame est pris en charge par l'appareil */
};
enum mlx5dv_context_comp_mask {
MLX5DV_CONTEXT_MASK_CQE_COMPRESION      = 1 << 0,
MLX5DV_CONTEXT_MASK_SWP                 = 1 << 1,
MLX5DV_CONTEXT_MASK_STRIDING_RQ         = 1 << 2,
MLX5DV_CONTEXT_MASK_TUNNEL_OFFLOADS     = 1 << 3,
MLX5DV_CONTEXT_MASK_DYN_BFREGS          = 1 << 4,
MLX5DV_CONTEXT_MASK_CLOCK_INFO_UPDATE   = 1 << 5,
MLX5DV_CONTEXT_MASK_FLOW_ACTION_FLAGS   = 1 << 6,
MLX5DV_CONTEXT_MASK_DC_ODP_CAPS         = 1 << 7,
MLX5DV_CONTEXT_MASK_HCA_CORE_CLOCK      = 1 << 8,
MLX5DV_CONTEXT_MASK_NUM_LAG_PORTS       = 1 << 9,
MLX5DV_CONTEXT_MASK_SIGNATURE_OFFLOAD   = 1 << 10,
MLX5DV_CONTEXT_MASK_DCI_STREAMS         = 1 << 11,
MLX5DV_CONTEXT_MASK_WR_MEMCPY_LENGTH    = 1 << 12,
MLX5DV_CONTEXT_MASK_CRYPTO_OFFLOAD      = 1 << 13,
MLX5DV_CONTEXT_MASK_MAX_DC_RD_ATOM      = 1 << 14,
MLX5DV_CONTEXT_MASK_REG_C0              = 1 << 15,
MLX5DV_CONTEXT_MASK_OOO_RECV_WRS        = 1 << 16,
};
enum enum mlx5dv_sw_parsing_offloads {
MLX5DV_SW_PARSING         = 1 << 0,
MLX5DV_SW_PARSING_CSUM    = 1 << 1,
MLX5DV_SW_PARSING_LSO     = 1 << 2,
};
enum mlx5dv_tunnel_offloads {
MLX5DV_RAW_PACKET_CAP_TUNNELED_OFFLOAD_VXLAN  = 1 << 0,
MLX5DV_RAW_PACKET_CAP_TUNNELED_OFFLOAD_GRE    = 1 << 1,
MLX5DV_RAW_PACKET_CAP_TUNNELED_OFFLOAD_GENEVE = 1 << 2,
};
enum mlx5dv_flow_action_cap_flags {
MLX5DV_FLOW_ACTION_FLAGS_ESP_AES_GCM                = 1 << 0, /* L'action de flux ESP (avec clé AES_GCM) est prise en charge */
MLX5DV_FLOW_ACTION_FLAGS_ESP_AES_GCM_REQ_METADATA   = 1 << 1, /* L'action de flux ESP renvoie toujours les métadonnées dans la charge utile */
MLX5DV_FLOW_ACTION_FLAGS_ESP_AES_GCM_SPI_STEERING   = 1 << 2, /* ESP (avec clé AESGCM) prend en charge la correspondance par SPI (plutôt que le hachage sur SPI) */
MLX5DV_FLOW_ACTION_FLAGS_ESP_AES_GCM_FULL_OFFLOAD   = 1 << 3, /* L'action de flux ESP prend en charge la décharge complète (avec clé AES_GCM) */
MLX5DV_FLOW_ACTION_FLAGS_ESP_AES_GCM_TX_IV_IS_ESN   = 1 << 4, /* Action de flux ESP (avec clé AES_GCM), ESN provient implicitement de IV. */
};
struct mlx5dv_sig_caps {
uint64_t block_size; /* utiliser enum mlx5dv_block_size_caps */
uint32_t block_prot; /* utiliser enum mlx5dv_sig_prot_caps */
uint16_t t10dif_bg; /* utiliser enum mlx5dv_sig_t10dif_bg_caps */
uint16_t crc_type; /* utiliser enum mlx5dv_sig_crc_type_caps */
};
enum mlx5dv_sig_prot_caps {
MLX5DV_SIG_PROT_CAP_T10DIF = 1 << MLX5DV_SIG_TYPE_T10DIF,
MLX5DV_SIG_PROT_CAP_CRC = 1 << MLX5DV_SIG_TYPE_CRC,
};
enum mlx5dv_sig_t10dif_bg_caps {
MLX5DV_SIG_T10DIF_BG_CAP_CRC = 1 << MLX5DV_SIG_T10DIF_CRC,
MLX5DV_SIG_T10DIF_BG_CAP_CSUM = 1 << MLX5DV_SIG_T10DIF_CSUM,
};
enum mlx5dv_sig_crc_type_caps {
MLX5DV_SIG_CRC_TYPE_CAP_CRC32 = 1 << MLX5DV_SIG_CRC_TYPE_CRC32,
MLX5DV_SIG_CRC_TYPE_CAP_CRC32C = 1 << MLX5DV_SIG_CRC_TYPE_CRC32C,
MLX5DV_SIG_CRC_TYPE_CAP_CRC64_XP10 = 1 << MLX5DV_SIG_CRC_TYPE_CRC64_XP10,
};
enum mlx5dv_block_size_caps {
MLX5DV_BLOCK_SIZE_CAP_512 = 1 << MLX5DV_BLOCK_SIZE_512,
MLX5DV_BLOCK_SIZE_CAP_520 = 1 << MLX5DV_BLOCK_SIZE_520,
MLX5DV_BLOCK_SIZE_CAP_4048 = 1 << MLX5DV_BLOCK_SIZE_4048,
MLX5DV_BLOCK_SIZE_CAP_4096 = 1 << MLX5DV_BLOCK_SIZE_4096,
MLX5DV_BLOCK_SIZE_CAP_4160 = 1 << MLX5DV_BLOCK_SIZE_4160,
};
struct mlx5dv_crypto_caps {
/*
 * si failed_selftests != 0, cela signifie qu'il y a des erreurs d'auto-tests
 * qui peuvent rendre certains moteurs cryptographiques inutilisables. La signification exacte du code
 * doit être vérifiée auprès de NVIDIA.
 */
uint16_t failed_selftests;
uint8_t crypto_engines; /* utiliser enum mlx5dv_crypto_engines_caps */
uint8_t wrapped_import_method; /* utiliser enum mlx5dv_crypto_wrapped_import_method_caps */
uint8_t log_max_num_deks;
uint32_t flags; /* utiliser enum mlx5dv_crypto_caps_flags */
};
/* Ce bitmap indique quels moteurs cryptographiques sont pris en charge par l'appareil. */
enum mlx5dv_crypto_engines_caps {
/* Obsolète, remplacé par MLX5DV_CRYPTO_ENGINES_CAP_AES_XTS_SINGLE_BLOCK */
 MLX5DV_CRYPTO_ENGINES_CAP_AES_XTS = 1 << 0,
/*
 * Indique que AES-XTS ne prend en charge que le chiffrement d'un seul bloc
 * à la fois.
 */
 MLX5DV_CRYPTO_ENGINES_CAP_AES_XTS_SINGLE_BLOCK = 1 << 1,
/* Indique que AES-XTS prend en charge le chiffrement multi-blocs. */
 MLX5DV_CRYPTO_ENGINES_CAP_AES_XTS_MULTI_BLOCK = 1 << 2,
};
/*
 * Ce bitmap indique la méthode d'importation de chaque moteur cryptographique.
 *
 * Si un bit n'est pas défini, le moteur cryptographique correspondant utilise la méthode d'importation en clair et
 * l'application doit utiliser des DEK en clair pour ce moteur cryptographique.
 *
 * Sinon, le moteur cryptographique correspondant utilise la méthode d'importation enveloppée et l'application
 * doit utiliser des DEK enveloppés pour ce moteur cryptographique. Pour charger des DEK enveloppés, l'application doit effectuer
 * une connexion cryptographique, ce qui nécessite que MLX5DV_CRYPTO_CAPS_WRAPPED_CRYPTO_OPERATIONAL ci-dessous soit défini.
 */
enum mlx5dv_crypto_wrapped_import_method_caps {
 MLX5DV_CRYPTO_WRAPPED_IMPORT_METHOD_CAP_AES_XTS = 1 << 0,
};
enum mlx5dv_crypto_caps_flags {
/* Indique si les capacités cryptographiques sont activées sur l'appareil. */
 MLX5DV_CRYPTO_CAPS_CRYPTO = 1 << 0,
/* Indique si les moteurs cryptographiques qui utilisent la méthode d'importation enveloppée sont opérationnels. */
 MLX5DV_CRYPTO_CAPS_WRAPPED_CRYPTO_OPERATIONAL = 1 << 1,
/*
 * Si défini, indique qu'après le prochain redémarrage du firmware, l'appareil reviendra
 * en mode de mise en service, ce qui signifie que MLX5DV_CRYPTO_CAPS_WRAPPED_CRYPTO_OPERATIONAL
 * sera défini sur 0.
 */
 MLX5DV_CRYPTO_CAPS_WRAPPED_CRYPTO_GOING_TO_COMMISSIONING = 1 << 2,
};
struct mlx5dv_ooo_recv_wrs_caps {
uint32_t max_rc;
uint32_t max_xrc;
uint32_t max_dct;
uint32_t max_ud;
uint32_t max_uc;
};
