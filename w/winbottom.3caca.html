NAME

caca_conio_text_info - structure for text information

SYNOPSIS

#include <libcaca.h>

struct caca_conio_text_info {
  unsigned char attribute;
  unsigned char attribute_text;
  unsigned char attribute_back;
  unsigned char attribute_bright;
  unsigned char attribute_blink;
  unsigned short text_attr;
  unsigned short norm_attr;
  unsigned short curtype;
  unsigned short screen_width;
  unsigned short screen_height;
  unsigned short screen_x;
  unsigned short screen_y;
  unsigned char *text;
  unsigned char *attribute;
  unsigned char c_curx;
  unsigned char c_cury;
  unsigned short rx;
  unsigned char cury;
};

Detailed Description

This structure is used to store information about the text mode of the libcaca library.
